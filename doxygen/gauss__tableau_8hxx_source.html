3/*!*************************************************************************************************
4 * \brief Gauss-Legendre collocation tableau in the eigenbasis of the Butcher matrix.
5 *
6 * Port of the tableau setup of hoRK's horkirk.c (HORKIRKBuildTableau): Gauss nodes/weights by
7 * Newton iteration on the Legendre polynomial, Butcher matrix A = W V^{-1} from the collocation
8 * conditions, eigendecomposition A = T diag(theta) T^{-1} via LAPACK zgeev (trick 1), conjugate
9 * eigenvalue pairing with enforced v_partner = conj(v_rep) and phase-fixed real eigenvector
10 * columns (trick 2) so only the ceil(s/2) representatives are ever solved.
11 *
12 * Difference to horkirk: horkirk works with stage DERIVATIVES (update weights b^T T); the HDG
13 * Gauss stepping (doc/hdg_gauss.pdf) works with stage VALUES, so the recombination weights here are
14 * w = (b^T T) / theta = (b^T A^{-1} T) and the endpoint update reads
15 * y+ = affine * y^n + sum_reps mult_l * Re(w_l * y_l), affine = 1 - b^T A^{-1} 1 = (-1)^s.
16 **************************************************************************************************/
17

44/*!*************************************************************************************************
45 * \brief Gauss-Legendre nodes c (mapped to [0,1]) and weights b (sum to 1).
46 **************************************************************************************************/
47inline void gauss_legendre(const unsigned int s, std::vector<double>& c, std::vector<double>& b)
48{
49 c.resize(s);
50 b.resize(s);
51 for (unsigned int i = 0; i < s; i++)
52 {
53 double x = std::cos(M_PI * (i + 0.75) / (s + 0.5)); // initial guess
54 double p0 = 1., p1 = x, dp = 1.;
55 for (unsigned int it = 0; it < 100; it++)
56 {
57 p0 = 1.;
58 p1 = x;
59 for (unsigned int k = 2; k <= s; k++) // P_k recurrence; p1 = P_s, p0 = P_{s-1}
60 {
61 const double p2 = ((2. * k - 1.) * x * p1 - (k - 1.) * p0) / k;
62 p0 = p1;
63 p1 = p2;
64 }
65 dp = s * (x * p1 - p0) / (x * x - 1.); // P_s'(x)
66 const double dx = -p1 / dp;
67 x += dx;
68 if (std::abs(dx) < 1e-15)
69 break;
70 }
71 const double weight = 2. / ((1. - x * x) * dp * dp); // weight on [-1,1]
72 c[i] = 0.5 * (x + 1.); // map to [0,1]
73 b[i] = 0.5 * weight;
74 }
75}

77/*!*************************************************************************************************
78 * \brief Eigenbasis tableau data of the s-stage Gauss collocation method (order 2s).
79 **************************************************************************************************/
80struct Tableau
81{
82 unsigned int s; // number of stages
83 unsigned int n_reps; // number of solved representatives, ceil(s/2)
84 std::vector<double> c; // collocation nodes in (0,1), length s
85 std::vector<std::complex<double>> theta; // representative Butcher eigenvalues, length n_reps
86 std::vector<std::complex<double>> omega; // load weights (T^{-1} 1)_rep
87 std::vector<std::complex<double>> w; // value-form recombination weights (b^T T)_rep/theta
88 std::vector<std::complex<double>> tinv; // T^{-1} rows of the reps, row-major n_reps x s
89 std::vector<double> mult; // multiplicity: 2 for a conjugate pair, else 1
90 double affine; // 1 - b^T A^{-1} 1 = R(infinity) = (-1)^s
91};

93/*!*************************************************************************************************
94 * \brief Build the tableau; see the file header. All dense work is on tiny s x s systems.
95 **************************************************************************************************/
96inline Tableau build_tableau(const unsigned int s)
97{
98 using cplx = std::complex<double>;
99 constexpr double tol = 1e-9;
100 Tableau tab;
101 tab.s = s;
102
103 std::vector<double> c, b;
104 gauss_legendre(s, c, b);
105 tab.c = c;
106
107 // Butcher A = W V^{-1}, V_{jk} = c_j^k, W_{jk} = c_j^{k+1}/(k+1) (column-major, complexified)
108 std::vector<cplx> V(s * s), W(s * s), Vinv(s * s, 0.), A(s * s, 0.);
109 for (unsigned int k = 0; k < s; k++)
110 for (unsigned int j = 0; j < s; j++)
111 {
112 V[j + s * k] = std::pow(c[j], (double)k);
113 W[j + s * k] = std::pow(c[j], (double)(k + 1)) / (k + 1.);
114 }
115 std::vector<int> ipiv(s);
116 for (unsigned int i = 0; i < s; i++)
117 Vinv[i + s * i] = 1.;
118 Wrapper::lapack_factorize((int)s, V.data(), ipiv.data());
119 Wrapper::lapack_solve_factored((int)s, (int)s, V.data(), ipiv.data(), Vinv.data());
120 for (unsigned int j = 0; j < s; j++)
121 for (unsigned int i = 0; i < s; i++)
122 for (unsigned int k = 0; k < s; k++)
123 A[i + s * j] += W[i + s * k] * Vinv[k + s * j];
124
125 // eigendecomposition A = T diag(theta) T^{-1}
126 std::vector<cplx> T(s * s), eigval(s);
127 Wrapper::lapack_eig((int)s, A.data(), eigval.data(), T.data());
128
129 // pair conjugate eigenvalues; enforce v_partner = conj(v_rep) so the conjugation of the
130 // stage solutions is exact and the partners never need solving
131 std::vector<bool> taken(s, false), is_rep(s, false);
132 std::vector<int> partner(s, -1);
133 std::vector<unsigned int> reps;
134 for (unsigned int i = 0; i < s; i++)
135 {
136 if (taken[i])
137 continue;
138 reps.push_back(i);
139 is_rep[i] = true;
140 taken[i] = true;
141 for (unsigned int j = i + 1; j < s; j++)
142 if (!taken[j] && std::abs(eigval[j] - std::conj(eigval[i])) < tol)
143 {
144 for (unsigned int k = 0; k < s; k++)
145 T[k + s * j] = std::conj(T[k + s * i]);
146 partner[i] = (int)j;
147 taken[j] = true;
148 break;
149 }
150 }
151
152 // real (unpaired) eigenvalues: zgeev returns their eigenvector with an arbitrary complex
153 // phase; rotate it real so conj(T) = T P holds exactly (only occurs for odd s)
154 for (const unsigned int i : reps)
155 {
156 if (partner[i] >= 0)
157 continue;
158 double maxa = -1.;
159 unsigned int km = 0;
160 for (unsigned int k = 0; k < s; k++)
161 if (std::abs(T[k + s * i]) > maxa)
162 {
163 maxa = std::abs(T[k + s * i]);
164 km = k;
165 }
166 const cplx ph = T[km + s * i] / maxa; // unit phase of largest entry
167 for (unsigned int k = 0; k < s; k++)
168 T[k + s * i] /= ph;
169 }
170
171 // omega = T^{-1} 1 and the full T^{-1} (both via the factored T; gesv-style copy)
172 std::vector<cplx> Tcopy(T), omega_full(s, 1.), Tinv(s * s, 0.);
173 for (unsigned int i = 0; i < s; i++)
174 Tinv[i + s * i] = 1.;
175 Wrapper::lapack_factorize((int)s, Tcopy.data(), ipiv.data());
176 Wrapper::lapack_solve_factored((int)s, 1, Tcopy.data(), ipiv.data(), omega_full.data());
177 Wrapper::lapack_solve_factored((int)s, (int)s, Tcopy.data(), ipiv.data(), Tinv.data());
178
179 // value-form recombination weights w_j = (b^T T)_j / theta_j and the affine coefficient
180 // 1 - b^T A^{-1} 1 = 1 - sum_j w_j omega_j (real by conjugate symmetry; must be (-1)^s)
181 std::vector<cplx> w_full(s);
182 cplx affine = 1.;
183 for (unsigned int j = 0; j < s; j++)
184 {
185 cplx acc = 0.;
186 for (unsigned int i = 0; i < s; i++)
187 acc += b[i] * T[i + s * j];
188 w_full[j] = acc / eigval[j];
189 affine -= w_full[j] * omega_full[j];
190 }
191 hy_assert(std::abs(affine.imag()) < 1e-10,
192 "affine coefficient must be real; got imag = " << affine.imag());
193 hy_assert(std::abs(affine.real() - (s % 2 ? -1. : 1.)) < 1e-10,
194 "affine coefficient must be R(inf) = (-1)^s; got " << affine.real());
195
196 tab.n_reps = (unsigned int)reps.size();
197 hy_assert(tab.n_reps == (s + 1) / 2, "expected ceil(s/2) representatives");
198 tab.affine = affine.real();
199 for (const unsigned int i : reps)
200 {
201 tab.theta.push_back(eigval[i]);
202 tab.omega.push_back(omega_full[i]);
203 tab.w.push_back(w_full[i]);
204 tab.mult.push_back(partner[i] >= 0 ? 2. : 1.);
205 for (unsigned int j = 0; j < s; j++)
206 tab.tinv.push_back(Tinv[i + s * j]); // row i of T^{-1}
207 }
208 return tab;
209}

26 bool from_python(handle src, uint8_t flags, cleanup_list *cleanup) noexcept {
27 value.clear();
28
29 if (!PyDict_CheckExact(src.ptr()) && !PyMapping_Check(src.ptr()))
30 return false;
31
32 PyObject *items = PyMapping_Items(src.ptr());
33 if (items == nullptr) {
34 PyErr_Clear();
35 return false;
36 }
37
38 // 'items' is safe to access without locking and reference counting, it
39 // is unique to this thread
40 Py_ssize_t size = NB_LIST_GET_SIZE(items);
41 bool success = size >= 0;
42
43 uint8_t flags_key = flags_for_local_caster<Key>(flags),
44 flags_val = flags_for_local_caster<Val>(flags);
45
46 KeyCaster key_caster;
47 ValCaster val_caster;
48 for (Py_ssize_t i = 0; i < size; ++i) {
49 PyObject *item = NB_LIST_GET_ITEM(items, i);
50 PyObject *key = NB_TUPLE_GET_ITEM(item, 0);
51 PyObject *val = NB_TUPLE_GET_ITEM(item, 1);
52
53 if (!key_caster.from_python(key, flags_key, cleanup) ||
54 !key_caster.template can_cast<Key>()) {
55 success = false;
56 break;
57 }
58
59 if (!val_caster.from_python(val, flags_val, cleanup) ||
60 !val_caster.template can_cast<Val>()) {
61 success = false;
62 break;
63 }
64
65 value.emplace(key_caster.operator cast_t<Key>(),
66 val_caster.operator cast_t<Val>());
67 }
68
69 Py_DECREF(items);
70
71 return success;
72 }

74 template <typename T>
75 static handle from_cpp(T &&src, rv_policy policy, cleanup_list *cleanup) {
76 dict ret;
77
78 if (ret.is_valid()) {
79 for (auto &item : src) {
80 object k = steal(KeyCaster::from_cpp(
81 forward_like_<T>(item.first), policy, cleanup));
82 object e = steal(ValCaster::from_cpp(
83 forward_like_<T>(item.second), policy, cleanup));
84
85 if (!k.is_valid() || !e.is_valid() ||
86 PyDict_SetItem(ret.ptr(), k.ptr(), e.ptr()) != 0) {
87 ret.reset();
88 break;
89 }
90 }
91 }
92
93 return ret.release();
94 }

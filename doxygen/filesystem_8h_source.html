37 template <typename Char = typename std::filesystem::path::value_type>
38 bool from_python(handle src, uint8_t, cleanup_list *) noexcept {
39 bool success = false;
40
41 /* PyUnicode_FSConverter and PyUnicode_FSDecoder normally take care of
42 calling PyOS_FSPath themselves, but that's broken on PyPy (see PyPy
43 issue #3168) so we do it ourselves instead. */
44 PyObject *buf = PyOS_FSPath(src.ptr());
45 if (buf) {
46 PyObject *native = nullptr;
47 if constexpr (std::is_same_v<Char, char>) {
48 if (PyUnicode_FSConverter(buf, &native)) {
49 if (char* s = PyBytes_AsString(native)) {
50 value = s; // Points to internal buffer, no need to free
51 success = true;
52 }
53 }
54 } else {
55 if (PyUnicode_FSDecoder(buf, &native)) {
56 if (wchar_t *s = PyUnicode_AsWideCharString(native, nullptr)) {
57 value = s;
58 PyMem_Free(s); // New string, must free
59 success = true;
60 }
61 }
62 }
63
64 Py_DECREF(buf);
65 Py_XDECREF(native);
66 }
67
68 if (!success)
69 PyErr_Clear();
70
71 return success;
72 }

15/*!*************************************************************************************************
16 * \brief Layouts of local operators: what a local solver stores of its condensed local problem.
17 *
18 * A local solver of the hybridized kind (Diffusion, ...) describes the local problem of a
19 * hyperedge by its blocks (\c Blocks)
20 *
21 * A x = B lambda + l, flux = C x + D lambda,
22 *
23 * with the local unknowns x (n_ldofs), the trace lambda (n_loc), and the local load l, which
24 * depends on the hyperedge, the time and the data (local_load of the solver). If the leading
25 * n_eliminate unknowns form a diagonal block of A (e.g. the flux unknowns of an LDG-H method with
26 * an orthonormal basis, affine elements and coefficients constant on the element), the solver says
27 * so in the type of its blocks; the factorization then eliminates them explicitly and factorizes
28 * the Schur complement of the others only (\c detail::Factors), in every layout.
29 *
30 * A layout makes the operator GlobalLoop::Generic applies: any type with
31 *
32 * static auto make(const SolverT& solver, Edge& edge, Kind kind, const Args& args);
33 *
34 * returning an operator with apply<k>(solver, edge, in, args, out),
35 * residual(solver, edge, lambda, args, r) and optionally local_solution(solver, edge, lambda,
36 * args). The layout asks the solver for what it needs (local_blocks(edge, kind, args), and
37 * local_load(edge, args) in residual and local_solution); the operator is handed the solver and
38 * the hyperedge in every call, so it keeps neither. The solver is templated on its layout and has
39 * a default; users can write their own (e.g. single-precision or symmetric storage) without
40 * touching the solver.
41 *
42 * Provided: \c Condensed (the default of Diffusion) stores S = C A^-1 B + D and K = C A^-1, so
43 * that apply is S lambda and residual is S lambda + K l, without factors; \c Factored stores the
44 * factors of A with B, C, D, as the local problem is set up, and also provides local_solution;
45 * \c Streaming stores nothing and sets up the local problem in every call.
46 *
47 * Which hyperedges read the same stored operator is not the layout's business: the solver numbers
48 * them (operator_slot, cf. global_loop/generic.hxx), and operators without state, as those of
49 * Streaming, are one for all hyperedges.
50 **************************************************************************************************/

53/*!*************************************************************************************************
54 * \brief The blocks of a condensed local problem, A x = B lambda + l, flux = C x + D lambda.
55 *
56 * \tparam n_eliminateT The leading n_eliminateT unknowns form a diagonal block of A, uncoupled
57 * among themselves, and are eliminated explicitly; 0: factorize A whole.
58 **************************************************************************************************/
59template <unsigned int n_ldofsT, unsigned int n_locT, typename T, unsigned int n_eliminateT = 0>
60struct Blocks
61{
62 static constexpr unsigned int n_ldofs = n_ldofsT, n_loc = n_locT, n_eliminate = n_eliminateT;
63 using value_type = T;
64 static_assert(n_eliminate < n_ldofs, "At least one unknown is left to the factorization.");
65
66 SmallSquareMat<n_ldofs, T> A;
67 SmallMat<n_ldofs, n_loc, T> B;
68 SmallMat<n_loc, n_ldofs, T> C;
69 SmallSquareMat<n_loc, T> D;
70};

88/*!*************************************************************************************************
89 * \brief Factors of the local matrix A, with the diagonal block of the leading n_eliminate
90 * unknowns eliminated explicitly.
91 *
92 * With A = [D_A, A_DR; A_RD, A_RR] (D_A diagonal, of size n_eliminate), A x = b is
93 * x_R = S^-1 (b_R - A_RD D_A^-1 b_D), x_D = D_A^-1 (b_D - A_DR x_R),
94 * with the Schur complement S = A_RR - A_RD D_A^-1 A_DR, of which LU factors are kept. Building
95 * costs about 2 n_D n_R^2 + 2/3 n_R^3 instead of 2/3 n^3 flops, a solve about 2 n_R^2 + 4 n_D n_R
96 * instead of 2 n^2 per right-hand side, and the factors (2 n_D + n_R) n_R + n_D numbers instead of
97 * n^2. Debug builds check that D_A is diagonal (up to 100 epsilon of T, relative).
98 **************************************************************************************************/
99template <unsigned int n, unsigned int n_eliminate, typename T>
100struct Factors
101{
102 static constexpr unsigned int n_d = n_eliminate, n_r = n - n_eliminate;
103
104 SmallVec<n_d, T> d_inv;
105 SmallMat<n_d, n_r, T> a_dr;
106 SmallMat<n_r, n_d, T> a_rd;
107 SmallSquareMat<n_r, T> lu;
108 std::array<int, n_r> ipiv;
109
110 Factors() = default;
111 explicit Factors(const SmallSquareMat<n, T>& a)
112 {
113 for (unsigned int i = 0; i < n_d; ++i)
114 {
115 if constexpr (hy_debug_mode)
116 for (unsigned int j = 0; j < n_d; ++j)
117 hy_assert(
118 i == j ||
119 std::abs(a(i, j)) <= 100 * std::numeric_limits<T>::epsilon() * std::abs(a(i, i)),
120 "Eliminated unknowns must form a diagonal block of the local matrix, but entry ("
121 << i << ", " << j << ") is " << (double)std::abs(a(i, j)) << " in magnitude.");
122 d_inv[i] = (T)1. / a(i, i);
123 for (unsigned int j = 0; j < n_r; ++j)
124 {
125 a_dr(i, j) = a(i, n_d + j);
126 a_rd(j, i) = a(n_d + j, i);
127 }
128 }
129 for (unsigned int j = 0; j < n_r; ++j)
130 for (unsigned int i = 0; i < n_r; ++i)
131 {
132 T schur = a(n_d + i, n_d + j);
133 for (unsigned int k = 0; k < n_d; ++k)
134 schur -= a_rd(i, k) * d_inv[k] * a_dr(k, j);
135 lu(i, j) = schur;
136 }
137 LU::factorize<n_r, T>(lu.data(), ipiv);
138 }

144 template <unsigned int k>
145 void solve(SmallMat<n, k, T>& rhs, const char trans = 'N') const
146 {
147 if (trans == 'N')
148 solve_impl<false>(rhs);
149 else
150 solve_impl<true>(rhs);
151 }
152
153 private:
154 template <bool transposedT, unsigned int k>
155 void solve_impl(SmallMat<n, k, T>& rhs) const
156 {
157 // the blocks of A (or of A^T) below and above the eliminated diagonal block
158 auto lower = [this](const unsigned int i, const unsigned int j)
159 {
160 if constexpr (transposedT)
161 return a_dr(j, i);
162 else
163 return a_rd(i, j);
164 };
165 auto upper = [this](const unsigned int i, const unsigned int j)
166 {
167 if constexpr (transposedT)
168 return a_rd(j, i);
169 else
170 return a_dr(i, j);
171 };
172 SmallMat<n_r, k, T> x_r;
173 for (unsigned int col = 0; col < k; ++col)
174 for (unsigned int i = 0; i < n_r; ++i)
175 {
176 T value = rhs(n_d + i, col);
177 for (unsigned int j = 0; j < n_d; ++j)
178 value -= lower(i, j) * d_inv[j] * rhs(j, col);
179 x_r(i, col) = value;
180 }
181 lu_solve(lu, ipiv, x_r, transposedT ? 'T' : 'N');
182 for (unsigned int col = 0; col < k; ++col)
183 {
184 for (unsigned int i = 0; i < n_d; ++i)
185 {
186 T value = rhs(i, col);
187 for (unsigned int j = 0; j < n_r; ++j)
188 value -= upper(i, j) * x_r(j, col);
189 rhs(i, col) = d_inv[i] * value;
190 }
191 for (unsigned int i = 0; i < n_r; ++i)
192 rhs(n_d + i, col) = x_r(i, col);
193 }
194 }
195};

224/*!*************************************************************************************************
225 * \brief The blocks as the local problem is set up: the factors of A, B, C, D.
226 *
227 * Cheap to build (one factorization), expensive to apply (a solve with A per application), and
228 * it provides local_solution.
229 **************************************************************************************************/
230struct Factored
231{
232 template <class BlocksT>
233 struct Operator
234 {
235 static constexpr unsigned int n_ldofs = BlocksT::n_ldofs, n_loc = BlocksT::n_loc;
236 using T = typename BlocksT::value_type;
237
238 detail::FactorsOf<BlocksT> factors;
239 SmallMat<n_ldofs, n_loc, T> B;
240 SmallMat<n_loc, n_ldofs, T> C;
241 SmallSquareMat<n_loc, T> D;
242
243 template <unsigned int k, class SolverT, class Edge, class Args>
244 void apply(const SolverT&,
245 Edge&,
246 const SmallMat<n_loc, k, T>& in,
247 const Args&,
248 SmallMat<n_loc, k, T>& out) const
249 {
250 SmallMat<n_ldofs, k, T> x = B * in;
251 factors.solve(x);
252 out = C * x + D * in;
253 }
254 template <class SolverT, class Edge, class Args>
255 void residual(const SolverT& solver,
256 Edge& edge,
257 const SmallVec<n_loc, T>& lambda,
258 const Args& args,
259 SmallVec<n_loc, T>& r) const
260 {
261 r = C * local_solution(solver, edge, lambda, args) + D * lambda;
262 }
263 template <class SolverT, class Edge, class Args>
264 SmallVec<n_ldofs, T> local_solution(const SolverT& solver,
265 Edge& edge,
266 const SmallVec<n_loc, T>& lambda,
267 const Args& args) const
268 {
269 SmallVec<n_ldofs, T> x = B * lambda + solver.local_load(edge, args);
270 factors.solve(x);
271 return x;
272 }
273 };
274
275 template <class SolverT, class Edge, class Kind, class Args>
276 static auto make(const SolverT& solver, Edge& edge, Kind kind, const Args& args)
277 {
278 using BlocksT = detail::BlocksOf<SolverT, Edge, Kind, Args>;
279 const BlocksT blocks = solver.local_blocks(edge, kind, args);
280 return Operator<BlocksT>{detail::FactorsOf<BlocksT>(blocks.A), blocks.B, blocks.C, blocks.D};
281 }
282};

284/*!*************************************************************************************************
285 * \brief The condensed operator: S = C A^-1 B + D and K = C A^-1, no factors.
286 *
287 * Building costs a factorization, a solve with n_loc right-hand sides, and one with A^T for K;
288 * applying costs one product with S (n_loc^2 instead of about 2 n_ldofs n_loc + n_ldofs^2 for
289 * Factored), the residual one with K in addition. Smaller than Factored, since n_loc < n_ldofs.
290 * No local_solution: reductions that need it build the Factored operator on the fly. Expensive
291 * to build: without prepare (an operator per hyperedge and sweep), Factored is the cheaper one.
292 **************************************************************************************************/
293struct Condensed
294{
295 template <class BlocksT>
296 struct Operator
297 {
298 static constexpr unsigned int n_ldofs = BlocksT::n_ldofs, n_loc = BlocksT::n_loc;
299 using T = typename BlocksT::value_type;
300
301 SmallSquareMat<n_loc, T> S;
302 SmallMat<n_loc, n_ldofs, T> K;
303
304 template <unsigned int k, class SolverT, class Edge, class Args>
305 void apply(const SolverT&,
306 Edge&,
307 const SmallMat<n_loc, k, T>& in,
308 const Args&,
309 SmallMat<n_loc, k, T>& out) const
310 {
311 out = S * in;
312 }
313 template <class SolverT, class Edge, class Args>
314 void residual(const SolverT& solver,
315 Edge& edge,
316 const SmallVec<n_loc, T>& lambda,
317 const Args& args,
318 SmallVec<n_loc, T>& r) const
319 {
320 r = S * lambda + K * solver.local_load(edge, args);
321 }
322 };
323
324 template <class SolverT, class Edge, class Kind, class Args>
325 static auto make(const SolverT& solver, Edge& edge, Kind kind, const Args& args)
326 {
327 using BlocksT = detail::BlocksOf<SolverT, Edge, Kind, Args>;
328 const BlocksT blocks = solver.local_blocks(edge, kind, args);
329 const detail::FactorsOf<BlocksT> factors(blocks.A);
330 Operator<BlocksT> op;
331 auto ainv_b = blocks.B;
332 factors.solve(ainv_b);
333 op.S = blocks.C * ainv_b + blocks.D;
334 // K = C A^-1, i.e. K^T = A^-T C^T: one solve with the transposed factors
335 auto kt = transposed(blocks.C);
336 factors.solve(kt, 'T');
337 op.K = transposed(kt);
338 return op;
339 }
340};

342/*!*************************************************************************************************
343 * \brief Nothing is stored: every call sets up the local problem of its hyperedge, factorizes
344 * it, uses the factors and drops them (the Factored operator, for one use).
345 *
346 * The operator is empty, so a prepared loop keeps one for all hyperedges and nothing per
347 * hyperedge; its results depend on the hyperedge it is applied to.
348 **************************************************************************************************/
349struct Streaming
350{
351 template <class Kind>
352 struct Operator
353 {
354 template <unsigned int k, unsigned int n_loc, typename T, class SolverT, class Edge, class Args>
355 void apply(const SolverT& solver,
356 Edge& edge,
357 const SmallMat<n_loc, k, T>& in,
358 const Args& args,
359 SmallMat<n_loc, k, T>& out) const
360 {
361 Factored::make(solver, edge, Kind{}, args).apply(solver, edge, in, args, out);
362 }
363 template <unsigned int n_loc, typename T, class SolverT, class Edge, class Args>
364 void residual(const SolverT& solver,
365 Edge& edge,
366 const SmallVec<n_loc, T>& lambda,
367 const Args& args,
368 SmallVec<n_loc, T>& r) const
369 {
370 Factored::make(solver, edge, Kind{}, args).residual(solver, edge, lambda, args, r);
371 }
372 template <unsigned int n_loc, typename T, class SolverT, class Edge, class Args>
373 auto local_solution(const SolverT& solver,
374 Edge& edge,
375 const SmallVec<n_loc, T>& lambda,
376 const Args& args) const
377 {
378 return Factored::make(solver, edge, Kind{}, args).local_solution(solver, edge, lambda, args);
379 }
380 };
381
382 template <class SolverT, class Edge, class Kind, class Args>
383 static Operator<Kind> make(const SolverT&, Edge&, Kind, const Args&)
384 {
385 return Operator<Kind>{};
386 }
387};

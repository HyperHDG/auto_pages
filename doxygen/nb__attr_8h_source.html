26// Basic function argument descriptor (no default value, not locked)
27struct arg {
28 NB_INLINE constexpr explicit arg(const char *name = nullptr) : name_(name), signature_(nullptr) { }
29
30 // operator= can be used to provide a default value
31 template <typename T> NB_INLINE arg_v operator=(T &&value) const;
32
33 // Mutators that don't change default value or locked state
34 NB_INLINE arg &noconvert(bool value = true) {
35 convert_ = !value;
36 return *this;
37 }
38 NB_INLINE arg &none(bool value = true) {
39 none_ = value;
40 return *this;
41 }
42 NB_INLINE arg &sig(const char *value) {
43 signature_ = value;
44 return *this;
45 }
46
47 // After lock(), this argument is locked
48 NB_INLINE arg_locked lock();
49
50 const char *name_, *signature_;
51 uint8_t convert_{ true };
52 bool none_{ false };
53};
54
55// Function argument descriptor with default value (not locked)
56struct arg_v : arg {
57 object value;
58 NB_INLINE arg_v(const arg &base, object &&value)
59 : arg(base), value(std::move(value)) {}
60
61 private:
62 // Inherited mutators would slice off the default, and are not generally needed
63 using arg::noconvert;
64 using arg::none;
65 using arg::sig;
66 using arg::lock;
67};

69// Function argument descriptor that is locked (no default value)
70struct arg_locked : arg {
71 NB_INLINE constexpr explicit arg_locked(const char *name = nullptr) : arg(name) { }
72 NB_INLINE constexpr explicit arg_locked(const arg &base) : arg(base) { }
73
74 // operator= can be used to provide a default value
75 template <typename T> NB_INLINE arg_locked_v operator=(T &&value) const;
76
77 // Mutators must be respecified in order to not slice off the locked status
78 NB_INLINE arg_locked &noconvert(bool value = true) {
79 convert_ = !value;
80 return *this;
81 }
82 NB_INLINE arg_locked &none(bool value = true) {
83 none_ = value;
84 return *this;
85 }
86 NB_INLINE arg_locked &sig(const char *value) {
87 signature_ = value;
88 return *this;
89 }
90
91 // Redundant extra lock() is allowed
92 NB_INLINE arg_locked &lock() { return *this; }
93};

169enum class func_flags : uint32_t {
170 /* Low 3 bits reserved for return value policy */
171
173 has_name = (1 << 4),
175 has_scope = (1 << 5),
177 has_doc = (1 << 6),
179 has_args = (1 << 7),
181 has_var_args = (1 << 8),
183 has_var_kwargs = (1 << 9),
185 is_method = (1 << 10),
187 is_constructor = (1 << 11),
189 is_implicit = (1 << 12),
191 is_operator = (1 << 13),
193 has_free = (1 << 14),
195 return_ref = (1 << 15),
197 has_signature = (1 << 16),

203enum cast_flags : uint8_t {
204 // Enable implicit conversions (code assumes this has value 1, don't reorder..)
205 convert = (1 << 0),
206
207 // Passed to the 'self' argument in a constructor call (__init__)
208 construct = (1 << 1),
209
210 // Indicates that the function dispatcher should accept 'None' arguments
211 accepts_none = (1 << 2),
212
215 none_disallowed = (1 << 3),
216
217 // Indicates that this cast is performed by nb::cast or nb::try_cast.
218 // This implies that objects added to the cleanup list may be
219 // released immediately after the caster's final output value is
220 // obtained, i.e., before it is used.
221 manual = (1 << 4),
222
225 trusted = (1 << 5)
226};

337NB_INLINE void func_extra_apply(F &f, const arg &a, size_t &index) {
338 uint8_t flag = 0;

438template <size_t NArgs, size_t Nurse, size_t Patient>
439NB_INLINE void
440process_postcall(PyObject **args, std::integral_constant<size_t, NArgs>,
441 PyObject *result, nanobind::keep_alive<Nurse, Patient> *) {
442 static_assert(Nurse != Patient,
443 "keep_alive with the same argument as both nurse and patient "
444 "doesn't make sense");
445 static_assert(Nurse <= NArgs && Patient <= NArgs,
446 "keep_alive template parameters must be in the range "
447 "[0, number of C++ function arguments]");
448 keep_alive(Nurse == 0 ? result : args[Nurse - 1],
449 Patient == 0 ? result : args[Patient - 1]);
450}

452template <size_t NArgs, typename Policy>
453NB_INLINE void
454process_postcall(PyObject **args, std::integral_constant<size_t, NArgs> nargs,
455 PyObject *&result, call_policy<Policy> *) {
456 // result_guard avoids leaking a reference to the return object
457 // if postcall throws an exception
458 object result_guard = steal(result);
459 Policy::postcall(args, nargs, result);
460 result_guard.release();
461}

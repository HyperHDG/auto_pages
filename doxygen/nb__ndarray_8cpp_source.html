103 template <bool Versioned> PyObject *make_capsule() {
104 using MT = std::conditional_t<Versioned, managed_dltensor_versioned,
105 managed_dltensor>;
106 const char *name = Versioned ? "dltensor_versioned" : "dltensor";
107
108 // Reuse nanobind's own managed tensor if its flavor already matches;
109 // otherwise allocate a fresh one wrapping a copy of the DLTensor.
110 MT *mt = nullptr;
111 if (versioned == Versioned) {
112 if constexpr (Versioned)
113 mt = mt_versioned;
114 else
115 mt = mt_unversioned;
116 if (mt->manager_ctx != this)
117 mt = nullptr;
118 }
119
120 PyObject *capsule;
121 if (mt) {
122 capsule = PyCapsule_New(mt, name, capsule_delete<Versioned>);
123 } else {
124 scoped_pymalloc<MT> fresh;
125 if constexpr (Versioned) {
126 fresh->version = {dlpack::major_version, dlpack::minor_version};
127 fresh->flags = ro ? dlpack::flag_bitmask_read_only : 0;
128 }
129 fresh->manager_ctx = this;
130 fresh->deleter = mt_from_handle_delete<MT>;
131 memcpy(&fresh->dltensor, &tensor(), sizeof(dlpack::dltensor));
132 capsule = PyCapsule_New(fresh.release(), name,
133 capsule_delete<Versioned>);
134 }
135 check(capsule, "Could not make capsule");
136 refcount++;
137 return capsule;
138 }

150static int nb_ndarray_getbuffer(PyObject *self, Py_buffer *view, int flags) {
151 // The buffer protocol requires that 'view->obj' be set to NULL whenever
152 // the exporter signals failure by returning -1.
153 view->obj = nullptr;
154
155 ndarray_handle *th = ((nb_ndarray *) self)->th;
156 dlpack::dltensor &t = th->tensor();
157
158 if (t.device.device_type != device::cpu::value) {
159 PyErr_SetString(PyExc_BufferError, "Only CPU-allocated ndarrays can be "
160 "accessed via the buffer protocol!");
161 return -1;
162 }
163
164 // Honor a writable request: refuse to expose read-only memory as writable.
165 if ((flags & PyBUF_WRITABLE) == PyBUF_WRITABLE && th->ro) {
166 PyErr_SetString(PyExc_BufferError,
167 "Cannot provide writable access to a read-only ndarray!");
168 return -1;
169 }
170
171 const char *format = nullptr;
172 switch ((dlpack::dtype_code) t.dtype.code) {
173 case dlpack::dtype_code::Int:
174 switch (t.dtype.bits) {
175 case 8: format = "b"; break;
176 case 16: format = "h"; break;
177 case 32: format = "i"; break;
178 case 64: format = "q"; break;
179 }
180 break;
181
182 case dlpack::dtype_code::UInt:
183 switch (t.dtype.bits) {
184 case 8: format = "B"; break;
185 case 16: format = "H"; break;
186 case 32: format = "I"; break;
187 case 64: format = "Q"; break;
188 }
189 break;
190
191 case dlpack::dtype_code::Float:
192 switch (t.dtype.bits) {
193 case 16: format = "e"; break;
194 case 32: format = "f"; break;
195 case 64: format = "d"; break;
196 }
197 break;
198
199 case dlpack::dtype_code::Complex:
200 switch (t.dtype.bits) {
201 case 32: format = "Ze"; break;
202 case 64: format = "Zf"; break;
203 case 128: format = "Zd"; break;
204 }
205 break;
206
207 case dlpack::dtype_code::Bool:
208 format = "?";
209 break;
210
211 default:
212 break;
213 }
214
215 if (!format || t.dtype.lanes != 1) {
216 PyErr_SetString(PyExc_BufferError,
217 "Cannot convert DLPack dtype into buffer protocol format!");
218 return -1;
219 }
220
221 const Py_ssize_t itemsize = t.dtype.bits / 8;
222 Py_ssize_t len = itemsize, size = 1;
223 for (size_t i = 0; i < (size_t) t.ndim; ++i) {
224 len *= (Py_ssize_t) t.shape[i];
225 size *= (Py_ssize_t) t.shape[i];
226 }
227
228 // When the consumer cannot handle strides, only C-contiguous data may be
229 // exported -- otherwise it would interpret 'buf'..'buf + len' as a packed
230 // C-contiguous block and silently read the wrong elements. Arrays with one
231 // or fewer elements are trivially contiguous.
232 if ((flags & PyBUF_STRIDES) != PyBUF_STRIDES && size > 1) {
233 bool c_contig = true;
234 for (int64_t i = t.ndim - 1, accum = 1; i >= 0; --i) {
235 c_contig &= t.shape[i] == 1 || t.strides[i] == accum;
236 accum *= t.shape[i];
237 }
238
239 if (!c_contig) {
240 PyErr_SetString(PyExc_BufferError,
241 "Cannot provide a contiguous buffer for a non-C-contiguous "
242 "ndarray!");
243 return -1;
244 }
245 }
246
247 scoped_pymalloc<Py_ssize_t> shape_and_strides(2 * (size_t) t.ndim);
248 Py_ssize_t* shape = shape_and_strides.get();
249 Py_ssize_t* strides = shape + t.ndim;
250
251 for (size_t i = 0; i < (size_t) t.ndim; ++i) {
252 shape[i] = (Py_ssize_t) t.shape[i];
253 strides[i] = (Py_ssize_t) t.strides[i] * itemsize;
254 }
255
256 view->buf = (void *) ((uintptr_t) t.data + t.byte_offset);
257 view->obj = self;
258 Py_INCREF(self);
259 view->len = len;
260 view->itemsize = itemsize;
261 view->readonly = th->ro;
262 view->ndim = t.ndim;
263 view->format =
264 ((flags & PyBUF_FORMAT) == PyBUF_FORMAT) ? (char *) format : nullptr;
265 view->shape = ((flags & PyBUF_ND) == PyBUF_ND) ? shape : nullptr;
266 view->strides =
267 ((flags & PyBUF_STRIDES) == PyBUF_STRIDES) ? strides : nullptr;
268 view->suboffsets = nullptr;
269 view->internal = shape_and_strides.release();
270
271 return 0;
272}

278// This function implements __dlpack__() for a nanobind.nb_ndarray.
279static PyObject *nb_ndarray_dlpack(PyObject *self, PyObject *const *args,
280 Py_ssize_t nargs, PyObject *kwnames) {
281 if (nargs != 0) {
282 PyErr_SetString(PyExc_TypeError,
283 "__dlpack__() does not accept positional arguments");
284 return nullptr;
285 }
286 Py_ssize_t nkwargs = (kwnames) ? NB_TUPLE_GET_SIZE(kwnames) : 0;
287
288 // Match a keyword name against an interned reference.
289 auto key_is = [](PyObject *key, PyObject *r) -> bool {
290 return key == r;
291 };
292
293 // Match a keyword name against an interned reference, falling back to a
294 // string comparison since kwnames passed via f(**d) are not guaranteed to
295 // be identical to the interned objects.
296 auto key_equals = [](PyObject *key, PyObject *r) -> bool {
297 return key == r || PyObject_RichCompareBool(key, r, Py_EQ) == 1;
298 };
299
300 ndarray_handle *th = ((nb_ndarray *) self)->th;
301 dlpack::dltensor &t = th->tensor();
302
303 long max_major_version = 0;
304
305 // Return nkwargs on success, -1 on error, else index of unmatched kwarg.
306 auto parse_kwargs = [&kwnames, &nkwargs, &args, &t, &max_major_version](
307 Py_ssize_t begin, auto compare) -> Py_ssize_t {
308 // Extract a 2-tuple of integers; returns false (with no error set)
309 // for any other input.
310 auto get_int_pair = [](PyObject *value, long *a, long *b) -> bool {
311 if (!PyTuple_Check(value) || NB_TUPLE_GET_SIZE(value) != 2)
312 return false;
313 *a = PyLong_AsLong(NB_TUPLE_GET_ITEM(value, 0));
314 *b = PyLong_AsLong(NB_TUPLE_GET_ITEM(value, 1));
315 if (PyErr_Occurred()) {
316 PyErr_Clear();
317 return false;
318 }
319 return true;
320 };
321
322 for (Py_ssize_t i = begin; i < nkwargs; ++i) {
323 PyObject* key = NB_TUPLE_GET_ITEM(kwnames, i);
324 PyObject* value = args[i];
325 long a, b;
326 if (compare(key, NB_INTERNED(copy))) {
327 // The capsule aliases C++-owned storage; a copy cannot be made
328 if (value == Py_True) {
329 PyErr_SetString(PyExc_BufferError,
330 "__dlpack__(): copy=True is not supported.");
331 return -1;
332 }
333 } else if (compare(key, NB_INTERNED(dl_device))) {
334 // Reject requests for a device other than the array's own
335 if (value != Py_None &&
336 (!get_int_pair(value, &a, &b) ||
337 a != (long) t.device.device_type ||
338 b != (long) t.device.device_id)) {
339 PyErr_SetString(PyExc_BufferError,
340 "__dlpack__(): unsupported dl_device.");
341 return -1;
342 }
343 } else if (compare(key, NB_INTERNED(max_version))) {
344 if (value != Py_None) {
345 if (!get_int_pair(value, &a, &b)) {
346 PyErr_SetString(PyExc_TypeError,
347 "max_version must be None or tuple[int, int]");
348 return -1;
349 }
350 max_major_version = a;
351 }
352 } else if (compare(key, NB_INTERNED(stream))) {
353 // Accepted but ignored: nanobind tracks no producer stream and
354 // has no backend dependency, so it cannot synchronize. See docs.
355 } else {
356 return i;
357 }
358 }
359 return nkwargs;
360 };
361
362 Py_ssize_t result = parse_kwargs(0, key_is);
363 if (NB_UNLIKELY(result < 0))
364 return nullptr;
365 if (NB_UNLIKELY(result < nkwargs)) {
366 result = parse_kwargs(result, key_equals);
367 if (NB_UNLIKELY(result < 0))
368 return nullptr;
369 if (NB_UNLIKELY(result < nkwargs)) {
370 PyErr_Format(PyExc_TypeError,
371 "__dlpack__(): unsupported keyword argument '%S'",
372 NB_TUPLE_GET_ITEM(kwnames, result));
373 return nullptr;
374 }
375 }
376
377 PyObject *capsule;
378 if (max_major_version >= (long)dlpack::major_version)
379 capsule = th->make_capsule<true>();
380 else
381 capsule = th->make_capsule<false>();
382
383 return capsule;
384}

386// This function implements __dlpack_device__() for a nanobind.nb_ndarray.
387static PyObject *nb_ndarray_dlpack_device(PyObject *self, PyObject *) {
388 ndarray_handle *th = ((nb_ndarray *) self)->th;
389 dlpack::dltensor& t = th->tensor();
390 PyObject *r;
391 if (t.device.device_type == 1 && t.device.device_id == 0) {
392 r = NB_INTERNED(dl_cpu_tpl);
393 Py_INCREF(r);
394 } else {
395 r = PyTuple_New(2);
396 PyObject *r0 = PyLong_FromLong(t.device.device_type);
397 PyObject *r1 = PyLong_FromLong(t.device.device_id);
398 if (!r || !r0 || !r1) {
399 Py_XDECREF(r);
400 Py_XDECREF(r0);
401 Py_XDECREF(r1);
402 return nullptr;
403 }
404 NB_TUPLE_SET_ITEM(r, 0, r0);
405 NB_TUPLE_SET_ITEM(r, 1, r1);
406 }
407 return r;
408}

456static mt_unique_ptr_t make_mt_from_buffer_protocol(PyObject *o, bool ro) {
457 mt_unique_ptr_t mt_unique_ptr(nullptr, &mt_from_buffer_delete);
458 scoped_pymalloc<Py_buffer> view;
459 if (PyObject_GetBuffer(o, view.get(),
460 ro ? PyBUF_RECORDS_RO : PyBUF_RECORDS)) {
461 PyErr_Clear();
462 return mt_unique_ptr;
463 }
464
465 char format_c = 'B';
466 const char *format_str = view->format;
467 if (format_str)
468 format_c = *format_str;
469
470 bool skip_first = format_c == '@' || format_c == '=';
471
472 int32_t num = 1;
473 if (*(uint8_t *) &num == 1) {
474 if (format_c == '<')
475 skip_first = true;
476 } else {
477 if (format_c == '!' || format_c == '>')
478 skip_first = true;
479 }
480
481 if (skip_first && format_str)
482 format_c = *++format_str;
483
484 bool is_complex = format_str && format_str[0] == 'Z';
485 if (is_complex)
486 format_c = *++format_str;
487
488 dlpack::dtype dt { };
489 bool fail = format_str && format_str[0] != '\0' && format_str[1] != '\0';
490
491 if (!fail) {
492 switch (format_c) {
493 case 'c':
494 case 'b':
495 case 'h':
496 case 'i':
497 case 'l':
498 case 'q':
499 case 'n': dt.code = (uint8_t) dlpack::dtype_code::Int; break;
500
501 case 'B':
502 case 'H':
503 case 'I':
504 case 'L':
505 case 'Q':
506 case 'N': dt.code = (uint8_t) dlpack::dtype_code::UInt; break;
507
508 case 'e':
509 case 'f':
510 case 'd': dt.code = (uint8_t) dlpack::dtype_code::Float; break;
511
512 case '?': dt.code = (uint8_t) dlpack::dtype_code::Bool; break;
513
514 default: fail = true;
515 }
516
517 if (is_complex) {
518 fail |= dt.code != (uint8_t) dlpack::dtype_code::Float;
519 dt.code = (uint8_t) dlpack::dtype_code::Complex;
520 }
521
522 dt.lanes = 1;
523 dt.bits = (uint8_t) (view->itemsize * 8);
524 }
525
526 if (fail) {
527 PyBuffer_Release(view.get());
528 return mt_unique_ptr;
529 }
530
531 int32_t ndim = view->ndim;
532 if (ndim < 0 || ndim > max_ndim) {
533 PyBuffer_Release(view.get());
534 return mt_unique_ptr;
535 }
536
537 static_assert(alignof(managed_dltensor_versioned) >= alignof(int64_t));
538 scoped_pymalloc<managed_dltensor_versioned> mt(1, 2 * sizeof(int64_t) * (size_t) ndim);
539 int64_t* shape = nullptr;
540 int64_t* strides = nullptr;
541 if (ndim > 0) {
542 shape = new ((void*) (mt.get() + 1)) int64_t[2 * (size_t) ndim];
543 strides = shape + ndim;
544 }
545
546 mt->dltensor.data = view->buf;
547 mt->dltensor.device = { device::cpu::value, 0 };
548 mt->dltensor.ndim = ndim;
549 mt->dltensor.dtype = dt;
550 mt->dltensor.shape = shape;
551 mt->dltensor.strides = strides;
552 mt->dltensor.byte_offset = 0UL;
553
554 const int64_t itemsize = (int64_t) view->itemsize;
555 for (int32_t i = 0; i < ndim; ++i) {
556 int64_t stride = view->strides[i] / itemsize;
557 if (stride * itemsize != view->strides[i]) {
558 PyBuffer_Release(view.get());
559 return mt_unique_ptr;
560 }
561 strides[i] = stride;
562 shape[i] = (int64_t) view->shape[i];
563 }
564
565 mt->version = {dlpack::major_version, dlpack::minor_version};
566 mt->manager_ctx = view.release();
567 mt->deleter = mt_from_buffer_delete;
568 mt->flags = (ro) ? dlpack::flag_bitmask_read_only : 0;
569
570 mt_unique_ptr.reset(mt.release());
571 return mt_unique_ptr;
572}

594// Detect the source framework from __module__, returning its ndarray.h `value`
595// (no_framework::value if unrecognized). Never raises.
596static int detect_framework(PyTypeObject *tp) noexcept {
597 object mod = steal(PyObject_GetAttr((PyObject *) tp,
598 NB_INTERNED(__module__)));
599 const char *name =
600 mod.is_valid() ? PyUnicode_AsUTF8AndSize(mod.ptr(), nullptr) : nullptr;
601 if (!name) {
602 PyErr_Clear();
603 return no_framework::value;
604 }
605 for (int i = 1; i < importer_count; ++i) {
606 const char *p = importers[i].module_prefix;
607 if (strncmp(name, p, strlen(p)) == 0)
608 return i;
609 }
610 return no_framework::value;
611}

613// Convert to the requested dtype (and order, where the framework supports it).
614static object convert_array(int framework, PyObject *src, const char *dtype,
615 char order) {
616 object converted;
617 try {
618 switch (framework) {
619 case numpy::value:
620 case cupy::value:
621 converted = handle(src).attr(NB_INTERNED(astype))(dtype, order);
622 break;
623
624 case pytorch::value: {
625 module_ torch = module_::import_("torch");
626 converted = handle(src).attr(NB_INTERNED(to))(torch.attr(dtype));
627 if (order == 'C')
628 converted = converted.attr(NB_INTERNED(contiguous))();
629 break;
630 }
631
632 case tensorflow::value: {
633 module_ tensorflow = module_::import_("tensorflow");
634 converted =
635 tensorflow.attr(NB_INTERNED(cast))(handle(src), dtype);
636 break;
637 }
638
639 case jax::value:
640 converted = handle(src).attr(NB_INTERNED(astype))(dtype);
641 break;
642
643 default:
644 break;
645 }
646 } catch (...) {
647 converted.reset();
648 }
649 return converted;
650}

677bool ndarray_check(PyObject *o) noexcept {
678 PyTypeObject *tp = Py_TYPE(o);
679 if (dlpack_method(tp).is_valid() || obj_has_buffer(o, tp))
680 return true;
681
682 if (tp == &PyCapsule_Type)
683 return true;
684
685 PyObject *name = nb_type_name((PyObject *) tp);
686 check(name, "Could not obtain type name! (1)");
687
688 const char *tp_name = PyUnicode_AsUTF8AndSize(name, nullptr);
689 check(tp_name, "Could not obtain type name! (2)");
690
691 bool result = false;
692 for (int i = 1; i < importer_count; ++i) {
693 if (strcmp(tp_name, importers[i].type_name) == 0) {
694 result = true;
695 break;
696 }
697 }
698
699 Py_DECREF(name);
700 return result;
701}

709ndarray_handle *ndarray_import(PyObject *src, const ndarray_config *c,
710 bool convert, cleanup_list *cleanup) noexcept {
711 object capsule;
712 mt_unique_ptr_t mt_unique_ptr(nullptr, &mt_from_buffer_delete);
713
714 // Capsule flavor (versioned or not) to probe for first during extraction.
715 // Defaults to unversioned: the right guess for an unknown user capsule.
716 bool expect_versioned = false;
717
718 PyTypeObject *tp = Py_TYPE(src);
719 const bool src_is_pycapsule = tp == &PyCapsule_Type;
720
721 if (src_is_pycapsule) {
722 capsule = borrow(src);
723 } else {
724 // __dlpack__ is by contract a plain method, so call the looked-up
725 // descriptor directly (args[0] is self) rather than re-resolving it.
726 object dlpack_descr = dlpack_method(tp);
727
728 if (dlpack_descr.is_valid()) {
729 PyObject* args[] = {src, NB_INTERNED(dl_version_tpl)};
730 size_t nargsf = 1 | PY_VECTORCALL_ARGUMENTS_OFFSET;
731
732 // max_version_kw requests a versioned capsule, nullptr the cheaper
733 // unversioned one.
734 PyObject *max_version_kw = NB_INTERNED(max_version_tpl);
735 auto dlpack = [&](PyObject *kwnames) {
736 return steal(PyObject_Vectorcall(dlpack_descr.ptr(), args,
737 nargsf, kwnames));
738 };
739
740 // The unversioned path is generally faster to handle for the target
741 // framework. Try that first if the user only requested readonly input.
742 capsule = dlpack(c->ro ? nullptr : max_version_kw);
743 expect_versioned = !c->ro;
744
745 // Fall back to the other variant on failure: a read-only source
746 // refusing unversioned export raises BufferError, and producers
747 // predating max_version (array API < v2023) raise TypeError.
748 if (!capsule.is_valid() &&
749 (PyErr_ExceptionMatches(PyExc_BufferError) ||
750 PyErr_ExceptionMatches(PyExc_TypeError))) {
751 PyErr_Clear();
752 capsule = dlpack(c->ro ? max_version_kw : nullptr);
753 expect_versioned = c->ro;
754 }
755
756 if (!capsule.is_valid())
757 PyErr_Clear();
758 }
759
760 // Fall back to the buffer protocol, again gated on a non-raising probe.
761 if (!capsule.is_valid() && obj_has_buffer(src, tp))
762 mt_unique_ptr = make_mt_from_buffer_protocol(src, c->ro);
763
764 // Try the function to_dlpack(), already obsolete in array API v2021
765 if (!mt_unique_ptr && !capsule.is_valid()) {
766 const char *pkg = importers[detect_framework(tp)].to_dlpack_pkg;
767 if (pkg) {
768 try {
769 object package = module_::import_(pkg);
770 if (package.is_valid())
771 capsule = package.attr("to_dlpack")(handle(src));
772 } catch (...) {
773 capsule.reset();
774 }
775 }
776 if (!capsule.is_valid())
777 return nullptr;
778 }
779 }
780
781 void* mt; // can be versioned or unversioned
782 bool versioned;
783 if (mt_unique_ptr) {
784 mt = mt_unique_ptr.get();
785 versioned = true;
786 } else {
787 // Probe the expected capsule name first
788 static const char *names[2] = { "dltensor", "dltensor_versioned" };
789 versioned = expect_versioned;
790 mt = PyCapsule_GetPointer(capsule.ptr(), names[(int) versioned]);
791 if (!mt) {
792 PyErr_Clear();
793 versioned = !versioned;
794 mt = PyCapsule_GetPointer(capsule.ptr(), names[(int) versioned]);
795 if (!mt) {
796 PyErr_Clear();
797 return nullptr;
798 }
799 }
800 }
801
802 dlpack::dltensor& t = (versioned)
803 ? ((managed_dltensor_versioned *) mt)->dltensor
804 : ((managed_dltensor *) mt)->dltensor;
805
806 uint64_t flags = (versioned) ? ((managed_dltensor_versioned *) mt)->flags
807 : 0UL;
808
809 if (t.ndim < 0 || t.ndim > max_ndim)
810 return nullptr;
811
812 // Reject a read-only ndarray if a writable one is required, and
813 // reject an ndarray not on the required device.
814 if ((!c->ro && (flags & dlpack::flag_bitmask_read_only))
815 || (c->device_type != 0 && t.device.device_type != c->device_type)) {
816 return nullptr;
817 }
818
819 // Check if the ndarray satisfies the remaining requirements.
820 bool has_dtype = c->dtype != dlpack::dtype(),
821 has_shape = c->ndim != -1,
822 has_order = c->order != '\0';
823
824 bool pass_dtype = true, pass_shape = true, pass_order = true;
825
826 if (has_dtype)
827 pass_dtype = t.dtype == c->dtype;
828
829 if (has_shape) {
830 pass_shape = t.ndim == c->ndim;
831 if (pass_shape) {
832 for (int32_t i = 0; i < c->ndim; ++i) {
833 if (c->shape[i] != -1 && t.shape[i] != c->shape[i]) {
834 pass_shape = false;
835 break;
836 }
837 }
838 }
839 }
840
841 // Only the order check below needs the element count, so skip it otherwise.
842 int64_t size = 1;
843 if (has_order)
844 for (int32_t i = 0; i < t.ndim; ++i)
845 size *= t.shape[i];
846
847 // Tolerate any strides if the array has 1 or fewer elements
848 if (pass_shape && has_order && size > 1) {
849 char order = c->order;
850
851 bool c_order = order == 'C' || order == 'A',
852 f_order = order == 'F' || order == 'A';
853
854 if (!t.strides) {
855 /* When the provided tensor does not have a valid
856 strides field, it uses the C ordering convention */
857 if (c_order) {
858 pass_order = true;
859 } else {
860 int nontrivial_dims = 0;
861 for (int i = 0; i < t.ndim; ++i)
862 nontrivial_dims += (int) (t.shape[i] > 1);
863 pass_order = nontrivial_dims <= 1;
864 }
865 } else {
866 if (c_order) {
867 for (int64_t i = t.ndim - 1, accum = 1; i >= 0; --i) {
868 c_order &= t.shape[i] == 1 || t.strides[i] == accum;
869 accum *= t.shape[i];
870 }
871 }
872
873 if (f_order) {
874 for (int64_t i = 0, accum = 1; i < t.ndim; ++i) {
875 f_order &= t.shape[i] == 1 || t.strides[i] == accum;
876 accum *= t.shape[i];
877 }
878 }
879
880 pass_order = c_order || f_order;
881 }
882 }
883
884 // Do not convert shape and do not convert complex numbers to non-complex.
885 convert &= pass_shape &
886 !(dtype_code_is_complex(t.dtype.code) &&
887 has_dtype && !dtype_code_is_complex(c->dtype.code));
888
889 // Support implicit conversion of dtype and order.
890 if (convert && (!pass_dtype || !pass_order) && !src_is_pycapsule) {
891 int fw = detect_framework(tp);
892
893 char order = 'K'; // for NumPy. 'K' means 'keep'
894 if (c->order)
895 order = c->order;
896
897 dlpack::dtype dt = has_dtype ? c->dtype : t.dtype;
898 if (dt.lanes != 1)
899 return nullptr;
900
901 char dtype[12];
902 if (dt.code == (uint8_t) dlpack::dtype_code::Bool) {
903 std::strcpy(dtype, "bool");
904 } else {
905 const char *prefix = nullptr;
906 switch (dt.code) {
907 case (uint8_t) dlpack::dtype_code::Int:
908 prefix = "int";
909 break;
910 case (uint8_t) dlpack::dtype_code::UInt:
911 prefix = "uint";
912 break;
913 case (uint8_t) dlpack::dtype_code::Float:
914 prefix = "float";
915 break;
916 case (uint8_t) dlpack::dtype_code::Bfloat:
917 prefix = "bfloat";
918 break;
919 case (uint8_t) dlpack::dtype_code::Complex:
920 prefix = "complex";
921 break;
922 case (uint8_t) dlpack::dtype_code::Bcomplex:
923 prefix = "bcomplex";
924 break;
925 default:
926 return nullptr;
927 }
928 snprintf(dtype, sizeof(dtype), "%s%u", prefix, dt.bits);
929 }
930
931 object converted = convert_array(fw, src, dtype, order);
932
933 // Potentially try once again, recursively
934 if (converted.is_valid()) {
935 ndarray_handle *h =
936 ndarray_import(converted.ptr(), c, false, nullptr);
937 if (h && cleanup)
938 cleanup->append(converted.release().ptr());
939 return h;
940 }
941 }
942
943 if (!pass_dtype || !pass_shape || !pass_order)
944 return nullptr;
945
946 // Create a reference-counted wrapper
947 scoped_pymalloc<ndarray_handle> result;
948 if (versioned)
949 result->mt_versioned = (managed_dltensor_versioned *) mt;
950 else
951 result->mt_unversioned = (managed_dltensor *) mt;
952
953 result->refcount = 0;
954 result->owner = nullptr;
955 result->versioned = versioned;
956 result->call_deleter = true;
957 result->ro = c->ro;
958
959 if (src_is_pycapsule) {
960 result->self = nullptr;
961 } else {
962 result->self = src;
963 Py_INCREF(src);
964 }
965
966 // If ndim > 0, ensure that the strides member is initialized.
967 if (t.strides || t.ndim == 0) {
968 result->free_strides = false;
969 } else {
970 result->free_strides = true;
971
972 scoped_pymalloc<int64_t> strides((size_t) t.ndim);
973 int64_t accum = 1;
974 for (int32_t i = t.ndim - 1; i >= 0; --i) {
975 strides[(size_t) i] = accum;
976 accum *= t.shape[i];
977 }
978 t.strides = strides.release();
979 }
980
981 if (capsule.is_valid()) {
982 // Neutralize the producer's capsule so its destructor won't free the
983 // DLManagedTensor that nanobind now owns. Clearing the destructor is
984 // sufficient and is the only step the common __dlpack__() path needs:
985 // nanobind holds the sole reference to that capsule and never re-reads
986 // its name. A user-supplied raw capsule, by contrast, is still
987 // referenced by the caller, so it is additionally renamed to the
988 // conventional "used" name to stop a second import from re-consuming it.
989 bool fail = PyCapsule_SetDestructor(capsule.ptr(), nullptr);
990 if (src_is_pycapsule && !fail) {
991 const char* used_name = (versioned) ? "used_dltensor_versioned"
992 : "used_dltensor";
993 fail = PyCapsule_SetName(capsule.ptr(), used_name);
994 }
995 if (fail)
996 check(false, "ndarray_import(): could not mark capsule as used");
997 }
998
999 mt_unique_ptr.release();
1000 return result.release();
1001}

1010// Final teardown of a handle whose refcount reached zero.
1011static void ndarray_dec_ref_free(ndarray_handle *th) noexcept {
1012 Py_XDECREF(th->owner);
1013 Py_XDECREF(th->self);
1014 if (th->versioned) {
1015 managed_dltensor_versioned *mt = th->mt_versioned;
1016 if (th->free_strides) {
1017 PyMem_Free(mt->dltensor.strides);
1018 mt->dltensor.strides = nullptr;
1019 }
1020 if (th->call_deleter) {
1021 if (mt->deleter)
1022 mt->deleter(mt);
1023 } else {
1024 PyMem_Free(mt); // This also frees shape and size arrays.
1025 }
1026 } else {
1027 managed_dltensor *mt = th->mt_unversioned;
1028 if (th->free_strides) {
1029 PyMem_Free(mt->dltensor.strides);
1030 mt->dltensor.strides = nullptr;
1031 }
1032 assert(th->call_deleter);
1033 if (mt->deleter)
1034 mt->deleter(mt);
1035 }
1036 PyMem_Free(th);
1037}

1039void ndarray_dec_ref(ndarray_handle *th) noexcept {
1040 if (!th)
1041 return;
1042 size_t rc_value = th->refcount--;
1043
1044 if (rc_value == 0) {
1045 check(false, "ndarray_dec_ref(): reference count became negative!");
1046 } else if (rc_value == 1) {
1047 // Don't run the cleanup if the interpreter has been shut down
1048 if (!is_alive())
1049 return;
1050
1051#if !defined(Py_LIMITED_API)
1052 // Avoid further GIL calls if we already hold it. (Slightly faster)
1053 if (PyGILState_Check()) {
1054 ndarray_dec_ref_free(th);
1055 return;
1056 }
1057#endif
1058 gil_scoped_acquire guard;
1059 ndarray_dec_ref_free(th);
1060 }

1063ndarray_handle *ndarray_create(void *data, size_t ndim, const size_t *shape_in,
1064 PyObject *owner, const int64_t *strides_in,
1065 dlpack::dtype dtype, bool ro, int device_type,
1066 int device_id, char order,
1067 uint64_t byte_offset) {
1068 check(ndim <= (size_t) max_ndim,
1069 "ndarray_create(): ndim is too large!");
1070
1071 /* A comment in the DLPack header file suggests 256-byte alignment of the
1072 DLTensor::data field, but this is generally (and necessarily) ignored.
1073 Note that the pointer dltensor.data must point to allocated memory
1074 (i.e., memory that can be accessed), so it cannot simply be rounded
1075 down by zeroing its lowest 8 bits.
1076 A byte_offset can be used to support array slicing when data is an
1077 opaque device pointer or handle, on which arithmetic is impossible.
1078 See also: https://github.com/data-apis/array-api/discussions/779 */
1079
1080 if (device_type == 0)
1081 device_type = device::cpu::value;
1082
1083 static_assert(alignof(managed_dltensor_versioned) >= alignof(int64_t));
1084 scoped_pymalloc<managed_dltensor_versioned> mt(1, 2 * sizeof(int64_t)*ndim);
1085 int64_t* shape = nullptr;
1086 int64_t* strides = nullptr;
1087 if (ndim > 0) {
1088 shape = new ((void*) (mt.get() + 1)) int64_t[2 * ndim];
1089 strides = shape + ndim;
1090 }
1091
1092 for (size_t i = 0; i < ndim; ++i)
1093 shape[i] = (int64_t) shape_in[i];
1094
1095 if (ndim > 0) {
1096 int64_t prod = 1;
1097 if (strides_in) {
1098 for (size_t i = 0; i < ndim; ++i)
1099 strides[i] = strides_in[i];
1100 } else if (order == 'F') {
1101 for (size_t i = 0; i < ndim; ++i) {
1102 strides[i] = prod;
1103 prod *= (int64_t) shape_in[i];
1104 }
1105 } else if (order == '\0' || order == 'A' || order == 'C') {
1106 for (ssize_t i = (ssize_t) ndim - 1; i >= 0; --i) {
1107 strides[i] = prod;
1108 prod *= (int64_t) shape_in[i];
1109 }
1110 } else {
1111 check(false, "ndarray_create(): unknown memory order requested!");
1112 }
1113 }
1114
1115 scoped_pymalloc<ndarray_handle> result;
1116
1117 mt->version = {dlpack::major_version, dlpack::minor_version};
1118 mt->manager_ctx = result.get();
1119 mt->deleter = [](managed_dltensor_versioned *self) {
1120 ndarray_dec_ref((ndarray_handle *) self->manager_ctx);
1121 };
1122 mt->flags = (ro) ? dlpack::flag_bitmask_read_only : 0;
1123 mt->dltensor.data = data;
1124 mt->dltensor.device.device_type = (int32_t) device_type;
1125 mt->dltensor.device.device_id = (int32_t) device_id;
1126 mt->dltensor.ndim = (int32_t) ndim;
1127 mt->dltensor.dtype = dtype;
1128 mt->dltensor.shape = shape;
1129 mt->dltensor.strides = strides;
1130 mt->dltensor.byte_offset = byte_offset;
1131 result->mt_versioned = mt.release();
1132 result->refcount = 0;
1133 result->owner = owner;
1134 result->self = nullptr;
1135 result->versioned = true;
1136 result->free_strides = false;
1137 result->call_deleter = false;
1138 result->ro = ro;
1139 Py_XINCREF(owner);
1140 return result.release();
1141}

1156static PyObject *ndarray_export_fn(nb_internals *internals_,
1157 ndarray_export_slot slot) {
1158 PyObject *fn = internals_->ndarray_export[slot].load_acquire();
1159 if (NB_LIKELY(fn))
1160 return fn;
1161
1162 lock_internals guard(internals_);
1163 fn = internals_->ndarray_export[slot].load_relaxed();
1164 if (fn)
1165 return fn;
1166
1167 object obj = steal(module_import(ndarray_export_spec[slot].pkg))
1168 .attr(ndarray_export_spec[slot].attr);
1169 fn = obj.release().ptr();
1170 new_object(internals_, fn);
1171 internals_->ndarray_export[slot].store_release(fn);
1172 return fn;

1175PyObject *ndarray_export(ndarray_handle *th, int framework,
1176 rv_policy policy, cleanup_list *cleanup) noexcept {
1177 if (!th)
1178 return none().release().ptr();
1179
1180 bool copy;
1181 switch (policy) {
1182 case rv_policy::reference_internal:
1183 if (cleanup && cleanup->self() != th->owner && !th->self) {
1184 if (th->owner) {
1185 PyErr_SetString(PyExc_RuntimeError,
1186 "nanobind::detail::ndarray_export(): "
1187 "reference_internal policy cannot be "
1188 "applied (ndarray already has an owner)");
1189 return nullptr;
1190 } else {
1191 th->owner = cleanup->self();
1192 Py_INCREF(th->owner);
1193 }
1194 }
1195 [[fallthrough]];
1196
1197 case rv_policy::automatic:
1198 case rv_policy::automatic_reference:
1199 copy = th->owner == nullptr && th->self == nullptr;
1200 break;
1201
1202 case rv_policy::copy:
1203 case rv_policy::move:
1204 copy = true;
1205 break;
1206
1207 default:
1208 copy = false;
1209 break;
1210 }
1211
1212 if (!copy) {
1213 if (th->self) {
1214 Py_INCREF(th->self);
1215 return th->self;
1216 } else if (policy == rv_policy::none) {
1217 return nullptr;
1218 }
1219 }
1220
1221 // These frameworks export a raw DLPack capsule or buffer view rather than
1222 // a framework array with a copy method, so the requested copy cannot be
1223 // performed. Refuse the cast rather than returning a view that would
1224 // alias (and possibly outlive) the original storage.
1225 if (copy && !th->self &&
1226 (framework == no_framework::value || framework == tensorflow::value ||
1227 framework == memview::value || framework == array_api::value)) {
1228 PyErr_SetString(PyExc_RuntimeError,
1229 "nanobind::detail::ndarray_export(): copying the "
1230 "array contents is not supported for this framework; "
1231 "please specify an 'owner' so that the array can be "
1232 "returned without a copy.");
1233 return nullptr;
1234 }
1235
1236 nb_internals *internals_ = internals;
1237
1238 object o;
1239 if (copy && framework == no_framework::value && th->self) {
1240 o = borrow(th->self);
1241 } else if (framework == no_framework::value ||
1242 framework == tensorflow::value) {
1243 // Make a new capsule wrapping an unversioned managed_dltensor.
1244 o = steal(th->make_capsule<false>());
1245 } else {
1246 // Make a Python object providing the buffer interface and having
1247 // the two DLPack methods __dlpack__() and __dlpack_device__().
1248 nb_ndarray *h = PyObject_New(nb_ndarray, nb_ndarray_tp(internals_));
1249 if (!h)
1250 return nullptr;
1251 h->th = th;
1252 ndarray_inc_ref(th);
1253 o = steal((PyObject *) h);
1254 }
1255
1256 if (framework == numpy::value) {
1257 try {
1258 // Call nump.asarray(o) to create a view, and numpy.copy(o) to copy
1259 PyObject *export_fn = ndarray_export_fn(
1260 internals_, copy ? nd_export_numpy_copy : nd_export_numpy_view);
1261 PyObject *stack[] = {nullptr, o.ptr()};
1262 size_t nargsf = 1 | PY_VECTORCALL_ARGUMENTS_OFFSET;
1263 return PyObject_Vectorcall(export_fn, stack + 1, nargsf, nullptr);
1264 } catch (const std::exception &e) {
1265 PyErr_Format(PyExc_TypeError,
1266 "could not export nanobind::ndarray: %s", e.what());
1267 return nullptr;
1268 }
1269 }
1270
1271 // The DLPack frameworks build a view via <pkg>.from_dlpack(o); no_framework
1272 // and array_api leave `o` as-is; memview returns a memoryview directly.
1273 try {
1274 ndarray_export_slot slot;
1275 switch (framework) {
1276 case pytorch::value: slot = nd_export_pytorch; break;
1277 case tensorflow::value: slot = nd_export_tensorflow; break;
1278 case jax::value: slot = nd_export_jax; break;
1279 case cupy::value: slot = nd_export_cupy; break;
1280 case mlx::value: slot = nd_export_mlx; break;
1281 case memview::value: return PyMemoryView_FromObject(o.ptr());
1282 default: slot = nd_export_count; // no export call
1283 }
1284
1285 if (slot != nd_export_count) {
1286 PyObject *export_fn = ndarray_export_fn(internals_, slot);
1287 PyObject *stack[] = {nullptr, o.ptr()};
1288 size_t nargsf = 1 | PY_VECTORCALL_ARGUMENTS_OFFSET;
1289 o = steal(PyObject_Vectorcall(export_fn, stack + 1, nargsf, nullptr));
1290 if (!o.is_valid())
1291 return nullptr;
1292 }
1293 } catch (const std::exception &e) {
1294 PyErr_Format(PyExc_TypeError,
1295 "could not export nanobind::ndarray: %s",
1296 e.what());
1297 return nullptr;
1298 }
1299
1300 // MLX has no copy()/clone() method; mlx.core.array() already returned an
1301 // owned copy, so the copy policy is satisfied without an extra step.
1302 if (copy && framework != mlx::value) {
1303 PyObject* copy_fn_name = framework == pytorch::value ? NB_INTERNED(clone)
1304 : NB_INTERNED(copy);
1305 try {
1306 o = o.attr(copy_fn_name)();
1307 } catch (std::exception &e) {
1308 PyErr_Format(PyExc_RuntimeError,
1309 "copying nanobind::ndarray failed: %s",
1310 e.what());
1311 return nullptr;
1312 }
1313 }
1314
1315 return o.release().ptr();
1316}

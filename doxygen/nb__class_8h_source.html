382template <typename... Args> struct init : def_visitor<init<Args...>> {
383 template <typename T, typename... Ts> friend class class_;
384 NB_INLINE init() {}
385
386private:
387 template <typename Class, typename... Extra>
388 NB_INLINE static void execute(Class &cl, const Extra&... extra) {
389 using Type = typename Class::Type;
390 using Alias = typename Class::Alias;
391 cl.def(
392 "__init__",
393 [](pointer_and_handle<Type> v, Args... args) {
394 if constexpr (!std::is_same_v<Type, Alias> &&
395 std::is_constructible_v<Type, Args...>) {
396 if (!detail::nb_inst_python_derived(v.h.ptr())) {
397 new (v.p) Type((detail::forward_t<Args>) args...);
398 return;
399 }
400 }
401 // Prefer direct- over list-initialization
402 if constexpr (std::is_constructible_v<Alias, Args...>)
403 new ((void *) v.p) Alias((detail::forward_t<Args>) args...);
404 else
405 new ((void *) v.p) Alias{(detail::forward_t<Args>) args...};
406 },
407 extra...);
408 }
409};

411template <typename Arg> struct init_implicit : def_visitor<init_implicit<Arg>> {
412 template <typename T, typename... Ts> friend class class_;
413 NB_INLINE init_implicit() { }
414
415private:
416 template <typename Class, typename... Extra>
417 NB_INLINE static void execute(Class &cl, const Extra&... extra) {
418 using Type = typename Class::Type;
419 using Alias = typename Class::Alias;
420
421 cl.def(
422 "__init__",
423 [](pointer_and_handle<Type> v, Arg arg) {
424 if constexpr (!std::is_same_v<Type, Alias> &&
425 std::is_constructible_v<Type, Arg>) {
426 if (!detail::nb_inst_python_derived(v.h.ptr())) {
427 new ((Type *) v.p) Type((detail::forward_t<Arg>) arg);
428 return;
429 }
430 }
431 // Prefer direct- over list-initialization
432 if constexpr (std::is_constructible_v<Alias, Arg>)
433 new ((Alias *) v.p) Alias((detail::forward_t<Arg>) arg);
434 else
435 new ((Alias *) v.p) Alias{ (detail::forward_t<Arg>) arg };
436 }, is_implicit(), extra...);
437
438 using Caster = detail::make_caster<Arg>;

440 if constexpr (!detail::is_class_caster_v<Caster>) {
441 detail::implicitly_convertible(
442 [](PyTypeObject *, PyObject *src,
443 detail::cleanup_list *cleanup) noexcept -> bool {
444 return Caster().from_python(
445 src, detail::cast_flags::convert, cleanup);
446 },
447 &typeid(Type));
448 }
449 }

453 // This is 'inline' so we can define it in a header and not pay
454 // for it if unused, and also 'noinline' so we don't generate
455 // multiple copies and produce code bloat.
456 NB_NOINLINE inline void wrap_base_new(handle cls, bool do_wrap) {
457 if (PyCFunction_Check(cls.attr("__new__").ptr())) {
458 if (do_wrap) {
459 cpp_function_def(
460 [](handle type) {
461 if (!type_check(type))
462 throw cast_error();
463 return inst_alloc(type);
464 },
465 scope(cls), name("__new__"));
466 }
467 } else {
468 if (!do_wrap) {
469 // We already defined the wrapper, so this zero-arg overload
470 // would be unreachable. Raise an error rather than hiding it.
471 raise("nanobind: %s must define its zero-argument __new__ "
472 "before any other overloads", type_name(cls).c_str());
473 }
474 }
475 }

477 // Call policy that ensures __new__ returns an instance of the correct
478 // Python type, even when deriving from the C++ class in Python
479 struct new_returntype_fixup_policy {
480 static inline void precall(PyObject **, size_t,
481 detail::cleanup_list *) {}
482 NB_NOINLINE static inline void postcall(PyObject **args, size_t,
483 PyObject *&ret) {
484 handle type_requested = args[0];
485 if (ret == nullptr || !type_requested.is_type())
486 return; // somethign strange about this call; don't meddle
487 handle type_created = Py_TYPE(ret);
488 if (type_created.is(type_requested))
489 return; // already created the requested type so no fixup needed
490
491 if (type_check(type_created) &&
492 PyType_IsSubtype((PyTypeObject *) type_requested.ptr(),
493 (PyTypeObject *) type_created.ptr()) &&
494 type_info(type_created) == type_info(type_requested)) {
495 // The new_ constructor returned an instance of a bound type T.
496 // The user wanted an instance of some python subclass S of T.
497 // Since both wrap the same C++ type, we can satisfy the request
498 // by returning a pyobject of type S that wraps a C++ T*, and
499 // handling the lifetimes by having that pyobject keep the
500 // already-created T pyobject alive.
501 object wrapper = inst_reference(type_requested,
502 inst_ptr<void>(ret),
503 /* parent = */ ret);
504 handle(ret).dec_ref();
505 ret = wrapper.release().ptr();
506 }
507 }
508 };
509}

514template <typename Func, typename Return, typename... Args>
515struct new_<Func, Return(Args...)> : def_visitor<new_<Func, Return(Args...)>> {
516 std::remove_reference_t<Func> func;
517
518 new_(Func &&f) : func((detail::forward_t<Func>) f) {}
519
520 template <typename Class, typename... Extra>
521 NB_INLINE void execute(Class &cl, const Extra&... extra) {
522 // If this is the first __new__ overload we're defining, then wrap
523 // nanobind's built-in __new__ so we overload with it instead of
524 // replacing it; this is important for pickle support.
525 // We can't do this if the user-provided __new__ takes no
526 // arguments, because it would make an ambiguous overload set.
527 constexpr size_t num_defaults =
528 ((std::is_same_v<Extra, arg_v> ||
529 std::is_same_v<Extra, arg_locked_v>) + ... + 0);
530 constexpr size_t num_varargs =
531 ((std::is_same_v<detail::intrinsic_t<Args>, args> ||
532 std::is_same_v<detail::intrinsic_t<Args>, kwargs>) + ... + 0);
533 detail::wrap_base_new(cl, sizeof...(Args) > num_defaults + num_varargs);
534
535 auto wrapper = [func_ = (detail::forward_t<Func>) func](handle, Args... args) {
536 return func_((detail::forward_t<Args>) args...);
537 };
538
539 auto policy = call_policy<detail::new_returntype_fixup_policy>();
540 if constexpr ((std::is_base_of_v<arg, Extra> || ...)) {
541 // If any argument annotations are specified, add another for the
542 // extra class argument that we don't forward to Func, so visible
543 // arg() annotations stay aligned with visible function arguments.
544 cl.def_static("__new__", std::move(wrapper), arg("cls"), extra...,
545 policy);
546 } else {
547 cl.def_static("__new__", std::move(wrapper), extra..., policy);
548 }
549 cl.def("__init__", [](handle, Args...) {}, extra...);
550 }
551};

577template <typename T, typename... Ts>
578class class_ : public object {
579public:
580 NB_OBJECT_DEFAULT(class_, object, "type", PyType_Check)
581 using Type = T;
582 using Base = typename detail::extract<T, detail::is_base, Ts...>::type;
583 using Alias = typename detail::extract<T, detail::is_alias, Ts...>::type;
584
585 static_assert(sizeof(Alias) < (((uint64_t) 1) << 32), "Instance size is too big!");
586 static_assert(alignof(Alias) < (1 << 8), "Instance alignment is too big!");
587 static_assert(
588 sizeof...(Ts) == !std::is_same_v<Base, T> + !std::is_same_v<Alias, T>,
589 "nanobind::class_<> was invoked with extra arguments that could not be handled");
590
591 static_assert(
592 detail::is_base_caster_v<detail::make_caster<Type>>,
593 "You attempted to bind a type that is already intercepted by a type "
594 "caster. Having both at the same time is not allowed. Are you perhaps "
595 "binding an STL type, while at the same time including a matching "
596 "type caster from <nanobind/stl/*>? Or did you perhaps forget to "
597 "declare NB_MAKE_OPAQUE(..) to specifically disable the type caster "
598 "catch-all for a specific type? Please review the documentation "
599 "to learn about the difference between bindings and type casters.");
600
601 template <typename... Extra>
602 NB_INLINE class_(handle scope, const char *name, const Extra &... extra) {
603 detail::type_init_data d;
604
605 d.flags = 0;
606 d.align = (uint8_t) alignof(Alias);
607 d.size = (uint32_t) sizeof(Alias);
608 d.name = name;
609 d.scope = scope.ptr();
610 d.type = &typeid(T);
611
612 if constexpr (!std::is_same_v<Base, T>) {
613 d.base = &typeid(Base);
614 d.flags |= (uint32_t) detail::type_init_flags::has_base;
615 }
616
617 if constexpr (detail::is_copy_constructible_v<T>) {
618 d.flags |= (uint32_t) detail::type_flags::is_copy_constructible;
619
620 if constexpr (!std::is_trivially_copy_constructible_v<T>) {
621 d.flags |= (uint32_t) detail::type_flags::has_copy;
622 d.copy = detail::wrap_copy<T>;
623 }
624 }
625
626 if constexpr (std::is_move_constructible<T>::value) {
627 d.flags |= (uint32_t) detail::type_flags::is_move_constructible;
628
629 if constexpr (!std::is_trivially_move_constructible_v<T>) {
630 d.flags |= (uint32_t) detail::type_flags::has_move;
631 d.move = detail::wrap_move<T>;
632 }
633 }
634
635 constexpr bool has_never_destruct = (std::is_same_v<Extra, never_destruct> || ...);
636
637 if constexpr (std::is_destructible_v<T> && !has_never_destruct) {
638 d.flags |= (uint32_t) detail::type_flags::is_destructible;
639
640 if constexpr (!std::is_trivially_destructible_v<T>) {
641 d.flags |= (uint32_t) detail::type_flags::has_destruct;
642 d.destruct = detail::wrap_destruct<T>;
643 }
644 }
645
646 if constexpr (detail::has_shared_from_this_v<T>) {
647 d.flags |= (uint32_t) detail::type_flags::has_shared_from_this;
648 d.keep_shared_from_this_alive = [](PyObject *self) noexcept {
649 // weak_from_this().lock() is equivalent to shared_from_this(),
650 // except that it returns an empty shared_ptr instead of
651 // throwing an exception if there is no active shared_ptr
652 // for this object. (Added in C++17.)
653 if (auto sp = inst_ptr<T>(self)->weak_from_this().lock()) {
654 detail::keep_alive(self, new auto(std::move(sp)),
655 [](void *p) noexcept {
656 delete (decltype(sp) *) p;
657 });
658 return true;
659 }
660 return false;
661 };
662 }
663
664 (detail::type_extra_apply(d, extra), ...);
665
666 m_ptr = detail::nb_type_new(&d);
667 }

119template <typename T, typename /* SFINAE */ = int> struct dtype_traits {
120 using traits = ndarray_traits<T>;
121
122 static constexpr int matches = traits::is_bool + traits::is_complex +
123 traits::is_float + traits::is_int;
124 static_assert(matches <= 1, "dtype matches multiple type categories!");
125
126 static constexpr dlpack::dtype value{
127 (uint8_t) ((traits::is_bool ? (int) dlpack::dtype_code::Bool : 0) +
128 (traits::is_complex ? (int) dlpack::dtype_code::Complex : 0) +
129 (traits::is_float ? (int) dlpack::dtype_code::Float : 0) +
130 (traits::is_int && traits::is_signed ? (int) dlpack::dtype_code::Int : 0) +
131 (traits::is_int && !traits::is_signed ? (int) dlpack::dtype_code::UInt : 0)),
132 (uint8_t) matches ? sizeof(T) * 8 : 0,
133 matches ? 1 : 0
134 };

136 static constexpr auto name =
137 const_name<traits::is_complex>("complex", "") +
138 const_name<traits::is_int && traits::is_signed>("int", "") +
139 const_name<traits::is_int && !traits::is_signed>("uint", "") +
140 const_name<traits::is_float>("float", "") +
141 const_name<traits::is_bool>(const_name("bool"), const_name<sizeof(T) * 8>());
142};

273template <typename Scalar, size_t Dim, char Order> struct ndarray_view {
274 ndarray_view() = default;
275 ndarray_view(const ndarray_view &) = default;
276 ndarray_view(ndarray_view &&) = default;
277 ndarray_view &operator=(const ndarray_view &) = default;
278 ndarray_view &operator=(ndarray_view &&) noexcept = default;
279 ~ndarray_view() noexcept = default;
280
281 template <typename... Args> NB_INLINE Scalar &operator()(Args... indices) const {
282 static_assert(
283 sizeof...(Args) == Dim,
284 "ndarray_view::operator(): invalid number of arguments");
285
286 const int64_t indices_i64[] { (int64_t) indices... };
287 int64_t offset = 0;
288 for (size_t i = 0; i < Dim; ++i)
289 offset += indices_i64[i] * m_strides[i];
290
291 return *(m_data + offset);
292 }
293
294 size_t ndim() const { return Dim; }
295 size_t shape(size_t i) const { return (size_t) m_shape[i]; }
296 int64_t stride(size_t i) const { return m_strides[i]; }
297 Scalar *data() const { return m_data; }
298
299private:
300 template <typename...> friend class ndarray;
301
302 template <size_t... I1, ssize_t... I2>
303 ndarray_view(Scalar *data, const int64_t *shape, const int64_t *strides,
304 std::index_sequence<I1...>, nanobind::shape<I2...>)
305 : m_data(data) {
306
307 /* Initialize shape/strides with compile-time knowledge if
308 available (to permit vectorization, loop unrolling, etc.) */
309 ((m_shape[I1] = (I2 == -1) ? shape[I1] : (int64_t) I2), ...);
310 ((m_strides[I1] = strides[I1]), ...);
311
312 if constexpr (Order == 'F') {
313 m_strides[0] = 1;
314 for (size_t i = 1; i < Dim; ++i)
315 m_strides[i] = m_strides[i - 1] * m_shape[i - 1];
316 } else if constexpr (Order == 'C') {
317 m_strides[Dim - 1] = 1;
318 for (Py_ssize_t i = (Py_ssize_t) Dim - 2; i >= 0; --i)
319 m_strides[i] = m_strides[i + 1] * m_shape[i + 1];
320 }
321 }
322
323 Scalar *m_data = nullptr;
324 int64_t m_shape[Dim] { };
325 int64_t m_strides[Dim] { };
326};

502 NB_INLINE int64_t compute_byte_offset(Args2... indices) const {
503 constexpr bool has_scalar = !std::is_void_v<Scalar>,
504 has_shape = Config::N != -1;
505
506 static_assert(has_scalar,
507 "To use ndarray::operator(), you must add a scalar type "
508 "annotation (e.g. 'float') to the ndarray template parameters.");
509
510 static_assert(has_shape,
511 "To use ndarray::operator(), you must add a shape<> or "
512 "ndim<> annotation to the ndarray template parameters.");
513
514 if constexpr (has_scalar && has_shape) {
515 static_assert(sizeof...(Args2) == (size_t) Config::N,
516 "ndarray::operator(): invalid number of arguments");

518 size_t counter = 0;
519 int64_t index = 0;
520 ((index += int64_t(indices) * m_dltensor.strides[counter++]), ...);
521
522 return (int64_t) m_dltensor.byte_offset + index * (int64_t) sizeof(Scalar);

525 }

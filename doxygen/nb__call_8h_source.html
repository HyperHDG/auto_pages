35template <typename T>
36NB_INLINE void call_analyze(size_t &nargs, size_t &nkwargs, const T &value) {
37 using D = std::decay_t<T>;
38 static_assert(!std::is_base_of_v<arg_locked, D>,
39 "nb::arg().lock() may be used only when defining functions, "
40 "not when calling them");
41
42 if constexpr (std::is_same_v<D, arg_v>)
43 nkwargs++;
44 else if constexpr (std::is_same_v<D, args_proxy>)
45 nargs += len(value);
46 else if constexpr (std::is_same_v<D, kwargs_proxy>)
47 nkwargs += len(value);
48 else
49 nargs += 1;
50
51 (void) nargs; (void) nkwargs; (void) value;
52}

55template <rv_policy policy, typename T>
56NB_INLINE void call_init(PyObject **args, PyObject *kwnames, size_t &nargs,
57 size_t &nkwargs, const size_t kwargs_offset,
58 T &&value) {
59 using D = std::decay_t<T>;
60
61 if constexpr (std::is_same_v<D, arg_v>) {
62 args[kwargs_offset + nkwargs] = value.value.release().ptr();
63 NB_TUPLE_SET_ITEM(kwnames, (Py_ssize_t) nkwargs++,
64 PyUnicode_InternFromString(value.name_));
65 } else if constexpr (std::is_same_v<D, args_proxy>) {
66 for (size_t i = 0, l = len(value); i < l; ++i)
67 args[nargs++] = borrow(value[i]).release().ptr();
68 } else if constexpr (std::is_same_v<D, kwargs_proxy>) {
69 PyObject *key, *entry;
70 Py_ssize_t pos = 0;
71#if defined(NB_FREE_THREADED)
72 ft_object_guard guard(value);
73#endif
74 while (PyDict_Next(value.ptr(), &pos, &key, &entry)) {
75 Py_INCREF(key); Py_INCREF(entry);
76 args[kwargs_offset + nkwargs] = entry;
77 NB_TUPLE_SET_ITEM(kwnames, (Py_ssize_t) nkwargs++, key);
78 }
79 } else {
80 args[nargs++] =
81 make_caster<T>::from_cpp((forward_t<T>) value, policy, nullptr).ptr();
82 }
83 (void) args; (void) kwnames; (void) nargs;
84 (void) nkwargs; (void) kwargs_offset;
85}

102template <typename Derived>
103template <rv_policy policy, typename... Args>
104object api<Derived>::operator()(Args &&...args_) const {
105 static constexpr bool method_call =
106 std::is_same_v<Derived, accessor<obj_attr>> ||
107 std::is_same_v<Derived, accessor<str_attr>>;
108
109 if constexpr (((std::is_same_v<Args, arg_v> ||
110 std::is_same_v<Args, args_proxy> ||
111 std::is_same_v<Args, kwargs_proxy>) || ...)) {
112 // Complex call with keyword arguments, *args/**kwargs expansion, etc.
113 size_t nargs = 0, nkwargs = 0, nargs2 = 0, nkwargs2 = 0;
114
115 // Determine storage requirements for positional and keyword args
116 (call_analyze(nargs, nkwargs, (const Args &) args_), ...);
117
118 // Allocate memory on the stack
119 PyObject **args =
120 (PyObject **) alloca((nargs + nkwargs + 1) * sizeof(PyObject *));
121
122 PyObject *kwnames =
123 nkwargs ? PyTuple_New((Py_ssize_t) nkwargs) : nullptr;
124
125 // Fill 'args' and 'kwnames' variables
126 (call_init<policy>(args + 1, kwnames, nargs2, nkwargs2, nargs,
127 (forward_t<Args>) args_), ...);
128
129 NB_DO_VECTORCALL();
130 } else {
131 // Simple version with only positional arguments
132 PyObject *args[sizeof...(Args) + 1], *kwnames = nullptr;
133 size_t nargs = 0;
134
135 ((args[1 + nargs++] =
136 detail::make_caster<Args>::from_cpp(
137 (detail::forward_t<Args>) args_, policy, nullptr)
138 .ptr()),
139 ...);
140
141 NB_DO_VECTORCALL();
142 }
143}

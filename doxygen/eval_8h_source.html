28template <eval_mode start = eval_expr>
29object eval(const str &expr, handle global = handle(), handle local = handle()) {
30 dict global_default;
31 if (!global.is_valid()) {
32 global_default = globals();
33 global = global_default;
34 }
35
36 if (!local.is_valid())
37 local = global;
38
39 // This used to be PyRun_String, but that function isn't in the stable ABI.
40 object codeobj = steal(Py_CompileString(expr.c_str(), "<string>", start));
41 if (!codeobj.is_valid())
42 raise_python_error();
43
44 PyObject *result = PyEval_EvalCode(codeobj.ptr(), global.ptr(), local.ptr());
45 if (!result)
46 raise_python_error();
47
48 return steal(result);
49}

51template <eval_mode start = eval_expr, size_t N>
52object eval(const char (&s)[N], handle global = handle(), handle local = handle()) {
53 // Support raw string literals by removing common leading whitespace
54 str expr = (s[0] == '\n') ? str(module_::import_("textwrap").attr("dedent")(s)) : str(s);
55 return eval<start>(expr, global, local);

18// shared_ptr deleter that reduces the reference count of a Python object
19struct py_deleter {
20 void operator()(void *) noexcept {
21 // Don't run the deleter if the interpreter has been shut down
22 if (!is_alive())
23 return;

73 bool from_python(handle src, uint8_t flags,
74 cleanup_list *cleanup) noexcept {
75 flags &= ~((uint8_t) cast_flags::convert);
76
77 Caster caster;
78 if (!caster.from_python(src, flags, cleanup))
79 return false;
80
81 Td *ptr = caster.operator Td *();
82 if constexpr (has_shared_from_this_v<T>) {
83 if (ptr) {
84 // Guard against concurrent conversions of the same object,
85 // which would race on its internal 'weak_this' member
86 ft_object_guard guard(src);
87 if (auto sp = ptr->weak_from_this().lock()) {
88 // There is already a C++ shared_ptr for this object. Use it.
89 value = std::static_pointer_cast<T>(std::move(sp));
90 return true;
91 }
92 // Otherwise create a new one. Use shared_from_python<T>(...)
93 // so that future calls to ptr->shared_from_this() can share
94 // ownership with it.
95 value = shared_from_python(ptr, src);
96 return true;
97 }
98 value = shared_from_python(ptr, src);
99 } else {
100 value = std::static_pointer_cast<T>(
101 shared_from_python(static_cast<void *>(ptr), src));
102 }
103 return true;
104 }

106 static handle from_cpp(const Value &value, rv_policy,
107 cleanup_list *cleanup) noexcept {
108 bool is_new = false;
109 handle result;
110
111 Td *ptr = (Td *) value.get();
112 const std::type_info *type = &typeid(Td);
113
114 constexpr bool has_type_hook =
115 !std::is_base_of_v<std::false_type, type_hook<Td>>;
116 if constexpr (has_type_hook)
117 type = type_hook<Td>::get(ptr);
118
119 if constexpr (!std::is_polymorphic_v<Td>) {
120 result = nb_type_put(type, ptr, rv_policy::reference,
121 cleanup, &is_new);
122 } else {
123 const std::type_info *type_p =
124 (!has_type_hook && ptr) ? &typeid(*ptr) : nullptr;
125
126 result = nb_type_put_p(type, type_p, ptr, rv_policy::reference,
127 cleanup, &is_new);
128 }
129
130 if (is_new) {
131 std::shared_ptr<void> pp;
132 if constexpr (std::is_const_v<T>)
133 pp = std::static_pointer_cast<void>(std::const_pointer_cast<Td>(value));
134 else
135 pp = std::static_pointer_cast<void>(value);
136 shared_from_cpp(std::move(pp), result.ptr());
137 }
138
139 return result;
140 }

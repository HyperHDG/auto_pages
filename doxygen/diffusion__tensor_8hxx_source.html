22/*!*************************************************************************************************
23 * \brief Eigendecomposition A = V diag(lambda) V^T of a small symmetric matrix (cyclic Jacobi).
24 *
25 * Row-major n x n; V orthogonal, its columns the eigenvectors. Free of LAPACK, so that it works
26 * for any floating point type.
27 **************************************************************************************************/
28template <unsigned int n, typename T>
29void symmetric_eigen(std::array<T, n * n> a, std::array<T, n * n>& v, std::array<T, n>& lambda)
30{
31 v.fill((T)0.);
32 for (unsigned int i = 0; i < n; ++i)
33 v[i * n + i] = (T)1.;
34 T scale = (T)0.;
35 for (const T& entry : a)
36 scale = std::max(scale, (T)std::abs(entry));
37 for (unsigned int sweep = 0; sweep < 64; ++sweep)
38 {
39 T off = (T)0.;
40 for (unsigned int p = 0; p < n; ++p)
41 for (unsigned int q = p + 1; q < n; ++q)
42 off = std::max(off, (T)std::abs(a[p * n + q]));
43 if (off <= std::numeric_limits<T>::epsilon() * scale)
44 break;
45 for (unsigned int p = 0; p < n; ++p)
46 for (unsigned int q = p + 1; q < n; ++q)
47 {
48 if (a[p * n + q] == (T)0.)
49 continue;
50 // rotation that annihilates a(p, q)
51 const T theta = (a[q * n + q] - a[p * n + p]) / ((T)2. * a[p * n + q]);
52 const T t = (theta >= (T)0. ? (T)1. : (T)-1.) /
53 ((T)std::abs(theta) + (T)std::sqrt(theta * theta + (T)1.));
54 const T c = (T)1. / (T)std::sqrt(t * t + (T)1.), s = t * c;
55 for (unsigned int k = 0; k < n; ++k)
56 {
57 const T akp = a[k * n + p], akq = a[k * n + q];
58 a[k * n + p] = c * akp - s * akq;
59 a[k * n + q] = s * akp + c * akq;
60 }
61 for (unsigned int k = 0; k < n; ++k)
62 {
63 const T apk = a[p * n + k], aqk = a[q * n + k];
64 a[p * n + k] = c * apk - s * aqk;
65 a[q * n + k] = s * apk + c * aqk;
66 }
67 for (unsigned int k = 0; k < n; ++k)
68 {
69 const T vkp = v[k * n + p], vkq = v[k * n + q];
70 v[k * n + p] = c * vkp - s * vkq;
71 v[k * n + q] = s * vkp + c * vkq;
72 }
73 }
74 }
75 for (unsigned int i = 0; i < n; ++i)
76 lambda[i] = a[i * n + i];
77}
78} // end of namespace detail

80/*!*************************************************************************************************
81 * \brief LDG-H diffusion with a sum-factorized, matrix-free local operator (fast
82 *diagonalization).
83 *
84 * The same discretization as \c Diffusion (same shape functions, quadrature and unknowns, so the
85 * same discrete solution), for hyperedges that are boxes (mat_r diagonal, axis-aligned local
86 * normals, as UnitCube and the one-dimensional hyperedges of File) and a diffusion coefficient that
87 * is constant on each hyperedge (the parameters declare constant_coefficients). Then the local
88 * problem splits into one-dimensional matrices:
89 * - the flux unknowns q_d have the mass k |E| I (k the inverse diffusion coefficient, the
90 * Legendre basis orthonormal) and are eliminated, q_d = (b_q + G_d u) / (k |E|);
91 * - the Schur complement in u is S_u = sum_d X_d acting in direction d (identity in the others),
92 * X_d = tau |F_d| sum_s b_s b_s^T + H_d G_d / (k |E|) with the 1D derivative matrix G_d, its
93 * partner H_d = -G_d + boundary terms (= G_d^T up to quadrature) and the endpoint values b_s;
94 * - S_u^-1 is applied by fast diagonalization: X_d = V_d Lambda_d V_d^T, transform with V_d^T
95 * direction by direction, divide by sum_d Lambda_d, transform back with V_d;
96 * - B and C (trace to volume and back) are extensions and restrictions along one direction.
97 * One application costs about 4 d (p+1)^(d+1) + O(d (p+1)^d) operations and the operator stores
98 * O(d (p+1)^2) numbers, against (2 d (p+1)^(d-1))^2 for the condensed matrix of Diffusion; nothing
99 * is factorized, and no LAPACK is needed.
100 *
101 * The load, the errors and the Dirichlet nodes are Diffusion's (composition: a Diffusion member).
102 * The local operator implements apply, residual and local_solution for GlobalLoop::Generic; the
103 * legacy interface of the other global loops is not provided.
104 **************************************************************************************************/

145 /*!***********************************************************************************************
146 * \brief The matrix-free local operator of one box-shaped hyperedge.
147 ************************************************************************************************/
148 struct Operator
149 {
150 using Mat1 = std::array<T, n1 * n1>; // row-major: (i_d, j_d)
151 T tau, kE_inv; // tau and 1 / (k |E|)
152 std::array<std::array<T, n1>, 2> b; // endpoint values b_s(i), s = 0, 1
153 std::array<T, dim> face_area; // |F_d|, both faces in direction d
154 std::array<std::array<T, 2>, dim> n; // normal component n_d of face 2 d + s
155 std::array<Mat1, dim> G, H, V; // 1D derivative, its partner, eigenvectors of X_d
156 std::array<std::array<T, n1>, dim> lambda;
157
158 template <unsigned int k, class SolverT, class Edge, class Args>
159 void apply(const SolverT&,
160 Edge&,
161 const SmallMat<n_loc, k, T>& in,
162 const Args&,
163 SmallMat<n_loc, k, T>& out) const
164 {
165 for (unsigned int col = 0; col < k; ++col)
166 {
167 SmallVec<n_loc, T> lambda_col, flux;
168 for (unsigned int i = 0; i < n_loc; ++i)
169 lambda_col[i] = in(i, col);
170 flux_of(lambda_col, SmallVec<n_ldofs, T>(), flux);
171 for (unsigned int i = 0; i < n_loc; ++i)
172 out(i, col) = flux[i];
173 }
174 }

186 Edge& edge,
187 const SmallVec<n_loc, T>& lambda_in,
188 const Args& args) const

193 private:
194 // r = C x + D lambda for the local solution x of the trace lambda and the load
195 void flux_of(const SmallVec<n_loc, T>& lambda_in,
196 const SmallVec<n_ldofs, T>& load,
197 SmallVec<n_loc, T>& r) const
198 {
199 const SmallVec<n_ldofs, T> x = solve(lambda_in, load);
200 // flux = C x + D lambda: restriction of -(n_d q_d + tau (u - lambda)) to every face, the
201 // sign with which the condensed operator is positive semi-definite (as Diffusion)
202 for (unsigned int d = 0; d < dim; ++d)
203 for (unsigned int s = 0; s < 2; ++s)
204 {
205 const unsigned int face = 2 * d + s;
206 for (unsigned int j = 0; j < n_face; ++j)
207 r[face * n_face + j] = tau * face_area[d] * lambda_in[face * n_face + j];
208 for (unsigned int i = 0; i < n_vol; ++i)
209 {
210 const unsigned int i_d = digit(i, d);
211 r[face * n_face + face_index(i, d)] -=
212 face_area[d] * b[s][i_d] * (n[d][s] * x[d * n_vol + i] + tau * x[dim * n_vol + i]);
213 }
214 }
215 }
216 // the local solution x = A^-1 (B lambda + l)
217 SmallVec<n_ldofs, T> solve(const SmallVec<n_loc, T>& lambda_in,
218 const SmallVec<n_ldofs, T>& load) const
219 {
220 // b = B lambda + l: extension of the face values along their direction
221 SmallVec<n_ldofs, T> rhs = load;
222 for (unsigned int d = 0; d < dim; ++d)
223 for (unsigned int s = 0; s < 2; ++s)
224 {
225 const unsigned int face = 2 * d + s;
226 for (unsigned int i = 0; i < n_vol; ++i)
227 {
228 const T value =
229 face_area[d] * b[s][digit(i, d)] * lambda_in[face * n_face + face_index(i, d)];
230 rhs[dim * n_vol + i] += tau * value;
231 rhs[d * n_vol + i] -= n[d][s] * value;
232 }
233 }
234 // u = S_u^-1 (b_u - sum_d H_d b_q_d / (k |E|)), by fast diagonalization
235 std::array<T, n_vol> u, work;
236 for (unsigned int i = 0; i < n_vol; ++i)
237 u[i] = rhs[dim * n_vol + i];
238 for (unsigned int d = 0; d < dim; ++d)
239 {
240 for (unsigned int i = 0; i < n_vol; ++i)
241 work[i] = rhs[d * n_vol + i];
242 along(H[d], false, d, work);
243 for (unsigned int i = 0; i < n_vol; ++i)
244 u[i] -= kE_inv * work[i];
245 }
246 for (unsigned int d = 0; d < dim; ++d)
247 along(V[d], true, d, u);
248 for (unsigned int i = 0; i < n_vol; ++i)
249 {
250 T sum = (T)0.;
251 for (unsigned int d = 0; d < dim; ++d)
252 sum += lambda[d][digit(i, d)];
253 u[i] /= sum;
254 }
255 for (unsigned int d = 0; d < dim; ++d)
256 along(V[d], false, d, u);
257 // q_d = (b_q_d + G_d u) / (k |E|)
258 SmallVec<n_ldofs, T> x;
259 for (unsigned int d = 0; d < dim; ++d)
260 {
261 work = u;
262 along(G[d], false, d, work);
263 for (unsigned int i = 0; i < n_vol; ++i)
264 x[d * n_vol + i] = kE_inv * (rhs[d * n_vol + i] + work[i]);
265 }
266 for (unsigned int i = 0; i < n_vol; ++i)
267 x[dim * n_vol + i] = u[i];
268 return x;
269 }
270
271 private:
272 // n1^d, the step of direction d in the volume index
273 static constexpr unsigned int step(const unsigned int d)
274 {
275 unsigned int result = 1;
276 for (unsigned int c = 0; c < d; ++c)
277 result *= n1;
278 return result;
279 }
280 // digit d of the volume index i (direction 0 fastest, as Hypercube::index_decompose)
281 static constexpr unsigned int digit(const unsigned int i, const unsigned int d)
282 {
283 return (i / step(d)) % n1;
284 }
285 // the face index of i on a face in direction d: its digits without digit d
286 static constexpr unsigned int face_index(const unsigned int i, const unsigned int d)
287 {
288 const unsigned int stride = step(d);
289 return i % stride + (i / (stride * n1)) * stride;
290 }
291 // v = M v (or M^T v) along direction d
292 static void along(const Mat1& m,
293 const bool transposed_m,
294 const unsigned int d,
295 std::array<T, n_vol>& v)
296 {
297 const unsigned int stride = step(d);
298 const std::array<T, n_vol> in = v;
299 for (unsigned int i = 0; i < n_vol; ++i)
300 {
301 const unsigned int i_d = digit(i, d), base = i - i_d * stride;
302 T sum = (T)0.;
303 for (unsigned int m_d = 0; m_d < n1; ++m_d)
304 sum += (transposed_m ? m[m_d * n1 + i_d] : m[i_d * n1 + m_d]) * in[base + m_d * stride];
305 v[i] = sum;
306 }
307 }
308 };

311 * \brief Set up the one-dimensional matrices and the eigendecompositions of one hyperedge.

313 template <class Edge>
314 Operator local_operator(Edge& edge, GlobalLoop::Stiffness, const Args& args) const
315 {
316 constexpr unsigned int space_dim = std::remove_cvref_t<decltype(Edge::geometry)>::space_dim();
317 using parameters = parametersT<space_dim, T>;
318 static_assert(
319 requires { parameters::constant_coefficients; },
320 "DiffusionTensor needs a diffusion coefficient constant on every hyperedge: the "
321 "parameters must declare constant_coefficients.");
322 static_assert(parameters::constant_coefficients,
323 "DiffusionTensor needs a diffusion coefficient constant on every hyperedge.");
324 const auto& geometry = edge.geometry;
325 const auto r = geometry.mat_r();
326 const T area = geometry.area();
327 const T k = parameters::inverse_diffusion_coeff(geometry.map_ref_to_phys(Point<dim, T>((T)0.5)),
328 args.time);
329
330 Operator op;
331 op.tau = tau_;
332 op.kE_inv = (T)1. / (k * area);
333 for (unsigned int s = 0; s < 2; ++s)
334 for (unsigned int a = 0; a < n1; ++a)
335 op.b[s][a] = integrator::shape_fcts_at_bdr[a][s];
336 for (unsigned int d = 0; d < dim; ++d)
337 {
338 if constexpr (hy_debug_mode)
339 for (unsigned int c = 0; c < dim; ++c)
340 {
341 hy_assert(c == d || std::abs(r(c, d)) <= 1e-12 * std::abs(r(d, d)),
342 "DiffusionTensor needs box-shaped hyperedges (mat_r diagonal).");
343 hy_assert(c == d || std::abs(geometry.local_normal(2 * d)[c]) <= 1e-12,
344 "DiffusionTensor needs axis-aligned local normals.");
345 }
346 op.face_area[d] = geometry.face_area(2 * d);
347 for (unsigned int s = 0; s < 2; ++s)
348 op.n[d][s] = geometry.local_normal(2 * d + s)[d];
349 // G_d(i, j) = |E| / r_d int phi_i' phi_j, and its partner H_d = -G_d + boundary terms
350 typename Operator::Mat1 x;
351 for (unsigned int i = 0; i < n1; ++i)
352 for (unsigned int j = 0; j < n1; ++j)
353 {
354 const T g = area / r(d, d) * integrator::integrate_1D_Dphiphi(i, j);
355 T boundary = (T)0., face_mass = (T)0.;
356 for (unsigned int s = 0; s < 2; ++s)
357 {
358 boundary += op.face_area[d] * op.n[d][s] * op.b[s][i] * op.b[s][j];
359 face_mass += op.face_area[d] * op.b[s][i] * op.b[s][j];
360 }
361 op.G[d][i * n1 + j] = g;
362 op.H[d][i * n1 + j] = -g + boundary;
363 x[i * n1 + j] = tau_ * face_mass;
364 }
365 // X_d = tau |F_d| sum_s b_s b_s^T + H_d G_d / (k |E|), symmetric up to quadrature rounding
366 for (unsigned int i = 0; i < n1; ++i)
367 for (unsigned int j = 0; j < n1; ++j)
368 for (unsigned int m = 0; m < n1; ++m)
369 x[i * n1 + j] += op.kE_inv * op.H[d][i * n1 + m] * op.G[d][m * n1 + j];
370 for (unsigned int i = 0; i < n1; ++i)
371 for (unsigned int j = i + 1; j < n1; ++j)
372 x[i * n1 + j] = x[j * n1 + i] = (T)0.5 * (x[i * n1 + j] + x[j * n1 + i]);
373 detail::symmetric_eigen<n1, T>(x, op.V[d], op.lambda[d]);
374 }
375 return op;
376 }

24template <typename T> constexpr bool is_eigen_sparse_matrix_v =
25 is_base_of_template_v<T, Eigen::SparseMatrixBase> &&
26 !std::is_base_of_v<Eigen::SparseMapBase<T, Eigen::ReadOnlyAccessors>, T>;

54 bool from_python(handle src, uint8_t flags, cleanup_list *cleanup) noexcept {
55 object obj = borrow(src);
56
57 try {
58 object matrix_type =
59 module_::import_("scipy.sparse")
60 .attr(RowMajor ? "csr_matrix" : "csc_matrix");
61 if (!obj.type().is(matrix_type)) {
62 if (!(flags & (uint8_t) cast_flags::convert))
63 return false;
64 obj = matrix_type(obj);
65 }
66
67 if (!cast<bool>(obj.attr("has_sorted_indices")))
68 obj.attr("sort_indices")();
69
70 if (object data_o = obj.attr("data");
71 !data_caster.from_python(data_o, flags, cleanup))
72 return false;
73
74 if (object indices_o = obj.attr("indices");
75 !indices_caster.from_python(indices_o, flags, cleanup))
76 return false;
77
78 if (object indptr_o = obj.attr("indptr");
79 !indptr_caster.from_python(indptr_o, flags, cleanup))
80 return false;
81
82 object shape_o = obj.attr("shape");
83 if (len(shape_o) != 2)
84 return false;
85
86 Index rows = cast<Index>(shape_o[0]),
87 cols = cast<Index>(shape_o[1]),
88 nnz = cast<Index>(obj.attr("nnz"));
89
90 value = SparseMap(rows, cols, nnz,
91 indptr_caster.value.data(),
92 indices_caster.value.data(),
93 data_caster.value.data());
94 return true;
95 } catch (const std::exception &) {
96 return false;
97 }
98 }

108 template <typename T2>
109 static handle from_cpp(T2 &&v, rv_policy policy, cleanup_list *cleanup) noexcept {
110 policy = infer_policy<T2>(policy);
111 if constexpr (std::is_pointer_v<T2>)
112 return from_cpp_internal((const T &) *v, policy, cleanup);
113 else

117 static handle from_cpp_internal(const T &v, rv_policy policy, cleanup_list *) noexcept {
118 if (!v.isCompressed()) {
119 PyErr_SetString(PyExc_ValueError,
120 "nanobind: unable to return an Eigen sparse matrix that is not in a compressed format. "
121 "Please call `.makeCompressed()` before returning the value on the C++ end.");
122 return handle();
123 }
124
125 object matrix_type;
126 try {
127 matrix_type = module_::import_("scipy.sparse").attr(RowMajor ? "csr_matrix" : "csc_matrix");
128 } catch (python_error &e) {
129 e.restore();
130 return handle();
131 }
132
133 const Index rows = v.rows(), cols = v.cols();
134 const size_t data_shape[] = { (size_t) v.nonZeros() };
135 const size_t outer_indices_shape[] = { (size_t) ((RowMajor ? rows : cols) + 1) };
136
137 T *src = std::addressof(const_cast<T &>(v));
138 object owner;
139 if (policy == rv_policy::move) {
140 // Eigen sparse matrices lack a move constructor; copy-constructing
141 // from a matrix flagged as an rvalue performs a cheap pointer swap.
142 const_cast<T &>(v).markAsRValue();
143 src = new T(v);
144 owner = capsule(src, [](void *p) noexcept { delete (T *) p; });
145 }
146
147 ScalarNDArray data(src->valuePtr(), 1, data_shape, owner);
148 StorageIndexNDArray outer_indices(src->outerIndexPtr(), 1, outer_indices_shape, owner);
149 StorageIndexNDArray inner_indices(src->innerIndexPtr(), 1, data_shape, owner);
150
151 try {
152 return matrix_type(nanobind::make_tuple(
153 std::move(data), std::move(inner_indices), std::move(outer_indices)),
154 nanobind::make_tuple(rows, cols))
155 .release();
156 } catch (python_error &e) {
157 e.restore();
158 return handle();
159 }
160 }

149void default_exception_translator(const std::exception_ptr &p, void *) {
150 try {
151 std::rethrow_exception(p);
152 } catch (const std::bad_alloc &e) {
153 PyErr_SetString(PyExc_MemoryError, e.what());
154 } catch (const std::domain_error &e) {
155 PyErr_SetString(PyExc_ValueError, e.what());
156 } catch (const std::invalid_argument &e) {
157 PyErr_SetString(PyExc_ValueError, e.what());
158 } catch (const std::length_error &e) {
159 PyErr_SetString(PyExc_ValueError, e.what());
160 } catch (const std::out_of_range &e) {
161 PyErr_SetString(PyExc_IndexError, e.what());
162 } catch (const std::range_error &e) {
163 PyErr_SetString(PyExc_ValueError, e.what());
164 } catch (const std::overflow_error &e) {
165 PyErr_SetString(PyExc_OverflowError, e.what());
166 } catch (const std::exception &e) {
167 PyErr_SetString(PyExc_RuntimeError, e.what());
168 }
169}

177// Reclaims a thread's state when it exits (the cleanup-key callback).
178static void nb_thread_state_destroy(void *p) noexcept {
179 nb_thread_state *ts = (nb_thread_state *) p;
180 if (!ts)
181 return;
182
183 // Reclaim this thread's instance pools if the runtime is still alive
184 if (internals && ts->pools) {
185 PyGILState_STATE state = PyGILState_Ensure();
186 for (uint32_t i = 0; i < ts->pools_size; ++i)
187 nb_pool_drain(&ts->pools[i], /* can_free = */ true);
188 PyGILState_Release(state);
189 }
190 PyMem_Free(ts->pools);
191
192 nb_thread_state_tls = nullptr;
193 delete ts;
194}
195
196// Slow path for nb_thread_state_get(): allocate the per-thread state with a cleanup callback
197nb_thread_state *nb_thread_state_alloc() noexcept {
198#if defined(_WIN32)
199 DWORD key = internals->thread_state_key;
200 nb_thread_state *ts = (nb_thread_state *) FlsGetValue(key);
201 if (!ts) {
202 ts = new nb_thread_state();
203 check(FlsSetValue(key, ts), "nanobind: FlsSetValue() failed!");
204 }
205#else
206 pthread_key_t key = internals->thread_state_key;
207 nb_thread_state *ts = (nb_thread_state *) pthread_getspecific(key);
208 if (!ts) {
209 ts = new nb_thread_state();
210 check(pthread_setspecific(key, ts) == 0,
211 "nanobind: pthread_setspecific() failed!");
212 }
213#endif
214 nb_thread_state_tls = ts;
215 return ts;
216}

238static void init_pyobjects(nb_internals *p) {
239 if (static_pyobjects[0] &&
240 static_pyobjects_generation == p->lifeline_generation)
241 return;
242
243 static_pyobjects_generation = p->lifeline_generation;

245 NB_NOUNROLL
246 for (int i = 0; i < pyobj_name::string_count; ++i)

248
249 new_constant(p, pyobj_name::interned_max_version_tpl,

260 Py_DECREF(minor);
261 Py_DECREF(major);

265static void init_internals(nb_internals *p) {
266 if (p->lifeline)
267 return;
268
269 p->lifeline = PyList_New(0);
270 check(p->lifeline, "nanobind::detail::nb_module_exec(): "
271 "could not create lifeline list!");
272
273 str nb_name("nanobind");
274 p->nb_module = PyModule_NewObject(nb_name.ptr());
275 new_object(p, p->nb_module);
276
277 // Construct nanobind's meta-meta class
278 nb_meta_slots[0].pfunc = (PyObject *) &PyType_Type;
279 PyTypeObject *nb_meta = new_type(p, &nb_meta_spec);
280
281 p->nb_func = new_type(p, &nb_func_spec);
282 p->nb_method = new_type(p, &nb_method_spec);
283 p->nb_bound_method = new_type(p, &nb_bound_method_spec);
284
285 check(p->nb_module && nb_meta && p->nb_func &&
286 p->nb_method && p->nb_bound_method,
287 "nanobind::detail::nb_module_exec(): initialization failed!");
288
289#if defined(Py_LIMITED_API)
290 p->PyType_Type_tp_free = (freefunc) PyType_GetSlot(&PyType_Type, Py_tp_free);
291 p->PyType_Type_tp_init = (initproc) PyType_GetSlot(&PyType_Type, Py_tp_init);
292 p->PyType_Type_tp_dealloc =
293 (destructor) PyType_GetSlot(&PyType_Type, Py_tp_dealloc);
294 p->PyType_Type_tp_setattro =
295 (setattrofunc) PyType_GetSlot(&PyType_Type, Py_tp_setattro);
296 p->PyProperty_Type_tp_descr_get =
297 (descrgetfunc) PyType_GetSlot(&PyProperty_Type, Py_tp_descr_get);
298 p->PyProperty_Type_tp_descr_set =
299 (descrsetfunc) PyType_GetSlot(&PyProperty_Type, Py_tp_descr_set);
300
301 PyType_Slot dummy_slots[] = {
302 { Py_tp_base, &PyType_Type },
303 { 0, nullptr }
304 };
305
306 PyType_Spec dummy_spec = {
307 /* .name = */ "nanobind.dummy",
308 /* .basicsize = */ - (int) sizeof(void*),
309 /* .itemsize = */ 0,
310 /* .flags = */ Py_TPFLAGS_DEFAULT,
311 /* .slots = */ dummy_slots
312 };
313
314 PyObject *dummy = PyType_FromMetaclass(
315 nb_meta, p->nb_module, &dummy_spec, nullptr);
316 p->type_data_offset =
317 ((uint8_t *) PyObject_GetTypeData(dummy, nb_meta) - (uint8_t *) dummy);
318 Py_DECREF(dummy);
319#endif
320
321 // Create the single metaclass shared by all bound types. This may
322 // access 'type_data_offset' defined just above.
323 p->nb_type = nb_type_create_metaclass(p, nb_meta);
324 check(p->nb_type, "nanobind::detail::nb_module_exec(): "
325 "nb_type metaclass creation failed!");
326}

332void internals_dec_ref() {
333 nb_internals *p = internals;
334 auto value = --p->shared_ref_count.value;
335 if (value != 0)
336 return;
337
338 // Invalidate every library's cached 'static_pyobjects' array: destroying
339 // the lifeline frees the objects that these arrays reference.
340 p->lifeline_generation++;
341
342 Py_CLEAR(p->lifeline);
343
344 p->nb_module = nullptr;
345 p->nb_type = nullptr;
346 p->nb_func = nullptr;
347 p->nb_method = nullptr;
348 p->nb_bound_method = nullptr;
349 p->nb_static_property.store_release(nullptr);
350 p->nb_ndarray.store_release(nullptr);
351 for (auto &entry : p->ndarray_export)
352 entry.store_release(nullptr);
353
354 for (int i = 0; i < pyobj_name::total_count; ++i)
355 static_pyobjects[i] = nullptr;
356}

358void nb_module_free(void *) {
359 internals_dec_ref();
360}

369static void internals_cleanup() {
370 nb_internals *p = internals;
371 if (!p)
372 return;
373
374 *is_alive_ptr = false;
375
376#if !defined(PYPY_VERSION) && !defined(NB_FREE_THREADED)
377 /* The memory leak checker is unsupported on PyPy, see
378 see https://foss.heptapod.net/pypy/pypy/-/issues/3855.
379
380 Leak reporting is explicitly disabled on free-threaded builds
381 for now because of the decision to immortalize function and
382 type objects. This may change in the future. */
383
384 bool print_leak_warnings = p->print_leak_warnings;
385
386 // Unmap pooled instances to avoid false leaks (can_free=false: no thread state here).
387 for (const auto &kv : p->type_c2p_slow) {
388 type_data *td = kv.second;
389 if (td->flags & (uint32_t) type_flags::pooled)
390 nb_pool_drain(&td->pool, /* can_free = */ false);
391 }
392
393 size_t inst_leaks = 0, keep_alive_leaks = 0;
394
395 // Shard locking no longer needed, Py_AtExit is single-threaded
396 for (size_t i = 0; i < p->shard_count; ++i) {
397 nb_shard &s = p->shards[i];
398 inst_leaks += s.inst_c2p.size();
399 keep_alive_leaks += s.keep_alive.size();
400 }
401
402#ifdef _DEBUG
403// in debug mode, show all leak records
404#define INC_CTR do {} while(0)
405#else
406// otherwise show just the first 10 or 20
407#define INC_CTR ctr++
408#endif
409
410 bool leak = inst_leaks > 0 || keep_alive_leaks > 0;
411
412 if (print_leak_warnings && inst_leaks > 0) {
413 fprintf(stderr, "nanobind: leaked %zu instances!\n", inst_leaks);
414
415#if !defined(Py_LIMITED_API)
416 auto print_leak = [](void* k, PyObject* v) {
417 type_data *tp = nb_type_data(Py_TYPE(v));
418 fprintf(stderr, " - leaked instance %p of type \"%s\"\n", k, tp->name);
419 };
420
421 int ctr = 0;
422 for (size_t i = 0; i < p->shard_count && ctr < 20; ++i) {
423 for (auto [k, v]: p->shards[i].inst_c2p) {
424 if (NB_UNLIKELY(nb_is_seq(v))) {
425 nb_inst_seq* seq = nb_get_seq(v);
426 for(; seq != nullptr && ctr < 20; seq = seq->next) {
427 print_leak(k, seq->inst);
428 INC_CTR;
429 }
430 } else {
431 print_leak(k, (PyObject*)v);
432 INC_CTR;
433 }
434 if (ctr >= 20)
435 break;
436 }
437 }
438 if (ctr >= 20) {
439 fprintf(stderr, " - ... skipped remainder\n");
440 }
441#endif
442 }
443
444 if (print_leak_warnings && keep_alive_leaks > 0)
445 fprintf(stderr, "nanobind: leaked %zu keep_alive records!\n",
446 keep_alive_leaks);
447
448 // Only report function/type leaks if actual nanobind instances were leaked
449#if !defined(NB_ABORT_ON_LEAK)
450 if (!leak)
451 print_leak_warnings = false;
452#endif
453
454 if (!p->type_c2p_slow.empty()) {
455 if (print_leak_warnings) {
456 fprintf(stderr, "nanobind: leaked %zu types!\n",
457 p->type_c2p_slow.size());
458 int ctr = 0;
459 for (const auto &kv : p->type_c2p_slow) {
460 fprintf(stderr, " - leaked type \"%s\"\n", kv.second->name);
461 INC_CTR;
462 if (ctr == 10) {
463 fprintf(stderr, " - ... skipped remainder\n");
464 break;
465 }
466 }
467 }
468 leak = true;
469 }
470
471 if (!p->funcs.empty()) {
472 if (print_leak_warnings) {
473 fprintf(stderr, "nanobind: leaked %zu functions!\n",
474 p->funcs.size());
475 int ctr = 0;
476 for (auto [f, p2] : p->funcs) {
477 fprintf(stderr, " - leaked function \"%s\"\n",
478 nb_func_data(f)->name);
479 INC_CTR;
480 if (ctr == 10) {
481 fprintf(stderr, " - ... skipped remainder\n");
482 break;
483 }
484 }
485 }
486 leak = true;
487 }
488
489 if (!leak) {
490 nb_translator_seq* t = p->translators.load_relaxed();
491 while (t) {
492 nb_translator_seq *next = t->next;
493 delete t;
494 t = next;
495 }
496
497#if defined(NB_FREE_THREADED)
498 // This code won't run for now but is kept here for a time when
499 // immortalization isn't needed anymore.
500
501 PyThread_tss_delete(p->nb_static_property_disabled);
502 PyThread_tss_free(p->nb_static_property_disabled);
503 delete[] p->shards;
504#endif
505
506 delete p;
507 internals = nullptr;
508 } else {
509 if (print_leak_warnings) {
510 fprintf(stderr, "nanobind: this is likely caused by a reference "
511 "counting issue in the binding code.\n"
512 "See https://nanobind.readthedocs.io/en/latest/refleaks.html");
513 }
514
515 #if defined(NB_ABORT_ON_LEAK) && !defined(NB_FREE_THREADED)
516 abort(); // Extra-strict behavior for the CI server
517 #endif
518 }
519#endif
520}

522NB_NOINLINE void nb_module_exec(const char *name, PyObject *) {
523 if (internals) {
524 init_internals(internals);
525 init_pyobjects(internals);
526 internals_inc_ref();
527 return;
528 }
529
530#if defined(PYPY_VERSION)
531 PyObject *dict = PyEval_GetBuiltins();
532#else
533 PyObject *dict = PyInterpreterState_GetDict(PyInterpreterState_Get());
534#endif
535 check(dict, "nanobind::detail::nb_module_exec(): "
536 "could not access internals dictionary!");
537
538 PyObject *key = PyUnicode_FromFormat("__nb_internals_%s_%s__",
539 abi_tag(), name ? name : "");
540 check(key, "nanobind::detail::nb_module_exec(): "
541 "could not create dictionary key!");
542
543 PyObject *capsule = dict_getitem_or_default(dict, key, nullptr);
544 if (capsule) {
545 Py_DECREF(key);
546 internals = (nb_internals *) PyCapsule_GetPointer(capsule, "nb_internals");
547 check(internals, "nanobind::detail::nb_module_exec(): "
548 "capsule pointer is NULL!");
549 is_alive_ptr = internals->is_alive_ptr;
550
551 init_internals(internals);
552 init_pyobjects(internals);
553 internals_inc_ref();
554
555 Py_DECREF(capsule);
556 return;
557 }
558
559 nb_internals *p = new nb_internals();
560
561 size_t shard_count = 1;
562#if defined(NB_FREE_THREADED)
563 size_t hw_concurrency = std::thread::hardware_concurrency();
564 while (shard_count < hw_concurrency)
565 shard_count *= 2;
566 shard_count *= 2;
567 p->shards = new nb_shard[shard_count];
568 p->shard_mask = shard_count - 1;
569
570 // Per-domain key for reclaiming nb_thread_state at thread exit
571#if defined(_WIN32)
572 p->thread_state_key = FlsAlloc((PFLS_CALLBACK_FUNCTION) nb_thread_state_destroy);
573 check(p->thread_state_key != FLS_OUT_OF_INDEXES, "nanobind: FlsAlloc() failed!");
574#else
575 check(pthread_key_create(&p->thread_state_key, nb_thread_state_destroy) == 0,
576 "nanobind: pthread_key_create() failed!");
577#endif
578#endif
579 p->shard_count = shard_count;
580
581 internals = p;
582
583 init_internals(p);
584 init_pyobjects(p);
585
586#if defined(NB_FREE_THREADED)
587 p->nb_static_property_disabled = PyThread_tss_alloc();
588 PyThread_tss_create(p->nb_static_property_disabled);
589#endif
590
591 p->translators.store_release(
592 new nb_translator_seq{ default_exception_translator, nullptr, nullptr });
593
594 is_alive_value = true;
595 is_alive_ptr = &is_alive_value;
596 p->is_alive_ptr = is_alive_ptr;
597
598 internals_inc_ref();
599
600#if !defined(PYPY_VERSION)
601 // typing.py on CPython introduces spurious reference leaks that upset
602 // nanobind's leak checker. The following band-aid installs an 'atexit'
603 // handler that clears LRU caches used in typing.py. To be resilient to
604 // potential future changes in typing.py, the implementation fails silently
605 // if any step goes wrong. For context, see
606 // https://github.com/python/cpython/issues/98253 and
607 // https://github.com/python/cpython/issues/151728. */
608
609 const char *str =
610 "def cleanup():\n"
611 " try:\n"
612 " import sys\n"
613 " fs = getattr(sys.modules.get('typing'), '_cleanups', None)\n"
614 " if fs is not None:\n"
615 " for f in fs:\n"
616 " f()\n"
617 " except:\n"
618 " pass\n"
619 "import atexit\n"
620 "atexit.register(cleanup)\n"
621 "del atexit, cleanup";
622
623 PyObject *code = Py_CompileString(str, "<internal>", Py_file_input);
624 if (code) {
625 PyObject *result = PyEval_EvalCode(code, PyEval_GetGlobals(), nullptr);
626 if (!result)
627 PyErr_Clear();
628 Py_XDECREF(result);
629 Py_DECREF(code);
630 } else {
631 PyErr_Clear();
632 }

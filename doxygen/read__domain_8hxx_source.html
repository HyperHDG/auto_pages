12/*!*************************************************************************************************
13 * \brief Check whether a \c std::vector does not contain duplicate entries.
14 *
15 * This function returns true if the \c std::vector \c vec does not contain duplicates, and false if
16 * it contains duplicates.
17 *
18 * \tparam vectorT Class template for vector.
19 * \param vec General vector to be checked for duplicates.
20 * \retval is_unique True if vector does not contain duplicates and false otherwise.
21 *
22 * \authors Guido Kanschat, Heidelberg University, 2020.
23 * \authors Andreas Rupp, Heidelberg University, 2020.
24 **************************************************************************************************/
25template <typename vectorT>
26bool is_unique(const vectorT& vec)
27{
28 vectorT x(vec); // Copy vector enabling reorder.
29 std::sort(x.begin(), x.end()); // Sort vector in O(N log N).
30 return (std::adjacent_find(x.begin(), x.end()) == x.end()); // Check for duplicates in O(n).
31}

122 hy_assert(consistent, "At least one hypernode index is invalid!");

124 });

127 [&](std::array<pt_index_t, 1 << hyEdge_dim> hyEdge)

132 consistent = (hyEdge[i] < n_points && hyEdge[i] >= 0);
133 hy_assert(consistent, "At least one point index is invalid!");
134 }
135 });
136
137 consistent = is_unique(points);
138 hy_assert(consistent, "DomainInfo.points contains duplicate points!");
139 if (!consistent)
140 return false;
141
142 consistent = is_unique(hyNodes_hyEdge);
143 hy_assert(consistent, "DomainInfo.hyNodes_hyEdge contains duplicate hypernode!");
144 if (!consistent)
145 return false;
146
147 consistent = is_unique(points_hyEdge);
148 hy_assert(consistent, "DomainInfo.points_hyEdge contains duplicate points!");
149 if (!consistent)
150 return false;

152 return true;
153 } // end of check_consistency

374/*!*************************************************************************************************
375 * \brief General Function to read domain from input file.
376 *
377 * At the moment, only self-defined .geo files are supported. These files are supposed to comprise
378 * hypergraphs consisting of hypercuboids only. Other versions have not yet been implemented.
379 *
380 * \tparam hyEdge_dim The local dimension of a hyperedge.
381 * \tparam space_dim The dimension of the surrounding space.
382 * \tparam vectorT The typename of a large vector holding e.g. all points.
383 * \tparam pointT The typename of a point.
384 * \tparam hyEdge_index_t The index type for hyperedges. Default is \c unsigned \c int.
385 * \tparam hyNode_index_t The index type for hypernodes. Default is hyEdge_index_t.
386 * \tparam pt_index_t The index type for points. Default is hyNode_index_t.
387 *
388 * \param filename Name of the .geo file to be read.
389 * \retval domain_info Topological and geometrical information of hypergraph.
390 *
391 * \authors Guido Kanschat, Heidelberg University, 2020.
392 * \authors Andreas Rupp, Heidelberg University, 2020.
393 **************************************************************************************************/
394template <unsigned int hyEdge_dim,
395 unsigned int space_dim,
396 template <typename...> typename vectorT = std::vector,
397 typename pointT = Point<space_dim, double>,
398 typename hyEdge_index_t = unsigned int,
399 typename hyNode_index_t = hyEdge_index_t,
400 typename pt_index_t = hyNode_index_t>
401DomainInfo<hyEdge_dim, space_dim, vectorT, pointT, hyEdge_index_t, hyNode_index_t, pt_index_t>
402read_domain(std::string filename)
403{
404 hy_assert(std::filesystem::exists(filename), "File does not exist.");
405
406 if (filename.substr(filename.size() - 4, filename.size()) == ".pts")
407 {
408 hy_assert(hyEdge_dim == 1, "This only works for graphs, so far!");
409 make_epsilon_neighborhood_graph<space_dim, vectorT, pointT, hyEdge_index_t>(filename);
410 }
411
412 hy_assert(filename.substr(filename.size() - 4, filename.size()) == ".geo",
413 "The given file needs to be a .geo file, since no other input file types are currently"
414 << " implemented.");
415
416 DomainInfo<hyEdge_dim, space_dim, vectorT, pointT, hyEdge_index_t, hyNode_index_t, pt_index_t>
417 domain_info = read_domain_geo<hyEdge_dim, space_dim, vectorT, pointT, hyEdge_index_t,
418 hyNode_index_t, pt_index_t>(filename);
419
420 hy_assert(domain_info.check_consistency(),
421 "Domain info appears to be inconsistent!"
422 << std::endl
423 << "This assertion is never to be thrown since it can only be caused by internal "
424 << "assertions of DomainInfo.check_consistency()!");
425
426 return domain_info;

16void trampoline_new(void **data, size_t size, void *ptr) noexcept {
17 // GIL is held when the trampoline constructor runs. Lock the
18 // associated instance shard in GIL-less Python.
19
20 nb_shard &shard = internals->shard(ptr);
21 lock_shard lock(shard);
22
23 nb_ptr_map &inst_c2p = shard.inst_c2p;
24 nb_ptr_map::iterator it = inst_c2p.find(ptr);
25 check(it != inst_c2p.end() && (((uintptr_t) it->second) & 1) == 0,
26 "nanobind::detail::trampoline_new(): unique instance not found!");
27
28 data[0] = it->second;
29 memset(data + 1, 0, sizeof(void *) * 2 * size);
30}

32void trampoline_release(void **data, size_t size) noexcept {
33 // GIL is held when the trampoline destructor runs
34 for (size_t i = 0; i < size; ++i)
35 Py_XDECREF((PyObject *) data[i*2 + 2]);

38static void trampoline_enter_internal(void **data, size_t size,
39 const char *name, bool pure, ticket *t) {
40 const PyObject *None = Py_None;
41 PyGILState_STATE state{ };
42 const char *error = nullptr;
43 PyObject *key = nullptr, *value = nullptr;
44 PyObject *self = (PyObject *) data[0];
45 PyTypeObject *value_tp = nullptr;
46 size_t offset = 0;
47
48 // First, perform a quick sweep without holding the GIL. This is safe
49 // because pointer-sized reads are atomic on all architectures and
50 // compilers targeted by nanobind (x86, x86_64, ARM, AArch64). Slots are
51 // only written under the GIL or an object critical section, transitioning
52 // from (nullptr, nullptr) to (name, value) exactly once, so the worst case
53 // of a racy read is that we miss an already-populated entry and fall through
54 // to the locked sweep, which is correct. The stored values are either
55 // Py_None or interned strings, both of which are immortal, so the
56 // reference read here cannot become dangling.
57 for (size_t i = 0; i < size; i++) {
58 void *d_name = data[2*i + 1],
59 *d_value = data[2*i + 2];
60 if (name == d_name && d_value) {
61 if (d_value != None) {
62 t->state = PyGILState_Ensure();
63 t->key = (PyObject *) d_value;
64 return;
65 } else {
66 if (pure) {
67 error = "tried to call a pure virtual function";
68 break;
69 } else {
70 return;
71 }
72 }
73 }
74 }
75
76 // Nothing found -- retry, now with lock held. The object critical section
77 // must end (on every path) before PyGILState_Release() is called, since the
78 // latter may detach the thread state and suspend the critical section. We
79 // therefore enclose the guarded region in a block scope and defer any GIL
80 // release or error reporting until after it.
81 state = PyGILState_Ensure();
82 {
83 ft_object_guard guard(self);
84
85 if (error)
86 goto fail;
87
88 for (size_t i = 0; i < size; i++) {
89 void *d_name = data[2*i + 1],
90 *d_value = data[2*i + 2];
91 if (name == d_name && d_value) {
92 if (d_value != None) {
93 t->state = state;
94 t->key = (PyObject *) d_value;
95 return;
96 } else {
97 if (pure) {
98 error = "tried to call a pure virtual function";
99 goto fail;
100 } else {
101 goto release;
102 }
103 }
104 }
105 }
106
107 // Sill no luck -- perform a lookup and populate the trampoline
108 for (; offset < size; offset++) {
109 if (data[2 * offset + 1] == nullptr &&
110 data[2 * offset + 2] == nullptr)
111 break;
112 }
113
114 if (offset == size) {
115 error = "the trampoline ran out of slots (you will need to increase "
116 "the value provided to the NB_TRAMPOLINE() macro)";
117 goto fail;
118 }
119
120 key = PyUnicode_InternFromString(name);
121 if (!key) {
122 error = "could not intern string";
123 goto fail;
124 }
125
126 value = PyObject_GetAttr(self, key);
127 if (!value) {
128 error = "lookup failed";
129 goto fail;
130 }
131
132 value_tp = Py_TYPE(value);
133 Py_CLEAR(value);
134
135 if (value_tp == internals->nb_func || value_tp == internals->nb_method ||
136 value_tp == internals->nb_bound_method) {
137 Py_DECREF(key);
138
139 if (pure) {
140 error = "tried to call a pure virtual function";
141 goto fail;
142 }
143
144 key = none_ref();
145 }
146
147 data[2 * offset + 1] = (void *) name;
148 data[2 * offset + 2] = key;
149
150 if (key != None) {
151 t->state = state;
152 t->key = key;
153 return;
154 }
155 }
156
157release:
158 PyGILState_Release(state);
159 return;
160
161fail:
162 type_data *td = nb_type_data(Py_TYPE(self));
163 PyGILState_Release(state);
164
165 raise("nanobind::detail::get_trampoline('%s::%s()'): %s!",
166 td->name, name, error);
167}

171void trampoline_enter(void **data, size_t size, const char *name, bool pure, ticket *t) {
172 trampoline_enter_internal(data, size, name, pure, t);
173
174 if (t->key) {
175 t->self = (PyObject *) data[0];
176 t->prev = current_ticket;
177
178 if (t->prev && t->prev->self.is(t->self) && t->prev->key.is(t->key)) {
179 t->self = handle();
180 t->key = handle();
181 t->prev = nullptr;
182 PyGILState_Release(t->state);
183 if (pure)
184 raise("nanobind::detail::get_trampoline('%s()'): tried to call "
185 "a pure virtual function!", name);
186 return;
187 }
188
189 current_ticket = t;
190 }
191}

193void trampoline_leave(ticket *t) noexcept {
194 if (!t->key)
195 return;
196 current_ticket = t->prev;
197 PyGILState_Release(t->state);
198}

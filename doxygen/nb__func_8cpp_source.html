89 size_t size = (size_t) Py_SIZE(self);
90 if (size) {

94#if !defined(NB_FREE_THREADED)

96 check(n_deleted == 1,
97 "nanobind::detail::nb_func_dealloc(\"%s\"): function not found!",

100#endif

103 if (f->flags & (uint32_t) func_flags::has_free)
104 f->free_capture(f->capture);

107 for (size_t j = 0; j < f->nargs; ++j) {
108 const arg_data &arg = f->args[j];
109 Py_XDECREF(arg.value);
110 Py_XDECREF(arg.name_py);
111 free((char *) arg.signature);
112 }
113 }
114
115 if (f->flags & (uint32_t) func_flags::has_doc)
116 free((char *) f->doc);
117
118 free((char *) f->name);
119 free(f->args);
120 free((char *) f->descr);
121 free(f->descr_types);
122 free(f->signature);
123 ++f;
124 }
125 }
126
127 Py_XDECREF(((nb_func *) self)->module_name);
128
129 PyTypeObject *tp = Py_TYPE(self);
130 PyObject_GC_Del(self);
131 NB_DECREF_TYPE(tp);
132
133 internals_dec_ref();
134}

216PyObject *nb_func_new(const func_data_prelim_base *f) noexcept {
217 bool has_scope = f->flags & (uint32_t) func_flags::has_scope,
218 has_name = f->flags & (uint32_t) func_flags::has_name,
219 has_args = f->flags & (uint32_t) func_flags::has_args,
220 has_var_args = f->flags & (uint32_t) func_flags::has_var_args,
221 has_var_kwargs = f->flags & (uint32_t) func_flags::has_var_kwargs,
222 can_mutate_args = f->flags & (uint32_t) func_flags::can_mutate_args,
223 has_doc = f->flags & (uint32_t) func_flags::has_doc,
224 has_signature = f->flags & (uint32_t) func_flags::has_signature,
225 is_implicit = f->flags & (uint32_t) func_flags::is_implicit,
226 is_method = f->flags & (uint32_t) func_flags::is_method,
227 return_ref = f->flags & (uint32_t) func_flags::return_ref,
228 is_constructor = false,
229 is_init = false,
230 is_new = false,
231 is_setstate = false;
232
233 arg_data *args_in = nullptr;
234 if (has_args)
235 args_in = std::launder((arg_data*) ((func_data_prelim<1>*) f)->args);
236
237 PyObject *name = nullptr;
238 PyObject *func_prev = nullptr;
239
240 char *name_cstr;
241 if (has_signature) {
242 name_cstr = extract_name("nanobind::detail::nb_func_new", "def ", f->name);
243 has_name = *name_cstr != '\0';
244 } else {
245 name_cstr = strdup_check(has_name ? f->name : "");
246 }
247
248 // Check for previous overloads
249 nb_internals *internals_ = internals;
250 if (has_scope && has_name) {
251 name = PyUnicode_InternFromString(name_cstr);
252 check(name, "nb::detail::nb_func_new(\"%s\"): invalid name.", name_cstr);
253
254 func_prev = PyObject_GetAttr(f->scope, name);
255 if (func_prev) {
256 PyTypeObject *func_prev_tp = Py_TYPE(func_prev);
257 if (func_prev_tp == internals_->nb_func ||
258 func_prev_tp == internals_->nb_method) {
259 func_data *fp = nb_func_data(func_prev);
260
261 check((fp->flags & (uint32_t) func_flags::is_method) ==
262 (f->flags & (uint32_t) func_flags::is_method),
263 "nb::detail::nb_func_new(\"%s\"): mismatched static/"
264 "instance method flags in function overloads!",
265 name_cstr);
266
267 /* Never append a method to an overload chain of a parent class;
268 instead, hide the parent's overloads in this case */
269 if (fp->scope != f->scope)
270 NB_CLEAR_FUNC(func_prev);
271 } else if (name_cstr[0] == '_') {
272 NB_CLEAR_FUNC(func_prev);
273 } else {
274 check(false,
275 "nb::detail::nb_func_new(\"%s\"): cannot overload "
276 "existing non-function object of the same name!", name_cstr);
277 }
278 } else {
279 PyErr_Clear();
280 }
281
282 is_init = strcmp(name_cstr, "__init__") == 0;
283 is_new = strcmp(name_cstr, "__new__") == 0;
284 is_setstate = strcmp(name_cstr, "__setstate__") == 0;
285
286 // Is this method a constructor that takes a class binding as first parameter?
287 is_constructor = is_method && (is_init || is_setstate) &&
288 strncmp(f->descr, "({%}", 4) == 0;
289
290 // Don't use implicit conversions in copy constructors (causes infinite recursion)
291 // Notes:
292 // f->nargs = C++ argument count.
293 // f->descr_types = zero-terminated array of bound types among them.
294 // Hence of size >= 2 for constructors, where f->descr_types[1] my be null.
295 // args_in = array of Python arguments (nb::arg). Non-empty if has_args.
296 // By contrast, fc->args below has size f->nargs.
297 if (is_constructor && f->nargs == 2 && f->descr_types[0] &&
298 f->descr_types[0] == f->descr_types[1]) {
299 if (has_args) {
300 args_in[0].flag &= ~(uint8_t) cast_flags::convert;
301 } else {
302 args_in = method_args + 1;
303 has_args = true;
304 }
305 }
306 }
307
308 // Create a new function and destroy the old one
309 Py_ssize_t prev_overloads = func_prev ? Py_SIZE(func_prev) : 0;
310 nb_func *func = (nb_func *) PyType_GenericAlloc(
311 is_method ? internals_->nb_method : internals_->nb_func, prev_overloads + 1);
312 check(func, "nb::detail::nb_func_new(\"%s\"): alloc. failed (1).",
313 name_cstr);
314
315 make_immortal((PyObject *) func);
316 internals_inc_ref();
317
318 // Determine which dispatcher this overload needs
319 call_complexity complexity = call_complexity::simple;
320
321 if (has_var_kwargs || has_var_args || f->nargs > NB_MAXARGS_SIMPLE) {
322 complexity = call_complexity::complex;
323 } else {
324 bool medium_call = can_mutate_args;
325
326 if (has_args) {
327 for (size_t i = is_method; i < f->nargs; ++i) {
328 arg_data &a = args_in[i - is_method];
329 uint8_t dispatch_flags =
330 a.flag & ~(uint8_t) cast_flags::none_disallowed;
331 medium_call |= a.name != nullptr || a.value != nullptr ||
332 dispatch_flags != cast_flags::convert;
333 }
334 }
335
336 if (medium_call)
337 complexity = call_complexity::medium;
338 }
339
340 uint32_t max_nargs = f->nargs;
341
342 const char *prev_doc = nullptr;
343
344 if (func_prev) {
345 nb_func *nb_func_prev = (nb_func *) func_prev;
346 complexity = std::max(complexity, nb_func_prev->complexity);
347 max_nargs = std::max(max_nargs, nb_func_prev->max_nargs);
348
349 func_data *cur = nb_func_data(func),
350 *prev = nb_func_data(func_prev);
351
352 if (nb_func_prev->doc_uniform)
353 prev_doc = prev->doc;
354
355 memcpy(cur, prev, sizeof(func_data) * (size_t) prev_overloads);
356 memset(prev, 0, sizeof(func_data) * (size_t) prev_overloads);
357
358 ((PyVarObject *) func_prev)->ob_size = 0;
359
360#if !defined(NB_FREE_THREADED)
361 size_t n_deleted = internals_->funcs.erase(func_prev);
362 check(n_deleted == 1,
363 "nanobind::detail::nb_func_new(): internal update failed (1)!");
364#endif
365
366 NB_CLEAR_FUNC(func_prev);
367 }
368
369 func->max_nargs = max_nargs;
370 func->complexity = complexity;
371
372 // Snapshot the current '__module__'
373 if (has_scope) {
374 func->module_name =
375 PyObject_GetAttr(f->scope, PyModule_Check(f->scope)
376 ? NB_INTERNED(__name__)
377 : NB_INTERNED(__module__));
378 if (!func->module_name)
379 PyErr_Clear();
380 }
381
382 PyObject* (*vectorcall)(PyObject *, PyObject * const*, size_t, PyObject *);
383 if (complexity == call_complexity::complex) {
384 vectorcall = nb_func_vectorcall_complex;
385 } else if (complexity == call_complexity::medium) {
386 vectorcall = nb_func_vectorcall_medium;
387 } else {
388 if (f->nargs == 0 && !prev_overloads)
389 vectorcall = nb_func_vectorcall_simple_0;
390 else if (f->nargs == 1 && !prev_overloads)
391 vectorcall = nb_func_vectorcall_simple_1;
392 else if (f->nargs == 2 && !prev_overloads)
393 vectorcall = nb_func_vectorcall_simple_2;
394 else
395 vectorcall = nb_func_vectorcall_simple;
396 }
397 func->vectorcall = vectorcall;
398
399#if !defined(NB_FREE_THREADED)
400 // Register the function
401 auto [it, success] = internals_->funcs.try_emplace(func, nullptr);
402 check(success,
403 "nanobind::detail::nb_func_new(): internal update failed (2)!");
404#endif
405
406 func_data *fc = nb_func_data(func) + prev_overloads;
407 memcpy(fc, f, sizeof(func_data_prelim_base));
408 if (has_doc) {
409 if (fc->doc[0] == '\n')
410 fc->doc++;
411 if (fc->doc[0] == '\0') {
412 fc->doc = nullptr;
413 fc->flags &= ~(uint32_t) func_flags::has_doc;
414 has_doc = false;
415 } else {
416 fc->doc = strdup_check(fc->doc);
417 }
418 }
419
420 // Detect when an entire overload chain has the same docstring
421 func->doc_uniform =
422 (has_doc && ((prev_overloads == 0) ||
423 (prev_doc && strcmp(fc->doc, prev_doc) == 0)));
424
425 if (is_constructor)
426 fc->flags |= (uint32_t) func_flags::is_constructor;
427 if (has_args)
428 fc->flags |= (uint32_t) func_flags::has_args;
429
430 fc->name = name_cstr;
431 fc->signature = has_signature ? strdup_check(f->name) : nullptr;
432
433 if (is_implicit) {
434 check(fc->flags & (uint32_t) func_flags::is_constructor,
435 "nb::detail::nb_func_new(\"%s\"): nanobind::is_implicit() "
436 "should only be specified for constructors.",
437 name_cstr);
438 check(f->nargs == 2,
439 "nb::detail::nb_func_new(\"%s\"): implicit constructors "
440 "should only have one argument.",
441 name_cstr);
442
443 if (f->descr_types[1])
444 implicitly_convertible(f->descr_types[1], f->descr_types[0]);
445 }
446
447 for (size_t i = 0;; ++i) {
448 if (!f->descr[i]) {
449 fc->descr = (char *) malloc_check(sizeof(char) * (i + 1));
450 memcpy((char *) fc->descr, f->descr, (i + 1) * sizeof(char));
451 break;
452 }
453 }
454
455 for (size_t i = 0;; ++i) {
456 if (!f->descr_types[i]) {
457 fc->descr_types = (const std::type_info **)
458 malloc_check(sizeof(const std::type_info *) * (i + 1));
459 memcpy(fc->descr_types, f->descr_types,
460 (i + 1) * sizeof(const std::type_info *));
461 break;
462 }
463 }
464
465 if (has_args) {
466 fc->args = (arg_data *) malloc_check(sizeof(arg_data) * f->nargs);
467
468 if (is_method) // add implicit 'self' argument annotation
469 fc->args[0] = method_args[0];
470 for (size_t i = is_method; i < fc->nargs; ++i)
471 fc->args[i] = args_in[i - is_method];
472
473 for (size_t i = 0; i < fc->nargs; ++i) {
474 arg_data &a = fc->args[i];
475 if (a.name) {
476 a.name_py = PyUnicode_InternFromString(a.name);
477 a.name = PyUnicode_AsUTF8AndSize(a.name_py, nullptr);
478 } else {
479 a.name_py = nullptr;
480 }
481 if (a.value == Py_None)
482 a.flag |= (uint8_t) cast_flags::accepts_none;
483 a.signature = a.signature ? strdup_check(a.signature) : nullptr;
484 Py_XINCREF(a.value);
485 }
486 }
487
488 // Fast path for vector call object construction
489 if (((is_init && is_method) || (is_new && !is_method)) &&
490 nb_type_check(f->scope)) {
491 type_data *td = nb_type_data((PyTypeObject *) f->scope);
492 bool has_new = td->flags & (uint32_t) type_flags::has_new;
493
494 if (is_init) {
495 if (!has_new) {
496 td->init = func;
497 } else {
498 // Keep track of whether we have a __init__ overload that
499 // accepts no arguments (except self). If not, then we
500 // shouldn't allow calling the type object with no arguments,
501 // even though (for unpickling support) we probably do have
502 // a __new__ overload that accepts no arguments (except cls).
503 // This check is necessary because our type vectorcall shortcut
504 // skips Python's usual logic where __init__ is always called
505 // if __new__ returns an instance of the type.
506 bool noargs_ok = true;
507 for (uint32_t i = 1; i < fc->nargs - (uint32_t) has_var_kwargs; ++i) {
508 if (has_var_args && i == fc->nargs_pos)
509 continue; // skip `nb::args` since it can be empty
510 if (has_args && fc->args[i].value != nullptr)
511 continue; // arg with default is OK
512 noargs_ok = false;
513 break;
514 }
515 if (noargs_ok)
516 td->flags |= (uint32_t) type_flags::has_nullary_new;
517 }
518 } else if (is_new) {
519 td->init = func;
520 td->flags |= (uint32_t) type_flags::has_new;
521 }
522 }
523
524 if (has_scope && name) {
525 int rv = PyObject_SetAttr(f->scope, name, (PyObject *) func);
526 check(rv == 0, "nb::detail::nb_func_new(\"%s\"): setattr. failed.",
527 name_cstr);
528 }
529
530 Py_XDECREF(name);
531
532 if (return_ref) {
533 return (PyObject *) func;
534 } else {
535 NB_DECREF_FUNC(func);
536 return nullptr;
537 }

541static NB_NOINLINE PyObject *
542nb_func_error_overload(PyObject *self, PyObject *const *args_in,
543 size_t nargs_in, PyObject *kwargs_in) noexcept {
544 uint32_t count = (uint32_t) Py_SIZE(self);
545 func_data *f = nb_func_data(self);
546
547 if (f->flags & (uint32_t) func_flags::is_operator)
548 return not_implemented().release().ptr();
549
550 // The buffer 'buf' is protected by 'internals.mutex'
551 lock_internals guard(internals);
552
553 buf.clear();
554 buf.put_dstr(f->name);
555 buf.put("(): incompatible function arguments. The following argument types "
556 "are supported:\n");
557
558 // Mask default __new__ overload created by nb::new_()
559 if (strcmp(f->name, "__new__") == 0 && count > 1 && f->nargs == 1) {
560 count -= 1;
561 f += 1;
562 }
563
564 for (uint32_t i = 0; i < count; ++i) {
565 buf.put(" ");
566 buf.put_uint32(i + 1);
567 buf.put(". ");
568 nb_func_render_signature(f + i);
569 buf.put('\n');
570 }
571
572 buf.put("\nInvoked with types: ");
573 for (size_t i = 0; i < nargs_in; ++i) {
574 str name = steal<str>(nb_inst_name(args_in[i]));
575 buf.put_dstr(name.c_str());
576 if (i + 1 < nargs_in)
577 buf.put(", ");
578 }
579
580 if (kwargs_in) {
581 if (nargs_in)
582 buf.put(", ");
583 buf.put("kwargs = { ");
584
585 Py_ssize_t nkwargs_in = NB_TUPLE_GET_SIZE(kwargs_in);
586 for (Py_ssize_t j = 0; j < nkwargs_in; ++j) {
587 PyObject *key = NB_TUPLE_GET_ITEM(kwargs_in, j),
588 *value = args_in[nargs_in + (size_t) j];
589
590 const char *key_cstr = PyUnicode_AsUTF8AndSize(key, nullptr);
591 if (!key_cstr) {
592 PyErr_Clear();
593 key_cstr = "?";
594 }
595 buf.put_dstr(key_cstr);
596 buf.put(": ");
597 str name = steal<str>(nb_inst_name(value));
598 buf.put_dstr(name.c_str());
599 buf.put(", ");
600 }
601 buf.rewind(2);
602 buf.put(" }");
603 }
604
605 PyErr_SetString(PyExc_TypeError, buf.get());
606 return nullptr;
607}

653 const size_t count = (size_t) Py_SIZE(self),
654 nargs_in = (size_t) NB_VECTORCALL_NARGS(nargsf),
655 nkwargs_in = kwargs_in ? (size_t) NB_TUPLE_GET_SIZE(kwargs_in) : 0;
656
657 func_data *fr = nb_func_data(self);
658
659 const bool is_method = fr->flags & (uint32_t) func_flags::is_method,
660 is_constructor = fr->flags & (uint32_t) func_flags::is_constructor;
661
662 PyObject *result = nullptr,
663 *self_arg = (is_method && nargs_in > 0) ? args_in[0] : nullptr;
664
665 /* The following lines allocate memory on the stack, which is very efficient
666 but also potentially dangerous since it can be used to generate stack
667 overflows. We refuse unrealistically large number of 'kwargs' (the
668 'max_nargs' value is fine since it is specified by the bindings) */
669 if (nkwargs_in > 1024) {
670 PyErr_SetString(PyExc_TypeError,
671 "nanobind::detail::nb_func_vectorcall(): too many (> "
672 "1024) keyword arguments.");
673 return nullptr;
674 }
675
676 // Handler routine that will be invoked in case of an error condition
677 PyObject *(*error_handler)(PyObject *, PyObject *const *, size_t,
678 PyObject *) noexcept = nullptr;
679
680 // Small array holding temporaries (implicit conversion/*args/**kwargs)
681 cleanup_list cleanup(self_arg);
682
683 // Preallocate stack memory for function dispatch
684 size_t max_nargs = ((nb_func *) self)->max_nargs;
685 PyObject **args = (PyObject **) alloca(max_nargs * sizeof(PyObject *));
686 uint8_t *args_flags = (uint8_t *) alloca(max_nargs * sizeof(uint8_t));
687 bool *kwarg_used = (bool *) alloca(nkwargs_in * sizeof(bool));
688
689 // Ensure that keyword argument names are interned. That makes it faster
690 // to compare them against pre-interned argument names in the overload chain.
691 // Normal function calls will have their keyword arguments already interned,
692 // but we can't rely on that; it fails for things like fn(**json.loads(...)).
693 PyObject **kwnames = nullptr;
694
695#if !defined(PYPY_VERSION) && !defined(Py_LIMITED_API)
696 bool kwnames_interned = true;
697 for (size_t i = 0; i < nkwargs_in; ++i) {
698 PyObject *key = NB_TUPLE_GET_ITEM(kwargs_in, (Py_ssize_t) i);
699 kwnames_interned &= ((PyASCIIObject *) key)->state.interned != 0;
700 }
701 if (kwargs_in && NB_LIKELY(kwnames_interned)) {
702 kwnames = ((PyTupleObject *) kwargs_in)->ob_item;
703 goto traverse_overloads;
704 }
705#endif
706
707 kwnames = (PyObject **) alloca(nkwargs_in * sizeof(PyObject *));
708 for (size_t i = 0; i < nkwargs_in; ++i) {
709 PyObject *key = NB_TUPLE_GET_ITEM(kwargs_in, (Py_ssize_t) i);
710 Py_INCREF(key);
711
712 kwnames[i] = key;
713 PyUnicode_InternInPlace(&kwnames[i]);
714 PyObject *key_interned = kwnames[i];
715
716 if (NB_LIKELY(key == key_interned)) // string was already interned
717 Py_DECREF(key);
718 else
719 cleanup.append(key_interned);
720 }
721
722#if !defined(PYPY_VERSION) && !defined(Py_LIMITED_API)
723 traverse_overloads:
724#endif
725
726 /* The logic below tries to find a suitable overload using two passes
727 of the overload chain (or 1, if there are no overloads). The first pass
728 is strict and permits no implicit conversions, while the second pass
729 allows them.
730
731 The following is done per overload during a pass
732
733 1. Copy individual arguments while checking that named positional
734 arguments weren't *also* specified as kwarg. Substitute missing
735 entries using keyword arguments or default argument values provided
736 in the bindings, if available.
737
738 2. Ensure that either all keyword arguments were "consumed", or that
739 the function takes a kwargs argument to accept unconsumed kwargs.
740
741 3. Any positional arguments still left get put into a tuple (for args),
742 and any leftover kwargs get put into a dict.
743
744 4. Pack everything into a vector; if we have nb::args or nb::kwargs,
745 they become a tuple or dict at the end of the positional arguments.
746
747 5. Call the function call dispatcher (func_data::impl)
748
749 If one of these fail, move on to the next overload and keep trying
750 until we get a result other than NB_NEXT_OVERLOAD.
751 */
752
753 for (size_t pass = (count > 1) ? 0 : 1; pass < 2; ++pass) {
754 for (size_t k = 0; k < count; ++k) {
755 const func_data *f = fr + k;
756
757 const bool has_args = f->flags & (uint32_t) func_flags::has_args,
758 has_var_args = f->flags & (uint32_t) func_flags::has_var_args,
759 has_var_kwargs = f->flags & (uint32_t) func_flags::has_var_kwargs;
760
761 // Number of C++ parameters eligible to be filled from individual
762 // Python positional arguments
763 size_t nargs_pos = f->nargs_pos;
764
765 // Number of C++ parameters in total, except for a possible trailing
766 // nb::kwargs. All of these are eligible to be filled from individual
767 // Python arguments (keyword always, positional until index nargs_pos)
768 // except for a potential nb::args, which exists at index nargs_pos
769 // if has_var_args is true. We'll skip that one in the individual-args
770 // loop, and go back and fill it later with the unused positionals.
771 size_t nargs_step1 = f->nargs - has_var_kwargs;
772
773 if (nargs_in > nargs_pos && !has_var_args)
774 continue; // Too many positional arguments given for this overload
775
776 if (nargs_in < nargs_pos && !has_args)
777 continue; // Not enough positional arguments, insufficient
778 // keyword/default arguments to fill in the blanks
779
780 memset(kwarg_used, 0, nkwargs_in * sizeof(bool));
781
782 // 1. Copy individual arguments, potentially substitute kwargs/defaults
783 size_t i = 0;
784 for (; i < nargs_step1; ++i) {
785 if (has_var_args && i == nargs_pos)
786 continue; // skip nb::args parameter, will be handled below
787
788 PyObject *arg = nullptr;
789
790 uint8_t arg_flag = 1;
791
792 // If i >= nargs_pos, then this is a keyword-only parameter.
793 // (We skipped any *args parameter using the test above,
794 // and we set the bounds of nargs_step1 to not include any
795 // **kwargs parameter.) In that case we don't want to take
796 // a positional arg (which might validly exist and be
797 // destined for the *args) but we do still want to look for
798 // a matching keyword arg.
799 if (i < nargs_in && i < nargs_pos)
800 arg = args_in[i];
801
802 if (has_args) {
803 const arg_data &ad = f->args[i];
804
805 if (kwargs_in && ad.name_py) {
806 PyObject *hit = nullptr;
807 for (size_t j = 0; j < nkwargs_in; ++j) {
808 if (kwnames[j] == ad.name_py) {
809 hit = args_in[nargs_in + j];
810 kwarg_used[j] = true;
811 break;
812 }
813 }
814
815 if (hit) {
816 if (arg)
817 break; // conflict between keyword and positional arg.
818 arg = hit;
819 }
820 }
821
822 if (!arg)
823 arg = ad.value;
824 arg_flag = ad.flag;
825 }
826
827 if (!arg || (arg == Py_None && (arg_flag & cast_flags::accepts_none) == 0))
828 break;
829
830 // Implicit conversion only active in the 2nd pass
831 // Have to cast to uint8_t because of integer promotion (uint8_t promoted to int before ~ and & operations)
832 args_flags[i] = (uint8_t) (arg_flag & ~uint8_t(pass == 0));
833 args[i] = arg;
834 }
835
836 // Skip this overload if any arguments were unavailable
837 if (i != nargs_step1)
838 continue;
839
840 // Deal with remaining positional arguments
841 if (has_var_args) {
842 PyObject *tuple = PyTuple_New(
843 nargs_in > nargs_pos ? (Py_ssize_t) (nargs_in - nargs_pos) : 0);
844
845 for (size_t j = nargs_pos; j < nargs_in; ++j) {
846 PyObject *o = args_in[j];
847 Py_INCREF(o);
848 NB_TUPLE_SET_ITEM(tuple, (Py_ssize_t) (j - nargs_pos), o);
849 }
850
851 args[nargs_pos] = tuple;
852 args_flags[nargs_pos] = 0;
853 cleanup.append(tuple);
854 }
855
856 // Deal with remaining keyword arguments
857 if (has_var_kwargs) {
858 PyObject *dict = PyDict_New();
859 for (size_t j = 0; j < nkwargs_in; ++j) {
860 PyObject *key = kwnames[j];
861 if (!kwarg_used[j])
862 PyDict_SetItem(dict, key, args_in[nargs_in + j]);
863 }
864
865 args[nargs_step1] = dict;
866 args_flags[nargs_step1] = 0;
867 cleanup.append(dict);
868 } else if (kwargs_in) {
869 bool success = true;
870 for (size_t j = 0; j < nkwargs_in; ++j)
871 success &= kwarg_used[j];
872 if (!success)
873 continue;
874 }
875
876
877 // A constructor's 'self' may also arrive as a keyword argument,
878 // so it must be read back from args[0] rather than from args_in[0]
879 PyObject *self_arg_constructor = nullptr;
880 if (is_constructor) {
881 args_flags[0] |= (uint8_t) cast_flags::construct;
882 self_arg_constructor = args[0];
883 }
884
885 rv_policy policy = (rv_policy) (f->flags & 0b111);
886
887 try {
888 result = nullptr;
889
890 // Found a suitable overload, let's try calling it
891 result = f->impl((void *) f->capture, args, args_flags,
892 policy, &cleanup);
893
894 if (NB_UNLIKELY(!result))
895 error_handler = nb_func_error_noconvert;
896 } catch (builtin_exception &e) {
897 if (!set_builtin_exception_status(e))
898 result = NB_NEXT_OVERLOAD;
899 } catch (python_error &e) {
900 e.restore();
901 } catch (...) {
902 nb_func_convert_cpp_exception();
903 }
904
905 if (result != NB_NEXT_OVERLOAD) {
906 if (is_constructor && result != nullptr) {
907 nb_inst *self_arg_nb = (nb_inst *) self_arg_constructor;
908 self_arg_nb->state.destruct = true;
909 self_arg_nb->state.state = nb_inst_state::state_ready;
910 if (NB_UNLIKELY(self_arg_nb->state.intrusive))
911 nb_type_data(Py_TYPE(self_arg_constructor))
912 ->set_self_py(inst_ptr(self_arg_nb), self_arg_constructor);
913 }
914
915 goto done;
916 }
917 }
918 }
919
920 error_handler = nb_func_error_overload;
921
922done:
923 if (NB_UNLIKELY(cleanup.used()))
924 cleanup.release();
925
926 if (NB_UNLIKELY(error_handler))
927 result = error_handler(self, args_in, nargs_in, kwargs_in);
928
929 return result;
930}

1384static uint32_t nb_func_render_signature(const func_data *f,
1385 bool nb_signature_mode) noexcept {
1386 const bool is_method = f->flags & (uint32_t) func_flags::is_method,
1387 has_args = f->flags & (uint32_t) func_flags::has_args,
1388 has_var_args = f->flags & (uint32_t) func_flags::has_var_args,
1389 has_var_kwargs = f->flags & (uint32_t) func_flags::has_var_kwargs,
1390 has_signature = f->flags & (uint32_t) func_flags::has_signature;
1391
1392 nb_internals *internals_ = internals;
1393 if (has_signature) {
1394 const char *s = f->signature;
1395
1396 if (!nb_signature_mode) {
1397 // go to last line of manually provided signature, strip away 'def ' prefix
1398 const char *p = strrchr(s, '\n');
1399 s = p ? (p + 1) : s;
1400 if (strncmp(s, "def ", 4) == 0)
1401 s += 4;
1402 }
1403
1404 buf.put_dstr(s);
1405 return 0;
1406 }
1407
1408 if (nb_signature_mode)
1409 buf.put("def ");
1410
1411 const std::type_info **descr_type = f->descr_types;
1412 bool rv = false;
1413
1414 uint32_t arg_index = 0, n_default_args = 0;
1415 buf.put_dstr(f->name);
1416
1417 for (const char *pc = f->descr; *pc != '\0'; ++pc) {
1418 char c = *pc;
1419
1420 switch (c) {
1421 case '@':
1422 // Handle types that differ depending on whether they appear
1423 // in an argument or a return value position
1424 pc++;
1425 if (!rv) {
1426 while (*pc && *pc != '@')
1427 buf.put(*pc++);
1428 if (*pc == '@')
1429 pc++;
1430 while (*pc && *pc != '@')
1431 pc++;
1432 } else {
1433 while (*pc && *pc != '@')
1434 pc++;
1435 if (*pc == '@')
1436 pc++;
1437 while (*pc && *pc != '@')
1438 buf.put(*pc++);
1439 }
1440 break;
1441
1442 case '{':
1443 {
1444 const char *arg_name = has_args ? f->args[arg_index].name : nullptr;
1445
1446 // Argument name
1447 if (has_var_kwargs && arg_index + 1 == f->nargs) {
1448 buf.put("**");
1449 buf.put_dstr(arg_name ? arg_name : "kwargs");
1450 pc += 4; // strlen("dict")
1451 break;
1452 }
1453
1454 if (arg_index == f->nargs_pos) {
1455 buf.put("*");
1456 if (has_var_args) {
1457 buf.put_dstr(arg_name ? arg_name : "args");
1458 pc += 5; // strlen("tuple")
1459 break;
1460 } else {
1461 buf.put(", ");
1462 // fall through to render the first keyword-only arg
1463 }
1464 }
1465
1466 if (is_method && arg_index == 0) {
1467 buf.put("self");
1468
1469 // Skip over type
1470 while (*pc != '}') {
1471 if (*pc == '%')
1472 descr_type++;
1473 pc++;
1474 }
1475 arg_index++;
1476 continue;
1477 } else if (arg_name) {
1478 buf.put_dstr(arg_name);
1479 } else {
1480 buf.put("arg");
1481 if (f->nargs > 1 + (uint32_t) is_method)
1482 buf.put_uint32(arg_index - is_method);
1483 }
1484
1485 buf.put(": ");
1486 if (has_args && f->args[arg_index].flag &
1487 (uint8_t) cast_flags::accepts_none) {
1488#if PY_VERSION_HEX < 0x030A0000
1489 buf.put("typing.Optional[");
1490 #else
1491 // See below
1492 #endif
1493 }
1494 }
1495 break;
1496
1497 case '}':
1498 // Default argument
1499 if (has_args) {
1500 if (f->args[arg_index].flag & (uint8_t) cast_flags::accepts_none) {
1501 #if PY_VERSION_HEX < 0x030A0000
1502 buf.put(']');
1503 #else
1504 buf.put(" | None");
1505 #endif
1506 }
1507
1508 if (f->args[arg_index].value) {
1509 const arg_data &arg = f->args[arg_index];
1510 if (nb_signature_mode) {
1511 buf.put(" = \\");
1512 if (arg.signature)
1513 buf.put('=');
1514 buf.put_uint32(n_default_args++);
1515 } else if (arg.signature) {
1516 buf.put(" = ");
1517 buf.put_dstr(arg.signature);
1518 } else {
1519 PyObject *o = arg.value, *str;
1520
1521 {
1522 unlock_internals guard2(internals_);
1523 str = PyObject_Repr(o);
1524 }
1525
1526 if (str) {
1527 Py_ssize_t size = 0;
1528 const char *cstr =
1529 PyUnicode_AsUTF8AndSize(str, &size);
1530 if (!cstr) {
1531 PyErr_Clear();
1532 } else {
1533 buf.put(" = ");
1534 buf.put(cstr, (size_t) size);
1535 }
1536 Py_DECREF(str);
1537 } else {
1538 PyErr_Clear();
1539 }
1540 }
1541 }
1542 }
1543
1544 arg_index++;
1545
1546 if (arg_index == f->nargs_pos && !has_args)
1547 buf.put(", /");
1548
1549 break;
1550
1551 case '%':
1552 check(*descr_type,
1553 "nb::detail::nb_func_render_signature(): missing type!");
1554
1555 if (!(is_method && arg_index == 0)) {
1556 bool found = false;
1557 auto it = internals_->type_c2p_slow.find(*descr_type);
1558
1559 if (it != internals_->type_c2p_slow.end()) {
1560 handle th((PyObject *) it->second->type_py);
1561 buf.put_dstr((borrow<str>(
1562 th.attr(NB_INTERNED(__module__)))).c_str());
1563 buf.put('.');
1564 buf.put_dstr((borrow<str>(
1565 th.attr(NB_INTERNED(__qualname__)))).c_str());
1566 found = true;
1567 }
1568 if (!found) {
1569 if (nb_signature_mode)
1570 buf.put('"');
1571 char *name = type_name(*descr_type);
1572 buf.put_dstr(name);
1573 free(name);
1574 if (nb_signature_mode)
1575 buf.put('"');
1576 }
1577 }
1578
1579 descr_type++;
1580 break;
1581
1582 case '-':
1583 if (pc[1] == '>')
1584 rv = true;
1585 buf.put(c);
1586 break;
1587
1588
1589 default:
1590 buf.put(c);
1591 break;
1592 }
1593 }
1594
1595 check(arg_index == f->nargs && !*descr_type,
1596 "nanobind::detail::nb_func_render_signature(%s): arguments inconsistent.",
1597 f->name);
1598
1599 return n_default_args;
1600}

1638 PyObject *docstr = nullptr, *item = nullptr, *sigstr = nullptr,
1639 *defaults = nullptr;
1640
1641 func_data *f = nb_func_data(self);
1642 uint32_t count = (uint32_t) Py_SIZE(self);
1643 PyObject *result = PyTuple_New(count);
1644 if (!result)

1652 if (fi->flags & (uint32_t) func_flags::has_doc) {

1654 } else {

1664
1665 item = PyTuple_New(3);

1667 if (n_default_args) {

1669 } else {
1670 defaults = Py_None;
1671 Py_INCREF(defaults);
1672 }
1673
1674 if (!docstr || !sigstr || !item || !defaults)
1675 goto fail;
1676
1677 if (n_default_args) {
1678 size_t pos = 0;
1679 for (uint32_t j = 0; j < fi->nargs; ++j) {
1680 const arg_data &arg = fi->args[j];

1685 value = PyUnicode_FromString(arg.signature);
1686 if (!value)
1687 goto fail;
1688 } else {
1689 Py_INCREF(value);
1690 }
1691 NB_TUPLE_SET_ITEM(defaults, (Py_ssize_t) pos, value);
1692 pos++;
1693 }
1694
1695 check(pos == n_default_args,
1696 "__nb_signature__: default argument counting inconsistency!");
1697 }
1698
1699 NB_TUPLE_SET_ITEM(item, 0, sigstr);
1700 NB_TUPLE_SET_ITEM(item, 1, docstr);
1701 NB_TUPLE_SET_ITEM(item, 2, defaults);
1702 NB_TUPLE_SET_ITEM(result, (Py_ssize_t) i, item);
1703 }
1704
1705 return result;
1706
1707fail:
1708 Py_XDECREF(docstr);
1709 Py_XDECREF(sigstr);
1710 Py_XDECREF(defaults);
1711 Py_XDECREF(item);
1712 Py_DECREF(result);

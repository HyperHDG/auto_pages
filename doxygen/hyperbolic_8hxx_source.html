160 /*!***********************************************************************************************
161 * \brief Evaluate condensed matrix-vector product.
162 *
163 * Function that evaluates the condensed, matrix-free version of the matrix-vector product
164 * \f$A x = y\f$, where \f$A\f$ is the condensed matrix of the LDG-H method that needs to be
165 * inverted for a time step, \f$x\f$ is the vector of parameters to define the skeletal variable
166 * \f$\lambda\f$, and \f$y\f$ is the resulting vector, which has the same size as the input vector
167 * \f$x\f$.
168 *
169 * \param x_vec A vector containing the input vector \f$x\f$.
170 * \param time Time at which the new time step will end.
171 * \retval y_vec A vector containing the product \f$y = Ax\f$.
172 ************************************************************************************************/
173 template <typename hyNode_index_t = dof_index_t, typename time_t = double>
174 LargeVecT trace_to_flux(const LargeVecT& x_vec, const time_t time = 0.)
175 {
176 constexpr unsigned int hyEdge_dim = TopologyT::hyEdge_dim();
177 constexpr unsigned int n_dofs_per_node = LocalSolverT::n_glob_dofs_per_node();
178
179 LargeVecT vec_Ax(x_vec.size(), 0.);
180 SmallVec<2 * hyEdge_dim, hyNode_index_t> hyEdge_hyNodes;
181 std::array<std::array<dof_value_t, n_dofs_per_node>, 2 * hyEdge_dim> hyEdge_dofs_old,
182 hyEdge_dofs_new;
183
184 // Do matrix--vector multiplication by iterating over all hyperedges.
185 std::for_each(
186 hyper_graph_.begin(), hyper_graph_.end(),
187 [&](auto hyper_edge)
188 {
189 // Fill x_vec's degrees of freedom of a hyperedge into hyEdge_dofs array.
190 hyEdge_hyNodes = hyper_edge.topology.get_hyNode_indices();
191 for (unsigned int hyNode = 0; hyNode < hyEdge_hyNodes.size(); ++hyNode)
192 {
193 hyper_graph_.hyNode_factory().get_dof_values(hyEdge_hyNodes[hyNode], x_vec,
194 hyEdge_dofs_old[hyNode]);
195 hyEdge_dofs_new[hyNode].fill(0.);
196 }
197
198 // Turn degrees of freedom of x_vec that have been stored locally into those of vec_Ax.
199 if constexpr (requires {
200 local_solver_.trace_to_flux(hyEdge_dofs_old, hyEdge_dofs_new,
201 prvalue_of(time));
202 })
203 local_solver_.trace_to_flux(hyEdge_dofs_old, hyEdge_dofs_new, time);
204 else if constexpr (requires {
205 local_solver_.trace_to_flux(hyEdge_dofs_old, hyEdge_dofs_new,
206 hyper_edge, prvalue_of(time));
207 })
208 local_solver_.trace_to_flux(hyEdge_dofs_old, hyEdge_dofs_new, hyper_edge, time);
209 else
210 static_assert(always_false_v<LocalSolverT, decltype(hyper_edge)>,
211 "LocalSolverT implements no usable overload of trace_to_flux!");
212
213 // Fill hyEdge_dofs array degrees of freedom into vec_Ax.
214 for (unsigned int hyNode = 0; hyNode < hyEdge_hyNodes.size(); ++hyNode)
215 hyper_graph_.hyNode_factory().add_to_dof_values(hyEdge_hyNodes[hyNode], vec_Ax,
216 hyEdge_dofs_new[hyNode]);
217 });
218
219 return vec_Ax;
220 }

378

498 /*!***********************************************************************************************
499 * \brief Fill caller-provided span with per-edge energy components.
500 *
501 * Does not allocate. The span must hold exactly \c n_edges() * \c n_energy_components()
502 * entries; the local solver's array for edge \c e is written into
503 * \c out[e * n_energy_components() .. (e+1) * n_energy_components()).
504 ************************************************************************************************/
505 template <typename SpanT, typename LambdaSpanT>
506 void energy(const LambdaSpanT& x_vec, SpanT out, const double time = 0.)
507 {
508 constexpr unsigned int n_comp = n_energy_components();
509 hy_check(out.size() == static_cast<std::size_t>(n_edges()) * n_comp,
510 "energy span size mismatch: got " << out.size() << " expected "
511 << static_cast<std::size_t>(n_edges()) * n_comp);
512
513 std::array<std::array<dof_value_t, n_dofs_per_node>, 2 * hyEdge_dim> dofs;
514
515 const dof_index_t n_e = n_edges();
516 for (dof_index_t e = 0; e < n_e; ++e)
517 {
518 auto hyper_edge = hyper_graph_[e];
519 const auto hyNodes = hyper_edge.topology.get_hyNode_indices();
520 for (unsigned int node = 0; node < hyNodes.size(); ++node)
521 hyper_graph_.hyNode_factory().get_dof_values(hyNodes[node], x_vec, dofs[node]);
522
523 const auto local = local_solver_.energy(dofs, hyper_edge, time);
524 const std::size_t base = static_cast<std::size_t>(e) * n_comp;
525 for (unsigned int c = 0; c < n_comp; ++c)
526 out[base + c] = local[c];
527 }
528 }

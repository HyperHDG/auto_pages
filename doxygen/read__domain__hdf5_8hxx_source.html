25/*!*************************************************************************************************
26 * \brief The .geo.h5 domain file format and the HDF5 calls to read and write it.
27 *
28 * A .geo.h5 file holds the same information as a .geo file (cf. read_domain_geo) in the group
29 * \c /domain, as plain two-dimensional datasets that NumPy/h5py read directly:
30 *
31 * /domain/points (n_points, space_dim) float coordinates
32 * /domain/edges (n_hyEdges, 2^hyEdge_dim) int point indices per hyperedge
33 * /domain/hypernodes (n_hyEdges, 2 hyEdge_dim) int hypernode indices per hyperedge
34 * /domain/types_faces (n_hyEdges, 2 hyEdge_dim) int face types (boundary conditions)
35 * /domain/types_points (n_points, 1) int derived (hyEdge_dim 1): max
36 * type of the faces at a point
37 * /domain/properties (n_hyEdges, n_properties) float optional per-hyperedge data
38 *
39 * with the attributes \c space_dim, \c hyperedge_dim and \c n_hypernodes on \c /domain (integers;
40 * further attributes are ignored). \c hypernodes is optional: without it, hypernodes are numbered
41 * like points (one hypernode per point), which is the case for networks (hyEdge_dim 1) and the
42 * convention of the Python network generators. Files written by \c write_domain_hdf5 for
43 * hyEdge_dim 1 contain no \c hypernodes dataset if the numbering agrees.
44 *
45 * \c python/geo2h5.py converts a .geo file into this format. The group \c /VTKHDF (optional,
46 * written by default) is a VTKHDF unstructured grid view of the same data for ParaView: \c Points
47 * and the point/cell data are links into \c /domain, and the \c Connectivity of a network is a
48 * virtual dataset over \c /domain/edges, so the view costs nothing but a few small arrays.
49 * ParaView chooses its reader by file extension, so open a .geo.h5 file with File > Open and pick
50 * "VTKHDF Reader" when asked.
51 *
52 * Reading is one \c H5Dread per dataset into the (contiguous) containers of \c DomainInfo;
53 * types are converted by HDF5, so int32 files load into unsigned int indices and float32 points
54 * into double. The layout is row-major per hyperedge, which is what a later per-rank hyperslab
55 * read needs.
56 *
57 * Requires HyperHDG to be configured with HDF5 (CMake option \c HYPERHDG_HDF5, which defines
58 * \c HYPERHDG_HAVE_HDF5). The functions throw \c std::runtime_error on malformed files.
59 *
60 * \authors Joseph Holten, Karlsruhe Institute of Technology, 2026.
61 **************************************************************************************************/

88/*!*************************************************************************************************
89 * \brief The HDF5 memory type of a C++ arithmetic type.
90 **************************************************************************************************/
91template <typename T>
92hid_t native_type()
93{
94 if constexpr (std::is_same_v<T, double>)
95 return H5T_NATIVE_DOUBLE;
96 else if constexpr (std::is_same_v<T, float>)
97 return H5T_NATIVE_FLOAT;
98 else if constexpr (std::is_same_v<T, long double>)
99 return H5T_NATIVE_LDOUBLE;
100 else if constexpr (std::is_same_v<T, int>)
101 return H5T_NATIVE_INT;
102 else if constexpr (std::is_same_v<T, unsigned int>)
103 return H5T_NATIVE_UINT;
104 else if constexpr (std::is_same_v<T, long>)
105 return H5T_NATIVE_LONG;
106 else if constexpr (std::is_same_v<T, unsigned long>)
107 return H5T_NATIVE_ULONG;
108 else if constexpr (std::is_same_v<T, long long>)
109 return H5T_NATIVE_LLONG;
110 else if constexpr (std::is_same_v<T, unsigned long long>)
111 return H5T_NATIVE_ULLONG;
112 else if constexpr (std::is_same_v<T, short>)
113 return H5T_NATIVE_SHORT;
114 else if constexpr (std::is_same_v<T, unsigned short>)
115 return H5T_NATIVE_USHORT;
116 else if constexpr (std::is_same_v<T, signed char>)
117 return H5T_NATIVE_SCHAR;
118 else if constexpr (std::is_same_v<T, unsigned char>)
119 return H5T_NATIVE_UCHAR;
120 else
121 static_assert(!std::is_same_v<T, T>, "No HDF5 type for this C++ type.");
122}

132/*!*************************************************************************************************
133 * \brief Rows and columns of a one- or two-dimensional dataset (a 1d dataset has one column).
134 **************************************************************************************************/
135inline std::array<hsize_t, 2> shape(const hid_t loc, const char* name)
136{
137 Id dset(H5Dopen2(loc, name, H5P_DEFAULT), H5Dclose, std::string("open dataset ") + name);
138 Id space(H5Dget_space(dset), H5Sclose, "get dataspace");
139 const int rank = H5Sget_simple_extent_ndims(space);
140 hy_check(rank == 1 || rank == 2,
141 "HDF5: dataset " << name << " must be 1d or 2d, but has rank " << rank << ".");
142 hsize_t dims[2] = {0, 1};
143 H5Sget_simple_extent_dims(space, dims, nullptr);
144 return {dims[0], rank == 2 ? dims[1] : 1};
145}

147/*!*************************************************************************************************
148 * \brief Read a whole dataset of \c rows x \c cols entries into \c buffer (row-major).
149 **************************************************************************************************/
150template <typename T>
151void read(const hid_t loc, const char* name, const hsize_t rows, const hsize_t cols, T* buffer)
152{
153 const std::array<hsize_t, 2> dims = shape(loc, name);
154 hy_check(dims[0] == rows && dims[1] == cols,
155 "HDF5: dataset " << name << " has shape (" << dims[0] << ", " << dims[1]
156 << "), expected (" << rows << ", " << cols << ").");
157 if (rows * cols == 0)
158 return;
159 Id dset(H5Dopen2(loc, name, H5P_DEFAULT), H5Dclose, std::string("open dataset ") + name);
160 hy_check(H5Dread(dset, native_type<T>(), H5S_ALL, H5S_ALL, H5P_DEFAULT, buffer) >= 0,
161 "HDF5: reading dataset " << name << " failed.");
162}

204/*!*************************************************************************************************
205 * \brief Write a dataset of \c rows x \c cols entries from \c data (row-major).
206 *
207 * \c cols = 0 writes a one-dimensional dataset of \c rows entries. Datasets with at least one
208 * entry are chunked and gzip compressed (level 1, the level at which network files compress
209 * well without slowing down the write).
210 **************************************************************************************************/
211template <typename T>
212void write(const hid_t loc,
213 const char* name,
214 const hsize_t rows,
215 const hsize_t cols,
216 const T* data,
217 const bool compress = true)
218{
219 const int rank = cols == 0 ? 1 : 2;
220 const hsize_t dims[2] = {rows, cols};
221 Id space(H5Screate_simple(rank, dims, nullptr), H5Sclose, "create dataspace");
222 Id dcpl(H5Pcreate(H5P_DATASET_CREATE), H5Pclose, "create dataset property list");
223 if (compress && rows > 0)
224 {
225 const hsize_t chunk[2] = {std::min<hsize_t>(rows, hsize_t(1) << 16), cols};
226 H5Pset_chunk(dcpl, rank, chunk);
227 H5Pset_deflate(dcpl, 1);
228 }
229 Id dset(H5Dcreate2(loc, name, native_type<T>(), space, H5P_DEFAULT, dcpl, H5P_DEFAULT), H5Dclose,
230 std::string("create dataset ") + name);
231 if (rows * (cols == 0 ? 1 : cols) > 0)
232 hy_check(H5Dwrite(dset, native_type<T>(), H5S_ALL, H5S_ALL, H5P_DEFAULT, data) >= 0,
233 "HDF5: writing dataset " << name << " failed.");
234}

236/*!*************************************************************************************************
237 * \brief Write the VTKHDF unstructured-grid view of \c /domain into \c file.
238 *
239 * Hyperedges become VTK lines, quads or hexahedra (hyEdge_dim 1, 2, 3). Points are linked for
240 * three space dimensions and zero-padded otherwise (VTK wants three coordinates). For networks
241 * the connectivity is a virtual dataset over \c /domain/edges; for quads and hexahedra the
242 * lexicographic vertex order of a hypercube is permuted into VTK's order.
243 **************************************************************************************************/
244template <unsigned int hyEdge_dim, unsigned int space_dim, typename PointsT, typename EdgesT>
245void write_vtkhdf_view(const hid_t file,
246 const PointsT& points,
247 const EdgesT& edges,
248 const bool has_types_points,
249 const bool has_properties)
250{
251 static_assert(hyEdge_dim >= 1 && hyEdge_dim <= 3 && space_dim <= 3,
252 "The VTKHDF view exists for lines, quads and hexahedra in up to 3 dimensions.");
253 constexpr unsigned int n_verts = 1 << hyEdge_dim;
254 constexpr unsigned char vtk_type = hyEdge_dim == 1 ? 3 : hyEdge_dim == 2 ? 9 : 12;
255 const hsize_t n_points = points.size(), n_cells = edges.size(), n_conn = n_cells * n_verts;
256
257 Id root(H5Gcreate2(file, "VTKHDF", H5P_DEFAULT, H5P_DEFAULT, H5P_DEFAULT), H5Gclose,
258 "create group VTKHDF");
259 {
260 const long long version[2] = {2, 0};
261 const hsize_t two = 2;
262 Id space(H5Screate_simple(1, &two, nullptr), H5Sclose, "create dataspace");
263 Id attr(H5Acreate2(root, "Version", H5T_STD_I64LE, space, H5P_DEFAULT, H5P_DEFAULT), H5Aclose,
264 "create attribute Version");
265 hy_check(H5Awrite(attr, H5T_NATIVE_LLONG, version) >= 0, "HDF5: writing Version failed.");
266 }
267 write_attribute(root, "Type", std::string("UnstructuredGrid"));
268
269 if constexpr (space_dim == 3)
270 hy_check(H5Lcreate_soft("/domain/points", root, "Points", H5P_DEFAULT, H5P_DEFAULT) >= 0,
271 "HDF5: linking Points failed.");
272 else
273 {
274 using coord_t = typename PointsT::value_type::value_type;
275 std::vector<coord_t> padded(3 * n_points, coord_t(0));
276 for (hsize_t pt = 0; pt < n_points; ++pt)
277 for (unsigned int dim = 0; dim < space_dim; ++dim)
278 padded[3 * pt + dim] = points[pt][dim];
279 write(root, "Points", n_points, 3, padded.data());
280 }
281
282 if constexpr (hyEdge_dim == 1)
283 {
284 using index_t = typename EdgesT::value_type::value_type;
285 Id vspace(H5Screate_simple(1, &n_conn, nullptr), H5Sclose, "create virtual dataspace");
286 const hsize_t src_dims[2] = {n_cells, n_verts};
287 Id sspace(H5Screate_simple(2, src_dims, nullptr), H5Sclose, "create source dataspace");
288 Id dcpl(H5Pcreate(H5P_DATASET_CREATE), H5Pclose, "create dataset property list");
289 hy_check(H5Pset_virtual(dcpl, vspace, ".", "/domain/edges", sspace) >= 0,
290 "HDF5: virtual mapping of Connectivity failed.");
291 Id dset(H5Dcreate2(root, "Connectivity", native_type<index_t>(), vspace, H5P_DEFAULT, dcpl,
292 H5P_DEFAULT),
293 H5Dclose, "create dataset Connectivity");
294 }
295 else
296 {
297 // lexicographic (x fastest) hypercube vertices -> VTK_QUAD / VTK_HEXAHEDRON order
298 constexpr std::array<unsigned int, 8> order = {0, 1, 3, 2, 4, 5, 7, 6};
299 std::vector<long long> conn(n_conn);
300 for (hsize_t cell = 0; cell < n_cells; ++cell)
301 for (unsigned int v = 0; v < n_verts; ++v)
302 conn[cell * n_verts + v] = edges[cell][order[v]];
303 write(root, "Connectivity", n_conn, 0, conn.data());
304 }
305
306 std::vector<long long> offsets(n_cells + 1);
307 for (hsize_t cell = 0; cell <= n_cells; ++cell)
308 offsets[cell] = cell * n_verts;
309 write(root, "Offsets", n_cells + 1, 0, offsets.data());
310 const std::vector<unsigned char> types(n_cells, vtk_type);
311 write(root, "Types", n_cells, 0, types.data());
312 const long long counts[3] = {(long long)n_points, (long long)n_cells, (long long)n_conn};
313 write(root, "NumberOfPoints", 1, 0, counts, false);
314 write(root, "NumberOfCells", 1, 0, counts + 1, false);
315 write(root, "NumberOfConnectivityIds", 1, 0, counts + 2, false);
316
317 if (has_types_points)
318 {
319 Id pd(H5Gcreate2(root, "PointData", H5P_DEFAULT, H5P_DEFAULT, H5P_DEFAULT), H5Gclose,
320 "create group PointData");
321 H5Lcreate_soft("/domain/types_points", pd, "types_points", H5P_DEFAULT, H5P_DEFAULT);
322 }
323 if (has_properties)
324 {
325 Id cd(H5Gcreate2(root, "CellData", H5P_DEFAULT, H5P_DEFAULT, H5P_DEFAULT), H5Gclose,
326 "create group CellData");
327 H5Lcreate_soft("/domain/properties", cd, "properties", H5P_DEFAULT, H5P_DEFAULT);
328 }
329}

334 * \brief Whether \c filename names a .geo.h5 domain file (an HDF5 file with a group /domain).
335 **************************************************************************************************/
336inline bool is_domain_hdf5(const std::string& filename)
337{
338#if H5_VERSION_GE(1, 12, 0)
339 if (H5Fis_accessible(filename.c_str(), H5P_DEFAULT) <= 0)
340#else // HDF5 1.10 (e.g. Ubuntu 24.04) has only the predecessor of H5Fis_accessible
341 if (H5Fis_hdf5(filename.c_str()) <= 0)
342#endif
343 return false;
344 const hid_t file = H5Fopen(filename.c_str(), H5F_ACC_RDONLY, H5P_DEFAULT);
345 if (file < 0)
346 return false;
347 const bool has_domain = H5Lexists(file, "/domain", H5P_DEFAULT) > 0;
348 H5Fclose(file);
349 return has_domain;
350}

352/*!*************************************************************************************************
353 * \brief Read a .geo.h5 domain file (cf. namespace HDF5Domain for the format).
354 *
355 * \tparam hyEdge_dim The local dimension of a hyperedge.
356 * \tparam space_dim The dimension of the surrounding space.
357 * \tparam vectorT The typename of a large vector holding e.g. all points.
358 * \tparam pointT The typename of a point.
359 * \tparam hyEdge_index_t The index type for hyperedges.
360 * \tparam hyNode_index_t The index type for hypernodes.
361 * \tparam pt_index_t The index type for points.
362 * \param filename Name of the .geo.h5 file.
363 * \retval domain_info Topological and geometrical information of the hypergraph.
364 **************************************************************************************************/
365template <unsigned int hyEdge_dim,
366 unsigned int space_dim,
367 template <typename...>
368 typename vectorT,
369 typename pointT,
370 typename hyEdge_index_t,
371 typename hyNode_index_t,
372 typename pt_index_t>
373DomainInfo<hyEdge_dim, space_dim, vectorT, pointT, hyEdge_index_t, hyNode_index_t, pt_index_t>
374read_domain_hdf5(const std::string& filename)
375{
376 using namespace HDF5Domain;
377 using coord_t = typename pointT::value_type;
378 static_assert(sizeof(pointT) == space_dim * sizeof(coord_t),
379 "Points must be contiguous arrays of their coordinates.");
380 constexpr unsigned int n_verts = 1 << hyEdge_dim, n_faces = 2 * hyEdge_dim;
381
382 Id file(H5Fopen(filename.c_str(), H5F_ACC_RDONLY, H5P_DEFAULT), H5Fclose,
383 "open file " + filename);
384 Id domain(H5Gopen2(file, "/domain", H5P_DEFAULT), H5Gclose, "open group /domain of " + filename);
385
386 long long attr;
387 if (read_attribute(domain, "space_dim", attr))
388 hy_check(attr == space_dim, "space_dim in " << filename << " is " << attr << ", but should be "
389 << space_dim << "!");
390 if (read_attribute(domain, "hyperedge_dim", attr))
391 hy_check(attr == hyEdge_dim, "hyperedge_dim in " << filename << " is " << attr
392 << ", but should be " << hyEdge_dim << "!");
393
394 const std::array<hsize_t, 2> points_shape = shape(domain, "points");
395 const std::array<hsize_t, 2> edges_shape = shape(domain, "edges");
396 hy_check(points_shape[1] == space_dim, "points in " << filename << " have " << points_shape[1]
397 << " coordinates, but should have "
398 << space_dim << "!");
399 hy_check(edges_shape[1] == n_verts, "edges in " << filename << " have " << edges_shape[1]
400 << " points, but should have " << n_verts << "!");
401 const pt_index_t n_points = points_shape[0];
402 const hyEdge_index_t n_hyEdges = edges_shape[0];
403 const bool has_hypernodes = exists(domain, "hypernodes");
404 hy_check(has_hypernodes || hyEdge_dim == 1,
405 filename << " has no dataset hypernodes, which is only optional for hyperedge_dim 1.");
406
407 hyNode_index_t n_hyNodes = n_points;
408 if (has_hypernodes && !read_attribute(domain, "n_hypernodes", attr))
409 attr = -1;
410 DomainInfo<hyEdge_dim, space_dim, vectorT, pointT, hyEdge_index_t, hyNode_index_t, pt_index_t>
411 domain_info(n_points, n_hyEdges, has_hypernodes && attr >= 0 ? attr : n_hyNodes, n_points);
412
413 read(domain, "points", n_points, space_dim,
414 reinterpret_cast<coord_t*>(domain_info.points.data()));
415 read(domain, "edges", n_hyEdges, n_verts, domain_info.points_hyEdge.data()->data());
416 read(domain, "types_faces", n_hyEdges, n_faces, domain_info.hyFaces_hyEdge.data()->data());
417 if (has_hypernodes)
418 {
419 read(domain, "hypernodes", n_hyEdges, n_faces, domain_info.hyNodes_hyEdge.data()->data());
420 if (attr < 0) // no n_hypernodes attribute: count them
421 {
422 for (const auto& nodes : domain_info.hyNodes_hyEdge)
423 for (const hyNode_index_t node : nodes)
424 n_hyNodes = std::max<hyNode_index_t>(n_hyNodes, node + 1);
425 domain_info.n_hyNodes = n_hyNodes;
426 }
427 }
428 else
429 for (hyEdge_index_t edge = 0; edge < n_hyEdges; ++edge)
430 for (unsigned int i = 0; i < n_faces; ++i)
431 domain_info.hyNodes_hyEdge[edge][i] = domain_info.points_hyEdge[edge][i];
432
433 if (exists(domain, "properties"))
434 {
435 const std::array<hsize_t, 2> props_shape = shape(domain, "properties");
436 hy_check(props_shape[0] == n_hyEdges,
437 "properties in " << filename << " has " << props_shape[0] << " rows, but there "
438 << "are " << n_hyEdges << " hyperedges!");
439 domain_info.n_properties = props_shape[1];
440 std::vector<coord_t> flat(n_hyEdges * domain_info.n_properties);
441 read(domain, "properties", n_hyEdges, domain_info.n_properties, flat.data());
442 domain_info.hyEdge_properties.resize(n_hyEdges);
443 for (hyEdge_index_t edge = 0; edge < n_hyEdges; ++edge)
444 domain_info.hyEdge_properties[edge].assign(
445 flat.begin() + edge * domain_info.n_properties,
446 flat.begin() + (edge + 1) * domain_info.n_properties);
447 }
448
449 return domain_info;
450} // end of read_domain_hdf5

452/*!*************************************************************************************************
453 * \brief Write a domain as a .geo.h5 file (cf. namespace HDF5Domain for the format).
454 *
455 * An existing file is overwritten. With \c vtkhdf (default), the file also contains the VTKHDF
456 * view for ParaView, which exists for hyperedge dimensions 1 to 3 in up to 3 space dimensions
457 * and is skipped otherwise.
458 *
459 * \param filename Name of the file to write.
460 * \param domain_info Topological and geometrical information of the hypergraph.
461 * \param vtkhdf Also write the VTKHDF view.
462 **************************************************************************************************/
463template <unsigned int hyEdge_dim,
464 unsigned int space_dim,
465 template <typename...>
466 typename vectorT,
467 typename pointT,
468 typename hyEdge_index_t,
469 typename hyNode_index_t,
470 typename pt_index_t>
471void write_domain_hdf5(const std::string& filename,
472 const DomainInfo<hyEdge_dim,
473 space_dim,
474 vectorT,
475 pointT,
476 hyEdge_index_t,
477 hyNode_index_t,
478 pt_index_t>& domain_info,
479 const bool vtkhdf = true)
480{
481 using namespace HDF5Domain;
482 using coord_t = typename pointT::value_type;
483 static_assert(sizeof(pointT) == space_dim * sizeof(coord_t),
484 "Points must be contiguous arrays of their coordinates.");
485 constexpr unsigned int n_verts = 1 << hyEdge_dim, n_faces = 2 * hyEdge_dim;
486 const hsize_t n_points = domain_info.n_points, n_hyEdges = domain_info.n_hyEdges;
487
488 Id file(H5Fcreate(filename.c_str(), H5F_ACC_TRUNC, H5P_DEFAULT, H5P_DEFAULT), H5Fclose,
489 "create file " + filename);
490 Id domain(H5Gcreate2(file, "domain", H5P_DEFAULT, H5P_DEFAULT, H5P_DEFAULT), H5Gclose,
491 "create group domain");
492 write_attribute(domain, "space_dim", space_dim);
493 write_attribute(domain, "hyperedge_dim", hyEdge_dim);
494 write_attribute(domain, "n_hypernodes", (long long)domain_info.n_hyNodes);
495
496 write(domain, "points", n_points, space_dim,
497 reinterpret_cast<const coord_t*>(domain_info.points.data()));
498 write(domain, "edges", n_hyEdges, n_verts, domain_info.points_hyEdge.data()->data());
499 write(domain, "types_faces", n_hyEdges, n_faces, domain_info.hyFaces_hyEdge.data()->data());
500

532 flat.insert(flat.end(), props.begin(), props.end());
533 }
534 write(domain, "properties", n_hyEdges, domain_info.n_properties, flat.data());
535 }

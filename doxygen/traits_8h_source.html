18/* The builtin `std::is_copy_constructible` type trait merely checks whether
19 a copy constructor is present and returns `true` even when the this copy
20 constructor cannot be compiled. This a problem for older STL types like
21 `std::vector<T>` when `T` is noncopyable. The alternative below recurses
22 into STL types to work around this problem. */
23
24template <typename T>
25struct is_copy_constructible<
26 T, enable_if_t<
27 std::is_same_v<typename T::value_type &, typename T::reference> &&
28 std::is_copy_constructible_v<T> &&
29 !std::is_same_v<T, typename T::value_type>>> {
30 static constexpr bool value =
31 is_copy_constructible<typename T::value_type>::value;
32};
33
34// std::pair is copy-constructible <=> both constituents are copy-constructible
35template <typename T1, typename T2>
36struct is_copy_constructible<std::pair<T1, T2>> {
37 static constexpr bool value =
38 is_copy_constructible<T1>::value &&
39 is_copy_constructible<T2>::value;
40};

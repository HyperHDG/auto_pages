3/*!*************************************************************************************************
4 * \file tests/records.hxx
5 * \brief Golden records: the record lines a test prints, checked against an expected file.
6 *
7 * A record is one line `- {key: value, ...}`, printed by tests::record with a printf format. At the
8 * end, tests::check compares the records with RECORDS_FILE (tests/expected/<ctest name>.yaml, set
9 * by tests/CMakeLists.txt for the tests registered with RECORDS):
10 * - An expected record matches the one printed record that agrees with it in every integer and
11 * string key it has (except iterations); floats are compared to rtol, iterations to
12 * iterations_rtol |e| + 2. Strings such as the stop reason of CG are exact.
13 * - An expected record may hold fewer keys than the printed one, and printed records without an
14 * expected one are only counted, so a file can pin a part of the output (the seeds from the
15 * Cython-era scripts hold {dim, degree, n, error}).
16 * - With HYPERHDG_RECORD=1 in the environment, check writes the printed records to RECORDS_FILE
17 * instead: `HYPERHDG_RECORD=1 ctest -R timoshenko` re-records a set.
18 **************************************************************************************************/

38/*!*************************************************************************************************
39 * \brief The record of a line `- {key: value, ...}`, or false if the line is none.
40 **************************************************************************************************/
41inline bool parse_record(std::string line, Record& record)
42{
43 while (!line.empty() && (line.back() == '\n' || line.back() == '\r' || line.back() == ' '))
44 line.pop_back();
45 if (line.compare(0, 3, "- {") != 0 || line.back() != '}')
46 return false;
47 record.clear();
48 std::stringstream fields(line.substr(3, line.size() - 4));
49 for (std::string field; std::getline(fields, field, ',');)
50 {
51 const std::size_t begin = field.find_first_not_of(' '), colon = field.find(": ");
52 if (begin == std::string::npos || colon == std::string::npos)
53 return false;
54 record.emplace_back(field.substr(begin, colon - begin), field.substr(colon + 2));
55 }
56 return !record.empty();
57}

62inline std::string to_string(const Record& record)
63{
64 std::string line = "- {";
65 for (const auto& [key, value] : record)
66 line += (line.size() > 3 ? ", " : "") + key + ": " + value;
67 return line + "}";

78inline Kind kind(const std::string& value)
79{
80 char* end = nullptr;
81 std::strtol(value.c_str(), &end, 10);
82 if (!value.empty() && *end == '\0')
83 return Kind::integer;
84 std::strtod(value.c_str(), &end);
85 if (!value.empty() && *end == '\0')
86 return Kind::floating;
87 return Kind::string;

89// The keys that identify a record: integers and strings, but not iterations.
90inline bool identifies(const std::string& key, const std::string& value)
91{
92 return key != "iterations" && kind(value) != Kind::floating;
93}
94inline const std::string* find(const Record& record, const std::string& key)
95{
96 for (const auto& [k, v] : record)
97 if (k == key)
98 return &v;
99 return nullptr;
100}
101inline bool same(const std::string& a, const std::string& b)
102{
103 return kind(a) == Kind::integer && kind(b) == Kind::integer
104 ? std::strtoll(a.c_str(), nullptr, 10) == std::strtoll(b.c_str(), nullptr, 10)
105 : a == b;
106}
107// The identity keys of expected in which printed differs (missing or another value).
108inline std::vector<std::string> differences(const Record& expected, const Record& printed)
109{
110 std::vector<std::string> out;
111 for (const auto& [key, value] : expected)
112 {
113 if (!identifies(key, value))
114 continue;
115 const std::string* other = find(printed, key);
116 if (!other)
117 out.push_back(key + " missing");
118 else if (!same(*other, value))
119 out.push_back(key + " is " + *other + ", expected " + value);
120 }
121 return out;
122}

130/*!*************************************************************************************************
131 * \brief What compare found: the complaints, and how many printed records were checked or not.
132 **************************************************************************************************/
133struct Comparison
134{
135 std::vector<std::string> complaints;
136 unsigned int checked = 0, unchecked = 0;
137};
138
139/*!*************************************************************************************************
140 * \brief Compare printed record lines with the text of an expected file (rules: see the top).
141 **************************************************************************************************/
142inline Comparison compare(const std::vector<std::string>& printed_lines,
143 const std::string& expected_text,
144 const double rtol,
145 const double iterations_rtol)
146{
147 Comparison result;
148 std::vector<Record> printed, expected;
149 Record record;
150 for (const std::string& line : printed_lines)
151 if (parse_record(line, record))
152 printed.push_back(record);
153 std::stringstream text(expected_text);
154 for (std::string line; std::getline(text, line);)
155 if (parse_record(line, record))
156 expected.push_back(record);
157 if (expected.empty())
158 result.complaints.push_back("the expected file holds no records");
159
160 std::vector<bool> matched(printed.size(), false);
161 for (const Record& e : expected)
162 {
163 std::vector<std::size_t> candidates;
164 for (std::size_t k = 0; k < printed.size(); ++k)
165 if (detail::differences(e, printed[k]).empty())
166 candidates.push_back(k);
167 if (candidates.size() != 1)
168 {
169 std::string complaint = to_string(e) + ": ";
170 if (candidates.size() > 1)
171 complaint += std::to_string(candidates.size()) + " printed records match";
172 else if (printed.empty())
173 complaint += "no record printed";
174 else
175 {
176 // the nearest printed record: the fewest differing identity keys
177 std::vector<std::string> nearest;
178 for (const Record& p : printed)
179 {
180 const std::vector<std::string> d = detail::differences(e, p);
181 if (nearest.empty() || d.size() < nearest.size())
182 nearest = d;
183 }
184 complaint += "no printed record matches; the nearest:";
185 for (const std::string& d : nearest)
186 complaint += (&d == &nearest.front() ? " " : "; ") + d;
187 }
188 result.complaints.push_back(complaint);
189 continue;
190 }
191
192 const Record& p = printed[candidates.front()];
193 matched[candidates.front()] = true;
194 ++result.checked;
195 for (const auto& [key, value] : e)
196 {
197 if (detail::identifies(key, value))
198 continue;
199 const std::string* other = detail::find(p, key);
200 if (!other)
201 {
202 result.complaints.push_back(to_string(e) + ": " + key + " missing");
203 continue;
204 }
205 const double a = std::strtod(other->c_str(), nullptr),
206 x = std::strtod(value.c_str(), nullptr);
207 const double tolerance =
208 key == "iterations" ? iterations_rtol * std::abs(x) + 2. : rtol * std::abs(x);
209 if (!(std::abs(a - x) <= tolerance)) // also catches nan
210 result.complaints.push_back(to_string(e) + ": " + key + " is " + *other + ", expected " +
211 value);
212 }
213 }
214 result.unchecked = std::count(matched.begin(), matched.end(), false);
215 return result;
216}

218/*!*************************************************************************************************
219 * \brief Print a record line (printf format) and keep it for check.
220 **************************************************************************************************/
221[[gnu::format(printf, 1, 2)]] inline void record(const char* format, ...)
222{
223 std::va_list args, copy;
224 va_start(args, format);
225 va_copy(copy, args);
226 const int length = std::vsnprintf(nullptr, 0, format, copy);
227 va_end(copy);
228 std::vector<char> line(std::max(length, 0) + 1, '\0');
229 std::vsnprintf(line.data(), line.size(), format, args);
230 va_end(args);
231 std::fputs(line.data(), stdout);
232 detail::printed_lines().emplace_back(line.data());
233}

235/*!*************************************************************************************************
236 * \brief Check the kept records against RECORDS_FILE, or write it with HYPERHDG_RECORD=1; returns
237 * failures plus the number of complaints. Without RECORDS_FILE (a compile outside CMake)
238 * nothing is checked.
239 **************************************************************************************************/
240inline int check(const int failures, const double rtol = 1e-5, const double iterations_rtol = 0.1)
241{
242#ifndef RECORDS_FILE
243 (void)rtol;
244 (void)iterations_rtol;
245 std::printf("records: not checked (no RECORDS_FILE)\n");
246 return failures;
247#else
248 const std::vector<std::string>& lines = detail::printed_lines();
249 const char* record_mode = std::getenv("HYPERHDG_RECORD");
250 if (record_mode && std::string(record_mode) == "1")
251 {
252 if (failures > 0 || lines.empty())
253 {
254 std::printf("records: not written to %s (%d failures, %zu records)\n", RECORDS_FILE, failures,
255 lines.size());
256 return failures + 1;
257 }
258 std::ofstream file(RECORDS_FILE);
259 Record record;
260 for (const std::string& line : lines)
261 if (parse_record(line, record))
262 file << to_string(record) << '\n';
263 if (!file)
264 {
265 std::printf("records: cannot write %s\n", RECORDS_FILE);
266 return 1;
267 }
268 std::printf("records: %zu written to %s\n", lines.size(), RECORDS_FILE);
269 return 0;
270 }
271
272 std::ifstream file(RECORDS_FILE);
273 if (!file)
274 {
275 std::printf("records: cannot read %s (record it with HYPERHDG_RECORD=1)\n", RECORDS_FILE);
276 return failures + 1;
277 }
278 std::stringstream text;
279 text << file.rdbuf();
280 const Comparison result = compare(lines, text.str(), rtol, iterations_rtol);
281 for (const std::string& complaint : result.complaints)
282 std::printf(" FAIL: %s\n", complaint.c_str());
283 std::printf("records: %u checked, %u unchecked, %zu complaints\n", result.checked,
284 result.unchecked, result.complaints.size());
285 // an exit status counts modulo 256: keep the sum small
286 return std::min(failures + (int)result.complaints.size(), 100);
287#endif
288}

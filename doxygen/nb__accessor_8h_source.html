19 object tmp = steal(obj_op_2(ptr(), o.derived().ptr(), op)); \

24template <typename Impl> class accessor : public api<accessor<Impl>> {
25 template <typename T> friend void nanobind::del(accessor<T> &);
26 template <typename T> friend void nanobind::del(accessor<T> &&);
27public:
28 static constexpr auto Name = const_name("object");
29
30 template <typename Key>
31 accessor(handle obj, Key &&key)
32 : m_base(obj.ptr()), m_key(std::move(key)) { }
33 accessor(const accessor &) = delete;
34 accessor(accessor &&) = delete;
35 ~accessor() {
36 if constexpr (Impl::cache_dec_ref)
37 Py_XDECREF(m_cache);
38 }
39
40 template <typename T> accessor& operator=(T &&value);
41
42 template <typename T, enable_if_t<std::is_base_of_v<object, T>> = 0>
43 operator T() const { return borrow<T>(ptr()); }
44 NB_INLINE PyObject *ptr() const {
45 Impl::get(m_base, m_key, &m_cache);
46 return m_cache;
47 }
48 NB_INLINE handle base() const { return m_base; }
49 NB_INLINE object key() const { return steal(Impl::key(m_key)); }
50
51 NB_DECL_ACCESSOR_OP_I(operator+=)
52 NB_DECL_ACCESSOR_OP_I(operator-=)
53 NB_DECL_ACCESSOR_OP_I(operator*=)
54 NB_DECL_ACCESSOR_OP_I(operator/=)
55 NB_DECL_ACCESSOR_OP_I(operator%=)
56 NB_DECL_ACCESSOR_OP_I(operator|=)
57 NB_DECL_ACCESSOR_OP_I(operator&=)
58 NB_DECL_ACCESSOR_OP_I(operator^=)
59 NB_DECL_ACCESSOR_OP_I(operator<<=)
60 NB_DECL_ACCESSOR_OP_I(operator>>=)
61
62private:
63 NB_INLINE void del () { Impl::del(m_base, m_key); }
64
65private:
66 PyObject *m_base;
67 mutable PyObject *m_cache{nullptr};
68 typename Impl::key_type m_key;
69};

189 #if defined(Py_GIL_DISABLED)

193 #else
194 *cache = NB_LIST_GET_ITEM(obj, index);

197
198 NB_INLINE static void set(PyObject *obj, Py_ssize_t index, PyObject *v) {
199#if defined(Py_LIMITED_API) || defined(NB_FREE_THREADED)
200 Py_INCREF(v);
201 PyList_SetItem(obj, index, v);
202#else
203 PyObject *old = NB_LIST_GET_ITEM(obj, index);
204 Py_INCREF(v);
205 NB_LIST_SET_ITEM(obj, index, v);
206 Py_DECREF(old);
207#endif
208 }

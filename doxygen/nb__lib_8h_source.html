35struct NB_CORE cleanup_list {
36public:
37 static constexpr uint32_t Small = 6;
38
39 cleanup_list(PyObject *self) :
40 m_size{1},
41 m_capacity{Small},
42 m_data{m_local} {
43 m_local[0] = self;
44 }
45
46 ~cleanup_list() = default;
47
49 NB_INLINE void append(PyObject *value) noexcept {
50 if (m_size >= m_capacity)
51 expand();
52 m_data[m_size++] = value;
53 }
54
55 NB_INLINE PyObject *self() const {
56 return m_local[0];
57 }
58
60 void release() noexcept;
61
63 bool used() { return m_size != 1; }
64
66 size_t size() const { return m_size; }
67
69 PyObject *operator[](size_t index) const { return m_data[index]; }
70
71protected:
73 void expand() noexcept;
74
75protected:
76 uint32_t m_size;
77 uint32_t m_capacity;
78 PyObject **m_data;
79 PyObject *m_local[Small];
80};

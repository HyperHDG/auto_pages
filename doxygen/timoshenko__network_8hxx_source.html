382 inline std::array<std::array<double, 2 * space_dim>, 2 * hyEdge_dimT> node_dof_to_edge_dof(
383 const SmallMatT& glob_lambda,
384 hyEdgeT& hyper_edge) const
385 {
386 std::array<std::array<double, 2 * space_dim>, 2 * hyEdge_dimT> loc_lambda;
387 hy_assert(loc_lambda.size() == 2, "Only implemented in one dimension!");
388 for (unsigned int i = 0; i < loc_lambda.size(); ++i)
389 {
390 hy_assert(loc_lambda[i].size() == 6, "Only implemented in one dimension!");
391 loc_lambda[i].fill(0.);
392 }
393
394 Point<space_dim, lSol_float_t> normal_vec =
395 (Point<space_dim, lSol_float_t>)hyper_edge.geometry.inner_normal(0);
396
397 for (unsigned int i = 0; i < 2 * hyEdge_dimT; ++i)
398 for (unsigned int dim = 0; dim < space_dim; ++dim)
399 {
400 loc_lambda[i][0] += normal_vec[dim] * glob_lambda[i][dim];
401 loc_lambda[i][0 + space_dim] += normal_vec[dim] * glob_lambda[i][dim + space_dim];
402 }
403
404 for (unsigned int ind = 0; ind < space_dim - 1; ++ind)
405 {
406 normal_vec = (Point<space_dim, lSol_float_t>)hyper_edge.geometry.outer_normal(ind);
407
408 for (unsigned int i = 0; i < 2 * hyEdge_dimT; ++i)
409 for (unsigned int dim = 0; dim < space_dim; ++dim)
410 {
411 loc_lambda[i][1 + ind] += normal_vec[dim] * glob_lambda[i][dim];
412 loc_lambda[i][1 + ind + space_dim] += normal_vec[dim] * glob_lambda[i][space_dim + dim];
413 }
414 }
415
416 return loc_lambda;
417 }

477 inline SmallMat<2 * hyEdge_dimT, 2 * space_dim * n_shape_bdr_, lSol_float_t>
478 extract_fluxes_from_coeffs(const SmallVec<n_loc_dofs_, lSol_float_t>& coeffs,
479 hyEdgeT& hyper_edge) const
480 {
481 SmallMat<2 * hyEdge_dimT, 2 * space_dim * n_shape_bdr_, lSol_float_t> bdr_values;
482 lSol_float_t integral;
483
484 for (unsigned int i = 0; i < n_shape_fct_; ++i)
485 for (unsigned int j = 0; j < n_shape_bdr_; ++j)
486 for (unsigned int face = 0; face < 2 * hyEdge_dimT; ++face)
487 {
488 integral = integrator::template integrate_bdr_phipsi<decltype(hyEdgeT::geometry)>(
489 i, j, face, hyper_edge.geometry);
490 for (unsigned int dim = 0; dim < 2 * space_dim; ++dim)
491 bdr_values(face, j + dim) +=
492 hyper_edge.geometry.local_normal(face).operator[](0) * integral *
493 coeffs[dim * n_shape_fct_ + i] +
494 tau_ * coeffs[(2 * space_dim + dim) * n_shape_fct_ + i] * integral;
495 }
496
497 return bdr_values;

728 SmallSquareMat<n_loc_dofs_, lSol_float_t> local_mat;
729 lSol_float_t vol_integral, face_integral, helper;

731 SmallVec<2 * space_dim, lSol_float_t> extra_coeffs(1.);
732
733 if (hyper_edge.geometry.has_extra_data())
734 {
735 auto extra_data = hyper_edge.geometry.extra_data();
736 // (EA, kG_1A, kG_2A, G_xI_x, E_1I_1, E_2I_2): 6 structural constants
737 // (n_11,n_12,n_13) : normal 1
738 // (n_21,n_22,n_23) : normal 2
739 SmallVec<space_dim, lSol_float_t> normal1 =
740 std::array<lSol_float_t, space_dim>{{extra_data[6], extra_data[7], extra_data[8]}};
741 SmallVec<space_dim, lSol_float_t> normal2 =
742 std::array<lSol_float_t, space_dim>{{extra_data[9], extra_data[10], extra_data[11]}};

744 SmallVec<space_dim, lSol_float_t> outer2 = hyper_edge.geometry.outer_normal(1);

746 extra_coeffs[0] = extra_data[0];
747 extra_coeffs[1] = extra_data[1] * scalar_product(outer1, normal1) +
748 extra_data[2] * scalar_product(outer1, normal2);
749 extra_coeffs[2] = extra_data[1] * scalar_product(outer2, normal1) +
750 extra_data[2] * scalar_product(outer2, normal2);

753 extra_data[5] * scalar_product(outer1, normal2);
754 extra_coeffs[5] = extra_data[4] * scalar_product(outer2, normal1) +
755 extra_data[5] * scalar_product(outer2, normal2);

759 extra_coeffs[2] *= 1e4;
760 extra_coeffs[3] *= 1e12;
761 extra_coeffs[4] *= 1e12;
762 extra_coeffs[5] *= 1e12;

771 {

778 integrator::template integrate_vol_nablaphiphi<SmallVec<hyEdge_dimT, lSol_float_t>,
779 decltype(hyEdgeT::geometry)>(

787 i, j, face, hyper_edge.geometry);

801 (2 * space_dim + dim) * n_shape_fct_ + j) += tau_ * face_integral;
802 }
803

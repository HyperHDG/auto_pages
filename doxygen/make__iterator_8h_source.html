33// Note: these helpers take the iterator by non-const reference because some
34// iterators in the wild can't be dereferenced when const.
35template <typename Iterator> struct iterator_access {
36 using result_type = decltype(*std::declval<Iterator &>());
37 result_type operator()(Iterator &it) const { return *it; }
38};

40template <typename Iterator> struct iterator_key_access {
41 // Note double parens in decltype((...)) to capture the value category
42 // as well. This will be lvalue if the iterator's operator* returned an
43 // lvalue reference, and xvalue if the iterator's operator* returned an
44 // object (or rvalue reference but that's unlikely). decltype of an xvalue
45 // produces T&&, but we want to return a value T from operator() in that
46 // case, in order to avoid creating a Python object that references a
47 // C++ temporary. Thus, pass the result through remove_rvalue_ref.
48 using result_type = typename remove_rvalue_ref<
49 decltype(((*std::declval<Iterator &>()).first))>::type;
50 result_type operator()(Iterator &it) const { return (*it).first; }
51};

53template <typename Iterator> struct iterator_value_access {
54 using result_type = typename remove_rvalue_ref<
55 decltype(((*std::declval<Iterator &>()).second))>::type;
56 result_type operator()(Iterator &it) const { return (*it).second; }

59template <typename Access, rv_policy Policy, typename Iterator,
60 typename Sentinel, typename ValueType, typename... Extra>
61typed<iterator, ValueType> make_iterator_impl(handle scope, const char *name,
62 Iterator first, Sentinel last,
63 Extra &&...extra) {
64 using State = iterator_state<Access, Policy, Iterator, Sentinel, ValueType, Extra...>;
65
66 static_assert(
67 !detail::is_base_caster_v<detail::make_caster<ValueType>> ||
68 detail::is_copy_constructible_v<ValueType> ||
69 (Policy != rv_policy::automatic_reference &&
70 Policy != rv_policy::copy),
71 "make_iterator_impl(): the generated __next__ would copy elements, so the "
72 "element type must be copy-constructible");
73
74 {
75 static ft_mutex mu;
76 ft_lock_guard lock(mu);
77 if (!type<State>().is_valid()) {
78 class_<State>(scope, name)
79 .def("__iter__", [](handle h) { return h; })
80 .def("__next__",
81 [](State &s) -> ValueType {
82 if (!s.first_or_done)
83 ++s.it;
84 else
85 s.first_or_done = false;
86
87 if (s.it == s.end) {
88 s.first_or_done = true;
89 throw stop_iteration();
90 }
91
92 return Access()(s.it);
93 },
94 std::forward<Extra>(extra)...,
95 Policy);
96 }
97 }
98 return borrow<typed<iterator, ValueType>>(cast(State{
99 std::forward<Iterator>(first), std::forward<Sentinel>(last), true }));
100}

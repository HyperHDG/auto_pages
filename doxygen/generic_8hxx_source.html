40/*!*************************************************************************************************
41 * \brief The single global loop built on local operators.
42 *
43 * The loop is a scatter--gather engine over hyperedges. For every hyperedge it gathers the local
44 * trace from a global vector, hands it to the hyperedge's local operator, and scatters the result.
45 * It never sees block matrices, factorizations, or boundary conditions of the local solver, and
46 * it never allocates global vectors: every entry point writes into caller-provided vectors (any
47 * type with size() and operator[], e.g. std::vector or std::span).
48 *
49 * Local solver protocol (cf. doc/local_operator_design.md):
50 * - \c Args: runtime parameters (time, ...) forwarded to every call,
51 * - \c local_operator(edge, Kind, args): builds the immutable operator of that kind with
52 * \c apply<k>(solver, edge, in, args, out) and \c residual(solver, edge, lambda, args, r); what
53 * it computes and keeps is the solver's business (e.g. local_solver/local_operator.hxx). The
54 * loop hands the solver and the hyperedge to every call of the operator,
55 * - optional \c n_operator_slots(edge, Kind, args) and \c operator_slot(edge, Kind, args): the
56 * solver numbers its operators of that kind, and hyperedges with the same number share one,
57 * - optional \c Kinds: a std::tuple of the kinds of operators the solver builds (kinds.hxx), if
58 * there are others than Stiffness,
59 * - optional \c constrained_node<space_dim>(node_type): default predicate for constrain_dirichlet.
60 *
61 * Kinds: \c prepare and the sweeps are templates on the kind of operator, Stiffness by default
62 * (loop.apply<Mass>(x, y)); each kind has its own store, prepared on its own.
63 *
64 * Store: after \c prepare the operators live in a contiguous vector until \c release, one per
65 * slot; without operator_slot, the slot of a hyperedge is its index. A slot names the operator
66 * that is stored, not the local problem: hyperedges of one slot read the same operator, whose
67 * results may still depend on the hyperedge it is applied to. Operators without state (an empty
68 * type) are all the same, so there is one slot for all hyperedges. The loop keeps nothing per
69 * hyperedge beyond the operators.
70 * Without prepare every sweep builds the operator of a hyperedge, uses it and drops it. Every
71 * public sweep takes that decision once and runs a branch-free per-edge body for the chosen mode.
72 *
73 * Constraints: a sorted index set of global dofs owned by the loop. Constrained dofs are not part
74 * of the system (prescribed values are zero in this prototype, Dirichlet data enters via the
75 * local load): every sweep reads x as zero there, and apply and residual return zero there. So
76 * apply is P J P with the projection P onto the free dofs, symmetric if J is, and singular;
77 * assemble gives the same matrix, with a unit diagonal on the constrained dofs on request.
78 *
79 * Serial. Parallel sweeps (scatter strategy) are a separate step.
80 **************************************************************************************************/

277 /*!***********************************************************************************************
278 * \brief y = J x on free dofs, with x read as zero on constrained dofs; zero there. y is
279 * overwritten.
280 *
281 * x is the direction of the directional derivative. Where a nonlinear problem is linearized is
282 * not an argument: args reaches the local solver as it is, but a global vector does not (only
283 * the loop can gather its local traces), so the local solver keeps the point, e.g. in the data
284 * of the hyperedges.
285 ************************************************************************************************/
286 template <class Kind = Stiffness, class VecInT, class VecOutT>
287 void apply(const VecInT& x, VecOutT& y, const Args& args = Args())
288 {
289 check_sizes(x, y);
290 zero(y);
291 dispatch<Kind>(
292 [&](auto mode)
293 {
294 sweep<Kind, mode.value>(args,
295 [&](std::size_t, Edge& edge, const auto& op, const Dofs& dofs)
296 {
297 Traces<1> in, res;
298 gather(dofs, x, in);
299 op.apply(local_solver_, edge, in, args, res);
300 scatter(dofs, res, y);
301 });
302 });
303 for (dof_index_t i : constraints_)
304 y[i] = 0.;
305 }
306 /*!***********************************************************************************************
307 * \brief r = F(x): flux balance on free dofs, with x read as zero on constrained dofs; zero
308 * there. r is overwritten.
309 *
310 * x is the point at which the residual is evaluated.
311 ************************************************************************************************/
312 template <class Kind = Stiffness, class VecInT, class VecOutT>
313 void residual(const VecInT& x, VecOutT& r, const Args& args = Args())
314 {
315 check_sizes(x, r);
316 zero(r);
317 dispatch<Kind>(
318 [&](auto mode)
319 {
320 sweep<Kind, mode.value>(args,
321 [&](std::size_t, Edge& edge, const auto& op, const Dofs& dofs)
322 {
323 Traces<1> in, res;
324 gather(dofs, x, in);
325 op.residual(local_solver_, edge, in, args, res);

332 /*!***********************************************************************************************
333 * \brief Number of triplets assemble() writes: one block per hyperedge plus the identity rows.
334 ************************************************************************************************/
335 std::size_t n_triplets(const bool identity_rows = false) const
336 {
337 return hyper_graph_.n_hyEdges() * (std::size_t)n_loc * n_loc +
338 (identity_rows ? constraints_.size() : 0);
339 }
340 /*!***********************************************************************************************
341 * \brief Triplets of J into caller-provided arrays of size n_triplets(identity_rows).
342 *
343 * Every hyperedge's block is apply<n_loc> of the identity (so the matrix is the matrix-free
344 * operator by construction). Entries in constrained rows and columns are multiplied by zero
345 * (branch-free mask), so the matrix is symmetric if J is; with \c identity_rows a unit diagonal
346 * entry is appended for every constrained dof, which makes it non-singular.
347 * Duplicate (row, col) pairs have to be summed by the consumer (as scipy's coo_matrix does).
348 ************************************************************************************************/
349 template <class Kind = Stiffness, class IndexVecT, class ValueVecT>
350 void assemble(const Args& args,
351 const bool identity_rows,
352 IndexVecT& rows,
353 IndexVecT& cols,
354 ValueVecT& vals)
355 {
356 const std::size_t n_edges = hyper_graph_.n_hyEdges();
357 const std::size_t block = (std::size_t)n_loc * n_loc;
358 hy_assert(rows.size() == n_triplets(identity_rows) && cols.size() == rows.size() &&
359 vals.size() == rows.size(),
360 "Triplet arrays must have size n_triplets(identity_rows).");
361 Traces<n_loc> identity;
362 for (unsigned int i = 0; i < n_loc; ++i)
363 identity(i, i) = 1.;
364
365 auto write_block = [&](std::size_t e, const Dofs& dofs, const Traces<n_loc>& loc_mat)
366 {
367 std::size_t index = e * block;
368 for (unsigned int col = 0; col < n_loc; ++col)
369 for (unsigned int row = 0; row < n_loc; ++row, ++index)
370 {
371 rows[index] = dofs[row];
372 cols[index] = dofs[col];
373 vals[index] = loc_mat(row, col) * free_mask_[dofs[row]] * free_mask_[dofs[col]];
374 }
375 };
376
377 dispatch<Kind>(
378 [&](auto mode)
379 {
380 sweep<Kind, mode.value>(args,
381 [&](std::size_t e, Edge& edge, const auto& op, const Dofs& dofs)
382 {
383 Traces<n_loc> loc_mat;
384 op.apply(local_solver_, edge, identity, args, loc_mat);
385 write_block(e, dofs, loc_mat);
386 });
387 });
388
389 if (identity_rows)
390 {
391 std::size_t index = n_edges * block;
392 for (dof_index_t i : constraints_)
393 {
394 rows[index] = i;
395 cols[index] = i;
396 vals[index++] = 1.;
397 }
398 }
399 }
400 /*!***********************************************************************************************
401 * \brief Fold f(edge, operator, local trace) over all hyperedges: result = combine(result,
402 * f(...)), by default with operator+.
403 *
404 * \c combine is associative and \c result its neutral element (0 for +), so that partial
405 * results can be combined in any grouping.
406 ************************************************************************************************/
407 template <class Kind = Stiffness, class VecT, class F, class R, class Combine = std::plus<> >
408 R reduce(const VecT& x, const Args& args, F f, R result, Combine combine = Combine())
409 {
410 dispatch<Kind>(
411 [&](auto mode)
412 {
413 sweep<Kind, mode.value>(args,
414 [&](std::size_t, Edge& edge, const auto& op, const Dofs& dofs)
415 {
416 Traces<1> in;
417 gather(dofs, x, in);
418 result = combine(std::move(result), f(edge, op, in));
419 });
420 });
421 return result;
422 }

516 /*!***********************************************************************************************
517 * \brief The number of slots and the slot of a hyperedge: a single one if operators have no

519 ************************************************************************************************/
520 template <class Kind>

522 {

526 {

528 return 0;

537 {
538 if constexpr (std::is_empty_v<OperatorOf<Kind> >)
539 return 0;
540 else if constexpr (requires { local_solver_.operator_slot(edge, Kind{}, args); })
541 return local_solver_.operator_slot(edge, Kind{}, args);

545 /*!***********************************************************************************************
546 * \brief Run body(e, edge, operator, dofs) over all hyperedges; the operator comes from the
547 * store or is built on the fly, decided at compile time.
548 ************************************************************************************************/
549 template <class Kind, bool use_store, class Body>
550 void sweep(const Args& args, Body body)
551 {
552 const std::size_t n_edges = hyper_graph_.n_hyEdges();
553 for (std::size_t e = 0; e < n_edges; ++e)
554 {
555 Edge edge = hyper_graph_[e];
556 Dofs dofs;
557 local_dof_indices(edge, dofs);
558 if constexpr (use_store)
559 body(e, edge, store<Kind>().operators[operator_slot<Kind>(e, edge, args)], dofs);
560 else
561 {
562 const OperatorOf<Kind> op = local_solver_.local_operator(edge, Kind{}, args);
563 body(e, edge, op, dofs);
564 }
565 }

569/*!*************************************************************************************************
570 * \brief Errors of a solution: the sum of the local solver's error_sq(edge, operator, lambda,
571 * args) over all hyperedges (a reduction of the loop), square root taken componentwise.
572 **************************************************************************************************/
573template <class LoopT, class VecT>
574auto errors(LoopT& loop, const VecT& x, const typename LoopT::Args& args = typename LoopT::Args())
575{
576 using error_t = std::remove_cvref_t<decltype(loop.local_solver().error_sq(
577 std::declval<typename LoopT::Edge&>(), std::declval<const typename LoopT::Operator&>(),
578 std::declval<const typename LoopT::template Traces<1>&>(), args))>;
579 error_t sum = loop.reduce(
580 x, args, [&](auto& edge, const auto& op, const auto& lambda)
581 { return loop.local_solver().error_sq(edge, op, lambda, args); }, error_t());
582 for (unsigned int k = 0; k < sum.size(); ++k)
583 sum[k] = std::sqrt(sum[k]);
584 return sum;
585}

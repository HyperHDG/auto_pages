260 * \brief Block-diagonal decomposition of the local matrix into independent components.
261 *
262 * The local matrix decouples into 4 connected components (see assemble_loc_matrix). Each
263 * variable group dim in 0..2*space_dim-1 forms a self-coupled triplet of variable-blocks
264 * {sigma, w, wdot} = {dim, 2*space_dim+dim, 4*space_dim+dim}; the only links between triplets
265 * are the t x n cross-product terms, which (for space_dim==3) couple the transverse shear
266 * force with the perpendicular bending rotation. The components are, in units of variable-
267 * blocks of width n_shape_fct_:

272 * Level 2 (assemble_schur / solve_local_problem) further eliminates each triplet's diagonal

274 * nw_ x nw_ system, C,D to an n2w_ x n2w_ system.

321 /*!***********************************************************************************************
322 * \brief Step parameters of the local solve (see also doc/hdg_gauss.pdf).
323 *
324 * The only runtime time-discretisation inputs are:
325 * tau_ HDG stabilisation parameter (globally constant), tau > 0.
326 * delta_t_ time step Delta t.
327 * The local system is assembled in the STAGE NORMALIZATION of doc/hdg_gauss.pdf (eq 5): the step
328 * enters the matrix only through sigma_l = 1/(theta_l*Delta t), which weights the (y,z)/(z,y)
329 * couplings (see assemble_loc_matrix / assemble_schur). Deviation from the note: the z-rows keep
330 * their C_u scaling -- (z,z) = M, (z,y) = -sigma*C_u*M -- instead of the note's C_u^{-1} d-form,
331 * so massless welds (C_u = 0) stay regular (z = 0 there). This costs the literal symmetry of the
332 * local saddle (row scaling), but leaves the condensed trace operator unchanged.
333 ************************************************************************************************/

336 /*!***********************************************************************************************
337 * \brief Gauss tableau data in the eigenbasis of the Butcher matrix (doc/hdg_gauss.pdf).
338 *
339 * Built at construction by Gauss::build_tableau (ported from hoRK's horkirk.c): nodes and
340 * weights by Newton on the Legendre polynomial, Butcher A = W V^{-1}, eigendecomposition
341 * A = T diag(theta) T^{-1} via zgeev with conjugate-pair enforcement, so only the ceil(s/2)
342 * representatives are solved. Per representative l: eigenvalue theta_l (stage shift
343 * sigma_l = 1/(theta_l*dt)), load weight omega_l = (T^{-1} 1)_l, data-combination row
344 * tinv_l = (T^{-1})_{l,:}, value-form recombination weight w_l = (b^T A^{-1} T)_l and
345 * multiplicity mult_l (2 for a conjugate pair). Endpoint update:
346 * y+ = affine * y^n + sum_l mult_l * Re(w_l * y_l), affine = 1 - b^T A^{-1} 1 = (-1)^s.
347 * s = 1 is the (all-real) implicit midpoint whose trajectory is Crank-Nicolson\'s; s >= 2 has
348 * complex representatives, hence the stage scalar type stage_float_t.
349 ************************************************************************************************/

371 static stage_data_t make_stage_data()
372 {
373 const Gauss::Tableau tab = Gauss::build_tableau(n_stages);
374 hy_assert(tab.n_reps == n_reps_, "expected ceil(s/2) stage representatives");
375 stage_data_t sd;
376 sd.affine = tab.affine;
377 for (unsigned int j = 0; j < n_stages; ++j)
378 sd.c[j] = tab.c[j];
379 for (unsigned int l = 0; l < n_reps_; ++l)
380 {
381 sd.theta[l] = stage_cast(tab.theta[l]);
382 sd.omega[l] = stage_cast(tab.omega[l]);
383 sd.w[l] = stage_cast(tab.w[l]);
384 sd.mult[l] = tab.mult[l];
385 for (unsigned int j = 0; j < n_stages; ++j)
386 sd.tinv[l][j] = stage_cast(tab.tinv[l * n_stages + j]);
387 }
388 return sd;

543 i, bdr, comps[j], geom, time);
544 }

549 /*!***********************************************************************************************
550 * \brief The local saddle system in stage normalization (doc/hdg_gauss.pdf eq 5).
551 *
552 * Blocks (per variable group dim, in units of n_shape_fct_; see the block decomposition above):
553 * (q,q) = M/C_sig, (q,y) = -G, (y,q) = B-G, (y,y) = tau*F [always]
554 * (y,z) = +sigma*M, (z,y) = -sigma*C_u*M, (z,z) = M [with_z only]
555 * plus the t x n cross-product couplings. with_z = false yields the STATIC system (the sigma
556 * couplings and z rows dropped) used by the initializers; sigma is ignored then. The z-rows keep
557 * their C_u scaling (deviation from the note's C_u^{-1} d-form) so massless welds stay regular:
558 * C_u = 0 gives z = 0 and a vanishing mass shift, not a division by zero.
559 ************************************************************************************************/
560 template <bool with_z, typename mat_float_t = lSol_float_t, typename hyEdgeT>
561 inline SmallSquareMat<(with_z ? 6 : 4) * space_dim * n_shape_fct_, mat_float_t>
562 assemble_loc_matrix(hyEdgeT& hyper_edge, const mat_float_t sigma = 0.) const
563 {
564 SmallSquareMat<(with_z ? 6 : 4) * space_dim * n_shape_fct_, mat_float_t> local_mat;
565 SmallVec<4 * space_dim, lSol_float_t> extra_coeffs = get_extra_coeffs(hyper_edge);
566
567 for (unsigned int i = 0; i < n_shape_fct_; ++i)
568 for (unsigned int j = 0; j < n_shape_fct_; ++j)
569 {
570 const lSol_float_t vol_integral =
571 integrator::integrate_vol_phiphi(i, j, hyper_edge.geometry);
572 const auto grad_int_vec =
573 integrator::template integrate_vol_nablaphiphi<SmallVec<hyEdge_dimT, lSol_float_t>,
574 decltype(hyEdgeT::geometry)>(
575 i, j, hyper_edge.geometry);
576 lSol_float_t face_integral = 0.;
577 SmallVec<hyEdge_dimT, lSol_float_t> normal_int_vec(0.);
578 for (unsigned int face = 0; face < 2 * hyEdge_dimT; ++face)
579 {
580 const lSol_float_t helper =
581 integrator::template integrate_bdr_phiphi<decltype(hyEdgeT::geometry)>(
582 i, j, face, hyper_edge.geometry);
583 face_integral += helper;
584 normal_int_vec += helper * hyper_edge.geometry.local_normal(face);
585 }
586
587 for (unsigned int dim = 0; dim < 2 * space_dim; ++dim)
588 {
589 // compliance a(q,q) and equilibrium -b(q,w_y)
590 local_mat(dim * n_shape_fct_ + i, dim * n_shape_fct_ + j) +=
591 vol_integral / extra_coeffs[dim];
592 local_mat(dim * n_shape_fct_ + i, (2 * space_dim + dim) * n_shape_fct_ + j) -=
593 grad_int_vec[0];
594 // b^T and tau<y,w_y>
595 local_mat((2 * space_dim + dim) * n_shape_fct_ + i, dim * n_shape_fct_ + j) +=
596 (normal_int_vec[0] - grad_int_vec[0]);
597 local_mat((2 * space_dim + dim) * n_shape_fct_ + i,
598 (2 * space_dim + dim) * n_shape_fct_ + j) += tau_ * face_integral;
599 if constexpr (with_z)
600 {
601 // inertia (z,z) = M and the sigma-weighted couplings; z-rows carry the C_u scaling
602 local_mat((4 * space_dim + dim) * n_shape_fct_ + i,
603 (4 * space_dim + dim) * n_shape_fct_ + j) += vol_integral;
604 local_mat((4 * space_dim + dim) * n_shape_fct_ + i,
605 (2 * space_dim + dim) * n_shape_fct_ + j) -=
606 sigma * extra_coeffs[2 * space_dim + dim] * vol_integral;
607 local_mat((2 * space_dim + dim) * n_shape_fct_ + i,
608 (4 * space_dim + dim) * n_shape_fct_ + j) += sigma * vol_integral;
609 }
610 }
611
612 // cross product (textbook orientation: + i x r and + i x n)
613 local_mat(2 * n_shape_fct_ + i, (3 * space_dim + 1) * n_shape_fct_ + j) += vol_integral;
614 local_mat(1 * n_shape_fct_ + i, (3 * space_dim + 2) * n_shape_fct_ + j) -= vol_integral;
615 local_mat((3 * space_dim + 2) * n_shape_fct_ + i, 1 * n_shape_fct_ + j) += vol_integral;
616 local_mat((3 * space_dim + 1) * n_shape_fct_ + i, 2 * n_shape_fct_ + j) -= vol_integral;
617 }
618
619 return local_mat;
620 }

627 /*!***********************************************************************************************
628 * \brief Body loads and Dirichlet data of one collocation node (real, single time).
629 *
630 * The f-loads land on the y-rows, the Dirichlet data on the q- and y-rows (bit-6 static-only
631 * faces keep their trace). Shared by all stage representatives, which combine these vectors
632 * with their (complex) T^{-1} row weights in assemble_rhs_stage.
633 ************************************************************************************************/
634 template <typename hyEdgeT>
635 inline SmallVec<n_loc_dofs_, lSol_float_t> assemble_rhs_data(hyEdgeT& hyper_edge,
636 const lSol_float_t time) const
637 {
638 using parameters = parametersT<decltype(hyEdgeT::geometry)::space_dim(), lSol_float_t>;
639 SmallVec<n_loc_dofs_, lSol_float_t> rhs;
640
641 // body loads act on material only if the parameters opt in (massless welds unloaded)
642 bool loaded = true;
643 if constexpr (requires { parameters::massless_unloaded; })
644 if (parameters::massless_unloaded && hyper_edge.geometry.has_extra_data())
645 loaded = hyper_edge.geometry.extra_data()[0] > 0;
646
647 for (unsigned int i = 0; i < n_shape_fct_; ++i)
648 {
649 if (loaded)
650 {
651 auto integrals = integrate_vol_phivecfunccomp_beam<
652 Point<decltype(hyEdgeT::geometry)::space_dim(), lSol_float_t>,
653 decltype(hyEdgeT::geometry), parameters::right_hand_side_n>(i, {1, -1, -2},
654 hyper_edge.geometry, time);
655 for (unsigned int comp = 0; comp < 3; ++comp)
656 rhs[(2 * space_dim + comp) * n_shape_fct_ + i] += integrals[comp];
657 integrals = integrate_vol_phivecfunccomp_beam<
658 Point<decltype(hyEdgeT::geometry)::space_dim(), lSol_float_t>,
659 decltype(hyEdgeT::geometry), parameters::right_hand_side_m>(i, {1, -1, -2},
660 hyper_edge.geometry, time);
661 for (unsigned int comp = 0; comp < 3; ++comp)
662 rhs[(3 * space_dim + comp) * n_shape_fct_ + i] += integrals[comp];
663 }
664
665 // Dirichlet data (bit-6 static-only faces keep their trace)
666 for (unsigned int face = 0; face < 2 * hyEdge_dimT; ++face)
667 {
668 if (hyper_edge.node_descriptor[face] & (1 << 6))
669 continue;
670 if (!hyper_edge.node_descriptor[face])
671 continue;
672 auto integrals = integrate_bdr_phivecfunccomp_beam<
673 Point<decltype(hyEdgeT::geometry)::space_dim(), lSol_float_t>,
674 decltype(hyEdgeT::geometry), parameters::dirichlet_value_u>(i, face, {1, -1, -2},
675 hyper_edge.geometry, time);
676 for (unsigned int comp = 0; comp < 3; ++comp)
677 {
678 rhs[(0 * space_dim + comp) * n_shape_fct_ + i] -=
679 hyper_edge.geometry.local_normal(face).operator[](0) * integrals[comp];
680 rhs[(2 * space_dim + comp) * n_shape_fct_ + i] += tau_ * integrals[comp];
681 }
682 integrals = integrate_bdr_phivecfunccomp_beam<
683 Point<decltype(hyEdgeT::geometry)::space_dim(), lSol_float_t>,
684 decltype(hyEdgeT::geometry), parameters::dirichlet_value_phi>(i, face, {1, -1, -2},
685 hyper_edge.geometry, time);
686 for (unsigned int comp = 0; comp < 3; ++comp)
687 {
688 rhs[(1 * space_dim + comp) * n_shape_fct_ + i] -=
689 hyper_edge.geometry.local_normal(face).operator[](0) * integrals[comp];
690 rhs[(3 * space_dim + comp) * n_shape_fct_ + i] += tau_ * integrals[comp];
691 }
692 }
693 }
694
695 return rhs;
696 }

698 /*!***********************************************************************************************
699 * \brief Gauss stage loads (doc/hdg_gauss.pdf eq 7): history enters through (y^n, z^n) only.
700 *
701 * Eigenbasis stage l of the collocation system: the data terms mix the collocation nodes with
702 * the T^{-1} row of the representative, rhs_l = sum_j tinv_{l,j} * data(t^n + c_j*dt), and the
703 * history enters as sigma_l*omega_l mass pairings of (y^n, z^n):
704 * g_y += sigma*omega*(z^n, w_y), g_z -= sigma*omega*C_u*(y^n, w_z) in our C_u-scaled z-row
705 * normalization. At s = 1 (tinv = {1}, c = 1/2) the single midpoint sample is the O(dt^2) load
706 * difference to the retired trapezoid scheme (exact match for f = 0 and time-independent BC).
707 * No flux caches, no old trace: q and lambda are algebraic and never history.
708 ************************************************************************************************/
709 template <typename hyEdgeT>
710 inline SmallVec<n_loc_dofs_, stage_float_t> assemble_rhs_stage(hyEdgeT& hyper_edge,
711 const lSol_float_t time,
712 const unsigned int stage = 0) const
713 {
714 SmallVec<n_loc_dofs_, stage_float_t> rhs;
715
716 for (unsigned int j = 0; j < n_stages; ++j)
717 {
718 const lSol_float_t t_node = time - (1. - stages_.c[j]) * delta_t_;
719 const SmallVec<n_loc_dofs_, lSol_float_t> data_j = assemble_rhs_data(hyper_edge, t_node);
720 const stage_float_t weight = stages_.tinv[stage][j];
721 for (unsigned int i = 0; i < n_loc_dofs_; ++i)
722 rhs[i] += weight * data_j[i];
723 }
724
725 // history: sigma*omega mass pairings of (y^n, z^n); M = area * Id (orthonormal basis)
726 const auto extra = get_extra_coeffs(hyper_edge);
727 const auto& c_old = hyper_edge.data.coeffs_old;
728 const stage_float_t sig_om = sigma(stage) * stages_.omega[stage];
729 const lSol_float_t mass = hyper_edge.geometry.area();
730 for (unsigned int d = 0; d < space_dim; ++d)
731 for (unsigned int i = 0; i < n_shape_fct_; ++i)
732 {
733 rhs[(2 * space_dim + d) * n_shape_fct_ + i] +=
734 sig_om * mass * c_old[coeff_idx(fld_v, d, i)];
735 rhs[(3 * space_dim + d) * n_shape_fct_ + i] +=
736 sig_om * mass * c_old[coeff_idx(fld_s, d, i)];
737 rhs[(4 * space_dim + d) * n_shape_fct_ + i] -=
738 sig_om * mass * extra[2 * space_dim + d] * c_old[coeff_idx(fld_u, d, i)];
739 rhs[(5 * space_dim + d) * n_shape_fct_ + i] -=
740 sig_om * mass * extra[3 * space_dim + d] * c_old[coeff_idx(fld_r, d, i)];
741 }
742
743 return rhs;
744 }

866 /*!***********************************************************************************************
867 * \brief Assemble and factorize the displacement Schur systems of the four components.
868 *
869 * Builds the shared per-edge integral blocks M (diagonal), G = int (grad phi) phi, B = boundary
870 * normal flux and F = boundary mass; eliminates each triplet's sigma and wdot blocks (both pure
871 * mass) explicitly to obtain the displacement Schur operator
872 * S(C_sig, C_u) = theta*tau*F + theta*C_sig*(B-G) M^{-1} G + (C_u/(theta*dt^2)) M.
873 * Components A,B are single triplets (nw_ x nw_); C,D couple a force group (C_sig=C_sig_n) with a
874 * rotation group via the cross product into an n2w_ x n2w_ system. M_inv, G and (B-G) are cached
875 * for the reduction / back-substitution in solve_local_problem.
876 ************************************************************************************************/
877 template <typename hyEdgeT>
878 inline void assemble_schur(hyEdgeT& hyper_edge) const
879 {
880 auto& data = hyper_edge.data;
881 constexpr unsigned int n = n_shape_fct_;
882 data.extra = get_extra_coeffs(hyper_edge); // C_n, C_m, C_u, C_r
883
884 std::array<lSol_float_t, nw_> M_diag;
885 std::array<lSol_float_t, nw_ * nw_> G, Bm, F; // row-major; Bm = B - G
886 M_diag.fill(0.);
887 G.fill(0.);
888 Bm.fill(0.);
889 F.fill(0.);
890 for (unsigned int i = 0; i < n; ++i)
891 for (unsigned int j = 0; j < n; ++j)
892 {
893 const lSol_float_t vol = integrator::integrate_vol_phiphi(i, j, hyper_edge.geometry);
894 const auto grad =
895 integrator::template integrate_vol_nablaphiphi<SmallVec<hyEdge_dimT, lSol_float_t>,
896 decltype(hyEdgeT::geometry)>(
897 i, j, hyper_edge.geometry);
898 lSol_float_t face_integral = 0., normal_integral = 0.;
899 for (unsigned int face = 0; face < 2 * hyEdge_dimT; ++face)
900 {
901 const lSol_float_t h =
902 integrator::template integrate_bdr_phiphi<decltype(hyEdgeT::geometry)>(
903 i, j, face, hyper_edge.geometry);
904 face_integral += h;
905 normal_integral += h * hyper_edge.geometry.local_normal(face).operator[](0);
906 }
907 if (i == j)
908 M_diag[i] = vol; // M is diagonal: orthogonal Legendre basis on affine geometry
909 G[i * n + j] = grad[0];
910 Bm[i * n + j] = normal_integral - grad[0];
911 F[i * n + j] = face_integral;
912 }
913 for (unsigned int i = 0; i < n; ++i)
914 data.M_inv[i] = 1. / M_diag[i];
915 data.Gmat = G;
916 data.BmG = Bm;
917
918 // K = (B-G) M^{-1} G (row-major), the dense part of the Schur operator.
919 std::array<lSol_float_t, nw_ * nw_> K;
920 for (unsigned int r = 0; r < n; ++r)
921 for (unsigned int c = 0; c < n; ++c)
922 {
923 lSol_float_t acc = 0.;
924 for (unsigned int k = 0; k < n; ++k)
925 acc += Bm[r * n + k] * data.M_inv[k] * G[k * n + c];
926 K[r * n + c] = acc;
927 }
928
929 // Entry (r,c) of the single-triplet Schur operator S(C_sig, C_u) in stage normalization:
930 // eliminating q and z from assemble_loc_matrix<true> leaves S = tau*F + C_sig*K +
931 // sigma^2*C_u*M.
932 const lSol_float_t sig = sigma();
933 auto S_entry = [&](lSol_float_t C_sig, lSol_float_t C_u, unsigned int r, unsigned int c)
934 {
935 lSol_float_t v = tau_ * F[r * n + c] + C_sig * K[r * n + c];
936 if (r == c)
937 v += sig * sig * C_u * M_diag[r];
938 return v;
939 };
940
941 // Single component: factorize S directly (column-major for LAPACK).
942 auto fill_single = [&](std::array<lSol_float_t, nw_ * nw_>& lu, unsigned int dim)
943 {
944 const lSol_float_t C_sig = data.extra[sig_blk(dim)];
945 const lSol_float_t C_u = data.extra[w_blk(dim)];
946 for (unsigned int c = 0; c < nw_; ++c)
947 for (unsigned int r = 0; r < nw_; ++r)
948 lu[c * nw_ + r] = S_entry(C_sig, C_u, r, c);
949 };
950 fill_single(data.lu_A, compA_dim);
951 Wrapper::lapack_factorize(nw_, data.lu_A.data(), data.ipiv_A.data());
952 fill_single(data.lu_B, compB_dim);
953 Wrapper::lapack_factorize(nw_, data.lu_B.data(), data.ipiv_B.data());
954
955 // Coupled component (force group dim_n, rotation group dim_r), cross sign s as it appears in
956 // the full matrix (sigma_n row: +s*M*w_r; w_r row: +s*M*sigma_n up to sign). Eliminating
957 // sigma_n = C_sn*(M^{-1}b + M^{-1}G w_n - s*w_r) puts -s on both off-diagonal Schur blocks:
958 // [ S_n -s*C_sig_n*(B-G) ]
959 // [ -s*C_sig_n*G S_r + C_sig_n*M ]
960 auto fill_coupled = [&](std::array<lSol_float_t, n2w_ * n2w_>& lu, lSol_float_t s,
961 unsigned int dim_n, unsigned int dim_r)
962 {
963 const lSol_float_t Csn = data.extra[sig_blk(dim_n)], Cun = data.extra[w_blk(dim_n)];
964 const lSol_float_t Csr = data.extra[sig_blk(dim_r)], Cur = data.extra[w_blk(dim_r)];
965 for (unsigned int c = 0; c < n2w_; ++c)
966 for (unsigned int r = 0; r < n2w_; ++r)
967 {
968 lSol_float_t v;
969 if (r < nw_ && c < nw_)
970 v = S_entry(Csn, Cun, r, c);
971 else if (r < nw_ && c >= nw_)
972 v = -s * Csn * Bm[r * n + (c - nw_)];
973 else if (r >= nw_ && c < nw_)
974 v = -s * Csn * G[(r - nw_) * n + c];
975 else
976 {
977 v = S_entry(Csr, Cur, r - nw_, c - nw_);
978 if (r == c)
979 v += Csn * M_diag[r - nw_];
980 }
981 lu[c * n2w_ + r] = v;
982 }
983 };
984 fill_coupled(data.lu_C, +1., compC_dim_n, compC_dim_r);
985 Wrapper::lapack_factorize(n2w_, data.lu_C.data(), data.ipiv_C.data());
986 fill_coupled(data.lu_D, -1., compD_dim_n, compD_dim_r);
987 Wrapper::lapack_factorize(n2w_, data.lu_D.data(), data.ipiv_D.data());
988 }

990 /*!***********************************************************************************************
991 * \brief Reduce triplet \c dim to its displacement right-hand side; stash mass-scaled rhs.
992 *
993 * rhs_w = b_w - C_sig*(B-G) M^{-1} b_sig - sigma*b_wdot (stage normalization). Outputs
994 * M^{-1} b_sig and M^{-1} b_wdot for the back-substitution.
995 ************************************************************************************************/
996 template <typename DataT>
997 inline std::array<lSol_float_t, nw_> triplet_reduce(
998 const DataT& data,
999 unsigned int dim,
1000 const SmallVec<n_loc_dofs_, lSol_float_t>& rhs,
1001 std::array<lSol_float_t, nw_>& m_inv_bsig,
1002 std::array<lSol_float_t, nw_>& m_inv_bwdot) const
1003 {
1004 constexpr unsigned int n = n_shape_fct_;
1005 const lSol_float_t C_sig = data.extra[sig_blk(dim)];
1006 std::array<lSol_float_t, nw_> rhs_w, bw;
1007 for (unsigned int k = 0; k < n; ++k)
1008 {
1009 m_inv_bsig[k] = data.M_inv[k] * rhs[sig_blk(dim) * n + k];
1010 m_inv_bwdot[k] = data.M_inv[k] * rhs[wdot_blk(dim) * n + k];
1011 bw[k] = rhs[w_blk(dim) * n + k];
1012 }
1013 for (unsigned int r = 0; r < n; ++r)
1014 {
1015 lSol_float_t bmg_x = 0.;
1016 for (unsigned int c = 0; c < n; ++c)
1017 bmg_x += data.BmG[r * n + c] * m_inv_bsig[c];
1018 rhs_w[r] = bw[r] - C_sig * bmg_x - sigma() * rhs[wdot_blk(dim) * n + r];
1019 }
1020 return rhs_w;
1021 }

1023 /*!***********************************************************************************************
1024 * \brief Recover sigma, w, wdot of triplet \c dim from the solved displacement w and scatter.
1025 *
1026 * sigma_f = C_sig (M^{-1} b_sig + M^{-1} G w - s*w_partner), wdot = M^{-1} b_wdot +
1027 * sigma*C_u*w (stage normalization; C_u = 0 on massless welds gives wdot from b_wdot alone).
1028 * \c s_cross is 0 for single components; for a coupled force group it is the cross sign of the
1029 * full matrix (+s*M*w_partner on the sigma_f row) and \c w_partner the rotation group's
1030 * displacement.
1031 ************************************************************************************************/
1032 template <typename DataT>
1033 inline void triplet_backsub(const DataT& data,
1034 unsigned int dim,
1035 lSol_float_t s_cross,
1036 const std::array<lSol_float_t, nw_>& w,
1037 const std::array<lSol_float_t, nw_>& w_partner,
1038 const std::array<lSol_float_t, nw_>& m_inv_bsig,
1039 const std::array<lSol_float_t, nw_>& m_inv_bwdot,
1040 SmallVec<n_loc_dofs_, lSol_float_t>& result) const
1041 {
1042 constexpr unsigned int n = n_shape_fct_;
1043 const lSol_float_t C_sig = data.extra[sig_blk(dim)];
1044 const lSol_float_t C_u = data.extra[w_blk(dim)];
1045 for (unsigned int k = 0; k < n; ++k)
1046 {
1047 lSol_float_t Gw = 0.;
1048 for (unsigned int c = 0; c < n; ++c)
1049 Gw += data.Gmat[k * n + c] * w[c];
1050 result[sig_blk(dim) * n + k] =
1051 C_sig * (m_inv_bsig[k] + data.M_inv[k] * Gw - s_cross * w_partner[k]);
1052 result[w_blk(dim) * n + k] = w[k];
1053 result[wdot_blk(dim) * n + k] = m_inv_bwdot[k] + sigma() * C_u * w[k];
1054 }
1055 }

1057 template <typename hyEdgeT, typename SmallMatT>
1058 inline SmallVec<n_loc_dofs_, stage_float_t> solve_local_problem(
1059 const SmallMatT& lambda_values,
1060 const unsigned int solution_type,
1061 hyEdgeT& hyper_edge,
1062 const lSol_float_t time,
1063 const unsigned int stage = 0) const
1064 {
1065 try
1066 {
1067 auto& data = hyper_edge.data;
1068
1069 SmallVec<n_loc_dofs_, stage_float_t> rhs;
1070 {
1071 const auto rhs_lambda = assemble_rhs_from_lambda(lambda_values, hyper_edge);
1072 for (unsigned int i = 0; i < n_loc_dofs_; ++i)
1073 rhs[i] = rhs_lambda[i];
1074 }
1075 if (solution_type == 1)
1076 rhs += assemble_rhs_stage(hyper_edge, time, stage);
1077 else
1078 hy_assert(solution_type == 0, "This has not been implemented!");
1079
1080 if constexpr (n_stages > 1)
1081 {
1082 // complex stage operator: cached per-representative full-matrix LU (the Schur reduction
1083 // machinery below stays real / s = 1 only)
1084 auto& slu = data.stage_lu;
1085 if (!slu.factorized[stage])
1086 {
1087 slu.lu[stage] = assemble_loc_matrix<true, stage_float_t>(hyper_edge, sigma(stage));
1088 Wrapper::lapack_factorize(n_loc_dofs_, slu.lu[stage].data().data(),
1089 slu.ipiv[stage].data());
1090 slu.factorized[stage] = true;
1091 }
1092 Wrapper::lapack_solve_factored(n_loc_dofs_, 1, slu.lu[stage].data().data(),
1093 slu.ipiv[stage].data(), rhs.data().data());
1094 return rhs;
1095 }
1096 else
1097 {
1098 if (full_lu_)
1099 {
1100 if (!data.full_lu_factorized)
1101 {
1102 data.full_lu = assemble_loc_matrix<true>(hyper_edge, sigma());
1103 Wrapper::lapack_factorize(n_loc_dofs_, data.full_lu.data().data(),
1104 data.full_ipiv.data());
1105 data.full_lu_factorized = true;
1106 }
1107 Wrapper::lapack_solve_factored(n_loc_dofs_, 1, data.full_lu.data().data(),
1108 data.full_ipiv.data(), rhs.data().data());
1109 return rhs;
1110 }
1111
1112 if (!data.loc_mat_factorized)
1113 {
1114 assemble_schur(hyper_edge);
1115 data.loc_mat_factorized = true;
1116 }
1117
1118 constexpr unsigned int n = n_shape_fct_;
1119 SmallVec<n_loc_dofs_, lSol_float_t> result;
1120 const std::array<lSol_float_t, nw_> dummy{}; // unused w_partner for single components
1121
1122 // Single components A, B.
1123 auto solve_single =
1124 [&](std::array<lSol_float_t, nw_ * nw_>& lu, std::array<int, nw_>& ipiv, unsigned int dim)
1125 {
1126 std::array<lSol_float_t, nw_> mis, miw;
1127 auto rw = triplet_reduce(data, dim, rhs, mis, miw);
1128 Wrapper::lapack_solve_factored(nw_, 1, lu.data(), ipiv.data(), rw.data());
1129 triplet_backsub(data, dim, 0., rw, dummy, mis, miw, result);
1130 };
1131 solve_single(data.lu_A, data.ipiv_A, compA_dim);
1132 solve_single(data.lu_B, data.ipiv_B, compB_dim);
1133
1134 // Coupled components C (s=+1), D (s=-1): force group dim_n, rotation group dim_r.
1135 auto solve_coupled = [&](std::array<lSol_float_t, n2w_ * n2w_>& lu,
1136 std::array<int, n2w_>& ipiv, lSol_float_t s, unsigned int dim_n,
1137 unsigned int dim_r)
1138 {
1139 std::array<lSol_float_t, nw_> mis_n, miw_n, mis_r, miw_r;
1140 auto rw_n = triplet_reduce(data, dim_n, rhs, mis_n, miw_n);
1141 auto rw_r = triplet_reduce(data, dim_r, rhs, mis_r, miw_r);
1142 const lSol_float_t Csn = data.extra[sig_blk(dim_n)];
1143 std::array<lSol_float_t, n2w_> rw;
1144 for (unsigned int k = 0; k < n; ++k)
1145 {
1146 rw[k] = rw_n[k];
1147 rw[nw_ + k] = rw_r[k] + s * Csn * rhs[sig_blk(dim_n) * n + k];
1148 }
1149 Wrapper::lapack_solve_factored(n2w_, 1, lu.data(), ipiv.data(), rw.data());
1150 std::array<lSol_float_t, nw_> w_n, w_r;
1151 for (unsigned int k = 0; k < n; ++k)
1152 {
1153 w_n[k] = rw[k];
1154 w_r[k] = rw[nw_ + k];
1155 }
1156 triplet_backsub(data, dim_n, s, w_n, w_r, mis_n, miw_n, result); // force group: +s*w_r
1157 triplet_backsub(data, dim_r, 0., w_r, dummy, mis_r, miw_r, result); // rotation group
1158 };
1159 solve_coupled(data.lu_C, data.ipiv_C, +1., compC_dim_n, compC_dim_r);
1160 solve_coupled(data.lu_D, data.ipiv_D, -1., compD_dim_n, compD_dim_r);
1161
1162 return result;
1163 }
1164 }
1165 catch (Wrapper::LAPACKexception& exc)
1166 {
1167 std::cout << hyper_edge.geometry.area() << std::endl;
1168 hy_assert(0 == 1, exc.what() << std::endl
1169 << "This can happen if quadrature is too inaccurate!");
1170 throw exc;
1171 }
1172 }

1174 template <typename hyEdgeT, typename flux_float_t>
1175 inline SmallMat<2 * hyEdge_dimT, 2 * space_dim * n_shape_bdr_, flux_float_t>
1176 extract_fluxes_from_coeffs(const SmallVec<n_loc_dofs_, flux_float_t>& coeffs,
1177 hyEdgeT& hyper_edge) const
1178 {
1179 SmallMat<2 * hyEdge_dimT, 2 * space_dim * n_shape_bdr_, flux_float_t> bdr_values;
1180 lSol_float_t integral;
1181
1182 for (unsigned int i = 0; i < n_shape_fct_; ++i)
1183 for (unsigned int j = 0; j < n_shape_bdr_; ++j)
1184 for (unsigned int face = 0; face < 2 * hyEdge_dimT; ++face)
1185 {
1186 integral = integrator::template integrate_bdr_phipsi<decltype(hyEdgeT::geometry)>(
1187 i, j, face, hyper_edge.geometry);
1188 for (unsigned int dim = 0; dim < 2 * space_dim; ++dim)
1189 bdr_values(face, j + dim) +=
1190 hyper_edge.geometry.local_normal(face).operator[](0) * integral *
1191 coeffs[dim * n_shape_fct_ + i] +
1192 tau_ * coeffs[(2 * space_dim + dim) * n_shape_fct_ + i] * integral;
1193 }
1194
1195 return bdr_values;

1198 /*!***********************************************************************************************
1199 * \brief Zero every dynamic-Dirichlet trace dof of \c lambda in place (leaves bit-6 faces).
1200 *
1201 * A face flagged static-only (bit 6) keeps its trace; on every other face the components whose
1202 * Dirichlet bit is set are constrained, so their trace contribution is removed. Shared by the
1203 * two flux entry points below (and mirrors the masking done in the error evaluation).
1204 ************************************************************************************************/
1205 template <typename hyEdgeT, typename SmallMatT>
1206 void zero_dirichlet_trace(SmallMatT& lambda, hyEdgeT& hyper_edge) const
1207 {
1208 for (unsigned int node = 0; node < 2 * hyEdge_dimT; ++node)
1209 for (unsigned int dof = 0; dof < 2 * space_dim; ++dof)
1210 if (is_dirichlet(hyper_edge, node, dof))
1211 lambda[node][dof] = 0.;
1212 }

1214 /*!***********************************************************************************************
1215 * \brief Shared core of trace_to_flux / residual_flux: apply the condensed local operator.
1216 *
1217 * \c solution_type selects the local-solve right-hand side (see solve_local_problem): 0 is the
1218 * homogeneous operator action A*lambda (used to assemble the time-constant system matrix); 1 is
1219 * the residual, i.e. operator action plus the old-time-step / body-load / Dirichlet data. The
1220 * Dirichlet trace dofs are removed on input and output so constrained faces stay pinned.
1221 ************************************************************************************************/
1222 template <typename hyEdgeT, typename SmallMatInT, typename SmallMatOutT>
1223 SmallMatOutT& apply_local_flux(const SmallMatInT& lambda_values_in,
1224 SmallMatOutT& lambda_values_out,
1225 const unsigned int solution_type,
1226 hyEdgeT& hyper_edge,
1227 const lSol_float_t time = 0.,
1228 const unsigned int stage = 0) const
1229 {
1230 // The condensed stage operator is stage_float_t-valued (complex for n_stages > 1), so the
1231 // output arrays must be stage_float_t; real inputs (unit-vector probing) are fine.
1232 static_assert(std::is_same_v<typename SmallMatOutT::value_type::value_type, stage_float_t>,
1233 "apply_local_flux output must be stage_float_t-valued");
1234 hy_assert(lambda_values_in.size() == lambda_values_out.size() &&
1235 lambda_values_in.size() == 2 * hyEdge_dimT,
1236 "Both matrices must be of same size which corresponds to the number of faces!");
1237 for (unsigned int i = 0; i < lambda_values_in.size(); ++i)
1238 hy_assert(
1239 lambda_values_in[i].size() == lambda_values_out[i].size() &&
1240 lambda_values_in[i].size() == n_glob_dofs_per_node(),
1241 "Both matrices must be of same size which corresponds to the number of dofs per face!");
1242
1243 SmallMatInT lambda_in = lambda_values_in;
1244 zero_dirichlet_trace(lambda_in, hyper_edge);
1245
1246 SmallMatInT lambda_values_loc = node_dof_to_edge_dof(lambda_in, hyper_edge);
1247
1248 SmallVec<n_loc_dofs_, stage_float_t> coeffs =
1249 solve_local_problem(lambda_values_loc, solution_type, hyper_edge, time, stage);
1250
1251 auto result = extract_fluxes_from_coeffs(coeffs, hyper_edge);
1252
1253 SmallMatOutT out_loc;
1254 for (unsigned int i = 0; i < 2 * hyEdge_dimT; ++i)
1255 for (unsigned int j = 0; j < 2 * space_dim; ++j)
1256 out_loc[i][j] = tau_ * lambda_values_loc[i][j] - result(i, j);
1257
1258 lambda_values_out = edge_dof_to_node_dof(out_loc, lambda_values_out, hyper_edge);
1259
1260 zero_dirichlet_trace(lambda_values_out, hyper_edge);
1261
1262 return lambda_values_out;
1263 }

1265 /*!***********************************************************************************************
1266 * \brief Bridge for complex dof vectors over a REAL stage operator (complex-PETSc at s = 1):
1267 * narrow the trace to its real part, run the real machinery, widen the result.
1268 *
1269 * The trace of the real s = 1 operator is real-valued by construction; the imaginary parts of
1270 * the input are solver round-off. Only instantiated when stage_float_t is real.
1271 ************************************************************************************************/
1272 template <typename hyEdgeT, typename SmallMatInT, typename SmallMatOutT>
1273 SmallMatOutT& apply_local_flux_widened(const SmallMatInT& lambda_values_in,
1274 SmallMatOutT& lambda_values_out,
1275 const unsigned int solution_type,
1276 hyEdgeT& hyper_edge,
1277 const lSol_float_t time,
1278 const unsigned int stage) const
1279 {
1280 static_assert(std::is_same_v<stage_float_t, lSol_float_t>,
1281 "widening bridge only serves a real stage operator");
1282 std::array<std::array<lSol_float_t, n_glob_dofs_per_node()>, 2 * hyEdge_dimT> in_r, out_r;
1283 for (unsigned int i = 0; i < 2 * hyEdge_dimT; ++i)
1284 for (unsigned int j = 0; j < n_glob_dofs_per_node(); ++j)
1285 {
1286 in_r[i][j] = std::real(lambda_values_in[i][j]);
1287 out_r[i][j] = 0.;
1288 }
1289 apply_local_flux(in_r, out_r, solution_type, hyper_edge, time, stage);
1290 for (unsigned int i = 0; i < 2 * hyEdge_dimT; ++i)
1291 for (unsigned int j = 0; j < n_glob_dofs_per_node(); ++j)
1292 lambda_values_out[i][j] += out_r[i][j]; // callers accumulate into zeroed arrays
1293 return lambda_values_out;

1371 /*!***********************************************************************************************
1372 * \brief Local squared L2 error of the dual pair (n, m), or the squared norm of its
1373 * analytic solution (use_discrete == false).
1374 *
1375 * The dual fields are odd under flipping the edge orientation (n = -du/ds - d x r,
1376 * m = -dr/ds with d the axial direction), so the (point, normal, time) component interface
1377 * of the primal error integrals cannot express them: the parameters instead provide the
1378 * vector-valued analytic_result_n/_m(point, axial, time), receiving the edge's axial frame
1379 * vector d = inner_normal(0) (orientation included) and returning the global dual vector.
1380 * Parameters opt in by providing both functions; without them the contribution is zero.
1381 ************************************************************************************************/
1382 template <class hyEdgeT>
1383 lSol_float_t dual_error_sq(hyEdgeT& hyper_edge,
1384 const lSol_float_t time,
1385 const bool use_discrete) const
1386 {
1387 using parameters = parametersT<decltype(hyEdgeT::geometry)::space_dim(), lSol_float_t>;
1388 using point_t = Point<decltype(hyEdgeT::geometry)::space_dim(), lSol_float_t>;
1389 if constexpr (!requires(const point_t& p, const lSol_float_t t) {
1390 parameters::analytic_result_n(p, p, t);
1391 parameters::analytic_result_m(p, p, t);
1392 })
1393 return 0.;
1394 else
1395 {
1396 static_assert(hyEdge_dimT == 1, "dual error is implemented for beam edges only");
1397 // edge frame: component dim <-> {inner_normal(0), outer_normal(0), outer_normal(1)}
1398 std::array<point_t, space_dim> frame;
1399 for (unsigned int dim = 0; dim < space_dim; ++dim)
1400 frame[dim] = edge_frame_vector<hyEdgeT, lSol_float_t>(hyper_edge, -static_cast<int>(dim));
1401
1402 // The endpoint recombination of finalize_step carries an O(dt^2) defect in the algebraic
1403 // dual under time-dependent boundary data, so recover the exact endpoint dual from the
1404 // stored (u, r, lambda) and the Dirichlet data instead of reading the recombined
1405 // coefficients (the recovery recover_dual's doc anticipated for output-time diagnostics).
1406 // Only test problems providing the analytic dual pay for this.
1407 SmallVec<2 * space_dim * n_shape_fct_, lSol_float_t> dual;
1408 if (use_discrete)
1409 {
1410 auto lambda_loc = node_dof_to_edge_dof(hyper_edge.data.lambda_old, hyper_edge);
1411 SmallVec<n_loc_dofs_, lSol_float_t> rhs = assemble_rhs_from_lambda(lambda_loc, hyper_edge);
1412 add_dirichlet_rhs_static(rhs, hyper_edge, time);
1413 dual = compute_dual(hyper_edge, rhs);
1414 }
1415 SmallVec<n_shape_fct_, lSol_float_t> coeffs_n, coeffs_m;
1416 lSol_float_t integral = 0.;
1417 for (unsigned int q = 0; q < integrator::quad_weights.size(); ++q)
1418 {
1419 Point<hyEdge_dimT, lSol_float_t> ref_pt;
1420 ref_pt[0] = integrator::quad_points[q];
1421 const point_t phys = hyper_edge.geometry.map_ref_to_phys(ref_pt);
1422 const auto n_ana = parameters::analytic_result_n(phys, frame[0], time);
1423 const auto m_ana = parameters::analytic_result_m(phys, frame[0], time);
1424 lSol_float_t diff_sq = 0.;
1425 for (unsigned int dim = 0; dim < space_dim; ++dim)
1426 {
1427 lSol_float_t nh = 0., mh = 0.;
1428 if (use_discrete)
1429 {
1430 for (unsigned int i = 0; i < n_shape_fct_; ++i)
1431 {
1432 coeffs_n[i] = dual[coeff_idx(fld_n, dim, i)];
1433 coeffs_m[i] = dual[coeff_idx(fld_m, dim, i)];
1434 }
1435 nh = integrator::shape_fun_t::template lin_comb_fct_val<lSol_float_t>(coeffs_n, ref_pt);
1436 mh = integrator::shape_fun_t::template lin_comb_fct_val<lSol_float_t>(coeffs_m, ref_pt);
1437 }
1438 const lSol_float_t dn = nh - scalar_product(n_ana, frame[dim]);
1439 const lSol_float_t dm = mh - scalar_product(m_ana, frame[dim]);
1440 diff_sq += dn * dn + dm * dm;
1441 }
1442 integral += integrator::quad_weights[q] * diff_sq;
1443 }
1444 return integral * hyper_edge.geometry.area();
1445 }
1446 }

1448 /*!***********************************************************************************************
1449 * \brief Local squared contribution to the L2 error.
1450 *
1451 * \tparam hyEdgeT The geometry type / typename of the considered hyEdge's geometry.
1452 * \param lambda_values The values of the skeletal variable's coefficients.
1453 * \param hyper_edge The geometry of the considered hyperedge (of typename GeomT).
1454 * \param time Time at which analytic functions are evaluated.
1455 * \retval vec_b Local part of vector b.
1456 ************************************************************************************************/
1457 template <class hyEdgeT, typename lambda_float_t = lSol_float_t>
1458 std::array<lSol_float_t, 3U> errors(
1459 const std::array<std::array<lambda_float_t, n_glob_dofs_per_node()>, 2 * hyEdge_dimT>&
1460 /*lambda_values: unused -- the endpoint trace is read from data.lambda_old*/,
1461 hyEdgeT& hyper_edge,
1462 const lSol_float_t time = 0.) const
1463 {
1464 using parameters = parametersT<decltype(hyEdgeT::geometry)::space_dim(), lSol_float_t>;
1465 std::array<lSol_float_t, 3> comps = {1, -1, -2};
1466 std::array<lSol_float_t, n_shape_fct_> coeffs;
1467 std::array<lSol_float_t, n_shape_bdr_> bcoeffs;
1468 lSol_float_t error = 0, trace = 0;
1469 // Weight each endpoint contribution by this edge's length: summed over all edges this
1470 // accumulates w_v = sum_{e per v} l_e at every node, i.e. the HDG skeleton norm
1471 // sum_e l_e ||.||^2_{de} -- a function-space (lumped L2) norm instead of a bare nodal
1472 // l2 sum whose value scales with the node count.
1473 const lSol_float_t len = hyper_edge.geometry.area();
1474 const auto& c_old = hyper_edge.data.coeffs_old;
1475
1476 // The stored endpoint trace is in node-frame (global). Convert to edge-frame so we can
1477 // compare against analytic_result_u/phi evaluated on edge-local normals (comps = {1,-1,-2}).
1478 auto lambda_loc = node_dof_to_edge_dof(hyper_edge.data.lambda_old, hyper_edge);
1479
1480 for (unsigned int dim = 0; dim < space_dim; dim++)
1481 {
1482 for (unsigned int i = 0; i < coeffs.size(); ++i)
1483 coeffs[i] = c_old[coeff_idx(fld_u, dim, i)];
1484 error += integrator::template integrate_vol_diffsquare_discanacomp<
1485 Point<decltype(hyEdgeT::geometry)::space_dim(), lSol_float_t>, decltype(hyEdgeT::geometry),
1486 parameters::analytic_result_u, Point<hyEdge_dimT, lSol_float_t>>(coeffs, comps[dim],
1487 hyper_edge.geometry, time);
1488 }
1489
1490 for (unsigned int dim = 0; dim < space_dim; dim++)
1491 {
1492 for (unsigned int i = 0; i < coeffs.size(); ++i)
1493 coeffs[i] = c_old[coeff_idx(fld_r, dim, i)];
1494 error += integrator::template integrate_vol_diffsquare_discanacomp<
1495 Point<decltype(hyEdgeT::geometry)::space_dim(), lSol_float_t>, decltype(hyEdgeT::geometry),
1496 parameters::analytic_result_phi, Point<hyEdge_dimT, lSol_float_t>>(
1497 coeffs, comps[dim], hyper_edge.geometry, time);
1498 }
1499
1500 // Lambda is zeroed at dynamic-Dirichlet DOFs (see residual_flux), so for those DOFs we
1501 // substitute the projected Dirichlet value here -- otherwise the trace error picks up the
1502 // full analytic value at the boundary. Bit-6 faces (static-only dirichlet) keep their
1503 // lambda since make_initial_from_static fills it with the projected bulk solution.
1504 // For n_shape_bdr_ == 1 the projection coefficient equals dirichlet_value at the face point.
1505 for (unsigned int bdr = 0; bdr < 2 * hyEdge_dimT; ++bdr)
1506 {
1507 const bool dyn_dirichlet =
1508 hyper_edge.node_descriptor[bdr] && !(hyper_edge.node_descriptor[bdr] & (1u << 6));
1509 for (unsigned int dim = 0; dim < space_dim; ++dim)
1510 {
1511 for (unsigned int i = 0; i < bcoeffs.size(); ++i)
1512 {
1513 const unsigned int dof_j = dim * n_shape_bdr_ + i;
1514 if (dyn_dirichlet && (hyper_edge.node_descriptor[bdr] & (1u << dof_j)))
1515 bcoeffs[i] = integrator::template integrate_bdr_phivecfunccomp<
1516 Point<decltype(hyEdgeT::geometry)::space_dim(), lSol_float_t>,
1517 decltype(hyEdgeT::geometry), parameters::dirichlet_value_u,
1518 Point<hyEdge_dimT, lSol_float_t>>(i, bdr, comps[dim], hyper_edge.geometry, time);
1519 else
1520 bcoeffs[i] = lambda_loc[bdr][i + dim * n_shape_bdr_];
1521 }
1522 lSol_float_t contrib = integrator::template integrate_bdr_diffsquare_discanacomp<
1523 Point<decltype(hyEdgeT::geometry)::space_dim(), lSol_float_t>,
1524 decltype(hyEdgeT::geometry), parameters::analytic_result_u,
1525 Point<hyEdge_dimT, lSol_float_t>>(bcoeffs, bdr, comps[dim], hyper_edge.geometry, time);
1526 trace += len * contrib;
1527 }
1528 }
1529
1530 for (unsigned int bdr = 0; bdr < 2 * hyEdge_dimT; ++bdr)
1531 {
1532 const bool dyn_dirichlet =
1533 hyper_edge.node_descriptor[bdr] && !(hyper_edge.node_descriptor[bdr] & (1u << 6));
1534 for (unsigned int dim = 0; dim < space_dim; ++dim)
1535 {
1536 for (unsigned int i = 0; i < bcoeffs.size(); ++i)
1537 {
1538 const unsigned int dof_j = (3 + dim) * n_shape_bdr_ + i;
1539 if (dyn_dirichlet && (hyper_edge.node_descriptor[bdr] & (1u << dof_j)))
1540 bcoeffs[i] = integrator::template integrate_bdr_phivecfunccomp<
1541 Point<decltype(hyEdgeT::geometry)::space_dim(), lSol_float_t>,
1542 decltype(hyEdgeT::geometry), parameters::dirichlet_value_phi,
1543 Point<hyEdge_dimT, lSol_float_t>>(i, bdr, comps[dim], hyper_edge.geometry, time);
1544 else
1545 bcoeffs[i] = lambda_loc[bdr][i + (3 + dim) * n_shape_bdr_];
1546 }
1547 lSol_float_t contrib = integrator::template integrate_bdr_diffsquare_discanacomp<
1548 Point<decltype(hyEdgeT::geometry)::space_dim(), lSol_float_t>,
1549 decltype(hyEdgeT::geometry), parameters::analytic_result_phi,
1550 Point<hyEdge_dimT, lSol_float_t>>(bcoeffs, bdr, comps[dim], hyper_edge.geometry, time);
1551 trace += len * contrib;
1552 }
1553 }
1554
1555 return std::array<lSol_float_t, 3U>({error, trace, dual_error_sq(hyper_edge, time, true)});
1556 }

1560 *
1561 * Evaluates the same integrals as \c errors, but with a vanishing discrete solution, so that
1562 * the result is the squared L2 norm of the analytic solution. This allows to compute relative

1564 *
1565 * \tparam hyEdgeT The geometry type / typename of the considered hyEdge's geometry.

1567 * \param hyper_edge The geometry of the considered hyperedge (of typename GeomT).

1570 ************************************************************************************************/

1577 {

1580 std::array<lSol_float_t, n_shape_fct_> coeffs;
1581 coeffs.fill(0.);
1582 lSol_float_t norm = 0;
1583
1584 for (unsigned int dim = 0; dim < 3; dim++)
1585 {
1586 norm += integrator::template integrate_vol_diffsquare_discanacomp<
1587 Point<decltype(hyEdgeT::geometry)::space_dim(), lSol_float_t>, decltype(hyEdgeT::geometry),

1590 }
1591
1592 for (unsigned int dim = 0; dim < 3; dim++)
1593 {
1594 norm += integrator::template integrate_vol_diffsquare_discanacomp<
1595 Point<decltype(hyEdgeT::geometry)::space_dim(), lSol_float_t>, decltype(hyEdgeT::geometry),
1596 parameters::analytic_result_phi, Point<hyEdge_dimT, lSol_float_t>>(
1597 coeffs, comps[dim], hyper_edge.geometry, time);
1598 }

1601 // the trace error in errors(): zero boundary coefficients turn the diff-square integral
1602 // into the square of the analytic value at the endpoints
1603 std::array<lSol_float_t, n_shape_bdr_> bcoeffs;
1604 bcoeffs.fill(0.);
1605 const lSol_float_t len = hyper_edge.geometry.area();
1606 lSol_float_t trace = 0;
1607 for (unsigned int bdr = 0; bdr < 2 * hyEdge_dimT; ++bdr)
1608 for (unsigned int dim = 0; dim < space_dim; ++dim)
1609 {
1610 trace += len * integrator::template integrate_bdr_diffsquare_discanacomp<
1611 Point<decltype(hyEdgeT::geometry)::space_dim(), lSol_float_t>,
1612 decltype(hyEdgeT::geometry), parameters::analytic_result_u,
1613 Point<hyEdge_dimT, lSol_float_t>>(bcoeffs, bdr, comps[dim],

1616 Point<decltype(hyEdgeT::geometry)::space_dim(), lSol_float_t>,
1617 decltype(hyEdgeT::geometry), parameters::analytic_result_phi,
1618 Point<hyEdge_dimT, lSol_float_t>>(bcoeffs, bdr, comps[dim],
1619 hyper_edge.geometry, time);
1620 }
1621
1622 return std::array<lSol_float_t, 3U>({norm, trace, dual_error_sq(hyper_edge, time, false)});

1625 static constexpr unsigned int n_energy_components() { return 6 * space_dim; }
1626
1627 // Per-edge energy split into 6*space_dim components, ordered (block of size space_dim each):
1628 // 0: ½ ∫ n²/C_n 1: ½ ∫ m²/C_m 2: ½ ∫ v²/C_u 3: ½ ∫ s²/C_r
1629 // 4: ½ τ Σ_bdr ∫ (u-λ_u)² 5: ½ τ Σ_bdr ∫ (r-λ_r)²
1630 template <class hyEdgeT, typename lambda_float_t = lSol_float_t>
1631 std::array<lSol_float_t, n_energy_components()> energy(
1632 const std::array<std::array<lambda_float_t, n_glob_dofs_per_node()>, 2 * hyEdge_dimT>&
1633 /*lambda_values: unused -- the endpoint trace is read from data.lambda_old*/,
1634 hyEdgeT& hyper_edge,
1635 const lSol_float_t /*time*/ = 0.) const
1636 {
1637 std::array<lSol_float_t, n_energy_components()> result;
1638 result.fill(0.);
1639
1640 const auto& c_old = hyper_edge.data.coeffs_old;
1641
1642 auto extra = get_extra_coeffs(hyper_edge);
1643 auto lambda_loc = node_dof_to_edge_dof(hyper_edge.data.lambda_old, hyper_edge);
1644
1645 for (unsigned int d = 0; d < space_dim; ++d)
1646 {
1647 const lSol_float_t Cn = extra[0 * space_dim + d];
1648 const lSol_float_t Cm = extra[1 * space_dim + d];
1649 const lSol_float_t Cu = extra[2 * space_dim + d];
1650 const lSol_float_t Cr = extra[3 * space_dim + d];
1651
1652 lSol_float_t strain_n = 0, strain_m = 0, kin_v = 0, kin_s = 0;
1653 for (unsigned int i = 0; i < n_shape_fct_; ++i)
1654 for (unsigned int j = 0; j < n_shape_fct_; ++j)
1655 {
1656 const lSol_float_t mij =
1657 integrator::template integrate_vol_phiphi<decltype(hyEdgeT::geometry)>(
1658 i, j, hyper_edge.geometry);
1659 strain_n += c_old[coeff_idx(fld_n, d, i)] * c_old[coeff_idx(fld_n, d, j)] * mij;
1660 strain_m += c_old[coeff_idx(fld_m, d, i)] * c_old[coeff_idx(fld_m, d, j)] * mij;
1661 kin_v += c_old[coeff_idx(fld_v, d, i)] * c_old[coeff_idx(fld_v, d, j)] * mij;
1662 kin_s += c_old[coeff_idx(fld_s, d, i)] * c_old[coeff_idx(fld_s, d, j)] * mij;
1663 }
1664
1665 // massless welds have C == 0 -> 0/0 = NaN; zero mass carries zero energy
1666 result[0 * space_dim + d] = Cn > 0. ? 0.5 * strain_n / Cn : 0.;
1667 result[1 * space_dim + d] = Cm > 0. ? 0.5 * strain_m / Cm : 0.;
1668 result[2 * space_dim + d] = Cu > 0. ? 0.5 * kin_v / Cu : 0.;
1669 result[3 * space_dim + d] = Cr > 0. ? 0.5 * kin_s / Cr : 0.;
1670
1671 lSol_float_t hyb_u = 0, hyb_r = 0;
1672 for (unsigned int bdr = 0; bdr < 2 * hyEdge_dimT; ++bdr)
1673 {
1674 // (y - λ)² = y² - 2 y λ + λ² on face bdr, with y = u or r.
1675 for (unsigned int i = 0; i < n_shape_fct_; ++i)
1676 for (unsigned int j = 0; j < n_shape_fct_; ++j)
1677 {
1678 const lSol_float_t mij =
1679 integrator::template integrate_bdr_phiphi<decltype(hyEdgeT::geometry)>(
1680 i, j, bdr, hyper_edge.geometry);
1681 hyb_u += c_old[coeff_idx(fld_u, d, i)] * c_old[coeff_idx(fld_u, d, j)] * mij;
1682 hyb_r += c_old[coeff_idx(fld_r, d, i)] * c_old[coeff_idx(fld_r, d, j)] * mij;
1683 }
1684 for (unsigned int i = 0; i < n_shape_fct_; ++i)
1685 for (unsigned int j = 0; j < n_shape_bdr_; ++j)
1686 {
1687 const lSol_float_t mij =
1688 integrator::template integrate_bdr_phipsi<decltype(hyEdgeT::geometry)>(
1689 i, j, bdr, hyper_edge.geometry);
1690 hyb_u -=
1691 2 * c_old[coeff_idx(fld_u, d, i)] * lambda_loc[bdr][j + d * n_shape_bdr_] * mij;
1692 hyb_r -= 2 * c_old[coeff_idx(fld_r, d, i)] *
1693 lambda_loc[bdr][j + (space_dim + d) * n_shape_bdr_] * mij;
1694 }
1695 // hyEdge_dimT==1 ⇒ trace is 0-dimensional, ψ≡1 ⇒ ∫_∂e λ² = λ² directly.
1696 static_assert(hyEdge_dimT == 1, "trace-square shortcut only valid for hyEdge_dim==1");
1697 const lSol_float_t lam_u = lambda_loc[bdr][d * n_shape_bdr_];
1698 const lSol_float_t lam_r = lambda_loc[bdr][(space_dim + d) * n_shape_bdr_];
1699 hyb_u += lam_u * lam_u;
1700 hyb_r += lam_r * lam_r;
1701 }
1702 result[4 * space_dim + d] = 0.5 * tau_ * hyb_u;
1703 result[5 * space_dim + d] = 0.5 * tau_ * hyb_r;
1704 }
1705
1706 return result;
1707 }

1714 {
1715 if (idx >= 0)
1716 return (Point<space_dim, float_t>)hyper_edge.geometry.inner_normal(idx);
1717 else

1721 template <typename abscissa_float_t, std::size_t sizeT, class input_array_t, class hyEdgeT>
1722 std::array<std::array<lSol_float_t, Hypercube<hyEdge_dimT>::pow(sizeT)>, system_dimension()>
1723 bulk_values(const std::array<abscissa_float_t, sizeT>& abscissas,
1724 const input_array_t& lambda_values,
1725 hyEdgeT& hyper_edge,
1726 const lSol_float_t time = 0.) const
1727 {
1728 constexpr unsigned int n_pts = Hypercube<hyEdge_dimT>::pow(sizeT);
1729 constexpr unsigned int n_fields = 6; // n, m, u, r, v, s
1730
1731 // Read the stored state directly (edge-local frame); the coefficient layout's field index
1732 // coincides with the output field index c: 0=n, 1=m, 2=u, 3=r, 4=v, 5=s.
1733 SmallVec<n_shape_fct_, lSol_float_t> coeffs;
1734 SmallVec<static_cast<unsigned int>(sizeT), abscissa_float_t> helper(abscissas);
1735
1736 // Edge-local-frame component values at each abscissa, per field.
1737 // shape: [field][local_component][pt]
1738 std::array<std::array<std::array<lSol_float_t, n_pts>, space_dim>, n_fields> point_vals{};
1739
1740 for (unsigned int c = 0; c < n_fields; ++c)
1741 for (unsigned int dim = 0; dim < space_dim; ++dim)
1742 {
1743 for (unsigned int i = 0; i < coeffs.size(); ++i)
1744 coeffs[i] = hyper_edge.data.coeffs_old[coeff_idx(c, dim, i)];
1745 for (unsigned int pt = 0; pt < n_pts; ++pt)
1746 point_vals[c][dim][pt] = integrator::shape_fun_t::template lin_comb_fct_val<float>(
1747 coeffs, Hypercube<hyEdge_dimT>::template tensorial_pt<Point<hyEdge_dimT>>(pt, helper));
1748 }
1749
1750 std::array<std::array<lSol_float_t, n_pts>, system_dimension()> result{};
1751
1752 static_assert(2 <= n_fields);
1753
1754 // fields n m should be in local frame -> stay
1755 for (unsigned int c = 0; c < 2; ++c)
1756 for (unsigned int local = 0; local < space_dim; ++local)
1757 for (unsigned int q = 0; q < n_pts; ++q)
1758 result[c * space_dim + local][q] = point_vals[c][local][q];
1759
1760 // fields u r should be in global frame -> need to transform
1761 for (unsigned int c = 2; c < n_fields; ++c)
1762 for (unsigned int local = 0; local < space_dim; ++local)
1763 {
1764 const int idx = -static_cast<int>(local); // 0, -1, -2
1765 Point<space_dim, lSol_float_t> nv =
1766 edge_frame_vector<hyEdgeT, lSol_float_t>(hyper_edge, idx);
1767 for (unsigned int dim = 0; dim < space_dim; ++dim)
1768 for (unsigned int q = 0; q < n_pts; ++q)
1769 result[c * space_dim + dim][q] += point_vals[c][local][q] * nv[dim];
1770 }
1771
1772 return result;
1773 }

1775 template <class hyEdgeT>
1776 SmallVec<4 * space_dim, lSol_float_t> get_extra_coeffs(hyEdgeT& hyper_edge) const
1777 {
1778 SmallVec<4 * space_dim, lSol_float_t> extra_coeffs(1.); // C_n, C_m, C_u, C_r
1779
1780 if (!hyper_edge.geometry.has_extra_data())
1781 return extra_coeffs;
1782
1783 // 17 quantities associated with each fiber
1784 // column header description
1785 // 0 mass, mass
1786 // 1, 2, 3 EA,kG_1A,kG_2A, displacement stiffness
1787 // 4, 5, 6 G_xI_x,E_1I_1,E_2I_2, rotation stiffness
1788 // 7, 8, 9 n_11,n_12,n_13, normal 1
1789 // 10,11,12 n_21,n_22,n_23, normal 2
1790 // 13,14 width1,width2, widths in direction of normals
1791 // 15 fiber_id The fiber the edge is part of (-1 if a connection)
1792 // 16 fiber_edge_id Where along the fiber the edge is (-1 if a connection)
1793 // material coefficients are given in the tangent,normal1,normal2 basis
1794 // so must be transformed into local basis chosen by HyperHDG, tangent coincides upto sign
1795
1796 auto extra_data = hyper_edge.geometry.extra_data();
1797 hy_check(extra_data.size() == 17,
1798 "timowave expected 17 material coefficients, found " << extra_data.size());
1799
1800 lSol_float_t mass = extra_data[0];
1801 SmallVec<space_dim, lSol_float_t> normal1 =
1802 std::array<lSol_float_t, space_dim>{{extra_data[7], extra_data[8], extra_data[9]}};
1803 SmallVec<space_dim, lSol_float_t> normal2 =
1804 std::array<lSol_float_t, space_dim>{{extra_data[10], extra_data[11], extra_data[12]}};
1805 SmallVec<space_dim, lSol_float_t> outer1 = hyper_edge.geometry.outer_normal(0);
1806 SmallVec<space_dim, lSol_float_t> outer2 = hyper_edge.geometry.outer_normal(1);
1807
1808 auto length = hyper_edge.geometry.area();
1809 auto density = mass / (extra_data[13] * extra_data[14] * length);
1810
1811 auto w1 = extra_data[13];
1812 auto w2 = extra_data[14];
1813
1814 // approximation to second moment in direction of normal1, normal2
1815 // valid for rectangular cross section
1816 auto moment1 = w1 * w1 * w1 * w2 / 12;
1817 auto moment2 = w1 * w2 * w2 * w2 / 12;
1818
1819 // displacement stiffness C_n
1820 extra_coeffs[0] = extra_data[1];
1821 extra_coeffs[1] = extra_data[2] * scalar_product(outer1, normal1) +
1822 extra_data[3] * scalar_product(outer1, normal2);
1823 extra_coeffs[2] = extra_data[2] * scalar_product(outer2, normal1) +
1824 extra_data[3] * scalar_product(outer2, normal2);
1825
1826 // rotation stiffness C_m
1827 extra_coeffs[3] = extra_data[4];
1828 extra_coeffs[4] = extra_data[5] * scalar_product(outer1, normal1) +
1829 extra_data[6] * scalar_product(outer1, normal2);
1830 extra_coeffs[5] = extra_data[5] * scalar_product(outer2, normal1) +
1831 extra_data[6] * scalar_product(outer2, normal2);
1832
1833 // displacement inertia C_u
1834 extra_coeffs[6] = mass / length;
1835 extra_coeffs[7] = mass / length;
1836 extra_coeffs[8] = mass / length;
1837
1838 // rotation inertia
1839 extra_coeffs[9] = density * (moment1 + moment2);
1840 extra_coeffs[10] = density * (moment1 * scalar_product(outer1, normal1) +
1841 moment2 * scalar_product(outer1, normal2));
1842 extra_coeffs[11] = density * (moment1 * scalar_product(outer2, normal1) +
1843 moment2 * scalar_product(outer2, normal2));
1844
1845 for (unsigned int i = 0; i < extra_coeffs.size(); i++)
1846 {
1847 extra_coeffs[i] = std::abs(extra_coeffs[i]);
1848 // hy_check(std::isfinite(extra_coeffs[i]), "get_extra finite coeffs " << extra_coeffs);
1849 // hy_check(extra_coeffs[i] > 0, "get_extra zero coeffs " << extra_coeffs << " mass " <<
1850 // mass);
1851 }
1852
1853 // hy_check(false, "extra_coeffs " << extra_coeffs);
1854
1855 return extra_coeffs;
1856 }

1858 /*!***********************************************************************************************
1859 * \brief Solve the stage-local problem from the stage trace zeta and stash its solution.
1860 *
1861 * \c lambda_values_in is the stage trace zeta of the Gauss step protocol; the stage-local
1862 * solution (q,y,z) is stashed in slot \c stage, the state advances at stage == -1 (idempotent:
1863 * re-calling overwrites the slot). The sentinel stage == -1 means "all stages are in": the
1864 * trace argument is ignored and the stashed stage solutions recombine into the new state
1865 * (see finalize_step).
1866 ************************************************************************************************/
1867 template <class hyEdgeT, typename lambda_float_t, typename time_t = lSol_float_t>
1868 void set_data(const std::array<std::array<lambda_float_t, 2 * n_shape_bdr_ * space_dim>,
1869 2 * hyEdge_dimT>& lambda_values_in,
1870 hyEdgeT& hyper_edge,
1871 const time_t time = 0.) const
1872 {
1873 const auto [t, stage] = split_stage_time(time);
1874 if (stage < 0)
1875 return finalize_step(hyper_edge);
1876 if constexpr (std::is_same_v<lambda_float_t, stage_float_t>)
1877 {
1878 hyper_edge.data.stage_trace[stage] = lambda_values_in;
1879 auto lambda_values = node_dof_to_edge_dof(lambda_values_in, hyper_edge);
1880 hyper_edge.data.stage_coeffs[stage] =
1881 solve_local_problem(lambda_values, 1U, hyper_edge, t, stage);
1882 }
1883 else if constexpr (std::is_same_v<lambda_float_t, std::complex<lSol_float_t>> &&
1884 std::is_same_v<stage_float_t, lSol_float_t>)
1885 {
1886 // complex dof vectors over the real s = 1 operator (complex-PETSc build): the stage trace
1887 // is real-valued by construction; narrow and recurse into the real branch

1900 /*!***********************************************************************************************
1901 * \brief Gauss protocol: combine the stashed stage solutions into the new state.
1902 *
1903 * Endpoint update y+ = stage_affine_ * y^n + sum_l Re(w_l y_l); at s = 1 that is
1904 * y+ = 2 y_1 - y^n. q = (n, m) is algebraic and linear in (y, lambda), so the same extrapolation
1905 * is exact for it at s = 1 (multi-stage recovers q via recover_dual instead). Explicitly driver-
1906 * called once all stage set_data calls are in -- no per-edge completion tally. Reached either
1907 * directly (legacy loops) or through set_data with the stage == -1 sentinel.
1908 ************************************************************************************************/
1909 template <class hyEdgeT>
1910 void finalize_step(hyEdgeT& hyper_edge) const
1911 {
1912 auto& data = hyper_edge.data;
1913 data.coeffs_old *= stages_.affine;
1914 for (unsigned int l = 0; l < n_reps_; ++l)
1915 for (unsigned int i = 0; i < n_loc_dofs_; ++i)
1916 data.coeffs_old[i] += stages_.mult[l] * std::real(stages_.w[l] * data.stage_coeffs[l][i]);
1917 // the endpoint trace is the same recombination of the stashed stage traces (exact at s = 1,
1918 // Gauss-quadrature superconvergent otherwise); real by conjugate symmetry
1919 for (unsigned int face = 0; face < 2 * hyEdge_dimT; ++face)
1920 for (unsigned int dof = 0; dof < n_glob_dofs_per_node(); ++dof)
1921 {
1922 lSol_float_t val = stages_.affine * data.lambda_old[face][dof];
1923 for (unsigned int l = 0; l < n_reps_; ++l)
1924 val += stages_.mult[l] * std::real(stages_.w[l] * data.stage_trace[l][face][dof]);
1925 data.lambda_old[face][dof] = val;
1926 }
1927 }

1972 /*!***********************************************************************************************
1973 * \brief Recover the algebraic dual (n, m) from the stored (u, r) and static boundary data.
1974 *
1975 * The constitutive rows of the static system read (M/C) q - G w (+ cross terms) = b_q, so
1976 * q = C * M^{-1} * (b_q - (A_static w)_q). \c rhs must hold the q-row boundary data (lambda +
1977 * Dirichlet contributions); only its first 2*space_dim blocks are read. The dual is algebraic --
1978 * slaved to (y, lambda) at the same instant -- so the same recovery serves the initializers now
1979 * and the Gauss output-time diagnostics later. (This also fixes the formerly missing material
1980 * factor C: for unit coefficients, i.e. the wave4-type tests, nothing changes.)
1981 ************************************************************************************************/
1982 template <class hyEdgeT>
1983 SmallVec<2 * space_dim * n_shape_fct_, lSol_float_t> compute_dual(
1984 hyEdgeT& hyper_edge,
1985 const SmallVec<n_loc_dofs_, lSol_float_t>& rhs) const
1986 {
1987 const auto& data = hyper_edge.data;
1988 SmallVec<4 * space_dim * n_shape_fct_, lSol_float_t> coeffs_w;
1989 for (unsigned int i = 0; i < n_shape_fct_; ++i)
1990 for (unsigned int dim = 0; dim < space_dim; ++dim)
1991 {
1992 coeffs_w[coeff_idx(fld_u, dim, i)] = data.coeffs_old[coeff_idx(fld_u, dim, i)];
1993 coeffs_w[coeff_idx(fld_r, dim, i)] = data.coeffs_old[coeff_idx(fld_r, dim, i)];
1994 }
1995 const auto res = assemble_loc_matrix<false>(hyper_edge) * coeffs_w;
1996 const auto extra = get_extra_coeffs(hyper_edge); // C_n, C_m in components 0 .. 2*space_dim-1
1997
1998 SmallSquareMat<n_shape_fct_> mmat;
1999 for (unsigned int i = 0; i < n_shape_fct_; ++i)
2000 for (unsigned int j = 0; j < n_shape_fct_; ++j)
2001 mmat(i, j) = integrator::integrate_vol_phiphi(i, j, hyper_edge.geometry);
2002
2003 SmallVec<2 * space_dim * n_shape_fct_, lSol_float_t> dual;
2004 SmallVec<n_shape_fct_> temp;
2005 for (unsigned int d = 0; d < 2 * space_dim; ++d)
2006 {
2007 for (unsigned int i = 0; i < n_shape_fct_; ++i)
2008 temp[i] = rhs[d * n_shape_fct_ + i] - res[d * n_shape_fct_ + i];
2009 temp = temp / mmat;
2010 // d spans the q-rows, i.e. exactly the (n, m) blocks leading the coefficient layout
2011 for (unsigned int i = 0; i < n_shape_fct_; ++i)
2012 dual[d * n_shape_fct_ + i] = extra[d] * temp[i];
2013 }
2014 return dual;
2015 }

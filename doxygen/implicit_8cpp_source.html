16// Note: nb_type_get_implicit() reads the conversion arrays grown below without
17// holding the internals lock. This is safe because conversions are registered
18// while binding a type, which never overlaps with concurrent use of that type.
19
20void implicitly_convertible(const std::type_info *src,
21 const std::type_info *dst) noexcept {
22 nb_internals *internals_ = internals;
23 type_data *t = nb_type_c2p(internals_, dst);
24 check(t, "nanobind::detail::implicitly_convertible(src=%s, dst=%s): "
25 "destination type unknown!", type_name(src), type_name(dst));
26
27 lock_internals guard(internals_);
28 size_t size = 0;
29
30 if (t->flags & (uint32_t) type_flags::has_implicit_conversions) {
31 while (t->implicit.cpp && t->implicit.cpp[size])
32 size++;
33 } else {
34 t->implicit.cpp = nullptr;
35 t->implicit.py = nullptr;
36 t->flags |= (uint32_t) type_flags::has_implicit_conversions;
37 }
38
39 void **data = (void **) PyMem_Malloc(sizeof(void *) * (size + 2));
40 check(data, "nanobind::detail::implicitly_convertible(): out of memory!");
41
42 if (size)
43 memcpy(data, t->implicit.cpp, size * sizeof(void *));
44 data[size] = (void *) src;
45 data[size + 1] = nullptr;
46 PyMem_Free(t->implicit.cpp);
47 t->implicit.cpp = (decltype(t->implicit.cpp)) data;
48}

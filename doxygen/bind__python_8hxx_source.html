3// Explicit nanobind bindings for a global-loop instantiation.
4//
5// Usage: write a tiny module .cxx that instantiates the global loop you actually need and
6// registers it,
7//
8// NB_MODULE(my_module, m) { HyperHDG::bind_python<MyGlobalLoop>(m, "MyGlobalLoop"); }
9//
10// then build it with nanobind_add_module (python/hyperhdg.py generates such a CMake project and
11// builds it on demand, cf. python/README.md). This is an alternative
12// to the runtime cython compile machinery (import/ + cython/) for workflows that want full
13// control over the compile: the set of template instantiations is spelled out in the .cxx,
14// CMake produces a plain .so, python imports it.
15//
16// Every method is guarded with `if constexpr (requires ...)`, so the helper binds whatever
17// subset of the global-loop protocol the instantiation provides (an elliptic loop gets no
18// set_data/make_initial, an eigenvalue loop no residual_flux, ...) -- one helper for all
19// loops instead of a .pyx/.pxd template pair per loop. Vectors cross the boundary as
20// std::vector copies; the condensed system crosses as a COO triplet of NumPy arrays consumed by
21// scipy.sparse.
22

24// feature-test macros, else pyconfig.h redefines _POSIX_C_SOURCE/_XOPEN_SOURCE

42/*!*************************************************************************************************
43 * \brief Large vector type of a global loop: what zero_vector() returns, else the default.
44 *
45 * The loops keep \c LargeVecT private, but every loop that owns global vectors hands one out
46 * through \c zero_vector(). Eigenvalue loops do not; for those the default matches the
47 * default template argument of the loops themselves.
48 **************************************************************************************************/
49template <typename GlobalLoopT>
50struct large_vec
51{
52 using type = std::vector<double>;
53};
54template <typename GlobalLoopT>
55 requires requires(GlobalLoopT o) { o.zero_vector(); }
56struct large_vec<GlobalLoopT>
57{
58 using type = std::remove_cvref_t<decltype(std::declval<GlobalLoopT&>().zero_vector())>;
59};
60template <typename GlobalLoopT>
61using large_vec_t = typename large_vec<GlobalLoopT>::type;
62} // end of namespace detail

64/*!*************************************************************************************************
65 * \brief Bind a global-loop instantiation as a python class.
66 *
67 * \tparam GlobalLoopT The instantiated global loop, e.g. GlobalLoop::Elliptic<...>.
68 * \tparam TopoArgT Constructor argument of topology (and geometry): std::string for the
69 * file based classes, std::vector<unsigned int> for the cubic ones.
70 * \tparam SolverArgT Constructor argument of the local solver, e.g. the penalty tau.
71 * \tparam LargeVecT Global vector type; deduced from zero_vector() where available.
72 * \param m The nanobind module the class is added to.
73 * \param name Name of the class on the python side.
74 * \retval cls The nanobind class, so callers can add their own methods.
75 **************************************************************************************************/

87 // COO triplet (rows, cols, values) of NumPy arrays, ready for
88 // scipy.sparse.coo_matrix((vals, (rows, cols))): the loop emits one entry per local matrix
89 // position, duplicates are summed by scipy. The arrays take over the vectors (no copy).
90 auto to_numpy = []<typename T>(std::vector<T>&& vec)
91 {
92 auto* owned = new std::vector<T>(std::move(vec));
93 nb::capsule owner(owned, [](void* ptr) noexcept { delete static_cast<std::vector<T>*>(ptr); });
94 return nb::ndarray<nb::numpy, T, nb::ndim<1>>(owned->data(), {owned->size()}, owner);
95 };
96 auto to_coo = [to_numpy](auto&& mat)
97 {
98 return std::make_tuple(to_numpy(std::move(mat.row_vec)), to_numpy(std::move(mat.col_vec)),
99 to_numpy(std::move(mat.value_vec)));
100 };

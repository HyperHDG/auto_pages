27// Casts a std::chrono type (either a duration or a time_point) to/from
28// Python timedelta objects, or from a Python float representing seconds.
29template <typename type> class duration_caster {
30public:
31 using rep = typename type::rep;
32 using period = typename type::period;
33 using duration_t = std::chrono::duration<rep, period>;
34
35 bool from_python(handle src, uint8_t /*flags*/, cleanup_list*) noexcept {
36 namespace ch = std::chrono;
37
38 if (!src) return false;
39
40 // support for signed 25 bits is required by the standard
41 using days = ch::duration<int_least32_t, std::ratio<86400>>;
42
43 // If invoked with datetime.delta object, unpack it
44 int dd, ss, uu;
45 try {
46 if (unpack_timedelta(src.ptr(), &dd, &ss, &uu)) {
47 value = type(ch::duration_cast<duration_t>(
48 days(dd) + ch::seconds(ss) + ch::microseconds(uu)));
49 return true;
50 }
51 } catch (python_error& e) {
52 e.discard_as_unraisable(src.ptr());
53 return false;
54 }
55
56 // If invoked with a float we assume it is seconds and convert
57 int is_float;
58#if defined(Py_LIMITED_API)
59 is_float = PyType_IsSubtype(Py_TYPE(src.ptr()), &PyFloat_Type);
60#else
61 is_float = PyFloat_Check(src.ptr());
62#endif
63 if (is_float) {
64 value = type(ch::duration_cast<duration_t>(
65 ch::duration<double>(PyFloat_AsDouble(src.ptr()))));
66 return true;
67 }
68 return false;
69 }
70
71 // If this is a duration just return it back
72 static const duration_t& get_duration(const duration_t& src) {
73 return src;
74 }
75
76 // If this is a time_point get the time_since_epoch
77 template <typename Clock>
78 static duration_t get_duration(
79 const std::chrono::time_point<Clock, duration_t>& src) {
80 return src.time_since_epoch();
81 }
82
83 static handle from_cpp(const type& src, rv_policy, cleanup_list*) noexcept {
84 namespace ch = std::chrono;
85
86 // Use overloaded function to get our duration from our source
87 // Works out if it is a duration or time_point and get the duration
88 auto d = get_duration(src);
89
90 // Declare these special duration types so the conversions happen with the correct primitive types (int)
91 using dd_t = ch::duration<int, std::ratio<86400>>;
92 using ss_t = ch::duration<int, std::ratio<1>>;
93 using us_t = ch::duration<int, std::micro>;
94
95 auto dd = ch::duration_cast<dd_t>(d);
96 auto subd = d - dd;
97 auto ss = ch::duration_cast<ss_t>(subd);
98 auto us = ch::duration_cast<us_t>(subd - ss);
99 return pack_timedelta(dd.count(), ss.count(), us.count());
100 }
101
102 NB_TYPE_CASTER(type, io_name("datetime.timedelta | float",
103 "datetime.timedelta"))
104};

116template <class Time, class Buf>
117inline std::tm *localtime_thread_safe(const Time *time, Buf *buf) {
118 if constexpr (decltype(can_localtime_s(time, buf))::value) {
119 // C11 localtime_s
120 std::tm* ret = localtime_s(time, buf);
121 return ret;
122 } else if constexpr (decltype(can_localtime_s(buf, time))::value) {
123 // Microsoft localtime_s (with parameters switched and errno_t return)
124 int ret = localtime_s(buf, time);
125 return ret == 0 ? buf : nullptr;
126 } else {
127 static_assert(decltype(can_localtime_r(time, buf))::value,
128 "<nanobind/stl/chrono.h> type caster requires "
129 "that your C library support localtime_r or localtime_s");
130 std::tm* ret = localtime_r(time, buf);
131 return ret;
132 }
133}

135// Cast between times on the system clock and datetime.datetime instances
136// (also supports datetime.date and datetime.time for Python->C++ conversions)
137template <typename Duration>
138class type_caster<std::chrono::time_point<std::chrono::system_clock, Duration>> {
139public:
140 using type = std::chrono::time_point<std::chrono::system_clock, Duration>;
141 bool from_python(handle src, uint8_t /*flags*/, cleanup_list*) noexcept {
142 namespace ch = std::chrono;
143
144 if (!src)
145 return false;
146
147 std::tm cal;
148 ch::microseconds msecs;
149 int yy, mon, dd, hh, min, ss, uu;
150 try {
151 if (!unpack_datetime(src.ptr(), &yy, &mon, &dd,
152 &hh, &min, &ss, &uu)) {
153 return false;
154 }
155 } catch (python_error& e) {
156 e.discard_as_unraisable(src.ptr());
157 return false;
158 }
159 cal.tm_sec = ss;
160 cal.tm_min = min;
161 cal.tm_hour = hh;
162 cal.tm_mday = dd;
163 cal.tm_mon = mon - 1;
164 cal.tm_year = yy - 1900;
165 cal.tm_isdst = -1;
166 msecs = ch::microseconds(uu);
167 value = ch::time_point_cast<Duration>(
168 ch::system_clock::from_time_t(std::mktime(&cal)) + msecs);
169 return true;
170 }
171
172 static handle from_cpp(const type& src, rv_policy, cleanup_list*) noexcept {
173 namespace ch = std::chrono;
174
175 // Get out microseconds, and make sure they are positive, to
176 // avoid bug in eastern hemisphere time zones
177 // (cfr. https://github.com/pybind/pybind11/issues/2417). Note
178 // that if us_t is 32 bits and we get a time_point that also
179 // has a 32-bit time_since_epoch (perhaps because it's
180 // measuring time in minutes or something), then writing `src
181 // - us` below can lead to overflow based on how common_type
182 // is defined on durations. Defining us_t to store 64-bit
183 // microseconds works around this.
184 using us_t = ch::duration<std::int64_t, std::micro>;
185 auto us = ch::duration_cast<us_t>(src.time_since_epoch() %
186 ch::seconds(1));
187 if (us.count() < 0)
188 us += ch::seconds(1);
189
190 // Subtract microseconds BEFORE `system_clock::to_time_t`, because:
191 // > If std::time_t has lower precision, it is implementation-defined
192 // whether the value is rounded or truncated.
193 // (https://en.cppreference.com/w/cpp/chrono/system_clock/to_time_t)
194 std::time_t tt = ch::system_clock::to_time_t(
195 ch::time_point_cast<ch::system_clock::duration>(src - us));
196
197 std::tm localtime;
198 if (!localtime_thread_safe(&tt, &localtime)) {
199 PyErr_Format(PyExc_ValueError,
200 "Unable to represent system_clock in local time; "
201 "got time_t %ld", static_cast<std::int64_t>(tt));
202 return handle();
203 }
204 return pack_datetime(localtime.tm_year + 1900,
205 localtime.tm_mon + 1,
206 localtime.tm_mday,
207 localtime.tm_hour,
208 localtime.tm_min,
209 localtime.tm_sec,
210 (int) us.count());
211 }
212 NB_TYPE_CASTER(type, io_name("datetime.datetime | datetime.date | datetime.time",
213 "datetime.datetime"))
214};

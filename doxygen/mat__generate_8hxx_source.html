33/*!*************************************************************************************************
34 * \brief Forward exceptions thrown in the body of a parallel loop to the calling thread.
35 *
36 * An exception that leaves an OpenMP parallel region terminates the program. Therefore, the body of
37 * a parallel loop is executed via \c run, which catches all exceptions. Once an exception has been
38 * caught, the remaining iterations are skipped, and \c rethrow (called after the loop) rethrows the
39 * first caught exception. Without OpenMP, this results in the same behavior as a serial loop.
40 **************************************************************************************************/
41class ParallelExceptions
42{
43 public:
44 /*!***********************************************************************************************
45 * \brief Execute \c body unless an exception has been thrown before, and catch exceptions.
46 ************************************************************************************************/
47 template <typename BodyT>
48 void run(BodyT&& body) noexcept
49 {
50 if (failed_.load(std::memory_order_relaxed))
51 return;
52 try
53 {
54 body();
55 }
56 catch (...)
57 {
58 HYPERHDG_OMP_CRITICAL
59 {
60 if (!exception_)
61 exception_ = std::current_exception();
62 }
63 failed_.store(true, std::memory_order_relaxed);
64 }
65 }
66 /*!***********************************************************************************************
67 * \brief Rethrow the first caught exception (if any).
68 ************************************************************************************************/
69 void rethrow()
70 {
71 if (exception_)
72 std::rethrow_exception(exception_);
73 }
74
75 private:
76 std::atomic<bool> failed_{false};
77 std::exception_ptr exception_;
78};

82/*!*************************************************************************************************
83 * \brief Throw if the degrees of freedom of a hypergraph cannot be indexed by \c unsigned \c int.
84 *
85 * The global loops index degrees of freedom by \c unsigned \c int. Without this check, too large
86 * hypergraphs would result in silently wrapped indices.

89void check_dof_index_range(const HyperGraphT& hyper_graph)
90{
91 const std::size_t n_dofs =

102/*!*************************************************************************************************
103 * \brief Local condensed matrix of a hyperedge, i.e., the matrix representation of trace_to_flux.
104 *
105 * If the local solver provides \c trace_to_flux_mat(hyper_edge, time), which returns the local
106 * condensed matrix as a \c SmallSquareMat, this function is used. Local solvers can implement it by
107 * factorizing their local matrix once and reusing the factorization for all unit vectors (cf.
108 * local_solver/trace_to_flux_mat.hxx). Otherwise, the local condensed matrix is obtained by
109 * applying \c trace_to_flux to all local unit vectors.
110 *
111 * \param local_solver The local solver.
112 * \param hyper_edge The hyperedge.
113 * \param time Parameter of \c trace_to_flux (time or shift). It is passed through to the
114 * local solver, so it may also be of a non-scalar type.
115 * \param loc_mat Array of \f$n^2\f$ entries that is filled with the local matrix in
116 * column-major order, i.e., entry (row, col) is \c loc_mat[col * n + row].
117 **************************************************************************************************/
118template <typename LocalSolverT, typename hyEdgeT, typename timeT, typename dof_value_t>
119void local_condensed_matrix(const LocalSolverT& local_solver,
120 hyEdgeT& hyper_edge,
121 const timeT& time,
122 dof_value_t* loc_mat)
123{
124 constexpr unsigned int hyEdge_dim = LocalSolverT::hyEdge_dim();
125 constexpr unsigned int n_dofs_per_node = LocalSolverT::n_glob_dofs_per_node();
126 constexpr unsigned int n_loc = 2 * hyEdge_dim * n_dofs_per_node;
127 using dofs_t = std::array<std::array<dof_value_t, n_dofs_per_node>, 2 * hyEdge_dim>;
128
129 if constexpr (requires { local_solver.trace_to_flux_mat(hyper_edge, prvalue_of(time)); })
130 {
131 const auto helper = local_solver.trace_to_flux_mat(hyper_edge, time);
132 for (unsigned int col = 0; col < n_loc; ++col)
133 for (unsigned int row = 0; row < n_loc; ++row)
134 loc_mat[col * n_loc + row] = helper(row, col);
135 }
136 else
137 {
138 dofs_t dofs_old, dofs_new;
139 for (unsigned int node_j = 0; node_j < 2 * hyEdge_dim; ++node_j)
140 for (unsigned int dof_j = 0; dof_j < n_dofs_per_node; ++dof_j)
141 {
142 for (unsigned int node = 0; node < 2 * hyEdge_dim; ++node)
143 {
144 dofs_old[node].fill(0.);
145 dofs_new[node].fill(0.);
146 }
147 dofs_old[node_j][dof_j] = 1.;
148 if constexpr (requires {
149 local_solver.trace_to_flux(dofs_old, dofs_new, prvalue_of(time));
150 })
151 local_solver.trace_to_flux(dofs_old, dofs_new, time);
152 else if constexpr (requires {
153 local_solver.trace_to_flux(dofs_old, dofs_new, hyper_edge,
154 prvalue_of(time));
155 })
156 local_solver.trace_to_flux(dofs_old, dofs_new, hyper_edge, time);
157 else
158 static_assert(always_false_v<LocalSolverT, hyEdgeT>,
159 "LocalSolverT implements no usable overload of trace_to_flux!");
160
161 const unsigned int col = node_j * n_dofs_per_node + dof_j;
162 for (unsigned int node_i = 0; node_i < 2 * hyEdge_dim; ++node_i)
163 for (unsigned int dof_i = 0; dof_i < n_dofs_per_node; ++dof_i)
164 loc_mat[col * n_loc + node_i * n_dofs_per_node + dof_i] = dofs_new[node_i][dof_i];
165 }
166 }

169/*!*************************************************************************************************
170 * \brief Throw if a local matrix differs from the local matrix reused by \c factorize_once.
171 **************************************************************************************************/
172template <typename dof_value_t>
173void check_reused_local_matrix(const dof_value_t* computed,
174 const dof_value_t* reused,
175 const std::size_t size,
176 const std::size_t edge)
177{
178 // Absolute values are real also for complex dof values.
179 using real_t = decltype(std::abs(std::declval<dof_value_t>()));
180 real_t max_entry = 0., max_diff = 0.;
181 for (std::size_t k = 0; k < size; ++k)
182 {
183 max_entry = std::max(max_entry, (real_t)std::abs(reused[k]));
184 max_diff = std::max(max_diff, (real_t)std::abs(computed[k] - reused[k]));
185 }
186 const real_t tolerance =
187 std::max((real_t)1e-10, (real_t)1e3 * std::numeric_limits<real_t>::epsilon());
188 if (!(max_diff <= tolerance * max_entry)) // Also true if max_diff is NaN.
189 {
190 std::stringstream msg;
191 msg << "Option factorize_once requires all hyperedges with the same face types to have the "
192 << "same local matrix (e.g. uniform meshes and coefficients that are constant in space), "
193 << "but the local matrix of hyperedge " << edge << " differs by " << max_diff
194 << " (maximum entry " << max_entry << "). Use factorize_once = false instead!";
195 throw std::invalid_argument(msg.str());
196 }
197}

199/*!*************************************************************************************************
200 * \brief Compute the local condensed matrix of every hyperedge and hand it to \c sink.
201 *
202 * For every hyperedge \c edge, \c sink(edge, hyNodes, loc_mat) is called, where \c hyNodes are the
203 * indices of the hypernodes of the hyperedge and \c loc_mat points to the local condensed matrix
204 * (cf. \c local_condensed_matrix for its layout).
205 *
206 * If \c factorize_once is \c true, the local matrix is only computed once for every combination of
207 * face (hypernode) types, i.e., for every combination of boundary conditions, and reused for all
208 * other hyperedges with the same face types. This is only correct if all these hyperedges have the
209 * same local matrix, e.g. for uniform meshes with constant (in space) coefficients. It is wrong if,
210 * for example, the diffusion tensor is space dependent or the hyperedges differ in size or shape.
211 * Therefore, the local matrices of some hyperedges (the last one of every combination of face types
212 * and up to 16 hyperedges evenly spread over all hyperedges, at most one per 32 hyperedges) are
213 * recomputed and compared to the reused ones, and an \c std::invalid_argument is thrown if they
214 * differ. This costs at most one local matrix per combination of face types plus 16.
215 * If the code is compiled without \c NDEBUG, this check is done for every hyperedge (which is as
216 * expensive as \c factorize_once being false).
217 *
218 * The local matrices are computed in parallel if HyperHDG is compiled with OpenMP. Then, \c sink
219 * is called concurrently for different hyperedges. Exceptions thrown by the local solvers are
220 * forwarded to the caller.
221 *
222 * \tparam hyNode_index_t Typename of the hypernode index.
223 * \tparam dof_value_t Typename of the entries of the local matrices.
224 * \param hyper_graph The hypergraph.
225 * \param local_solver The local solver.
226 * \param time Parameter of \c trace_to_flux (time or shift, cf.
227 *local_condensed_matrix). \param factorize_once Compute the local matrix only once per
228 *combination of face types. \param sink Callable that receives the local matrices.
229 **************************************************************************************************/
230template <typename hyNode_index_t,
231 typename dof_value_t,
232 typename HyperGraphT,
233 typename LocalSolverT,
234 typename timeT,
235 typename SinkT>
236void for_each_local_matrix(HyperGraphT& hyper_graph,
237 const LocalSolverT& local_solver,
238 const timeT& time,
239 const bool factorize_once,
240 SinkT&& sink)
241{
242 constexpr unsigned int hyEdge_dim = HyperGraphT::hyEdge_dim();
243 constexpr unsigned int n_dofs_per_node = LocalSolverT::n_glob_dofs_per_node();
244 constexpr std::size_t loc_size =
245 (2 * hyEdge_dim * n_dofs_per_node) * (2 * hyEdge_dim * n_dofs_per_node);
246 using face_types_t = std::array<unsigned int, 2 * hyEdge_dim>;
247 using hyNodes_t = SmallVec<2 * hyEdge_dim, hyNode_index_t>;
248 const std::size_t n_hyEdges = hyper_graph.n_hyEdges();
249
250 // factorize_once: Compute the local matrix for every combination of face types (serially), and
251 // check that it can be reused for a sample of hyperedges.
252 std::map<face_types_t, std::vector<dof_value_t> > loc_mats_by_face_types;
253 std::vector<const dof_value_t*> loc_mat_of_edge(factorize_once ? n_hyEdges : 0, nullptr);
254 if (factorize_once)
255 {
256 std::vector<std::size_t> samples, first_edges;
257 std::map<face_types_t, std::size_t> last_edge_of_face_types;
258 for (std::size_t edge = 0; edge < n_hyEdges; ++edge)
259 {
260 auto hyper_edge = hyper_graph[edge];
261 face_types_t face_types;
262 for (unsigned int face = 0; face < 2 * hyEdge_dim; ++face)
263 face_types[face] = hyper_edge.node_descriptor[face];
264 auto it = loc_mats_by_face_types.find(face_types);
265 if (it == loc_mats_by_face_types.end())
266 {
267 it = loc_mats_by_face_types.emplace(face_types, std::vector<dof_value_t>(loc_size)).first;
268 local_condensed_matrix(local_solver, hyper_edge, time, it->second.data());
269 first_edges.push_back(edge);
270 }
271 loc_mat_of_edge[edge] = it->second.data();
272 last_edge_of_face_types[face_types] = edge;
273 }
274
275 if constexpr (!hy_debug_mode)
276 {
277 for (const auto& face_types_and_edge : last_edge_of_face_types)
278 samples.push_back(face_types_and_edge.second);
279 // At most one spread sample per 32 hyperedges, such that small meshes are checked cheaply.
280 const std::size_t n_spread = std::min<std::size_t>(n_hyEdges / 32, 16);
281 for (std::size_t k = 0; k < n_spread; ++k)
282 samples.push_back(n_spread == 1 ? 0 : k * (n_hyEdges - 1) / (n_spread - 1));
283 std::sort(samples.begin(), samples.end());
284 samples.erase(std::unique(samples.begin(), samples.end()), samples.end());
285 std::vector<dof_value_t> computed(loc_size);
286 for (const std::size_t edge : samples)
287 {
288 // The local matrix of the first hyperedge of its face types is the reused one.
289 if (std::binary_search(first_edges.begin(), first_edges.end(), edge))
290 continue;
291 auto hyper_edge = hyper_graph[edge];
292 local_condensed_matrix(local_solver, hyper_edge, time, computed.data());
293 check_reused_local_matrix(computed.data(), loc_mat_of_edge[edge], loc_size, edge);
294 }
295 }
296 }
297
298 // Local matrix of a hyperedge. It is computed into buffer (if necessary).
299 auto local_matrix = [&](const std::size_t edge, auto& hyper_edge,
300 dof_value_t* buffer) -> const dof_value_t*
301 {
302 if (!factorize_once)
303 {
304 local_condensed_matrix(local_solver, hyper_edge, time, buffer);
305 return buffer;
306 }
307 if constexpr (hy_debug_mode)
308 {
309 local_condensed_matrix(local_solver, hyper_edge, time, buffer);
310 check_reused_local_matrix(buffer, loc_mat_of_edge[edge], loc_size, edge);
311 }
312 return loc_mat_of_edge[edge];
313 };
314
315 ParallelExceptions exceptions;
316 HYPERHDG_OMP_PARALLEL_FOR
317 for (long long edge = 0; edge < (long long)n_hyEdges; ++edge)
318 exceptions.run(
319 [&]()
320 {
321 auto hyper_edge = hyper_graph[edge];
322 std::vector<dof_value_t> buffer(!factorize_once || hy_debug_mode ? loc_size : 0);
323 const hyNodes_t hyNodes = hyper_edge.topology.get_hyNode_indices();
324 sink((std::size_t)edge, hyNodes, local_matrix(edge, hyper_edge, buffer.data()));
325 });
326 exceptions.rethrow();
327}

329/*!*************************************************************************************************
330 * \brief Assemble the condensed matrix in coordinate (triplet) format.
331 *
332 * The local condensed matrix of each hyperedge is obtained at once (cf. \c for_each_local_matrix
333 * for details and the meaning of \c factorize_once), and its entries are stored column by column
334 * (and hyperedge by hyperedge). Entries with the same row and column index
335 * have to be summed up (as done by SciPy's \c csr_matrix). Every hyperedge writes its own block of
336 * triplets, so the result does not depend on the number of OpenMP threads.
337 *
338 * \tparam hyNode_index_t Typename of the hypernode index.
339 * \tparam LargeVecT Typename of the large vector.
340 * \param hyper_graph The hypergraph.
341 * \param local_solver The local solver.
342 * \param time Parameter of \c trace_to_flux (time or shift).
343 * \param factorize_once Compute the local matrix only once per combination of face types.
344 * \retval sparse_mat Condensed matrix in coordinate format.
345 **************************************************************************************************/
346template <typename hyNode_index_t,
347 typename LargeVecT,
348 typename HyperGraphT,
349 typename LocalSolverT,
350 typename timeT>
351sparse_mat<LargeVecT> assemble_triplets(HyperGraphT& hyper_graph,
352 const LocalSolverT& local_solver,
353 const timeT& time,
354 const bool factorize_once)
355{
356 constexpr unsigned int hyEdge_dim = HyperGraphT::hyEdge_dim();
357 constexpr unsigned int n_dofs_per_node = LocalSolverT::n_glob_dofs_per_node();
358 constexpr unsigned int n_loc = 2 * hyEdge_dim * n_dofs_per_node;
359 check_dof_index_range<n_dofs_per_node>(hyper_graph);
360
361 // The number of triplets may exceed the range of unsigned int (the type of the indices).
362 const std::size_t n_triplets = (std::size_t)hyper_graph.n_hyEdges() * n_loc * n_loc;
363 sparse_mat<LargeVecT> result_mat;
364 result_mat.value_vec.resize(n_triplets);
365 result_mat.row_vec.resize(n_triplets);
366 result_mat.col_vec.resize(n_triplets);
367
368 for_each_local_matrix<hyNode_index_t, typename LargeVecT::value_type>(
369 hyper_graph, local_solver, time, factorize_once,
370 [&](const std::size_t edge, const auto& hyNodes, const auto* loc_mat)
371 {
372 std::array<std::array<unsigned int, n_dofs_per_node>, 2 * hyEdge_dim> dof_indices;
373 for (unsigned int node = 0; node < hyNodes.size(); ++node)
374 hyper_graph.hyNode_factory().get_dof_indices(hyNodes[node], dof_indices[node]);
375
376 std::size_t index = edge * n_loc * n_loc;
377 for (unsigned int node_j = 0; node_j < hyNodes.size(); ++node_j)
378 for (unsigned int dof_j = 0; dof_j < n_dofs_per_node; ++dof_j)
379 for (unsigned int node_i = 0; node_i < hyNodes.size(); ++node_i)
380 for (unsigned int dof_i = 0; dof_i < n_dofs_per_node; ++dof_i, ++index)
381 {
382 result_mat.row_vec[index] = dof_indices[node_i][dof_i];
383 result_mat.col_vec[index] = dof_indices[node_j][dof_j];
384 result_mat.value_vec[index] = loc_mat[(node_j * n_dofs_per_node + dof_j) * n_loc +
385 node_i * n_dofs_per_node + dof_i];
386 }
387 });
388
389 return result_mat;
390}

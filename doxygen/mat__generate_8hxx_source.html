35/*!*************************************************************************************************
36 * \brief Forward exceptions thrown in the body of a parallel loop to the calling thread.
37 *
38 * An exception that leaves an OpenMP parallel region terminates the program. Therefore, the body of
39 * a parallel loop is executed via \c run, which catches all exceptions. Once an exception has been
40 * caught, the remaining iterations are skipped, and \c rethrow (called after the loop) rethrows the
41 * first caught exception. Without OpenMP, this results in the same behavior as a serial loop.
42 **************************************************************************************************/
43class ParallelExceptions
44{
45 public:
46 /*!***********************************************************************************************
47 * \brief Execute \c body unless an exception has been thrown before, and catch exceptions.
48 ************************************************************************************************/
49 template <typename BodyT>
50 void run(BodyT&& body) noexcept
51 {
52 if (failed_.load(std::memory_order_relaxed))
53 return;
54 try
55 {
56 body();
57 }
58 catch (...)
59 {
60 HYPERHDG_OMP_CRITICAL
61 {
62 if (!exception_)
63 exception_ = std::current_exception();
64 }
65 failed_.store(true, std::memory_order_relaxed);
66 }
67 }
68 /*!***********************************************************************************************
69 * \brief Rethrow the first caught exception (if any).
70 ************************************************************************************************/
71 void rethrow()
72 {
73 if (exception_)
74 std::rethrow_exception(exception_);
75 }
76
77 private:
78 std::atomic<bool> failed_{false};
79 std::exception_ptr exception_;
80};

104/*!*************************************************************************************************
105 * \brief Local condensed matrix of a hyperedge, i.e., the matrix representation of trace_to_flux.
106 *
107 * If the local solver provides \c trace_to_flux_mat(hyper_edge, time), which returns the local
108 * condensed matrix as a \c SmallSquareMat, this function is used. Local solvers can implement it by
109 * factorizing their local matrix once and reusing the factorization for all unit vectors (cf.
110 * local_solver/trace_to_flux_mat.hxx). Otherwise, the local condensed matrix is obtained by
111 * applying \c trace_to_flux to all local unit vectors.
112 *
113 * \param local_solver The local solver.
114 * \param hyper_edge The hyperedge.
115 * \param time Parameter of \c trace_to_flux (time or shift). It is passed through to the
116 * local solver, so it may also be of a non-scalar type.
117 * \param loc_mat Array of \f$n^2\f$ entries that is filled with the local matrix in
118 * column-major order, i.e., entry (row, col) is \c loc_mat[col * n + row].
119 **************************************************************************************************/
120template <typename LocalSolverT, typename hyEdgeT, typename timeT, typename dof_value_t>
121void local_condensed_matrix(const LocalSolverT& local_solver,
122 hyEdgeT& hyper_edge,
123 const timeT& time,
124 dof_value_t* loc_mat)
125{
126 constexpr unsigned int hyEdge_dim = LocalSolverT::hyEdge_dim();
127 constexpr unsigned int n_dofs_per_node = LocalSolverT::n_glob_dofs_per_node();
128 constexpr unsigned int n_loc = 2 * hyEdge_dim * n_dofs_per_node;
129 using dofs_t = std::array<std::array<dof_value_t, n_dofs_per_node>, 2 * hyEdge_dim>;
130
131 if constexpr (requires { local_solver.trace_to_flux_mat(hyper_edge, prvalue_of(time)); })
132 {
133 const auto helper = local_solver.trace_to_flux_mat(hyper_edge, time);
134 for (unsigned int col = 0; col < n_loc; ++col)
135 for (unsigned int row = 0; row < n_loc; ++row)
136 loc_mat[col * n_loc + row] = helper(row, col);
137 }
138 else
139 {
140 dofs_t dofs_old, dofs_new;
141 for (unsigned int node_j = 0; node_j < 2 * hyEdge_dim; ++node_j)
142 for (unsigned int dof_j = 0; dof_j < n_dofs_per_node; ++dof_j)
143 {
144 for (unsigned int node = 0; node < 2 * hyEdge_dim; ++node)
145 {
146 dofs_old[node].fill(0.);
147 dofs_new[node].fill(0.);
148 }
149 dofs_old[node_j][dof_j] = 1.;
150 if constexpr (requires {
151 local_solver.trace_to_flux(dofs_old, dofs_new, prvalue_of(time));
152 })
153 local_solver.trace_to_flux(dofs_old, dofs_new, time);
154 else if constexpr (requires {
155 local_solver.trace_to_flux(dofs_old, dofs_new, hyper_edge,
156 prvalue_of(time));
157 })
158 local_solver.trace_to_flux(dofs_old, dofs_new, hyper_edge, time);
159 else
160 static_assert(always_false_v<LocalSolverT, hyEdgeT>,
161 "LocalSolverT implements no usable overload of trace_to_flux!");
162
163 const unsigned int col = node_j * n_dofs_per_node + dof_j;
164 for (unsigned int node_i = 0; node_i < 2 * hyEdge_dim; ++node_i)
165 for (unsigned int dof_i = 0; dof_i < n_dofs_per_node; ++dof_i)
166 loc_mat[col * n_loc + node_i * n_dofs_per_node + dof_i] = dofs_new[node_i][dof_i];
167 }
168 }
169}

171/*!*************************************************************************************************
172 * \brief Throw if a local matrix differs from the local matrix reused by \c factorize_once.
173 **************************************************************************************************/
174template <typename dof_value_t>
175void check_reused_local_matrix(const dof_value_t* computed,
176 const dof_value_t* reused,
177 const std::size_t size,
178 const std::size_t edge)
179{
180 // Absolute values are real also for complex dof values.
181 using real_t = decltype(std::abs(std::declval<dof_value_t>()));
182 real_t max_entry = 0., max_diff = 0.;
183 for (std::size_t k = 0; k < size; ++k)
184 {
185 max_entry = std::max(max_entry, (real_t)std::abs(reused[k]));
186 max_diff = std::max(max_diff, (real_t)std::abs(computed[k] - reused[k]));
187 }
188 const real_t tolerance =
189 std::max((real_t)1e-10, (real_t)1e3 * std::numeric_limits<real_t>::epsilon());
190 if (!(max_diff <= tolerance * max_entry)) // Also true if max_diff is NaN.
191 {
192 std::stringstream msg;
193 msg << "Option factorize_once requires all hyperedges with the same face types to have the "
194 << "same local matrix (e.g. uniform meshes and coefficients that are constant in space), "
195 << "but the local matrix of hyperedge " << edge << " differs by " << max_diff
196 << " (maximum entry " << max_entry << "). Use factorize_once = false instead!";
197 throw std::invalid_argument(msg.str());
198 }
199}

201/*!*************************************************************************************************
202 * \brief Compute the local condensed matrix of every hyperedge and hand it to \c sink.
203 *
204 * For every hyperedge \c edge, \c sink(edge, hyNodes, loc_mat) is called, where \c hyNodes are the
205 * indices of the hypernodes of the hyperedge and \c loc_mat points to the local condensed matrix
206 * (cf. \c local_condensed_matrix for its layout).
207 *
208 * If \c factorize_once is \c true, the local matrix is only computed once for every combination of
209 * face (hypernode) types, i.e., for every combination of boundary conditions, and reused for all
210 * other hyperedges with the same face types. This is only correct if all these hyperedges have the
211 * same local matrix, e.g. for uniform meshes with constant (in space) coefficients. It is wrong if,
212 * for example, the diffusion tensor is space dependent or the hyperedges differ in size or shape.
213 * Therefore, the local matrices of some hyperedges (the last one of every combination of face types
214 * and up to 16 hyperedges evenly spread over all hyperedges, at most one per 32 hyperedges) are
215 * recomputed and compared to the reused ones, and an \c std::invalid_argument is thrown if they
216 * differ. This costs at most one local matrix per combination of face types plus 16.
217 * If the code is compiled without \c NDEBUG, this check is done for every hyperedge (which is as
218 * expensive as \c factorize_once being false).
219 *
220 * The local matrices are computed in parallel if HyperHDG is compiled with OpenMP. Then, \c sink
221 * is called concurrently for different hyperedges. Exceptions thrown by the local solvers are
222 * forwarded to the caller.
223 *
224 * \tparam hyNode_index_t Typename of the hypernode index.
225 * \tparam dof_value_t Typename of the entries of the local matrices.
226 * \param hyper_graph The hypergraph.
227 * \param local_solver The local solver.
228 * \param time Parameter of \c trace_to_flux (time or shift, cf.
229 *local_condensed_matrix). \param factorize_once Compute the local matrix only once per
230 *combination of face types. \param sink Callable that receives the local matrices.
231 **************************************************************************************************/
232template <typename hyNode_index_t,
233 typename dof_value_t,
234 typename HyperGraphT,
235 typename LocalSolverT,
236 typename timeT,
237 typename SinkT>
238void for_each_local_matrix(HyperGraphT& hyper_graph,
239 const LocalSolverT& local_solver,
240 const timeT& time,
241 const bool factorize_once,
242 SinkT&& sink)
243{
244 constexpr unsigned int hyEdge_dim = HyperGraphT::hyEdge_dim();
245 constexpr unsigned int n_dofs_per_node = LocalSolverT::n_glob_dofs_per_node();
246 constexpr std::size_t loc_size =
247 (2 * hyEdge_dim * n_dofs_per_node) * (2 * hyEdge_dim * n_dofs_per_node);
248 using face_types_t = std::array<unsigned int, 2 * hyEdge_dim>;
249 using hyNodes_t = SmallVec<2 * hyEdge_dim, hyNode_index_t>;
250 const std::size_t n_hyEdges = hyper_graph.n_hyEdges();
251
252 // factorize_once: Compute the local matrix for every combination of face types (serially), and
253 // check that it can be reused for a sample of hyperedges.
254 std::map<face_types_t, std::vector<dof_value_t> > loc_mats_by_face_types;
255 std::vector<const dof_value_t*> loc_mat_of_edge(factorize_once ? n_hyEdges : 0, nullptr);
256 if (factorize_once)
257 {
258 std::vector<std::size_t> samples, first_edges;
259 std::map<face_types_t, std::size_t> last_edge_of_face_types;
260 for (std::size_t edge = 0; edge < n_hyEdges; ++edge)
261 {
262 auto hyper_edge = hyper_graph[edge];
263 face_types_t face_types;
264 for (unsigned int face = 0; face < 2 * hyEdge_dim; ++face)
265 face_types[face] = hyper_edge.node_descriptor[face];
266 auto it = loc_mats_by_face_types.find(face_types);
267 if (it == loc_mats_by_face_types.end())
268 {
269 it = loc_mats_by_face_types.emplace(face_types, std::vector<dof_value_t>(loc_size)).first;
270 local_condensed_matrix(local_solver, hyper_edge, time, it->second.data());
271 first_edges.push_back(edge);
272 }
273 loc_mat_of_edge[edge] = it->second.data();
274 last_edge_of_face_types[face_types] = edge;
275 }
276
277 if constexpr (!hy_debug_mode)
278 {
279 for (const auto& face_types_and_edge : last_edge_of_face_types)
280 samples.push_back(face_types_and_edge.second);
281 // At most one spread sample per 32 hyperedges, such that small meshes are checked cheaply.
282 const std::size_t n_spread = std::min<std::size_t>(n_hyEdges / 32, 16);
283 for (std::size_t k = 0; k < n_spread; ++k)
284 samples.push_back(n_spread == 1 ? 0 : k * (n_hyEdges - 1) / (n_spread - 1));
285 std::sort(samples.begin(), samples.end());
286 samples.erase(std::unique(samples.begin(), samples.end()), samples.end());
287 std::vector<dof_value_t> computed(loc_size);
288 for (const std::size_t edge : samples)
289 {
290 // The local matrix of the first hyperedge of its face types is the reused one.
291 if (std::binary_search(first_edges.begin(), first_edges.end(), edge))
292 continue;
293 auto hyper_edge = hyper_graph[edge];
294 local_condensed_matrix(local_solver, hyper_edge, time, computed.data());
295 check_reused_local_matrix(computed.data(), loc_mat_of_edge[edge], loc_size, edge);
296 }
297 }
298 }
299
300 // Local matrix of a hyperedge. It is computed into buffer (if necessary).
301 auto local_matrix = [&](const std::size_t edge, auto& hyper_edge,
302 dof_value_t* buffer) -> const dof_value_t*
303 {
304 if (!factorize_once)
305 {
306 local_condensed_matrix(local_solver, hyper_edge, time, buffer);
307 return buffer;
308 }
309 if constexpr (hy_debug_mode)
310 {
311 local_condensed_matrix(local_solver, hyper_edge, time, buffer);
312 check_reused_local_matrix(buffer, loc_mat_of_edge[edge], loc_size, edge);
313 }
314 return loc_mat_of_edge[edge];
315 };
316
317 ParallelExceptions exceptions;
318 HYPERHDG_OMP_PARALLEL_FOR
319 for (long long edge = 0; edge < (long long)n_hyEdges; ++edge)
320 exceptions.run(
321 [&]()
322 {
323 auto hyper_edge = hyper_graph[edge];
324 std::vector<dof_value_t> buffer(!factorize_once || hy_debug_mode ? loc_size : 0);
325 const hyNodes_t hyNodes = hyper_edge.topology.get_hyNode_indices();
326 sink((std::size_t)edge, hyNodes, local_matrix(edge, hyper_edge, buffer.data()));
327 });
328 exceptions.rethrow();
329}

331/*!*************************************************************************************************
332 * \brief Assemble the condensed matrix in coordinate (triplet) format.
333 *
334 * The local condensed matrix of each hyperedge is obtained at once (cf. \c for_each_local_matrix
335 * for details and the meaning of \c factorize_once), and its entries are stored column by column
336 * (and hyperedge by hyperedge). Entries with the same row and column index
337 * have to be summed up (as done by SciPy's \c csr_matrix). Every hyperedge writes its own block of
338 * triplets, so the result does not depend on the number of OpenMP threads.
339 *
340 * If the local solver provides \c is_dirichlet(hyper_edge, node, dof), the column of a Dirichlet
341 * dof is replaced by the unit vector of that dof instead of the local solver's (vanishing)
342 * contribution, so the condensed matrix stays regular. Local solvers without it are unaffected.
343 *
344 * \tparam hyNode_index_t Typename of the hypernode index.
345 * \tparam LargeVecT Typename of the large vector.
346 * \param hyper_graph The hypergraph.
347 * \param local_solver The local solver.
348 * \param time Parameter of \c trace_to_flux (time or shift).
349 * \param factorize_once Compute the local matrix only once per combination of face types.
350 * \retval sparse_mat Condensed matrix in coordinate format.
351 **************************************************************************************************/
352template <typename hyNode_index_t,
353 typename LargeVecT,
354 typename HyperGraphT,
355 typename LocalSolverT,
356 typename timeT>
357sparse_mat<LargeVecT> assemble_triplets(HyperGraphT& hyper_graph,
358 const LocalSolverT& local_solver,
359 const timeT& time,
360 const bool factorize_once)
361{
362 constexpr unsigned int hyEdge_dim = HyperGraphT::hyEdge_dim();
363 constexpr unsigned int n_dofs_per_node = LocalSolverT::n_glob_dofs_per_node();
364 constexpr unsigned int n_loc = 2 * hyEdge_dim * n_dofs_per_node;
365 check_dof_index_range<n_dofs_per_node>(hyper_graph);
366
367 // The number of triplets may exceed the range of unsigned int (the type of the indices).
368 const std::size_t n_triplets = (std::size_t)hyper_graph.n_hyEdges() * n_loc * n_loc;
369 sparse_mat<LargeVecT> result_mat;
370 result_mat.value_vec.resize(n_triplets);
371 result_mat.row_vec.resize(n_triplets);
372 result_mat.col_vec.resize(n_triplets);
373
374 for_each_local_matrix<hyNode_index_t, typename LargeVecT::value_type>(
375 hyper_graph, local_solver, time, factorize_once,
376 [&](const std::size_t edge, const auto& hyNodes, const auto* loc_mat)
377 {
378 std::array<std::array<unsigned int, n_dofs_per_node>, 2 * hyEdge_dim> dof_indices;
379 for (unsigned int node = 0; node < hyNodes.size(); ++node)
380 hyper_graph.hyNode_factory().get_dof_indices(hyNodes[node], dof_indices[node]);
381
382 std::size_t index = edge * n_loc * n_loc;
383 for (unsigned int node_j = 0; node_j < hyNodes.size(); ++node_j)
384 for (unsigned int dof_j = 0; dof_j < n_dofs_per_node; ++dof_j)
385 for (unsigned int node_i = 0; node_i < hyNodes.size(); ++node_i)
386 for (unsigned int dof_i = 0; dof_i < n_dofs_per_node; ++dof_i, ++index)
387 {
388 result_mat.row_vec[index] = dof_indices[node_i][dof_i];
389 result_mat.col_vec[index] = dof_indices[node_j][dof_j];
390 result_mat.value_vec[index] = loc_mat[(node_j * n_dofs_per_node + dof_j) * n_loc +
391 node_i * n_dofs_per_node + dof_i];
392 }
393
394 if constexpr (requires(std::remove_cvref_t<decltype(hyper_graph[0])>& hyper_edge) {
395 {
396 local_solver.is_dirichlet(hyper_edge, 0U, 0U)
397 } -> std::convertible_to<bool>;
398 })
399 {
400 auto hyper_edge = hyper_graph[edge];
401 for (unsigned int node_j = 0; node_j < hyNodes.size(); ++node_j)
402 for (unsigned int dof_j = 0; dof_j < n_dofs_per_node; ++dof_j)
403 if (local_solver.is_dirichlet(hyper_edge, node_j, dof_j))
404 {
405 const unsigned int col = node_j * n_dofs_per_node + dof_j;
406 const std::size_t first = edge * n_loc * n_loc + col * n_loc;
407 for (unsigned int row = 0; row < n_loc; ++row)
408 result_mat.value_vec[first + row] = row == col ? 1. : 0.;
409 }
410 }
411 });
412
413 return result_mat;
414}

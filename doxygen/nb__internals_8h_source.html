676template <typename T> struct scoped_pymalloc {
677 scoped_pymalloc(size_t size = 1, size_t extra_bytes = 0) {
678 size_t total = size * sizeof(T);
679 if (NB_UNLIKELY(size > SIZE_MAX / sizeof(T) ||
680 total > SIZE_MAX - extra_bytes))
681 fail("scoped_pymalloc(): integer overflow!");
682 total += extra_bytes;
683 ptr = (T *) PyMem_Malloc(total);
684 if (!ptr)
685 fail("scoped_pymalloc(): could not allocate %llu bytes of memory!",
686 (unsigned long long) total);
687 }
688 ~scoped_pymalloc() { PyMem_Free(ptr); }
689 T *release() {
690 T *temp = ptr;
691 ptr = nullptr;
692 return temp;
693 }
694 T *get() const { return ptr; }
695 T &operator[](size_t i) { return ptr[i]; }
696 T *operator->() { return ptr; }
697private:
698 T *ptr{ nullptr };
699};

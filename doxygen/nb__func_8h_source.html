13template <typename Caster>
14bool from_python_remember_conv(Caster &c, PyObject **args, uint8_t *args_flags,
15 cleanup_list *cleanup, size_t index) {
16 size_t size_before = cleanup->size();
17 if (!c.from_python(args[index], args_flags[index], cleanup))
18 return false;
19
20 // If an implicit conversion took place, update the 'args' array so that
21 // any keep_alive annotation or postcall hook can be aware of this change
22 size_t size_after = cleanup->size();
23 if (size_after != size_before)
24 args[index] = (*cleanup)[size_after - 1];
25
26 return true;
27}

29// Return the number of nb::arg and nb::arg_v types in the first I types Ts.
30// Invoke with std::make_index_sequence<sizeof...(Ts)>() to provide
31// an index pack 'Is' that parallels the types pack Ts.
32template <size_t I, typename... Ts, size_t... Is>
33constexpr size_t count_args_before_index(std::index_sequence<Is...>) {
34 static_assert(sizeof...(Is) == sizeof...(Ts));
35 return ((Is < I && std::is_base_of_v<arg, Ts>) + ... + 0);
36}

88 // Detect locations of nb::args / nb::kwargs (if they exist).
89 // Find the first and last occurrence of each; we'll later make sure these
90 // match, in order to guarantee there's only one instance.
91 static constexpr size_t
92 args_pos_1 = index_1_v<std::is_same_v<intrinsic_t<Args>, args>...>,
93 args_pos_n = index_n_v<std::is_same_v<intrinsic_t<Args>, args>...>,
94 kwargs_pos_1 = index_1_v<std::is_same_v<intrinsic_t<Args>, kwargs>...>,
95 kwargs_pos_n = index_n_v<std::is_same_v<intrinsic_t<Args>, kwargs>...>,
96 nargs = sizeof...(Args);

98 constexpr bool has_arg_defaults = (detail::has_arg_defaults_v<Args> || ... || false);
99
100 // Determine the number of nb::arg/nb::arg_v annotations
101 constexpr size_t nargs_provided =
102 (std::is_base_of_v<arg, Extra> + ... + 0);
103 constexpr bool is_method_det =
104 (std::is_same_v<is_method, Extra> + ... + 0) != 0;
105 constexpr bool is_getter_det =
106 (std::is_same_v<is_getter, Extra> + ... + 0) != 0;
107 constexpr bool has_arg_annotations = has_arg_defaults || (nargs_provided > 0 && !is_getter_det);
108
109 // Determine the number of potentially-locked function arguments
110 constexpr bool lock_self_det =
111 (std::is_same_v<lock_self, Extra> + ... + 0) != 0;
112 static_assert(Info::nargs_locked <= 2,
113 "At most two function arguments can be locked");
114 static_assert(!(lock_self_det && !is_method_det),
115 "The nb::lock_self() annotation only applies to methods");
116
117 // Detect location of nb::kw_only annotation, if supplied. As with args/kwargs
118 // we find the first and last location and later verify they match each other.
119 // Note this is an index in Extra... while args/kwargs_pos_* are indices in
120 // Args... .
121 constexpr size_t
122 kwonly_pos_1 = index_1_v<std::is_same_v<kw_only, Extra>...>,
123 kwonly_pos_n = index_n_v<std::is_same_v<kw_only, Extra>...>;
124
125 // Arguments after nb::args are implicitly keyword-only even if there is no
126 // nb::kw_only annotation
127 constexpr bool explicit_kw_only = kwonly_pos_1 != sizeof...(Extra);
128 constexpr bool implicit_kw_only = args_pos_1 + 1 < kwargs_pos_1;
129
130 // A few compile-time consistency checks
131 static_assert(args_pos_1 == args_pos_n && kwargs_pos_1 == kwargs_pos_n,
132 "Repeated use of nb::kwargs or nb::args in the function signature!");
133 static_assert(!has_arg_annotations || has_arg_defaults || nargs_provided + is_method_det == nargs,
134 "The number of nb::arg annotations must match the argument count!");
135 static_assert(kwargs_pos_1 == nargs || kwargs_pos_1 + 1 == nargs,
136 "nb::kwargs must be the last element of the function signature!");
137 static_assert(args_pos_1 == nargs || args_pos_1 < kwargs_pos_1,
138 "nb::args must precede nb::kwargs if both are present!");
139 static_assert(has_arg_annotations || (!implicit_kw_only && !explicit_kw_only),
140 "Keyword-only arguments must have names!");
141
142 // Find the index in Args... of the first keyword-only parameter. Since
143 // the 'self' parameter doesn't get a nb::arg annotation, we must adjust
144 // by 1 for methods. Note that nargs_before_kw_only is only used if
145 // a kw_only annotation exists (i.e., if explicit_kw_only is true);
146 // the conditional is just to save the compiler some effort otherwise.
147 constexpr size_t nargs_before_kw_only =
148 explicit_kw_only
149 ? is_method_det + count_args_before_index<kwonly_pos_1, Extra...>(
150 std::make_index_sequence<sizeof...(Extra)>())
151 : nargs;

153 (void) kwonly_pos_n;
154
155 if constexpr (explicit_kw_only) {
156 static_assert(kwonly_pos_1 == kwonly_pos_n,
157 "Repeated use of nb::kw_only annotation!");
158
159 // If both kw_only and *args are specified, kw_only must be
160 // immediately after the nb::arg for *args.
161 static_assert(args_pos_1 == nargs || nargs_before_kw_only == args_pos_1 + 1,
162 "Arguments after nb::args are implicitly keyword-only; any "
163 "nb::kw_only() annotation must be positioned to reflect that!");
164
165 // If both kw_only and **kwargs are specified, kw_only must be
166 // before the nb::arg for **kwargs.
167 static_assert(nargs_before_kw_only < kwargs_pos_1,
168 "Variadic nb::kwargs are implicitly keyword-only; any "

306 // Set nargs_pos to the number of C++ function parameters (Args...) that
307 // can be filled from Python positional arguments in a one-to-one fashion.
308 // This ends at:
309 // - the location of the variadic *args parameter, if present; otherwise
310 // - the location of the first keyword-only parameter, if any; otherwise
311 // - the location of the variadic **kwargs parameter, if present; otherwise
312 // - the end of the parameter list
313 // It's correct to give *args priority over kw_only because we verified
314 // above that kw_only comes afterward if both are present. It's correct
315 // to give kw_only priority over **kwargs because we verified above that
316 // kw_only comes before if both are present.
317 f.nargs_pos = args_pos_1 < nargs ? args_pos_1 :
318 explicit_kw_only ? nargs_before_kw_only :
319 kwargs_pos_1 < nargs ? kwargs_pos_1 : nargs;
320
321 // Fill remaining fields of 'f'
322 size_t arg_index = 0;
323 (func_extra_apply(f, extra, arg_index), ...);
324
325 (void) arg_index;
326
327 // Apply implicit accepts_none for std::optional<> typed arguments
328 // after func_extra_apply, so that explicit nb::arg().noconvert() works.
329 if constexpr (has_arg_defaults) {
330 ((void)(Is >= (size_t)is_method_det && has_arg_defaults_v<Args> &&
331 (f.args[Is - is_method_det].flag |=
332 (uint8_t) cast_flags::accepts_none, true)), ...);
333 }
334
335 // Record 'none_disallowed' (nonzero only for value/reference bound-type
336 // targets) in the per-argument flag at bind time. The dispatcher carries
337 // it into the call via 'args_flags', so the heavily-inlined function
338 // trampoline need not OR it in at every from_python() invocation. Simple
339 // overloads ignore the per-argument flags but reject 'None' up front, so
340 // the bit is only consulted where 'None' can actually reach a caster.
341 if constexpr (has_arg_annotations) {
342 ((void)(Is >= (size_t)is_method_det &&
343 (f.args[Is - is_method_det].flag |=
344 none_disallowed_flag<Args>, true)), ...);
345 }
346
347 return nb_func_new(&f);
348}

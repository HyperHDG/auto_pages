18template <typename Map, typename Key, typename Value>
19inline void map_set(Map &m, const Key &k, const Value &v) {
20 if constexpr (detail::is_copy_assignable_v<Value>) {
21 m[k] = v;
22 } else {
23 auto r = m.emplace(k, v);
24 if (!r.second) {
25 // Value is not copy-assignable. Erase and retry
26 m.erase(r.first);
27 m.emplace(k, v);
28 }
29 }
30}

34template <typename Map,
35 rv_policy Policy = rv_policy::automatic_reference,
36 typename... Args>
37class_<Map> bind_map(handle scope, const char *name, Args &&...args) {
38 using Key = typename Map::key_type;
39 using Value = typename Map::mapped_type;
40
41 using ValueRef = typename detail::iterator_value_access<
42 typename Map::iterator>::result_type;
43
44 static_assert(
45 !detail::is_base_caster_v<detail::make_caster<Value>> ||
46 detail::is_copy_constructible_v<Value> ||
47 (Policy != rv_policy::automatic_reference &&
48 Policy != rv_policy::copy),
49 "bind_map(): the generated __getitem__ would copy elements, so the "
50 "value type must be copy-constructible");
51
52 handle cl_cur = type<Map>();
53 if (cl_cur.is_valid()) {
54 // Binding already exists, don't re-create
55 return borrow<class_<Map>>(cl_cur);
56 }
57
58 auto cl = class_<Map>(scope, name, std::forward<Args>(args)...)
59 .def(init<>(),
60 "Default constructor")
61
62 .def("__len__", [](const Map &m) { return m.size(); })
63
64 .def("__bool__",
65 [](const Map &m) { return !m.empty(); },
66 "Check whether the map is nonempty")
67
68 .def("__repr__",
69 [](handle_t<Map> h) {
70 return steal<str>(detail::repr_map(h.ptr()));
71 })
72
73 .def("__contains__",
74 [](const Map &m, const Key &k) { return m.find(k) != m.end(); })
75
76 .def("__contains__", // fallback for incompatible types
77 [](const Map &, handle) { return false; })
78
79 .def("__iter__",
80 [](Map &m) {
81 return make_key_iterator<Policy>(type<Map>(), "KeyIterator",
82 m.begin(), m.end());
83 },
84 keep_alive<0, 1>())
85
86 .def("__getitem__",
87 [](Map &m, const Key &k) -> ValueRef {
88 auto it = m.find(k);
89 if (it == m.end())
90 throw key_error();
91 return (*it).second;
92 }, Policy)
93
94 .def("__delitem__",
95 [](Map &m, const Key &k) {
96 auto it = m.find(k);
97 if (it == m.end())
98 throw key_error();
99 m.erase(it);
100 })
101
102 .def("clear", [](Map &m) { m.clear(); },
103 "Remove all items");
104
105
106 if constexpr (detail::is_copy_constructible_v<Map>) {
107 cl.def(init<const Map &>(), "Copy constructor");
108
109 cl.def("__init__", [](Map *m, typed<dict, Key, Value> d) {
110 new (m) Map();
111 try {
112 for (auto [k, v] : borrow<dict>(std::move(d)))
113 m->emplace(cast<Key>(k), cast<Value>(v));
114 } catch (...) {
115 m->~Map();
116 throw;
117 }
118 }, "Construct from a dictionary");
119
120 implicitly_convertible<dict, Map>();
121 }
122
123 // Assignment operator for copy-assignable/copy-constructible types
124 if constexpr (detail::is_copy_assignable_v<Value> ||
125 detail::is_copy_constructible_v<Value>) {
126 cl.def("__setitem__", [](Map &m, const Key &k, const Value &v) {
127 detail::map_set<Map, Key, Value>(m, k, v);
128 });
129
130 cl.def("update", [](Map &m, const Map &m2) {
131 // Updating a map with itself would be a no-op, but the underlying
132 // map_set() may erase and re-emplace nodes; doing so while
133 // iterating m2 == m leaves kv referencing freed storage (a
134 // dangling-reference for non-copy-assignable values). Skip it.
135 if (&m2 == &m)
136 return;
137 for (auto &kv : m2)
138 detail::map_set<Map, Key, Value>(m, kv.first, kv.second);
139 },
140 "Update the map with element from ``arg``");
141 }
142
143 if constexpr (detail::is_equality_comparable_v<Map>) {
144 cl.def(self == self, sig("def __eq__(self, arg: object, /) -> bool"))
145 .def(self != self, sig("def __ne__(self, arg: object, /) -> bool"));
146 }
147
148 // Item, value, and key views
149 struct KeyView { Map &map; };
150 struct ValueView { Map &map; };
151 struct ItemView { Map &map; };
152
153 class_<ItemView>(cl, "ItemView")
154 .def("__len__", [](ItemView &v) { return v.map.size(); })
155 .def("__iter__",
156 [](ItemView &v) {
157 return make_iterator<Policy>(type<Map>(), "ItemIterator",
158 v.map.begin(), v.map.end());
159 },
160 keep_alive<0, 1>());
161
162 class_<KeyView>(cl, "KeyView")
163 .def("__contains__", [](KeyView &v, const Key &k) { return v.map.find(k) != v.map.end(); })
164 .def("__contains__", [](KeyView &, handle) { return false; })
165 .def("__len__", [](KeyView &v) { return v.map.size(); })
166 .def("__iter__",
167 [](KeyView &v) {
168 return make_key_iterator<Policy>(type<Map>(), "KeyIterator",
169 v.map.begin(), v.map.end());
170 },
171 keep_alive<0, 1>());
172
173 class_<ValueView>(cl, "ValueView")
174 .def("__len__", [](ValueView &v) { return v.map.size(); })
175 .def("__iter__",
176 [](ValueView &v) {
177 return make_value_iterator<Policy>(type<Map>(), "ValueIterator",
178 v.map.begin(), v.map.end());
179 },
180 keep_alive<0, 1>());
181
182 cl.def("keys", [](Map &m) { return new KeyView{m}; }, keep_alive<0, 1>(),
183 "Returns an iterable view of the map's keys.");
184 cl.def("values", [](Map &m) { return new ValueView{m}; }, keep_alive<0, 1>(),
185 "Returns an iterable view of the map's values.");
186 cl.def("items", [](Map &m) { return new ItemView{m}; }, keep_alive<0, 1>(),
187 "Returns an iterable view of the map's items.");
188
189 return cl;
190}

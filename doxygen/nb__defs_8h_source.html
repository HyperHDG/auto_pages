179// Helper macros to ensure macro arguments are expanded before token pasting/stringification
180#define NB_MODULE_IMPL(name, variable) NB_MODULE_IMPL2(name, variable)
181#define NB_MODULE_IMPL2(name, variable) \
182 static void nanobind_##name##_exec_impl(nanobind::module_); \
183 static int nanobind_##name##_exec(PyObject *m) { \
184 nanobind::detail::nb_module_exec(NB_DOMAIN_STR, m); \
185 try { \
186 nanobind_##name##_exec_impl( \
187 nanobind::borrow<nanobind::module_>(m)); \
188 return 0; \
189 } catch (nanobind::python_error &e) { \
190 e.restore(); \
191 nanobind::chain_error( \
192 PyExc_ImportError, \
193 "Encountered an error while initializing the extension."); \
194 } catch (const std::exception &e) { \
195 PyErr_SetString(PyExc_ImportError, e.what()); \
196 } \
197 return -1; \
198 } \
199 static PyModuleDef_Slot nanobind_##name##_slots[] = { \
200 { Py_mod_exec, (void *) nanobind_##name##_exec }, \
201 NB_MODULE_SLOTS_2 \
202 }; \
203 static struct PyModuleDef nanobind_##name##_module = { \
204 PyModuleDef_HEAD_INIT, #name, nullptr, 0, nullptr, \
205 nanobind_##name##_slots, nullptr, nullptr, \
206 nanobind::detail::nb_module_free \
207 }; \
208 extern "C" [[maybe_unused]] NB_EXPORT PyObject *PyInit_##name(void); \
209 extern "C" PyObject *PyInit_##name(void) { \
210 return PyModuleDef_Init(&nanobind_##name##_module); \
211 } \
212 void nanobind_##name##_exec_impl(nanobind::module_ variable)

8/*!*************************************************************************************************
9 * \brief Assembly of the condensed matrix of a global loop.
10 *
11 * Base class (CRTP) of the linear global loops. The global loop \c GlobalLoopT, which is passed as
12 * a template together with its template arguments, has to declare this class a friend, and to
13 * provide the members \c hyper_graph_ and \c local_solver_. Passing the template (instead of the
14 * class) allows to derive via
15 * \c CondensedOperator<GlobalLoop, TopologyT, ...> without spelling out the derived class.
16 *
17 * \tparam GlobalLoopT The template of the global loop.
18 * \tparam TopologyT Class type containing topological information.
19 * \tparam GeometryT Class type containing geometrical information.
20 * \tparam NodeDescriptorT Class type containing the information of nodes of hyperedges.
21 * \tparam LocalSolverT Class type of the local solver.
22 * \tparam LargeVecT Type of large (global) vectors.
23 * \tparam dof_index_t Index type of hyperedges.
24 **************************************************************************************************/
25template <template <class, class, class, class, typename, typename> class GlobalLoopT,
26 class TopologyT,
27 class GeometryT,
28 class NodeDescriptorT,
29 class LocalSolverT,
30 typename LargeVecT,

35 * \brief The derived global loop.
36 ************************************************************************************************/

39 /*!***********************************************************************************************
40 * \brief Floating type is determined by floating type of large vector's entries.

42 using dof_value_t = typename LargeVecT::value_type;

44 public:
45 /*!***********************************************************************************************
46 * \brief Assemble the condensed matrix.
47 *
48 * Function that assembles the condensed matrix \f$A\f$ of the HDG method, i.e., the matrix of
49 * the linear map evaluated by \c trace_to_flux, in coordinate (triplet) format. Entries with the
50 * same row and column index have to be summed up (as done by SciPy's \c csr_matrix).
51 *
52 * If the local solver provides \c trace_to_flux_mat, its local matrix is factorized once per
53 * hyperedge. Otherwise, the local problem is solved for every local unit vector.
54 *
55 * \tparam hyNode_index_t Typename of the hypernode index. Defaults to \c unsigned \c int.
56 * \tparam paramT Type of the parameter (passed through to the local solver).
57 * \param param Parameter of trace_to_flux, i.e., the time at which given analytic
58 * functions will be evaluated (or the shift for eigenvalue problems).
59 * \param factorize_once If true, the local matrix is only computed (and factorized) once for
60 * every combination of face types (boundary conditions) and reused for
61 * all hyperedges. This is only correct if all hyperedges have the same
62 * local matrix, e.g., for uniform meshes with coefficients that are
63 * constant in space. It is wrong for space dependent diffusion tensors
64 * or non-uniform meshes, which is detected for a sample of hyperedges
65 * (for all hyperedges in debug mode) and results in an exception.
66 * Defaults to false (local matrix per hyperedge).
67 * \retval sparse_mat Condensed matrix in coordinate format.
68 ************************************************************************************************/
69 template <typename hyNode_index_t = dof_index_t, typename paramT = dof_value_t>
70 sparse_mat<LargeVecT> trace_to_flux_mat(const paramT& param = paramT(),
71 const bool factorize_once = false)
72 {
73 GlobalLoop& loop = static_cast<GlobalLoop&>(*this);
74 return MatGenerate::assemble_triplets<hyNode_index_t, LargeVecT>(
75 loop.hyper_graph_, loop.local_solver_, param, factorize_once);
76 }
77};

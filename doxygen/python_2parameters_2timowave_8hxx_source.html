17 static constexpr param_float_t omega = 2 * M_PI;
18
19 // u_k = au[k] cos(w sigma) cos(w t + pu[k]), r_k = ar[k] cos(w sigma) cos(w t + pr[k])
20 // on the axis-aligned cross (cross2.geo).
21 // arms must come in +-pairs so the nodal forces n(0) = -i x r(0) balance.
22 // distinct per-component phases: all of u, v, r, s, n, m are nonzero at t = 0
23 static constexpr std::array<param_float_t, 3> au{1, 2, 3}, pu{0.3, 0.8, 1.3};
24 static constexpr std::array<param_float_t, 3> ar{5, 7, 11}, pr{0.5, 1.1, 1.7};
25
26 // Runtime SPATIAL phase of the standing-wave factor cos(w*(x+y+z) + px). The time phases
27 // pu/pr cannot silence the boundary (cos(w t + p) never vanishes identically); px moves the
28 // spatial nodes: px = -pi/2 turns the factor into sin(w*s), which is zero at every tip of
29 // cross2 (s in {0, +-1}) -- Dirichlet data identically zero for all t. The per-arm
30 // forcing formulas below generalize by literally phase-shifting their trig factors (the
31 // u-parts and m' still cancel at unit wave speed; the +-arm-pair junction balance is
32 // pairwise smoothness through the center, independent of px). Default 0 = classic wave4.
33 static inline param_float_t px = 0.;

37 SmallVec<space_dimT, param_float_t> res(0.);

39 res[k] = au[k] * cos(omega * time + pu[k]);
40 return res;

170 // Dual fields of the manufactured solution (for the dual-variable error). Both are odd
171 // under flipping the edge orientation, so the evaluator passes the edge's axial frame
172 // vector d = inner_normal(0), orientation included. With sig = x+y+z (the signed arm
173 // coordinate on cross2) and (d.1) = sum_k d_k = +-1 on an axis-aligned arm:
174 // n = -du/ds - d x r = w sin(w sig + px) (d.1) uvec - cos(w sig + px) d x rvec
175 // m = -dr/ds = w sin(w sig + px) (d.1) rvec
176 static SmallVec<space_dimT, param_float_t> analytic_result_n(
177 const Point<space_dimT, param_float_t>& point,
178 const Point<space_dimT, param_float_t>& axial,
179 const param_float_t time = 0.)
180 {
181 const param_float_t sig = point[0] + point[1] + point[2];
182 const param_float_t dsum = axial[0] + axial[1] + axial[2];
183 auto res = uvec(time);
184 res *= omega * sin(omega * sig + px) * dsum;
185 const auto r = rvec(time);
186 const param_float_t c = cos(omega * sig + px);
187 for (unsigned int k = 0; k < 3; k++)
188 res[k] -= c * (axial[(k + 1) % 3] * r[(k + 2) % 3] - axial[(k + 2) % 3] * r[(k + 1) % 3]);
189 return res;
190 }
191 static SmallVec<space_dimT, param_float_t> analytic_result_m(
192 const Point<space_dimT, param_float_t>& point,
193 const Point<space_dimT, param_float_t>& axial,
194 const param_float_t time = 0.)
195 {
196 const param_float_t sig = point[0] + point[1] + point[2];
197 const param_float_t dsum = axial[0] + axial[1] + axial[2];
198 auto res = rvec(time);
199 res *= omega * sin(omega * sig + px) * dsum;
200 return res;
201 }

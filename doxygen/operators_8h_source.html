42template <op_id id, op_type ot, typename L, typename R> struct op_ {
43 template <typename Class, typename... Extra> void execute(Class &cl, const Extra&... extra) const {
44 using Type = typename Class::Type;
45 using Lt = std::conditional_t<std::is_same_v<L, self_t>, Type, L>;
46 using Rt = std::conditional_t<std::is_same_v<R, self_t>, Type, R>;
47 using Op = op_impl<id, ot, Type, Lt, Rt>;
48 cl.def(Op::name(), &Op::execute, is_operator(), Op::default_policy, extra...);
49 }
50
51 template <typename Class, typename... Extra> void execute_cast(Class &cl, const Extra&... extra) const {
52 using Type = typename Class::Type;
53 using Lt = std::conditional_t<std::is_same_v<L, self_t>, Type, L>;
54 using Rt = std::conditional_t<std::is_same_v<R, self_t>, Type, R>;
55 using Op = op_impl<id, ot, Type, Lt, Rt>;
56 cl.def(Op::name(), &Op::execute_cast, is_operator(), Op::default_policy, extra...);
57 }

333PyObject *enum_from_cpp(const std::type_info *tp, int64_t key) noexcept {
334 type_data *t = nb_type_c2p(internals, tp);
335 if (!t)
336 return nullptr;
337
338 enum_map *fwd = (enum_map *) t->enum_tbl.fwd;
339
340 enum_map::iterator it = fwd->find(key);
341 if (it != fwd->end()) {
342 PyObject *value = (PyObject *) it->second;
343 Py_INCREF(value);
344 return value;
345 }
346
347 uint32_t flags = t->flags;
348 if ((flags & (uint32_t) enum_flags::is_flag) != 0) {
349 PyObject *enum_tp = (PyObject *) t->type_py;
350
351 object val;
352 if (flags & (uint32_t) enum_flags::is_signed)
353 val = steal(PyLong_FromLongLong((long long) key));
354 else
355 val = steal(PyLong_FromUnsignedLongLong((unsigned long long) key));
356 if (!val.is_valid())
357 return nullptr;
358
359 object new_fn = steal(PyObject_GetAttr(
360 enum_tp, NB_INTERNED(__new__)));
361 if (!new_fn.is_valid())
362 return nullptr;
363
364 // May fail, e.g. for out-of-range bits with a STRICT flag boundary
365 PyObject *args[2] = { enum_tp, val.ptr() };
366 return PyObject_Vectorcall(new_fn.ptr(), args, 2, nullptr);
367 }
368
369 if (flags & (uint32_t) enum_flags::is_signed)
370 PyErr_Format(PyExc_ValueError, "%lld is not a valid %s.",
371 (long long) key, t->name);
372 else
373 PyErr_Format(PyExc_ValueError, "%llu is not a valid %s.",
374 (unsigned long long) key, t->name);
375
376 return nullptr;
377}

71 template <typename T>
72 bool try_variant(const handle &src, uint8_t flags, cleanup_list *cleanup) {
73 using CasterT = make_caster<T>;
74
75 CasterT caster;
76
77 if (!caster.from_python(src, flags_for_local_caster<T>(flags), cleanup) ||
78 !caster.template can_cast<T>())
79 return false;
80
81 this->store(caster.operator cast_t<T>());

83 return true;

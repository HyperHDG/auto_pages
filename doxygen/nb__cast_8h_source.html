102template <typename T>
103NB_INLINE uint8_t flags_for_local_caster(uint8_t flags) noexcept {
104 using Caster = make_caster<T>;
105 constexpr bool is_ref = std::is_pointer_v<T> || std::is_reference_v<T>;
106 if constexpr (is_base_caster_v<Caster>) {
107 if constexpr (is_ref) {
108 /* References/pointers to a type produced by implicit conversions
109 refer to storage owned by the cleanup_list. In a nb::cast() call,
110 that storage will be released before the reference can be used;
111 to prevent dangling, don't allow implicit conversions there. */
112 if (flags & ((uint8_t) cast_flags::manual))
113 flags &= ~((uint8_t) cast_flags::convert);
114 }
115 flags |= none_disallowed_flag<T>;
116 } else {
117 /* Any pointer produced by a non-base caster will generally point
118 into storage owned by the caster, which won't live long enough.
119 Exception: the 'char' caster produces a result that points to
120 storage owned by the incoming Python 'str' object, so it's OK. */
121 static_assert(!is_ref || std::is_same_v<T, const char*> ||
122 (std::is_pointer_v<T> && std::is_constructible_v<T*, Caster>),
123 "nanobind generally cannot produce objects that "
124 "contain interior pointers T* (or references T&) if "
125 "the pointee T is not handled by nanobind's regular "
126 "class binding mechanism. For example, you can write "
127 "a function that accepts int*, or std::vector<int>, "
128 "but not std::vector<int*>.");
129 }
130 return flags;
131}

133template <typename T>
134struct type_caster<T, enable_if_t<std::is_arithmetic_v<T> && !is_std_char_v<T>>> {
135 NB_INLINE bool from_python(handle src, uint8_t flags, cleanup_list *) noexcept {
136 if constexpr (std::is_floating_point_v<T>) {
137 if constexpr (std::is_same_v<T, double>) {
138 return detail::load_f64(src.ptr(), flags, &value);
139 } else if constexpr (std::is_same_v<T, float>) {
140 return detail::load_f32(src.ptr(), flags, &value);
141 } else {
142 double d;
143 if (!detail::load_f64(src.ptr(), flags, &d))
144 return false;
145 T result = (T) d;
146 if ((flags & (uint8_t) cast_flags::convert)
147 || (double) result == d
148 || (result != result && d != d)) {
149 value = result;
150 return true;
151 }
152 return false;
153 }
154 } else {
155 if constexpr (std::is_signed_v<T>) {
156 if constexpr (sizeof(T) == 8)
157 return detail::load_i64(src.ptr(), flags, (int64_t *) &value);
158 else if constexpr (sizeof(T) == 4)
159 return detail::load_i32(src.ptr(), flags, (int32_t *) &value);
160 else if constexpr (sizeof(T) == 2)
161 return detail::load_i16(src.ptr(), flags, (int16_t *) &value);
162 else
163 return detail::load_i8(src.ptr(), flags, (int8_t *) &value);
164 } else {
165 if constexpr (sizeof(T) == 8)
166 return detail::load_u64(src.ptr(), flags, (uint64_t *) &value);
167 else if constexpr (sizeof(T) == 4)
168 return detail::load_u32(src.ptr(), flags, (uint32_t *) &value);
169 else if constexpr (sizeof(T) == 2)
170 return detail::load_u16(src.ptr(), flags, (uint16_t *) &value);
171 else
172 return detail::load_u8(src.ptr(), flags, (uint8_t *) &value);
173 }
174 }
175 }

177 NB_INLINE static handle from_cpp(T src, rv_policy, cleanup_list *) noexcept {
178 if constexpr (std::is_floating_point_v<T>) {
179 return PyFloat_FromDouble((double) src);
180 } else {
181 if constexpr (std::is_signed_v<T>) {
182 if constexpr (sizeof(T) <= sizeof(long))
183 return PyLong_FromLong((long) src);

185 return PyLong_FromLongLong((long long) src);
186 } else {
187 if constexpr (sizeof(T) <= sizeof(unsigned long))
188 return PyLong_FromUnsignedLong((unsigned long) src);
189 else
190 return PyLong_FromUnsignedLongLong((unsigned long long) src);
191 }

455template <typename T> NB_INLINE rv_policy infer_policy(rv_policy policy) {
456 if constexpr (is_pointer_v<T>) {
457 if (policy == rv_policy::automatic)
458 policy = rv_policy::take_ownership;
459 else if (policy == rv_policy::automatic_reference)
460 policy = rv_policy::reference;
461 } else if constexpr (std::is_lvalue_reference_v<T>) {
462 if (policy == rv_policy::automatic ||
463 policy == rv_policy::automatic_reference)
464 policy = rv_policy::copy;
465 } else {
466 if (policy == rv_policy::automatic ||
467 policy == rv_policy::automatic_reference ||
468 policy == rv_policy::reference ||
469 policy == rv_policy::reference_internal)
470 policy = rv_policy::move;
471 }
472 return policy;
473}

477template <typename Type_> struct type_caster_base : type_caster_base_tag {
478 using Type = Type_;
479 static constexpr auto Name = const_name<Type>();
480 template <typename T> using Cast = precise_cast_t<T>;
481
482 NB_INLINE bool from_python(handle src, uint8_t flags,
483 cleanup_list *cleanup) noexcept {
484 // The 'trusted' fast path lives in the pointer_and_handle caster (the
485 // only one that is ever trusted) and, as a fallback, in nb_type_get.
486 // The generic base caster therefore need not test for it here, which
487 // would only add a never-taken branch to every bound-type argument.
488 return nb_type_get(&typeid(Type), src.ptr(), flags, cleanup,
489 (void **) &value);
490 }

492 template <typename T>
493 NB_INLINE static handle from_cpp(T &&value, rv_policy policy,
494 cleanup_list *cleanup) noexcept {
495 Type *ptr;
496 if constexpr (is_pointer_v<T>)
497 ptr = (Type *) value;
498 else
499 ptr = (Type *) &value;
500
501 policy = infer_policy<T>(policy);
502 const std::type_info *type = &typeid(Type);
503
504 constexpr bool has_type_hook =
505 !std::is_base_of_v<std::false_type, type_hook<Type>>;
506 if constexpr (has_type_hook)
507 type = type_hook<Type>::get(ptr);
508
509 if constexpr (!std::is_polymorphic_v<Type>) {
510 return nb_type_put(type, ptr, policy, cleanup);
511 } else {
512 const std::type_info *type_p =
513 (!has_type_hook && ptr) ? &typeid(*ptr) : nullptr;
514 return nb_type_put_p(type, type_p, ptr, policy, cleanup);
515 }
516 }

535template <bool Convert, typename T>
536T cast_impl(handle h) {
537 using Caster = detail::make_caster<T>;
538
539 // A returned reference/pointer would usually refer into the type_caster
540 // object, which will be destroyed before the returned value can be used,
541 // so we prohibit it by default, with two exceptions that we know are safe:
542 //
543 // - If we're casting to a bound object type, the returned pointer points
544 // into storage owned by that object, not the type caster. Note this is
545 // only safe if we don't allow implicit conversions, because the pointer
546 // produced after an implicit conversion points into storage owned by
547 // a temporary object in the cleanup list, and we have to release those
548 // temporaries before we return.
549 //
550 // - If we're casting to const char*, the caster was provided by nanobind,
551 // and we know it will only accept Python 'str' objects, producing
552 // a pointer to storage owned by that object.
553
554 constexpr bool is_ref = std::is_reference_v<T> || std::is_pointer_v<T>;
555 static_assert(
556 !is_ref ||
557 is_base_caster_v<Caster> ||
558 std::is_same_v<const char *, T>,
559 "nanobind::cast(): cannot return a reference to a temporary.");
560
561 Caster caster;
562 bool rv;
563 if constexpr (Convert && !is_ref) {
564 // Release the values in the cleanup list only after we
565 // initialize the return object, since the initialization
566 // might access those temporaries.
567 struct raii_cleanup {
568 cleanup_list list{nullptr};
569 ~raii_cleanup() { list.release(); }
570 } cleanup;
571 rv = caster.from_python(h.ptr(),
572 ((uint8_t) cast_flags::convert) |
573 ((uint8_t) cast_flags::manual),
574 &cleanup.list) &&
575 caster.template can_cast<T>();
576 if (!rv)
577 detail::raise_python_or_cast_error();
578 return caster.operator cast_t<T>();
579 } else {
580 rv = caster.from_python(h.ptr(), (uint8_t) cast_flags::manual, nullptr) &&
581 caster.template can_cast<T>();
582 if (!rv)
583 detail::raise_python_or_cast_error();
584 return caster.operator cast_t<T>();
585 }
586}

588template <bool Convert, typename T>
589bool try_cast_impl(handle h, T &out) noexcept {
590 using Caster = detail::make_caster<T>;
591
592 // See comments in cast_impl above
593 constexpr bool is_ref = std::is_reference_v<T> || std::is_pointer_v<T>;
594 static_assert(
595 !is_ref ||
596 is_base_caster_v<Caster> ||
597 std::is_same_v<const char *, T>,
598 "nanobind::try_cast(): cannot return a reference to a temporary.");
599
600 Caster caster;
601 bool rv;
602 if constexpr (Convert && !is_ref) {
603 cleanup_list cleanup(nullptr);
604 rv = caster.from_python(h.ptr(),
605 ((uint8_t) cast_flags::convert) |
606 ((uint8_t) cast_flags::manual),
607 &cleanup) &&
608 caster.template can_cast<T>();
609 if (rv) {
610 out = caster.operator cast_t<T>();
611 }
612 cleanup.release(); // 'from_python' is 'noexcept', so this always runs
613 } else {
614 rv = caster.from_python(h.ptr(), (uint8_t) cast_flags::manual, nullptr) &&
615 caster.template can_cast<T>();
616 if (rv) {
617 out = caster.operator cast_t<T>();
618 }
619 }
620
621 return rv;
622}

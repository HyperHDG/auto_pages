17// Deleter for std::unique_ptr<T> (handles ownership by both C++ and Python)
18template <typename T> struct deleter {
20 deleter() = default;
21
23 deleter(handle h) : o(h.ptr()) { }
24
26 bool owned_by_python() const { return o != nullptr; }
27
29 bool owned_by_cpp() const { return o == nullptr; }
30
32 void operator()(void *p) noexcept {
33 if (o) {
34 // Don't run the deleter if the interpreter has been shut down
35 if (!is_alive())
36 return;
37 gil_scoped_acquire guard;
38 Py_DECREF(o);
39 } else {
40 delete (T *) p;
41 }
42 }
43
44 PyObject *o{nullptr};
45};

78 /* If true, the Python object has relinquished ownership but we have
79 not yet yielded a unique_ptr that holds ownership on the C++ side.
80
81 `nb_type_relinquish_ownership()` can fail, so we must check it in
82 `can_cast()`. If we do so, but then wind up not executing the cast
83 operator, we must remember to undo our relinquishment and push the
84 ownership back onto the Python side. For example, this might be
85 necessary if the Python object `[(foo, foo)]` is converted to
86 `std::vector<std::pair<std::unique_ptr<T>, std::unique_ptr<T>>>`;
87 the pair caster won't know that it can't cast the second element
88 until after it's verified that it can cast the first one. */
89 mutable bool inflight = false;
90
91 ~type_caster() {
92 if (inflight)
93 nb_type_restore_ownership(src.ptr(), IsDefaultDeleter);
94 }

96 bool from_python(handle src_, uint8_t, cleanup_list *) noexcept {

98 src = src_;

100 /* Try casting to a pointer of the underlying type. We pass flags=0 and
101 cleanup=nullptr to prevent implicit type conversions (they are
102 problematic since the instance then wouldn't be owned by 'src') */
103 return caster.from_python(src_, 0, nullptr);

115 template <typename T2>
116 static handle from_cpp(T2 &&value,
117 rv_policy, cleanup_list *cleanup) noexcept {
118
119 bool cpp_delete = true;
120 if constexpr (IsNanobindDeleter)
121 cpp_delete = value.get_deleter().owned_by_cpp();
122
123 Td *ptr = (Td *) value.get();
124 const std::type_info *type = &typeid(Td);
125 if (!ptr)
126 return none().release();
127
128 constexpr bool has_type_hook =
129 !std::is_base_of_v<std::false_type, type_hook<Td>>;
130 if constexpr (has_type_hook)
131 type = type_hook<Td>::get(ptr);
132
133 handle result;
134 if constexpr (!std::is_polymorphic_v<Td>) {
135 result = nb_type_put_unique(type, ptr, cleanup, cpp_delete);
136 } else {
137 const std::type_info *type_p =
138 (!has_type_hook && ptr) ? &typeid(*ptr) : nullptr;
139
140 result = nb_type_put_unique_p(type, type_p, ptr, cleanup, cpp_delete);
141 }
142
143 if (result.is_valid()) {
144 if (cpp_delete)
145 value.release();
146 else
147 value.reset();
148 }
149
150 return result;
151 }

153 template <typename T_>
154 bool can_cast() const noexcept {
155 if (src.is_none() || inflight)
156 return true;
157 else if (!nb_type_relinquish_ownership(src.ptr(), IsDefaultDeleter))
158 return false;
159 inflight = true;
160 return true;
161 }
162
163 explicit operator Value() {
164 if (!inflight && !src.is_none() &&
165 !nb_type_relinquish_ownership(src.ptr(), IsDefaultDeleter))
166 throw next_overload();
167
168 Td *p = caster.operator Td *();
169
170 Value value;
171 if constexpr (IsNanobindDeleter)
172 value = Value(p, deleter<T>(src.inc_ref()));
173 else
174 value = Value(p);
175 inflight = false;
176 return value;
177 }

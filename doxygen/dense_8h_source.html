76/* This type alias builds the most suitable 'ndarray' for the given Eigen type.
77 In particular, it
78
79 - matches the underlying scalar type
80 - matches the number of dimensions (i.e. whether the type is a vector/matrix)
81 - matches the shape (if the row/column count is known at compile time)
82 - matches the in-memory ordering when the Eigen type is contiguous.
83
84 This is helpful because type_caster<ndarray<..>> will then perform the
85 necessary conversion steps (if given incompatible input) to enable data
86 exchange with Eigen.
87
88 A limitation of this approach is that ndarray does not support compile-time
89 strides besides c_contig and f_contig. If an Eigen type requires
90 non-contiguous strides (at compile-time) and we are given an ndarray with
91 unsuitable strides (at run-time), type casting will fail. Note, however, that
92 this is rather unusual, since the default stride type of Eigen::Map requires
93 contiguous memory, and the one of Eigen::Ref requires a contiguous inner
94 stride, while handling any outer stride.
95*/

97template <typename T, typename Scalar = typename T::Scalar>
98using array_for_eigen_t = ndarray<
99 Scalar,
100 numpy,
101 std::conditional_t<
102 ndim_v<T> == 1,
103 shape<T::SizeAtCompileTime>,
104 shape<T::RowsAtCompileTime,
105 T::ColsAtCompileTime>>,
106 std::conditional_t<
107 is_contiguous_v<T>,
108 std::conditional_t<
109 ndim_v<T> == 1 || T::IsRowMajor,
110 c_contig,
111 f_contig>,
112 unused>>;

129struct type_caster<T, enable_if_t<is_eigen_plain_v<T> &&
130 is_ndarray_scalar_v<typename T::Scalar>>> {

137 bool from_python(handle src, uint8_t flags, cleanup_list *cleanup) noexcept {
138 // We're in any case making a copy, so non-writable inputs area also okay
139 using NDArrayConst = array_for_eigen_t<T, const typename T::Scalar>;
140 make_caster<NDArrayConst> caster;
141 if (!caster.from_python(src, flags & ~(uint8_t)cast_flags::accepts_none, cleanup))
142 return false;
143
144 const NDArrayConst &array = caster.value;
145 if constexpr (ndim_v<T> == 1)
146 value.resize((Eigen::Index) array.shape(0));
147 else
148 value.resize((Eigen::Index) array.shape(0),
149 (Eigen::Index) array.shape(1));
150
151 // The layout is contiguous & compatible thanks to array_for_eigen_t<T>
152 memcpy(value.data(), array.data(), array.size() * sizeof(Scalar));
153
154 return true;
155 }

157 template <typename T2>
158 static handle from_cpp(T2 &&v, rv_policy policy, cleanup_list *cleanup) noexcept {
159 policy = infer_policy<T2>(policy);
160 if constexpr (std::is_pointer_v<T2>)
161 return from_cpp_internal((const T &) *v, policy, cleanup);
162 else
163 return from_cpp_internal((const T &) v, policy, cleanup);
164 }
165
166 static handle from_cpp_internal(const T &v, rv_policy policy, cleanup_list *cleanup) noexcept {
167 size_t shape[ndim_v<T>];
168 int64_t strides[ndim_v<T>];
169
170 if constexpr (ndim_v<T> == 1) {
171 shape[0] = (size_t) v.size();
172 strides[0] = v.innerStride();
173 } else {
174 shape[0] = (size_t) v.rows();
175 shape[1] = (size_t) v.cols();
176 strides[0] = v.rowStride();
177 strides[1] = v.colStride();
178 }
179
180 void *ptr = (void *) v.data();
181
182 if (policy == rv_policy::move) {
183 // Don't bother moving when the data is static or occupies <1KB
184 if ((T::SizeAtCompileTime != Eigen::Dynamic ||
185 (size_t) v.size() < (1024 / sizeof(Scalar))))
186 policy = rv_policy::copy;
187 }
188
189 object owner;
190 if (policy == rv_policy::move) {
191 T *temp = new T((T&&) v);
192 owner = capsule(temp, [](void *p) noexcept { delete (T *) p; });
193 ptr = temp->data();
194 policy = rv_policy::reference;
195 } else if (policy == rv_policy::reference_internal) {
196 // reference_internal needs a self pointer; give up if unavailable
197 if (!cleanup || !cleanup->self())
198 return handle();
199 owner = borrow(cleanup->self());
200 policy = rv_policy::reference;
201 }
202
203 object o = steal(NDArrayCaster::from_cpp(
204 NDArray(ptr, ndim_v<T>, shape, owner, strides),
205 policy, cleanup));
206
207 return o.release();
208 }
209};

260 bool from_python_(handle src, uint8_t flags, cleanup_list* cleanup) noexcept {
261 if (!caster.from_python(src, flags & ~(uint8_t)cast_flags::accepts_none, cleanup))
262 return false;
263
264 // Check for memory layout compatibility of non-contiguous 'Map' types
265 if constexpr (!is_contiguous_v<Map>) {
266 // Dynamic inner strides support any input, check the fixed case
267 if constexpr (StrideType::InnerStrideAtCompileTime != Eigen::Dynamic) {
268 // A compile-time stride of 0 implies "contiguous" ..
269 int64_t is_expected = StrideType::InnerStrideAtCompileTime == 0
270 ? 1 /* .. and equals 1 for the inner stride */
271 : StrideType::InnerStrideAtCompileTime,
272 is_actual = caster.value.stride(
273 (ndim_v<T> != 1 && T::IsRowMajor) ? 1 : 0);
274
275 if (is_expected != is_actual)
276 return false;
277 }
278
279 // Analogous check for the outer strides
280 if constexpr (ndim_v<T> == 2 && StrideType::OuterStrideAtCompileTime != Eigen::Dynamic) {
281 int64_t os_expected = StrideType::OuterStrideAtCompileTime == 0
282 ? caster.value.shape(T::IsRowMajor ? 1 : 0)
283 : StrideType::OuterStrideAtCompileTime,
284 os_actual = caster.value.stride(T::IsRowMajor ? 0 : 1);
285
286 if (os_expected != os_actual)
287 return false;
288 }
289 }
290 return true;
291 }

293 static handle from_cpp(const Map &v, rv_policy policy, cleanup_list *cleanup) noexcept {
294 size_t shape[ndim_v<T>];
295 int64_t strides[ndim_v<T>];
296
297 if constexpr (ndim_v<T> == 1) {
298 shape[0] = (size_t) v.size();
299 strides[0] = v.innerStride();
300 } else {
301 shape[0] = (size_t) v.rows();
302 shape[1] = (size_t) v.cols();
303 strides[0] = v.rowStride();
304 strides[1] = v.colStride();
305 }

316 StrideType strides() const {
317 constexpr int IS = StrideType::InnerStrideAtCompileTime,
318 OS = StrideType::OuterStrideAtCompileTime;
319
320 int64_t inner = caster.value.stride(0),
321 outer;
322
323 if constexpr (ndim_v<T> == 1)
324 outer = (int64_t) caster.value.shape(0);
325 else
326 outer = caster.value.stride(1);
327
328 (void) inner; (void) outer;
329 if constexpr (ndim_v<T> == 2 && T::IsRowMajor)
330 std::swap(inner, outer);
331
332 // Eigen may expect a stride of 0 to avoid an assertion failure
333 if constexpr (IS == 0)
334 inner = 0;
335
336 // Starting from numpy 2.4, dl_tensors' stride field is *always* set (for ndim > 0).
337 // This also includes when shape=(0,0), when numpy reports the stride to be zero.
338 // This creates an incompatibility with Eigen compile-time vectors, which expect
339 // runtime and compile-time strides to be identical (e.g. for Eigen::VectorXi, equal to 1).
340 // For dynamic strides (IS == Eigen::Dynamic), substitute a unit inner stride
341 if constexpr (ndim_v<T> == 1) {
342 if (caster.value.shape(0) == 0)
343 inner = IS == Eigen::Dynamic ? 1 : IS;
344 }
345
346 if constexpr (OS == 0)
347 outer = 0;
348
349 if constexpr (std::is_same_v<StrideType, Eigen::InnerStride<IS>>)
350 return StrideType(inner);
351 else if constexpr (std::is_same_v<StrideType, Eigen::OuterStride<OS>>)
352 return StrideType(outer);
353 else
354 return StrideType(outer, inner);
355 }
356
357 operator Map() {
358 NDArray &t = caster.value;
359 if constexpr (ndim_v<T> == 1)
360 return Map(t.data(), (Eigen::Index) t.shape(0), strides());
361 else
362 return Map(t.data(), (Eigen::Index) t.shape(0),
363 (Eigen::Index) t.shape(1), strides());
364 }

428 bool from_python(handle src, uint8_t flags, cleanup_list *cleanup) noexcept {
429 // Try a direct cast without implicit conversion first
430 if (caster.from_python(src, flags, cleanup))
431 return true;
432
433 // Potentially convert strides/dtype when casting constant references
434 if constexpr (MaybeConvert) {
435 /* Generating an implicit copy requires some object to assume
436 ownership. During a function call, ``dcaster`` can serve that
437 role (this case is detected by checking whether ``flags`` has
438 the ``manual`` flag set). When used in other situations (e.g.
439 ``nb::cast()``), the created ``Eigen::Ref<..>`` must take
440 ownership of the copy. This is only guranteed to work if
441 DMapConstructorOwnsData.
442
443 If neither of these is possible, we disable implicit
444 conversions. */
445
446 if ((flags & (uint8_t) cast_flags::manual) &&
447 !DMapConstructorOwnsData)
448 flags &= ~(uint8_t) cast_flags::convert;
449
450 if (dcaster.from_python_(src, flags, cleanup))
451 return true;
452 }
453
454 return false;
455 }

123const char *python_error::what() const noexcept {
124 using namespace nanobind::detail;
125
126 // Return the existing error message if already computed once
127 if (m_what)
128 return m_what;
129
130 gil_scoped_acquire acq;
131
132 // Try again with GIL held
133 if (m_what)
134 return m_what;
135
136#if PY_VERSION_HEX < 0x030C0000
137 PyErr_NormalizeException(&m_type, &m_value, &m_traceback);
138 check(m_type,
139 "nanobind::python_error::what(): PyErr_NormalizeException() failed!");
140
141 if (m_traceback) {
142 if (PyException_SetTraceback(m_value, m_traceback) < 0)
143 PyErr_Clear();
144 }
145
146 handle exc_type = m_type, exc_value = m_value;
147#else
148 handle exc_value = m_value, exc_type = exc_value.type();
149#endif
150 object exc_traceback = traceback();
151
152#if defined(Py_LIMITED_API) || defined(PYPY_VERSION)
153 char *tmp;
154 try {
155 object mod = module_::import_("traceback"),
156 result = mod.attr("format_exception")(exc_type, exc_value, exc_traceback);
157 str s = borrow<str>(str("\n").attr("join")(result));
158 const char *cstr = s.c_str();
159 if (!cstr) // e.g. lone surrogates from an unencodable file name
160 raise_python_error();
161 tmp = strdup_check(cstr);
162 } catch (...) {
163 PyErr_Clear();
164 tmp = strdup_check("<error while formatting exception>");
165 }
166#else
167 Buffer exc_buf(128);
168 if (exc_traceback.is_valid()) {
169 PyTracebackObject *to = (PyTracebackObject *) exc_traceback.ptr();
170
171 // Get the deepest trace possible
172 while (to->tb_next)
173 to = to->tb_next;
174
175 PyFrameObject *frame = to->tb_frame;
176 Py_XINCREF(frame);
177
178 std::vector<PyFrameObject *, py_allocator<PyFrameObject *>> frames;
179
180 while (frame) {
181 frames.push_back(frame);
182 frame = PyFrame_GetBack(frame);
183 }
184
185 exc_buf.put("Traceback (most recent call last):\n");
186 for (auto it = frames.rbegin(); it != frames.rend(); ++it) {
187 frame = *it;
188 PyCodeObject *f_code = PyFrame_GetCode(frame);
189 const char *filename = borrow<str>(f_code->co_filename).c_str();
190 if (!filename) {
191 PyErr_Clear();
192 filename = "<unencodable filename>";
193 }
194 const char *name = borrow<str>(f_code->co_name).c_str();
195 if (!name) {
196 PyErr_Clear();
197 name = "<unencodable name>";
198 }
199 exc_buf.put(" File \"");
200 exc_buf.put_dstr(filename);
201 exc_buf.put("\", line ");
202 exc_buf.put_uint32((uint32_t) PyFrame_GetLineNumber(frame));
203 exc_buf.put(", in ");
204 exc_buf.put_dstr(name);
205 exc_buf.put('\n');
206 Py_DECREF(f_code);
207 Py_DECREF(frame);
208 }
209 }
210
211 if (exc_type.is_valid()) {
212 try {
213 object name = exc_type.attr(NB_INTERNED(__name__));
214 exc_buf.put_dstr(borrow<str>(name).c_str());
215 exc_buf.put(": ");
216 } catch (...) { PyErr_Clear(); }
217 }
218
219 if (exc_value.is_valid()) {
220 try {
221 exc_buf.put_dstr(str(exc_value).c_str());
222 } catch (...) {
223 PyErr_Clear();
224 exc_buf.put("<exception str() failed>");
225 }
226 }
227
228 char *tmp = exc_buf.copy();
229#endif
230
231 // Publish the message with a CAS; if a concurrent call raced us to it,
232 // free our copy and return the winner's message instead.
233 char *expected = nullptr;
234#if defined(_MSC_VER)
235 expected = (char *) _InterlockedCompareExchangePointer(
236 (void *volatile *) &m_what, tmp, nullptr);
237 if (!expected)
238 return tmp;
239#else
240 if (__atomic_compare_exchange_n(&m_what, &expected, tmp, false,
241 __ATOMIC_RELEASE, __ATOMIC_ACQUIRE))
242 return tmp;
243#endif
244 free(tmp);
245 return expected;
246}

260NB_CORE PyObject *exception_new(PyObject *scope, const char *name,
261 PyObject *base) {
262 object modname;
263 if (PyModule_Check(scope))
264 modname = getattr(scope, "__name__", handle());
265 else
266 modname = getattr(scope, NB_INTERNED(__module__), handle());
267
268 if (!modname.is_valid())
269 raise("nanobind::detail::exception_new(): could not determine module "
270 "name!");
271
272 str combined =
273 steal<str>(PyUnicode_FromFormat("%U.%s", modname.ptr(), name));
274
275 object result = steal(PyErr_NewException(combined.c_str(), base, nullptr));
276 check(result, "nanobind::detail::exception_new(): creation failed!");
277
278 if (hasattr(scope, name))
279 raise("nanobind::detail::exception_new(): an object of the same name "
280 "already exists!");
281
282 setattr(scope, name, result);
283 return result.release().ptr();
284}

288static void chain_error_v(handle type, const char *fmt, va_list args) noexcept {
289#if PY_VERSION_HEX >= 0x030C0000
290 PyObject *value = PyErr_GetRaisedException();
291#else
292 PyObject *tp = nullptr, *value = nullptr, *traceback = nullptr;
293
294 PyErr_Fetch(&tp, &value, &traceback);
295
296 if (tp) {
297 PyErr_NormalizeException(&tp, &value, &traceback);
298 if (traceback) {
299 PyException_SetTraceback(value, traceback);
300 Py_DECREF(traceback);
301 }
302
303 Py_DECREF(tp);
304 tp = traceback = nullptr;
305 }
306#endif
307
308#if !defined(PYPY_VERSION)
309 PyErr_FormatV(type.ptr(), fmt, args);
310#else
311 PyObject *exc_str = PyUnicode_FromFormatV(fmt, args);
312 check(exc_str, "nanobind::detail::raise_from(): PyUnicode_FromFormatV() failed!");
313 PyErr_SetObject(type.ptr(), exc_str);
314 Py_DECREF(exc_str);
315#endif
316
317 if (!value)
318 return;
319
320 PyObject *value_2 = nullptr;
321#if PY_VERSION_HEX >= 0x030C0000
322 value_2 = PyErr_GetRaisedException();
323#else
324 PyErr_Fetch(&tp, &value_2, &traceback);
325 PyErr_NormalizeException(&tp, &value_2, &traceback);
326#endif
327
328 Py_INCREF(value);
329 PyException_SetCause(value_2, value); // steals
330 PyException_SetContext(value_2, value); // steals
331
332#if PY_VERSION_HEX >= 0x030C0000
333 PyErr_SetRaisedException(value_2);
334#else
335 PyErr_Restore(tp, value_2, traceback);
336#endif
337}
